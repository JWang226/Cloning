Repository identifier: thm:grassmann

\begin{theorem}[Cloning projector states]
\label{thm:grassmann}
For fixed \(1\le r<d\), let \(\Gr(r,d)\) denote the set of rank-\(r\)
orthogonal projectors on \(\C^d\).  Then
\begin{equation}\label{eq:grassmann-value}
  \lim_{n\to\infty}\sup_{\cM_n}\inf_{P\in\Gr(r,d)}
  F\!\left(\cM_n((P/r)^{\ot n}),(P/r)^{\ot m_n}\right)
  =\gamma^{-r(d-r)/2}.
\end{equation}
The limit is attained by our cloner \(\cC_n^{\mathrm{proj}}\) constructed in \cref{sec:grassmann-achievability}.
\end{theorem}
