167/-- The actual amplifier sends joint vacuum to the product thermal density. -/
168theorem channel_vacuum (q : Fin d → ℝ)
169    (hq0 : ∀ i, 0 ≤ q i) (hq1 : ∀ i, q i < 1) :
170    (channel q hq0 hq1).toLinearMap (coherentProjector (0 : Fin d → ℂ)) =
171      vectorMixture (Cloning.MultimodeCoherentGaussianMixture.numberBasis d)
172        (Cloning.ThermalWitness.productGeometric q) := by
173  have h := channel_hasSum q hq0 hq1
174    (vectorProjector (Cloning.MultimodeIdler.numberBasis d 0))
175  simp only [krausTerm_vectorProjector, kraus_numberBasis, zero_add, productLaw_zero] at h
176  have he (k : Occupation d) : vectorProjector
177      ((Real.sqrt (Cloning.ThermalWitness.productGeometric q k) : ℂ) •
178        Cloning.MultimodeIdler.numberBasis d k) =
179      (Cloning.ThermalWitness.productGeometric q k : ℂ) •
180        vectorProjector (Cloning.MultimodeIdler.numberBasis d k) :=
181    Cloning.BosonicAmplifier.vectorProjector_sqrt_smul _
182      (Cloning.ThermalWitness.productGeometric_nonneg hq0 hq1 k) _
183  simp only [he] at h
184  change (channel q hq0 hq1).toLinearMap (vectorProjector (coherentVector 0)) = _
185  rw [coherentVector_zero_eq_numberBasis]
