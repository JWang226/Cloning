14theorem modeFactor_hasDerivAt {g q : ℝ} (hg : 1 < g) (hq : 0 < q) :
15    HasDerivAt (Thermal.modeFactor g)
16      ((g-1)^2 / (2*Real.sqrt (q*(g-1+q)) *
17        (Real.sqrt (g*(g-1+q))+Real.sqrt q)^2)) q := by
18  have hg0 : 0 < g := by linarith
19  have hv : 0 < g-1+q := by linarith
20  have hrad : 0 < q*(g-1+q) := mul_pos hq hv
21  have hden : g+q ≠ 0 := ne_of_gt (add_pos hg0 hq)
22  have hd := (((hasDerivAt_const q (Real.sqrt g)).add
23    (((hasDerivAt_id q).mul ((hasDerivAt_id q).const_add (g-1))).sqrt hrad.ne')).div
24    ((hasDerivAt_id q).const_add g) hden)
25  have hr := Real.sq_sqrt hq.le
26  have hs := Real.sq_sqrt hv.le
27  have ht := Real.sq_sqrt hg0.le
28  have hrs : Real.sqrt (q*(g-1+q)) = Real.sqrt q*Real.sqrt (g-1+q) :=
29    Real.sqrt_mul hq.le _
30  have hts : Real.sqrt (g*(g-1+q)) = Real.sqrt g*Real.sqrt (g-1+q) :=
31    Real.sqrt_mul hg0.le _
32  have hr0 : Real.sqrt q ≠ 0 := ne_of_gt (Real.sqrt_pos.mpr hq)
33  have hs0 : Real.sqrt (g-1+q) ≠ 0 := ne_of_gt (Real.sqrt_pos.mpr hv)
34  have hsum : Real.sqrt g*Real.sqrt (g-1+q)+Real.sqrt q ≠ 0 := by positivity
35  have htsq : (Real.sqrt g*Real.sqrt (g-1+q))^2=g*(g-1+q) := by
36    rw [mul_pow,ht,hs]
37  have hrsq : (Real.sqrt q*Real.sqrt (g-1+q))^2=q*(g-1+q) := by
38    rw [mul_pow,hr,hs]
39  have hnum : (g-1+q+q)*(g+q)-2*Real.sqrt q*Real.sqrt (g-1+q)*
40      (Real.sqrt g+Real.sqrt q*Real.sqrt (g-1+q)) =
41      (Real.sqrt g*Real.sqrt (g-1+q)-Real.sqrt q)^2 := by
42    nlinarith [htsq,hrsq]
43  have hprod : (Real.sqrt g*Real.sqrt (g-1+q)-Real.sqrt q)*
44      (Real.sqrt g*Real.sqrt (g-1+q)+Real.sqrt q)=(g-1)*(g+q) := by
45    nlinarith [htsq]

52        (Real.sqrt g*Real.sqrt (g-1+q)+Real.sqrt q))^2 := by rw [hprod]; ring
53    _ = _ := by rw [hnum]; ring
