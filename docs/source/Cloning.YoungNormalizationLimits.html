19def flatTableauPMF (r N : ℕ) (hr : 0 < r) : PMF (Shape r N) :=

27theorem flatTableauPMF_dimension_moments (r k : ℕ) (hr : 0 < r) :
28    Tendsto (fun N => ∑ μ, (flatTableauPMF r N hr μ).toReal *
29      normalizedDimension r k N μ) atTop (𝓝 1) ∧
30    Tendsto (fun N => ∑ μ, (flatTableauPMF r N hr μ).toReal *
31      inverseNormalizedDimension r k N μ) atTop (𝓝 1) :=
32  dimension_moments_tendsto_one r k hr (fun N => flatTableauPMF r N hr)
33    (fun N => flatTableauPMF_toReal r N hr)

37theorem tableauPMF_fallback_errors {d : ℕ} (hd : 1 ≤ d)
38    (K : Set (Fin (d + 1) → ℝ)) (hK : IsCompact K)
39    (hpos : ∀ p ∈ K, ∀ i, 0 < p i) (hanti : ∀ p ∈ K, StrictAnti p)
40    (m : ℕ → ℤ) (γn : ℕ → ℝ) (γ : ℝ) (hγ : 1 < γ)
41    (hγn : Tendsto γn atTop (𝓝 γ))
42    (hm : ∀ n, (m n : ℝ) = γn n * (n : ℝ))
43    (p : ℕ → Fin (d + 1) → ℝ) (hp : ∀ n, p n ∈ K)
44    (hs : ∀ n, ∑ i, p n i = 1) (Q : ℕ → PMF (Fin (d + 1) → ℤ)) :
45    let P := fun N => tableauPMF N (p N) (fun i => (hpos (p N) (hp N) i).le) (hs N)
46    Tendsto (fun n => CountableScheffe.l1Distance

48      (probability ((P n).bind (fun μ => rawKernel (γn n) (m n) (integerShape μ)))))
