5/-!
6# Weighted hybrid fidelity bound
7 
8This is the continuous classical–quantum inequality in Appendix C. The output
9is an arbitrary positive operator-valued integrable density, the target is the
10actual field `g(y) T`, and the bounded positive witness may have no bounded
11inverse. Its inverse moment is controlled through genuine bounded regularizations.
12-/

36/-- The pointwise weighted bound for the actual product target density. -/
37theorem rootFidelity_product_sq_le (R : PositiveField (H := H) μ)
38    (g : Ω → ℝ) (hg : Integrable g μ) (hgn : ∀ x, 0 ≤ g x) (T : DensityState H)
39    {W : H →L[ℂ] H} (hW : 0 ≤ W) {M : ℝ} (hM : 0 ≤ M)
40    (hInv : ∀ ε : ℝ, 0 < ε →
41      witnessMoment (TraceClass.ofOperator T.op T.traceClass)
42        (CFC.rpow (regularizedWeight W ε) (-1)) ≤ M) (x : Ω) :
43    PositiveTraceClass.rootFidelity (R.value x) ((product g hg hgn T).value x) ^ 2 ≤
44      witnessMoment (R.value x).1 W * (g x * M) := by
45  apply fidelity_sq_le_of_regularized_inverse_moment (R.value x).2
46    ((product g hg hgn T).value x).2 hW (R.value x).1.2
47    ((product g hg hgn T).value x).1.2 (mul_nonneg (hgn x) hM)
48  intro ε hε
49  change witnessMoment ((product g hg hgn T).value x).1
50    (CFC.rpow (regularizedWeight W ε) (-1)) ≤ g x * M
51  rw [product_value, witnessMoment_real_smul]
52  exact mul_le_mul_of_nonneg_left (hInv ε hε) (hgn x)

54/-- Appendix C's weighted hybrid fidelity inequality for general measurable
55positive weights and arbitrary trace-class quantum output densities.
56The two integrability premises express precisely the finiteness of its RHS. -/
57theorem rootFidelity_sq_le_weighted_product (R : PositiveField (H := H) μ)
58    (g : Ω → ℝ) (hg : Integrable g μ) (hgn : ∀ x, 0 ≤ g x) (T : DensityState H)
59    {W : H →L[ℂ] H} (hW : 0 ≤ W) {M : ℝ} (hM : 0 ≤ M)
60    (hInv : ∀ ε : ℝ, 0 < ε →
61      witnessMoment (TraceClass.ofOperator T.op T.traceClass)
62        (CFC.rpow (regularizedWeight W ε) (-1)) ≤ M)
63    (χ : Ω → ℝ) (hχ : ∀ x, 0 < χ x)
64    (hweighted : Integrable (fun x => χ x * witnessMoment (R.value x).1 W) μ)
65    (hclassical : Integrable (fun x => g x / χ x) μ) :
66    R.rootFidelity (product g hg hgn T) ^ 2 ≤
67      (∫ x, χ x * witnessMoment (R.value x).1 W ∂μ) *
68      (∫ x, g x / χ x ∂μ) * M := by
69  let f : Ω → ℝ := fun x => χ x * witnessMoment (R.value x).1 W
70  let k : Ω → ℝ := fun x => g x / χ x
71  have hfn (x : Ω) : 0 ≤ f x := mul_nonneg (hχ x).le (witnessMoment_nonneg _ hW)
72  have hkn (x : Ω) : 0 ≤ k x := div_nonneg (hgn x) (hχ x).le
73  have hip := integrable_sqrt_mul_sqrt hweighted hclassical hfn hkn
74  have hpoint (x : Ω) :
75      PositiveTraceClass.rootFidelity (R.value x) ((product g hg hgn T).value x) ≤
76        Real.sqrt (f x) * Real.sqrt (k x) * Real.sqrt M := by
77    have hsq := rootFidelity_product_sq_le R g hg hgn T hW hM hInv x
78    have heq : f x * k x * M = witnessMoment (R.value x).1 W * (g x * M) := by
79      dsimp [f, k]
80      field_simp [ne_of_gt (hχ x)]
81    have hsqrt : (Real.sqrt (f x) * Real.sqrt (k x) * Real.sqrt M) ^ 2 =
82        witnessMoment (R.value x).1 W * (g x * M) := by
83      rw [mul_pow, mul_pow, Real.sq_sqrt (hfn x), Real.sq_sqrt (hkn x), Real.sq_sqrt hM]
84      exact heq
85    have hpos := mul_nonneg (mul_nonneg (Real.sqrt_nonneg (f x))
86      (Real.sqrt_nonneg (k x))) (Real.sqrt_nonneg M)
87    nlinarith [PositiveTraceClass.rootFidelity_nonneg (R.value x) ((product g hg hgn T).value x)]
88  have hroot : R.rootFidelity (product g hg hgn T) ≤
89      Real.sqrt (∫ x, f x ∂μ) * Real.sqrt (∫ x, k x ∂μ) * Real.sqrt M := by
90    calc
91      R.rootFidelity (product g hg hgn T) ≤
92          ∫ x, (Real.sqrt (f x) * Real.sqrt (k x) * Real.sqrt M) ∂μ :=
93        integral_mono (R.integrable_rootFidelity _) (hip.mul_const _) hpoint
94      _ = (∫ x, Real.sqrt (f x) * Real.sqrt (k x) ∂μ) * Real.sqrt M := integral_mul_const _ _
95      _ ≤ _ := mul_le_mul_of_nonneg_right
96        (integral_sqrt_mul_sqrt_le hweighted hclassical hfn hkn) (Real.sqrt_nonneg M)
97  have hfint : 0 ≤ ∫ x, f x ∂μ := integral_nonneg (fun x => hfn x)
98  have hkint : 0 ≤ ∫ x, k x ∂μ := integral_nonneg (fun x => hkn x)
99  have hsquared := (sq_le_sq₀ (R.rootFidelity_nonneg _) (by positivity)).mpr hroot
100  simpa only [mul_pow, Real.sq_sqrt hfint, Real.sq_sqrt hkint, Real.sq_sqrt hM] using hsquared
