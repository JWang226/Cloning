3/-!
4# Intrinsic finite tableau sums and independence of the storage bound
5 
6The entries of the concrete row-content arrays are stored in `Fin (N+1)`.
7Every semistandard array is automatically bounded by its row lengths, so this
8storage parameter does not change the Schur polynomial once it contains the
9shape.  We prove the actual equivalence of admissible arrays, preserving every
10monomial, rather than postulating polynomial stability.
11-/

21theorem content_entry_le_row {d N : ℕ} {μ : Fin d → ℕ} (A : SemistandardContent N μ)
22    (i j : Fin d) : (A.val i j).val ≤ μ i := by
23  rw [← A.property.1 i]
24  exact Finset.single_le_sum (f := fun k ↦ (A.val i k).val)
25    (fun _ _ ↦ Nat.zero_le _) (Finset.mem_univ j)

27/-- Re-encode an actual admissible tableau in any sufficient entry bound. -/
28def rebaseContent {d N : ℕ} {μ : Fin d → ℕ} (A : SemistandardContent N μ)
29    (M : ℕ) (hM : ∀ i, μ i ≤ M) : SemistandardContent M μ := by
30  let B : ContentArray d M := fun i j ↦
31    ⟨(A.val i j).val, Nat.lt_succ_of_le ((content_entry_le_row A i j).trans (hM i))⟩
32  refine ⟨B, ?_⟩
33  exact A.property

59theorem schurPolynomial_eq_content_sum {d : ℕ} (N : ℕ) (p : Fin d → ℝ) (μ : Fin d → ℕ) :
60    schurPolynomial N p μ = ∑ A : SemistandardContent N μ, contentWeight p A.val := by
61  unfold schurPolynomial
62  rw [← Finset.sum_filter]
63  exact Finset.sum_subtype _ (by simp) _

65/-- The Schur polynomial depends on the tableau shape, not its entry-bound
66implementation parameter. -/
67theorem schurPolynomial_rebase {d : ℕ} (N M : ℕ) (p : Fin d → ℝ) (μ : Fin d → ℕ)
68    (hN : ∀ i, μ i ≤ N) (hM : ∀ i, μ i ≤ M) :
69    schurPolynomial N p μ = schurPolynomial M p μ := by
70  rw [schurPolynomial_eq_content_sum, schurPolynomial_eq_content_sum]
71  exact (rebaseContentEquiv N M μ hN hM).sum_comp (fun A ↦ contentWeight p A.val)

73theorem row_le_sum {d : ℕ} (μ : Fin d → ℕ) (i : Fin d) : μ i ≤ ∑ j, μ j :=
74  Finset.single_le_sum (fun _ _ ↦ Nat.zero_le _) (Finset.mem_univ i)

76/-- Stability at a fixed box count, in the form required by the normalization
77induction's `N` to `N+1` step. -/
78theorem schurPolynomial_succ_of_boxes {d : ℕ} (N : ℕ) (p : Fin d → ℝ)
79    (μ : Fin d → ℕ) (hμ : ∑ i, μ i = N) :
80    schurPolynomial N p μ = schurPolynomial (N + 1) p μ := by
81  have hb i : μ i ≤ N := by simpa [hμ] using row_le_sum μ i
82  exact schurPolynomial_rebase N (N + 1) p μ hb (fun i ↦ (hb i).trans (by omega))

90    (A : SemistandardContent N μ) (c : ℝ) :

99    schurPolynomial N (fun i ↦ c * p i) μ = c ^ (∑ i, μ i) * schurPolynomial N p μ := by
