18lemma L1CompletelyPositive.norm_map_of_nonneg
19    {Λ : Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ)) →ₗ[ℂ]
20      Lp (TraceClass K) 1 (volume : Measure (Fin k → ℝ))}

22    (A : Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ)))
23    (hA : ∀ᵐ y ∂volume, 0 ≤ (A y).1) : ‖Λ A‖ = ‖A‖ := by
24  have hp := hΛ.map_nonneg A hA
25  rw [L1.norm_eq_integral_norm, L1.norm_eq_integral_norm]

27    _ = ∫ y, (traceCLM (Λ A y)).re := by

31      Complex.reCLM.integral_comp_comm (traceCLM.integrable_comp (L1.integrable_coeFn _))

56lemma weightedL1Trace_gaussian_average_le
57    (Λ : Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ)) →L[ℂ]
58      Lp (TraceClass K) 1 (volume : Measure (Fin k → ℝ)))
59    (a : Fin k → ℝ) (ha : ∀ i, 0 < a i) {g : ℝ} (hg : 1 ≤ g)
60    (hχ : AEStronglyMeasurable
61      (GaussianAffinity.productWitness a (fun i => a i / g)) volume)
62    {C : ℝ} (hC : 0 ≤ C) (hbound : ∀ y,
63      ‖GaussianAffinity.productWitness a (fun i => a i / g) y‖ ≤ C)
64    (b : (Fin k → ℝ) → ℝ)
65    (hb : Integrable (fun h => GaussianAffinity.productDensity a h * b h))
66    (herror : ∀ h R, ‖Λ (classicalTranslation h R) -
67      classicalTranslation (Real.sqrt g • h) (Λ R)‖ ≤ b h * ‖R‖)
68    (R : Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ))) :
69    weightedL1Trace (GaussianAffinity.productWitness a (fun i => a i / g)) hχ hbound
70      (Λ (classicalAverage (GaussianAffinity.productDensity a) R)) ≤
71      Thermal.classicalBase g ^ ((k : ℝ) / 2) * ‖Λ R‖ +
72        C * (∫ h, GaussianAffinity.productDensity a h * b h) * ‖R‖ := by
73  let G := GaussianAffinity.productDensity a
74  let χ := GaussianAffinity.productWitness a (fun i => a i / g)
75  let Q := weightedL1Trace (H := K) χ hχ hbound
76  have hG := GaussianAffinity.integrable_productDensity a ha
77  have hI := integrable_classicalAverage_integrand G hG R
78  have hi : Integrable (fun h => G h * Q (Λ (classicalTranslation h R))) := by
79    convert (Q.comp (Λ.restrictScalars ℝ)).integrable_comp hI using 1
80    funext h
81    simp only [ContinuousLinearMap.comp_apply, map_smul, Complex.coe_smul,
82      ContinuousLinearMap.coe_restrictScalars', smul_eq_mul]
83  have ho := integrable_mul_weightedL1Trace_translation G hG χ hχ hbound
84    (Real.sqrt g) (Λ R)
85  have he : Integrable (fun h => C * (G h * b h) * ‖R‖) :=
86    (hb.const_mul C).mul_const ‖R‖
87  have hpoint (h : Fin k → ℝ) :
88      Q (Λ (classicalTranslation h R)) ≤
89        Q (classicalTranslation (Real.sqrt g • h) (Λ R)) + C * b h * ‖R‖ := by
90    have hnorm := (norm_weightedL1Trace_le χ hχ hbound
91      (Λ (classicalTranslation h R) - classicalTranslation (Real.sqrt g • h) (Λ R))).trans
92      (mul_le_mul_of_nonneg_left (herror h R) hC)
93    have hr := (le_abs_self (Q (Λ (classicalTranslation h R) -
94      classicalTranslation (Real.sqrt g • h) (Λ R)))).trans hnorm
95    rw [map_sub] at hr
96    nlinarith
97  have heq : Q (Λ (classicalAverage G R)) =
98      ∫ h, G h * Q (Λ (classicalTranslation h R)) := by
99    change (Q.comp (Λ.restrictScalars ℝ)) (∫ h, (G h : ℂ) • classicalTranslation h R) = _
100    rw [← (Q.comp (Λ.restrictScalars ℝ)).integral_comp_comm hI]
101    apply integral_congr_ae
102    exact Eventually.of_forall fun h => by
103      simp only [ContinuousLinearMap.comp_apply, Complex.coe_smul, map_smul,
104        ContinuousLinearMap.coe_restrictScalars', smul_eq_mul]
105  rw [heq]
106  calc
107    _ ≤ ∫ h, (G h * Q (classicalTranslation (Real.sqrt g • h) (Λ R)) +
108        C * (G h * b h) * ‖R‖) := by
109      apply integral_mono hi (ho.add he)
110      intro h
111      have ht := mul_le_mul_of_nonneg_left (hpoint h)
112        (GaussianAffinity.productDensity_nonneg a h)
113      convert ht using 1
114      dsimp only [G, Q, Pi.add_apply]
115      ring
116    _ = (∫ h, G h * Q (classicalTranslation (Real.sqrt g • h) (Λ R))) +
117        C * (∫ h, G h * b h) * ‖R‖ := by
118      rw [integral_add ho he, integral_mul_const, integral_const_mul]
119    _ ≤ _ := add_le_add (gaussian_average_weightedL1Trace_le (Λ R) a ha hg hχ hbound) le_rfl

121/-- The sharp residual trace bound for any actual hybrid channel with a
122quantitative classical covariance defect. The input mollifier and its limit
123are explicitly constructed in operator-valued L¹. -/
124theorem weightedQuantumMap_gaussian_trace_le
125    (Λ : Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ)) →L[ℂ]
126      Lp (TraceClass K) 1 (volume : Measure (Fin k → ℝ)))
127    (hΛ : L1CompletelyPositive Λ.toLinearMap) (hTP : L1TracePreserving Λ.toLinearMap)
128    (a : Fin k → ℝ) (ha : ∀ i, 0 < a i) {g : ℝ} (hg : 1 ≤ g)
129    (hχ : AEStronglyMeasurable
130      (GaussianAffinity.productWitness a (fun i => a i / g)) volume)
131    {C : ℝ} (hC : 0 ≤ C) (hbound : ∀ y,
132      ‖GaussianAffinity.productWitness a (fun i => a i / g) y‖ ≤ C)
133    (b : (Fin k → ℝ) → ℝ)
134    (hb : Integrable (fun h => GaussianAffinity.productDensity a h * b h))
135    (herror : ∀ h R, ‖Λ (classicalTranslation h R) -
136      classicalTranslation (Real.sqrt g • h) (Λ R)‖ ≤ b h * ‖R‖)
137    (X : TraceClass H) (hX : 0 ≤ X.1) :
138    (traceCLM (weightedQuantumMap Λ.toLinearMap (GaussianAffinity.productDensity a)
139      (GaussianAffinity.integrable_productDensity a ha)
140      (GaussianAffinity.productWitness a (fun i => a i / g)) hχ hbound X)).re ≤
141      (Thermal.classicalBase g ^ ((k : ℝ) / 2) +
142        C * (∫ h, GaussianAffinity.productDensity a h * b h)) * (traceCLM X).re := by
143  let G := GaussianAffinity.productDensity a
144  let χ := GaussianAffinity.productWitness a (fun i => a i / g)
145  let Q := weightedL1Trace (H := K) χ hχ hbound
146  let U (n : ℕ) := prepareL1 (classicalApproxDensity k n)
147    (integrable_classicalApproxDensity k n) X
148  have hUn (n : ℕ) : ‖U n‖ = ‖X‖ :=
149    norm_prepareL1 _ _ (classicalApproxDensity_nonneg k n)
150      (integral_classicalApproxDensity k n) X
151  have hUp (n : ℕ) : ∀ᵐ y ∂volume, 0 ≤ (U n y).1 :=
152    prepareL1_nonneg _ _ (classicalApproxDensity_nonneg k n) X hX
153  have hLn (n : ℕ) : ‖Λ (U n)‖ = ‖X‖ :=
154    (hΛ.norm_map_of_nonneg hTP (U n) (hUp n)).trans (hUn n)
155  have hboundn (n : ℕ) : Q (Λ (classicalAverage G (U n))) ≤
156      (Thermal.classicalBase g ^ ((k : ℝ) / 2) + C * (∫ h, G h * b h)) *
157        (traceCLM X).re := by
158    have ht := weightedL1Trace_gaussian_average_le Λ a ha hg hχ hC hbound b hb herror (U n)
159    rw [hLn n, hUn n, TraceClass.norm_eq_trace_re_of_nonneg X hX] at ht
160    exact ht.trans_eq (by dsimp only [G]; rw [traceCLM_apply]; ring)
161  have hlim : Tendsto (fun n => Q (Λ (classicalAverage G (U n)))) atTop
162      (𝓝 (Q (Λ (prepareL1 G (GaussianAffinity.integrable_productDensity a ha) X)))) :=
163    (Q.continuous.comp Λ.continuous).continuousAt.tendsto.comp
164      (classicalAverage_prepareApprox_tendsto G (GaussianAffinity.integrable_productDensity a ha) X)
165  exact le_of_tendsto_of_tendsto hlim tendsto_const_nhds (Eventually.of_forall hboundn)

167/-- Dominated convergence discharges the Gaussian-averaged covariance error. -/
168theorem tendsto_gaussian_covariance_error_zero
169    (a : Fin k → ℝ) (ha : ∀ i, 0 < a i) (C : ℝ)
170    (b : ℕ → (Fin k → ℝ) → ℝ)
171    (hb : ∀ n, AEStronglyMeasurable (b n) volume)
172    {B : ℝ} (hbound : ∀ n h, ‖b n h‖ ≤ B)
173    (hlim : ∀ h, Tendsto (fun n => b n h) atTop (𝓝 0)) :
174    Tendsto (fun n => C * (∫ h, GaussianAffinity.productDensity a h * b n h))
175      atTop (𝓝 0) := by
176  have hG := GaussianAffinity.integrable_productDensity a ha
177  have ht : Tendsto (fun n => ∫ h, GaussianAffinity.productDensity a h * b n h)
178      atTop (𝓝 (∫ _h : Fin k → ℝ, (0 : ℝ))) := by
179    apply tendsto_integral_of_dominated_convergence
180      (fun h => GaussianAffinity.productDensity a h * B)
181      (fun n => hG.aestronglyMeasurable.mul (hb n)) (hG.mul_const B)
182    · intro n
183      exact Eventually.of_forall fun h => by
184        rw [Pi.mul_apply, norm_mul, Real.norm_of_nonneg (GaussianAffinity.productDensity_nonneg a h)]
185        exact mul_le_mul_of_nonneg_left (hbound n h)
186          (GaussianAffinity.productDensity_nonneg a h)
187    · exact Eventually.of_forall fun h => by
188        simpa using (hlim h).const_mul (GaussianAffinity.productDensity a h)
189  simpa using ht.const_mul C

191/-- The residual trace factor is eventually at most the sharp classical
192factor plus any positive tolerance, uniformly over every positive input. -/
193theorem eventually_weightedQuantumMap_gaussian_trace_le
194    (Λ : ℕ → Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ)) →L[ℂ]
195      Lp (TraceClass K) 1 (volume : Measure (Fin k → ℝ)))
196    (hΛ : ∀ n, L1CompletelyPositive (Λ n).toLinearMap)
197    (hTP : ∀ n, L1TracePreserving (Λ n).toLinearMap)
198    (a : Fin k → ℝ) (ha : ∀ i, 0 < a i) {g : ℝ} (hg : 1 ≤ g)
199    (hχ : AEStronglyMeasurable
200      (GaussianAffinity.productWitness a (fun i => a i / g)) volume)
201    {C : ℝ} (hC : 0 ≤ C) (hχbound : ∀ y,
202      ‖GaussianAffinity.productWitness a (fun i => a i / g) y‖ ≤ C)
203    (b : ℕ → (Fin k → ℝ) → ℝ)
204    (hb : ∀ n, AEStronglyMeasurable (b n) volume)
205    {B : ℝ} (hbbound : ∀ n h, ‖b n h‖ ≤ B)
206    (hlim : ∀ h, Tendsto (fun n => b n h) atTop (𝓝 0))
207    (herror : ∀ n h R, ‖Λ n (classicalTranslation h R) -
208      classicalTranslation (Real.sqrt g • h) (Λ n R)‖ ≤ b n h * ‖R‖) :
209    ∀ ε > 0, ∀ᶠ n in atTop, ∀ X : TraceClass H, 0 ≤ X.1 →
210      (traceCLM (weightedQuantumMap (Λ n).toLinearMap (GaussianAffinity.productDensity a)
211        (GaussianAffinity.integrable_productDensity a ha)
212        (GaussianAffinity.productWitness a (fun i => a i / g)) hχ hχbound X)).re ≤
213        (Thermal.classicalBase g ^ ((k : ℝ) / 2) + ε) * (traceCLM X).re := by
214  intro ε hε
215  have ht := tendsto_gaussian_covariance_error_zero a ha C b hb hbbound hlim
216  have he := ht.eventually_lt_const hε
217  filter_upwards [he] with n hn
218  intro X hX
219  have hbi : Integrable (fun h => GaussianAffinity.productDensity a h * b n h) :=
220    (GaussianAffinity.integrable_productDensity a ha).mul_bdd (hb n)
221      (Eventually.of_forall (hbbound n))
222  apply (weightedQuantumMap_gaussian_trace_le (Λ n) (hΛ n) (hTP n) a ha hg
223    hχ hC hχbound (b n) hbi (herror n) X hX).trans
224  exact mul_le_mul_of_nonneg_right (add_le_add le_rfl hn.le)
225    (trace_re_nonneg hX X.2)
