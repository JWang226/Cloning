16theorem rowReverseBump_exists_of_prefix {d : ℕ} (U V : RowContent d) (b : Fin d)
17    (hUV : ∀ k, rowPrefix V k ≤ rowStrictPrefix U k) (hb : 0 < V b) :
18    ∃ a, rowReverseBump U b = some a := by
19  have hb' : V b ≤ rowPrefix V b := by
20    exact Finset.single_le_sum (fun _ _ => Nat.zero_le _)
21      (show b ∈ Finset.univ.filter (fun j => j ≤ b) by simp)
22  have hp : 0 < rowStrictPrefix U b := hb.trans_le (hb'.trans (hUV b))
23  obtain ⟨a, ha, hUa⟩ := Finset.sum_pos_iff.mp hp
24  have hs : (rowReverseCandidates U b).Nonempty := by
25    refine ⟨a, ?_⟩
26    simpa only [rowReverseCandidates, Finset.mem_filter, Finset.mem_univ, true_and]
27      using And.intro ((Finset.mem_filter.mp ha).2) hUa
28  exact ⟨_, by rw [rowReverseBump, dif_pos hs]⟩

30theorem rowReversed_prefix {d : ℕ} (A : RowContent d) (y x k : Fin d)
31    (h : rowReverseBump A y = some x) :
32    rowPrefix (rowReversed A y) k + (if x ≤ k ∧ k < y then 1 else 0) =
33      rowPrefix A k := by
34  obtain ⟨hxy, hx, _⟩ := (rowReverseBump_eq_some_iff A y x).mp h
35  simp only [rowReversed, h]
36  have he := rowReplace_sum A y x (ne_of_lt hxy).symm hx
37    (Finset.univ.filter (fun j => j ≤ k))

40    rowPrefix A k + (if y ≤ k then 1 else 0) at he

72theorem rowPrefix_slack {d : ℕ} (A : RowContent d) (k b : Fin d)
73    (hkb : k < b) (hb : 0 < A b) : rowPrefix A k + 1 ≤ rowPrefix A b := by
74  have hs : insert b (Finset.univ.filter (fun j => j ≤ k)) ⊆
75      Finset.univ.filter (fun j => j ≤ b) := by
76    intro j hj
77    simp only [Finset.mem_insert, Finset.mem_filter, Finset.mem_univ, true_and] at hj ⊢
78    rcases hj with rfl | hj
79    · exact le_rfl
80    · exact hj.trans hkb.le
81  have he := Finset.sum_le_sum_of_subset (f := A) hs
82  rw [Finset.sum_insert (by simp [not_le_of_gt hkb])] at he
83  change A b + rowPrefix A k ≤ rowPrefix A b at he
84  omega

86theorem rowStrictPrefix_eq_of_no_entries {d : ℕ} (A : RowContent d) (k b : Fin d)
87    (hkb : k ≤ b) (hzero : ∀ j, k ≤ j → j < b → A j = 0) :
88    rowStrictPrefix A b = rowStrictPrefix A k := by
89  symm
90  apply Finset.sum_subset
91  · intro j hj
92    simp only [Finset.mem_filter, Finset.mem_univ, true_and] at hj ⊢
93    exact hj.trans_le hkb
94  · intro j hj hjnot
95    simp only [Finset.mem_filter, Finset.mem_univ, true_and, not_lt] at hj hjnot

98theorem rowStrictPrefix_eq_at_reverse {d : ℕ} (U : RowContent d) (b a k : Fin d)
99    (h : rowReverseBump U b = some a) (hak : a < k) (hkb : k ≤ b) :
100    rowStrictPrefix U b = rowStrictPrefix U k := by
101  apply rowStrictPrefix_eq_of_no_entries U k b hkb
102  intro j hkj hjb
103  have hmax := ((rowReverseBump_eq_some_iff U b a).mp h).2.2
104  by_contra hj
105  exact (not_le_of_gt (hak.trans_le hkj)) (hmax j hjb (Nat.pos_of_ne_zero hj))

107/-- A reverse step in the upper row preserves interlacing with any lower
108row whose prefixes decrease and whose prefix at the returned letter drops. -/
109theorem rowReversed_interlaces_of_prefix {d : ℕ}
110    (U V W : RowContent d) (b a : Fin d)
111    (hUV : ∀ k, rowPrefix V k ≤ rowStrictPrefix U k)
112    (hb : 0 < V b) (hrev : rowReverseBump U b = some a)
113    (hWV : ∀ k, rowPrefix W k ≤ rowPrefix V k)
114    (hdrop : rowPrefix W b + 1 ≤ rowPrefix V b) :
115    ∀ k, rowPrefix W k ≤ rowStrictPrefix (rowReversed U b) k := by
116  intro k
117  have he := rowReversed_strictPrefix U b a k hrev
118  by_cases hk : a < k ∧ k ≤ b
119  · rw [if_pos hk] at he
120    have hflat := rowStrictPrefix_eq_at_reverse U b a k hrev hk.1 hk.2
121    have hslack : rowPrefix W k + 1 ≤ rowPrefix V b := by
122      rcases lt_or_eq_of_le hk.2 with hkb | rfl
123      · exact (Nat.add_le_add_right (hWV k) 1).trans (rowPrefix_slack V k b hkb hb)
124      · exact hdrop
125    have hupper := hUV b
126    omega
127  · rw [if_neg hk, add_zero] at he
128    rw [he]
129    exact (hWV k).trans (hUV k)

131theorem rowReversed_interlaces_remove {d : ℕ} (U V : RowContent d) (b a : Fin d)
132    (hUV : ∀ k, rowPrefix V k ≤ rowStrictPrefix U k)
133    (hb : 0 < V b) (hrev : rowReverseBump U b = some a) :
134    ∀ k, rowPrefix (rowRemove V b) k ≤ rowStrictPrefix (rowReversed U b) k := by
135  apply rowReversed_interlaces_of_prefix U V (rowRemove V b) b a hUV hb hrev
136  · intro k

142theorem rowReversed_interlaces_reverse {d : ℕ} (U V : RowContent d) (c b a : Fin d)
143    (hUV : ∀ k, rowPrefix V k ≤ rowStrictPrefix U k)
144    (hV : rowReverseBump V c = some b) (hU : rowReverseBump U b = some a) :
