18# Polar decomposition of bounded operators
19 
20## i. Overview
21 
22For a bounded operator `T` with absolute value `|T|`, `‖T x‖ = ‖|T| x‖`. The resulting isometry from
23the range of `|T|` onto the range of `T` extends to the closure of that range and gives a partial
24isometry `U` with `T = U |T|` and `U⋆ T = |T|`.
25 
26## ii. Key results
27 
28- `Polar.polarFactor` : the partial isometry of the polar decomposition.
29- `Polar.polarFactor_mul_absOperator` : `T = U |T|`.
30- `Polar.star_polarFactor_mul_self` : `U⋆ T = |T|`.
31- `Polar.polarFactor_opNorm_le` : the polar factor is a contraction.

50/-- The absolute value used in the polar construction. -/
51noncomputable def absOperator (T : H →L[ℂ] H) : H →L[ℂ] H := CFC.abs T

53/-- The closed range subspace of the absolute value. -/
54noncomputable def rangeClosure (T : H →L[ℂ] H) : Submodule ℂ H :=
55  (LinearMap.range (absOperator T).toLinearMap).topologicalClosure

99lemma denseRange_absIntoRange (T : H →L[ℂ] H) : DenseRange (absIntoRange T) := by
100  let r : H →L[ℂ] (LinearMap.range (absOperator T).toLinearMap) :=
101    (absOperator T).toLinearMap.codRestrict (LinearMap.range (absOperator T).toLinearMap)
102      (fun x => LinearMap.mem_range_self _ x) |>.mkContinuous
103        ‖absOperator T‖ (by intro x; exact (absOperator T).le_opNorm x)
104  have hr : DenseRange r := by
105    apply Function.Surjective.denseRange
106    intro y
107    rcases y with ⟨y, ⟨x, hx⟩⟩
108    exact ⟨x, Subtype.ext hx⟩
109  have hcomp := DenseRange.comp (denseRange_rangeInClosure T) hr (rangeInClosure T).continuous
110  have heq : (rangeInClosure T ∘ r) = absIntoRange T := by
111    funext x; exact Subtype.ext rfl
112  rwa [heq] at hcomp

116lemma norm_apply_eq_norm_abs_apply (T : H →L[ℂ] H) (x : H) : ‖T x‖ = ‖absOperator T x‖ := by
117  have hsq : ‖T x‖ ^ 2 = ‖absOperator T x‖ ^ 2 := by
118    have hnormT : ‖T x‖ ^ 2 = (⟪T x, T x⟫_ℂ).re := by rw [inner_self_eq_norm_sq_to_K]; norm_cast
119    have hnormS : ‖absOperator T x‖ ^ 2 = (⟪absOperator T x, absOperator T x⟫_ℂ).re := by
120      rw [inner_self_eq_norm_sq_to_K]; norm_cast
121    have hT : ⟪T x, T x⟫_ℂ = ⟪x, (ContinuousLinearMap.adjoint T) (T x)⟫_ℂ :=
122      (ContinuousLinearMap.adjoint_inner_right T x (T x)).symm
123    have hstar : ContinuousLinearMap.adjoint T = star T :=
124      (ContinuousLinearMap.star_eq_adjoint T).symm
125    have habs : absOperator T * absOperator T = star T * T := by
126      unfold absOperator; exact CFC.abs_mul_abs T
127    have habsSelf : IsSelfAdjoint (absOperator T) := .of_nonneg (CFC.abs_nonneg T)
128    rw [hnormT, hnormS]
129    calc
130      (⟪T x, T x⟫_ℂ).re = (⟪x, (star T * T) x⟫_ℂ).re := by
131        rw [hT, hstar]; simp [ContinuousLinearMap.mul_def, ContinuousLinearMap.comp_apply]
132      _ = (⟪x, (absOperator T * absOperator T) x⟫_ℂ).re := by rw [habs]
133      _ = (⟪absOperator T x, absOperator T x⟫_ℂ).re := by
134        rw [ContinuousLinearMap.mul_def, ContinuousLinearMap.comp_apply,
135          ← ContinuousLinearMap.adjoint_inner_left (absOperator T) (absOperator T x) x,
136          (ContinuousLinearMap.star_eq_adjoint (absOperator T)).symm.trans habsSelf]
137  exact (sq_eq_sq₀ (norm_nonneg _) (norm_nonneg _)).mp hsq

149/-- The polar factor on the closed range of `|T|`. -/

153lemma polarOnRange_eq_on_abs (T : H →L[ℂ] H) (x : H) :
154    polarOnRange T (absIntoRange T x) = T x := by
155  unfold polarOnRange
156  apply LinearMap.extendOfNorm_eq (denseRange_absIntoRange T)
157  refine ⟨1, fun y => ?_⟩
158  rw [one_mul, norm_absIntoRange_apply]
159  exact (norm_apply_eq_norm_abs_apply T y).le

161lemma polarOnRange_norm_le (T : H →L[ℂ] H) (x : rangeClosure T) : ‖polarOnRange T x‖ ≤ ‖x‖ := by
162  have hnorm : ∀ y : H, ‖T y‖ ≤ 1 * ‖absIntoRange T y‖ := by
163    intro y
164    rw [one_mul, norm_absIntoRange_apply]
165    exact (norm_apply_eq_norm_abs_apply T y).le
166  change ‖T.toLinearMap.extendOfNorm (absIntoRange T) x‖ ≤ ‖x‖
167  simpa only [one_mul] using LinearMap.norm_extendOfNorm_apply_le (denseRange_absIntoRange T) 1
168    hnorm x

170/-- The bounded polar factor, obtained by extending the range isometry by zero on the orthogonal
171complement of `closure (range |T|)`. -/
172noncomputable def polarFactor (T : H →L[ℂ] H) : H →L[ℂ] H :=
173  polarOnRange T ∘L (rangeClosure T).orthogonalProjection
174 
175lemma polarFactor_eq_on_abs (T : H →L[ℂ] H) (x : H) : polarFactor T (absOperator T x) = T x := by
176  unfold polarFactor
177  rw [ContinuousLinearMap.comp_apply]
178  have hx : absOperator T x ∈ rangeClosure T :=
179    Submodule.le_topologicalClosure _ (LinearMap.mem_range_self _ x)
180  have hp := (rangeClosure T).orthogonalProjection_mem_subspace_eq_self ⟨absOperator T x, hx⟩
181  rw [hp]
182  exact polarOnRange_eq_on_abs T x
183 
184lemma polarFactor_norm_le (T : H →L[ℂ] H) (x : H) : ‖polarFactor T x‖ ≤ ‖x‖ := by
185  unfold polarFactor
186  rw [ContinuousLinearMap.comp_apply]
187  calc
188    ‖polarOnRange T ((rangeClosure T).orthogonalProjection x)‖ ≤
189        ‖(rangeClosure T).orthogonalProjection x‖ := polarOnRange_norm_le T _
190    _ ≤ ‖x‖ := (rangeClosure T).norm_orthogonalProjection_apply_le x

192lemma polarFactor_opNorm_le (T : H →L[ℂ] H) : ‖polarFactor T‖ ≤ 1 := by
193  apply ContinuousLinearMap.opNorm_le_bound _ zero_le_one
194  simpa only [one_mul] using polarFactor_norm_le T

196lemma polarFactor_mul_absOperator (T : H →L[ℂ] H) : polarFactor T * absOperator T = T := by

201lemma polarOnRange_norm_eq (T : H →L[ℂ] H) (y : rangeClosure T) :
202    ‖polarOnRange T y‖ = ‖y‖ := by
203  have heq : (fun y : rangeClosure T => ‖polarOnRange T y‖) ∘ (absIntoRange T) =
204      (fun y : rangeClosure T => ‖y‖) ∘ (absIntoRange T) := by
205    funext x
206    show ‖polarOnRange T (absIntoRange T x)‖ = ‖absIntoRange T x‖
207    rw [polarOnRange_eq_on_abs, norm_apply_eq_norm_abs_apply, norm_absIntoRange_apply]
208  have hres := (denseRange_absIntoRange T).equalizer ((polarOnRange T).continuous.norm)
209    continuous_norm heq
210  exact congrFun hres y

212/-- `polarOnRange T` is an isometry: its adjoint is a left inverse. -/
213lemma adjoint_polarOnRange_comp_self (T : H →L[ℂ] H) :
214    ContinuousLinearMap.adjoint (polarOnRange T) ∘L polarOnRange T = 1 :=
215  (ContinuousLinearMap.norm_map_iff_adjoint_comp_self (polarOnRange T)).mp
216    (polarOnRange_norm_eq T)

218/-- `star (polarFactor T) * polarFactor T` is exactly the orthogonal projection onto
219`rangeClosure T`. -/
220lemma star_polarFactor_mul_polarFactor (T : H →L[ℂ] H) :
221    star (polarFactor T) * polarFactor T = (rangeClosure T).starProjection := by
222  rw [ContinuousLinearMap.star_eq_adjoint]
223  show ContinuousLinearMap.adjoint
224      (polarOnRange T ∘L (rangeClosure T).orthogonalProjection) *
225    (polarOnRange T ∘L (rangeClosure T).orthogonalProjection) =
226      (rangeClosure T).starProjection
227  rw [ContinuousLinearMap.adjoint_comp, ContinuousLinearMap.mul_def,
228    ContinuousLinearMap.comp_assoc,
229    ← ContinuousLinearMap.comp_assoc (ContinuousLinearMap.adjoint (polarOnRange T)),
230    adjoint_polarOnRange_comp_self, ContinuousLinearMap.one_def,
231    Submodule.adjoint_orthogonalProjection]
232  simp only [ContinuousLinearMap.id_comp]
233  rfl

235/-- **`U⋆ T = |T|`** for the polar factor `U` of any bounded operator `T`. -/
236lemma star_polarFactor_mul_self (T : H →L[ℂ] H) : star (polarFactor T) * T = absOperator T := by
237  have hT : polarFactor T * absOperator T = T := polarFactor_mul_absOperator T
238  have hproj : star (polarFactor T) * T = (rangeClosure T).starProjection * absOperator T := by
239    calc
240      star (polarFactor T) * T = star (polarFactor T) * (polarFactor T * absOperator T) := by
241        rw [hT]
242      _ = (star (polarFactor T) * polarFactor T) * absOperator T := by rw [mul_assoc]
243      _ = (rangeClosure T).starProjection * absOperator T := by
244        rw [star_polarFactor_mul_polarFactor]
245  rw [hproj]
246  ext x
247  show (rangeClosure T).starProjection (absOperator T x) = absOperator T x
248  exact Submodule.starProjection_eq_self_iff.mpr
249    (Submodule.le_topologicalClosure _ (LinearMap.mem_range_self _ x))
