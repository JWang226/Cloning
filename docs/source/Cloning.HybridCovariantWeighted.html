42theorem weightedL1Integral_compLp (T : TraceClass H →L[ℂ] TraceClass K)
43    (χ : Ω → ℝ) (hχ : AEStronglyMeasurable χ μ)
44    {C : ℝ} (hbound : ∀ y, ‖χ y‖ ≤ C) (A : Lp (TraceClass H) 1 μ) :
45    weightedL1Integral χ hχ hbound (T.compLpL 1 μ A) =
46      T (weightedL1Integral χ hχ hbound A) := by
47  change (∫ y, (χ y : ℂ) • (T.compLpL 1 μ A) y ∂μ) =
48    T (∫ y, (χ y : ℂ) • A y ∂μ)

50  apply integral_congr_ae
51  filter_upwards [T.coeFn_compLpL A] with y hy
52  rw [hy, map_smul]

54theorem norm_prepareL1 (g : Ω → ℝ) (hg : Integrable g μ)

101/-- Preparation and weighted output integration give exact Weyl covariance
102whenever the actual hybrid map is covariant in its quantum fibres. -/
103theorem weightedQuantumMap_weyl_covariance
104    (Λ : Lp (TraceClass (Fock s)) 1 μ →ₗ[ℂ] Lp (TraceClass (Fock s)) 1 μ) (r : ℝ)
105    (hΛ : ∀ z A, Λ (quantumL1Action z A) = quantumL1Action (r • z) (Λ A))
106    (g : Ω → ℝ) (hg : Integrable g μ)
107    (χ : Ω → ℝ) (hχ : AEStronglyMeasurable χ μ)
108    {C : ℝ} (hbound : ∀ y, ‖χ y‖ ≤ C) (z : Fin s → ℂ) (X : TraceClass (Fock s)) :
109    weightedQuantumMap Λ g hg χ hχ hbound (displacementTraceMap z X) =
110      displacementTraceMap (r • z) (weightedQuantumMap Λ g hg χ hχ hbound X) := by
111  change weightedL1Integral χ hχ hbound (Λ (prepareL1 g hg (displacementTraceMap z X))) = _
112  rw [prepareL1_compLp]
113  change weightedL1Integral χ hχ hbound (Λ (quantumL1Action z (prepareL1 g hg X))) = _
114  rw [hΛ]
115  exact weightedL1Integral_compLp (displacementTraceMap (r • z)) χ hχ hbound _

117/-- Quantitative covariance passes to the extracted quantum map with exactly
118the norm of the classical weight, and no loss from preparation. -/
119theorem weightedQuantumMap_weyl_error_le
120    (Λ : Lp (TraceClass (Fock s)) 1 μ →ₗ[ℂ] Lp (TraceClass (Fock s)) 1 μ) (r : ℝ)
121    (g : Ω → ℝ) (hg : Integrable g μ) (hg0 : ∀ y, 0 ≤ g y)
122    (hprob : ∫ y, g y ∂μ = 1)
123    (χ : Ω → ℝ) (hχ : AEStronglyMeasurable χ μ)
124    {C : ℝ} (hC : 0 ≤ C) (hbound : ∀ y, ‖χ y‖ ≤ C)
125    (z : Fin s → ℂ) (b : ℝ)
126    (herror : ∀ A, ‖Λ (quantumL1Action z A) - quantumL1Action (r • z) (Λ A)‖ ≤ b * ‖A‖)
127    (X : TraceClass (Fock s)) :
128    ‖weightedQuantumMap Λ g hg χ hχ hbound (displacementTraceMap z X) -
129      displacementTraceMap (r • z) (weightedQuantumMap Λ g hg χ hχ hbound X)‖ ≤
130      C * b * ‖X‖ := by
131  change ‖weightedL1Integral χ hχ hbound (Λ (prepareL1 g hg (displacementTraceMap z X))) -
132    displacementTraceMap (r • z) (weightedL1Integral χ hχ hbound (Λ (prepareL1 g hg X)))‖ ≤ _
133  rw [prepareL1_compLp, ← weightedL1Integral_compLp, ← map_sub]
134  calc
135    _ ≤ C * ‖Λ (quantumL1Action z (prepareL1 g hg X)) -
136        quantumL1Action (r • z) (Λ (prepareL1 g hg X))‖ :=
137      norm_weightedL1Integral_le χ hχ hbound _
138    _ ≤ C * (b * ‖prepareL1 g hg X‖) := mul_le_mul_of_nonneg_left (herror _) hC
139    _ = _ := by rw [norm_prepareL1 g hg hg0 hprob]; ring
