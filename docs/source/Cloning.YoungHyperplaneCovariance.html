20/-- The covariance of the first `d` coordinates of a multinomial fluctuation. -/
21def headCovariance (d : ℕ) (p : Fin (d + 1) → ℝ) : Matrix (Fin d) (Fin d) ℝ :=
22  fun i j ↦ (if i = j then p i.castSucc else 0) - p i.castSucc * p j.castSucc
23 
24/-- The principal minor has the exact product normalization. -/
25theorem headCovariance_det (d : ℕ) (p : Fin (d + 1) → ℝ) (hp : ∑ i, p i = 1) :
26    Matrix.det (headCovariance d p) = ∏ i, p i := by
27  have heq : headCovariance d p =
28      Matrix.diagonal (fun i : Fin d ↦ p i.castSucc) *
29        (1 + Matrix.replicateCol Unit (fun _ : Fin d ↦ (1 : ℝ)) *
30          Matrix.replicateRow Unit (fun i : Fin d ↦ -p i.castSucc)) := by
31    ext i j
32    rw [Matrix.diagonal_mul]
33    simp only [headCovariance, Matrix.add_apply, Matrix.mul_apply, Matrix.one_apply]
34    simp only [Matrix.replicateCol_apply, Matrix.replicateRow_apply, one_mul,
35      Fintype.sum_unique]
36    by_cases hij : i = j
37    · simp [hij]; ring
38    · simp [hij]
39  rw [heq, Matrix.det_mul, Matrix.det_diagonal, Matrix.det_one_add_replicateCol_mul_replicateRow]
40  rw [Fin.sum_univ_castSucc] at hp
41  simp only [dotProduct, mul_one, Finset.sum_neg_distrib]
42  rw [Fin.prod_univ_castSucc]
43  congr 1
44  linarith

46/-- The induced covariance endomorphism in the nonorthonormal root basis has
47matrix `C G`, where `G=I+11ᵀ` is the root-basis Gram matrix. -/
48def rootCovarianceMatrix (d : ℕ) (p : Fin (d + 1) → ℝ) : Matrix (Fin d) (Fin d) ℝ :=
49  headCovariance d p * (show Matrix (Fin d) (Fin d) ℝ from
50    fun i j ↦ inner ℝ (rootBasis d i) (rootBasis d j))

52theorem rootCovarianceMatrix_det (d : ℕ) (p : Fin (d + 1) → ℝ) (hp : ∑ i, p i = 1) :
53    Matrix.det (rootCovarianceMatrix d p) = ((d : ℝ) + 1) * ∏ i, p i := by
54  rw [rootCovarianceMatrix, Matrix.det_mul, headCovariance_det d p hp, rootBasis_gram_det]
55  ring

57/-- The covariance is a genuine endomorphism of the Euclidean root hyperplane. -/
58def rootCovariance (d : ℕ) (p : Fin (d + 1) → ℝ) : rootSpace d →ₗ[ℝ] rootSpace d :=
59  (coordinates d).toLinearMap ∘ₗ (rootCovarianceMatrix d p).toLin' ∘ₗ
60    (coordinates d).symm.toLinearMap

62theorem rootCovariance_det (d : ℕ) (p : Fin (d + 1) → ℝ) (hp : ∑ i, p i = 1) :
63    LinearMap.det (rootCovariance d p) = ((d : ℝ) + 1) * ∏ i, p i := by
64  rw [rootCovariance, LinearMap.det_conj, LinearMap.det_toLin']

85/-- The constructed endomorphism is exactly the ambient covariance
86`diag(p) - p pᵀ`, restricted to the true Euclidean sum-zero hyperplane. -/
87theorem rootCovariance_apply (d : ℕ) (p : Fin (d + 1) → ℝ) (hp : ∑ i, p i = 1)
88    (x : rootSpace d) (i : Fin (d + 1)) :
89    (rootCovariance d p x).1 i = p i * x.1 i - p i * ∑ j, p j * x.1 j := by
90  have hhead (k : Fin d) : (rootCovariance d p x).1 k.castSucc =
91      p k.castSucc * x.1 k.castSucc - p k.castSucc * ∑ j, p j * x.1 j := by
92    change (coordinates d (((rootCovarianceMatrix d p).toLin') ((coordinates d).symm x))).1 k.castSucc = _
93    rw [coordinates_head, Matrix.toLin'_apply, rootCovarianceMatrix, ← Matrix.mulVec_mulVec]
94    rw [headCovariance_mulVec]
95    simp only [rootGram_mulVec, coordinates_symm_apply]
96    have hx : (∑ j : Fin d, x.1 j.castSucc) + x.1 (Fin.last d) = 0 := by
97      have hx0 : (∑ j, x.1 j) = 0 := x.2
98      simpa only [Fin.sum_univ_castSucc] using hx0
99    have hp' : (∑ j : Fin d, p j.castSucc) + p (Fin.last d) = 1 := by
100      simpa only [Fin.sum_univ_castSucc] using hp
101    rw [Fin.sum_univ_castSucc]
102    simp only [mul_add, Finset.sum_add_distrib, ← Finset.sum_mul]
103    have hpx := congrArg (fun t : ℝ ↦ t * (∑ j : Fin d, x.1 j.castSucc)) hp'
104    have hsum : (∑ j : Fin d, p j.castSucc) * (∑ j : Fin d, x.1 j.castSucc) =
105        (∑ j : Fin d, x.1 j.castSucc) + p (Fin.last d) * x.1 (Fin.last d) := by
106      have hlast := congrArg (fun t : ℝ ↦ p (Fin.last d) * t) hx
107      nlinarith only [hpx, hlast]
108    rw [hsum]
109    ring
110  refine Fin.lastCases ?_ (fun k ↦ hhead k) i
111  have hout : (∑ j : Fin d, (rootCovariance d p x).1 j.castSucc) +
112      (rootCovariance d p x).1 (Fin.last d) = 0 := by
113    have h0 : (∑ j, (rootCovariance d p x).1 j) = 0 := (rootCovariance d p x).2
114    simpa only [Fin.sum_univ_castSucc] using h0
115  have htar : (∑ j, (p j * x.1 j - p j * ∑ k, p k * x.1 k)) = 0 := by
116    rw [Finset.sum_sub_distrib, ← Finset.sum_mul, hp, one_mul, sub_self]
117  rw [Fin.sum_univ_castSucc] at htar
118  simp only [hhead] at hout
119  linarith only [hout, htar]

121/-- Explicit inverse vector, subtracting the mean to restore zero total. -/
122def inverseCovarianceVector (d : ℕ) (p : Fin (d + 1) → ℝ) (x : rootSpace d) : rootSpace d :=
123  ⟨WithLp.toLp 2 (fun i ↦ x.1 i / p i - (∑ j, x.1 j / p j) / ((d : ℝ) + 1)), by
124    change (∑ i, (x.1 i / p i - (∑ j, x.1 j / p j) / ((d : ℝ) + 1))) = 0

126    simp only [Finset.sum_const, Finset.card_univ, Fintype.card_fin, nsmul_eq_mul,

128    field_simp
129    ring⟩

136/-- The candidate inverse solves the covariance equation for every root-space
137vector; it is not merely a formula for a quadratic form. -/
138theorem rootCovariance_inverseVector (d : ℕ) (p : Fin (d + 1) → ℝ)
139    (hp : ∑ i, p i = 1) (hp0 : ∀ i, p i ≠ 0) (x : rootSpace d) :
140    rootCovariance d p (inverseCovarianceVector d p x) = x := by
141  have hsum : (∑ j, p j * (inverseCovarianceVector d p x).1 j) =
142      -(∑ j, x.1 j / p j) / ((d : ℝ) + 1) := by
143    simp only [inverseCovarianceVector_apply, mul_sub, mul_div_cancel₀ _ (hp0 _),
144      Finset.sum_sub_distrib, ← Finset.sum_mul, hp, one_mul]
145    have hx : (∑ j, x.1 j) = 0 := x.2
146    rw [hx]
147    ring
148  apply Subtype.ext
149  ext i
150  rw [rootCovariance_apply d p hp, hsum, inverseCovarianceVector_apply]
151  rw [mul_sub, mul_div_cancel₀ _ (hp0 i)]

199/-- Strict positivity of the inverse quadratic form follows from a strictly
200positive spectrum. -/
201theorem inverseCovariance_quadratic_pos (d : ℕ) (p : Fin (d + 1) → ℝ)
202    (hp : ∀ i, 0 < p i) (x : rootSpace d) (hx : x ≠ 0) :
203    0 < inner ℝ x (inverseCovarianceVector d p x) := by
204  rw [inverseCovariance_quadratic]
205  have hi : ∃ i, x.1 i ≠ 0 := by
206    by_contra h
207    push_neg at h
208    apply hx
209    apply Subtype.ext
210    ext i
211    simpa using h i
212  obtain ⟨i, hi⟩ := hi
213  exact Finset.sum_pos' (fun j _ ↦ div_nonneg (sq_nonneg _) (hp j).le)
214    ⟨i, Finset.mem_univ i, div_pos (sq_pos_of_ne_zero hi) (hp i)⟩

216/-- The root covariance is positive definite on the actual Euclidean
217hyperplane whenever `p` is a positive normalized spectrum. -/
218theorem rootCovariance_quadratic_pos (d : ℕ) (p : Fin (d + 1) → ℝ)
219    (hp : ∑ i, p i = 1) (hp0 : ∀ i, 0 < p i) (x : rootSpace d) (hx : x ≠ 0) :
220    0 < inner ℝ x (rootCovariance d p x) := by
221  let e := rootCovarianceEquiv d p hp (fun i ↦ (hp0 i).ne')
222  have hy : e x ≠ 0 := by
223    intro h
224    apply hx
225    exact e.injective (by simpa using h)
226  have h := inverseCovariance_quadratic_pos d p hp0 (e x) hy
227  rw [← rootCovarianceEquiv_symm_apply d p hp (fun i ↦ (hp0 i).ne')] at h
228  change 0 < inner ℝ (e x) (e.symm (e x)) at h
229  rw [LinearEquiv.symm_apply_apply, real_inner_comm] at h
230  simpa only [e, rootCovarianceEquiv_apply] using h

232/-- Bilinear form of the actual covariance restriction. -/
233theorem rootCovariance_inner (d : ℕ) (p : Fin (d + 1) → ℝ) (hp : ∑ i, p i = 1)
234    (x y : rootSpace d) :
235    inner ℝ x (rootCovariance d p y) =
236      (∑ i, p i * x.1 i * y.1 i) - (∑ i, p i * x.1 i) * (∑ i, p i * y.1 i) := by
237  change inner ℝ x.1 (rootCovariance d p y).1 = _
238  erw [EuclideanSpace.inner_eq_star_dotProduct]
239  simp only [dotProduct, Pi.star_apply, star_trivial, rootCovariance_apply d p hp]
240  calc
241    (∑ i, (p i * y.1 i - p i * ∑ j, p j * y.1 j) * x.1 i) =
242        ∑ i, (p i * x.1 i * y.1 i - (p i * x.1 i) * ∑ j, p j * y.1 j) := by
243      apply Finset.sum_congr rfl
244      intro i _
245      ring
246    _ = _ := by rw [Finset.sum_sub_distrib, ← Finset.sum_mul]

248/-- Symmetry, together with the preceding strict positivity theorem, verifies

250theorem rootCovariance_isSymmetric (d : ℕ) (p : Fin (d + 1) → ℝ) (hp : ∑ i, p i = 1) :
251    (rootCovariance d p).IsSymmetric := by
252  intro x y
