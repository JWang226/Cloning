133theorem eventually_uniform_lifted_fidelity_factorization
134    (K : Set (SimpleSpectrum (d+1))) (hK : IsCompact K)
135    (m : ℕ → ℕ) (γ : ℝ) (hγ : 1<γ)
136    (hgain : Tendsto (fun n => (m n : ℝ)/n) atTop (𝓝 γ))
137    (δ : ℕ → ℝ) (hδ : Tendsto δ atTop (𝓝 0))
138    (q : ∀ n,SimpleSpectrum (d+1) → SchurCopy n (d+1) → SchurCopy (m n) (d+1) → ℝ)
139    (hq : ∀ n p i j,0≤q n p i j) (hs : ∀ n p i,∑ j,q n p i j=1)
140    (hbad : ∀ η : ℝ,0<η → ∀ᶠ n in atTop,∀ p∈K,
141      kernelBadMass n (m n) (d+1) p.eigenvalue (δ n) (q n p)<η)
142    (ε : ℝ) (hε : 0<ε) :
143    ∀ᶠ n in atTop,∀ p∈K,∀ U : unitary (Matrix (Fin (d+1)) (Fin (d+1)) ℂ),
144      |spectrumPayoff n (m n) (channel n (m n) (d+1) (q n p) (hq n p) (hs n p)) p U-
145        orbitalValue γ p*kernelAffinity n (m n) (d+1) p.eigenvalue (q n p)|<ε := by

148  have hh := hn p hp (q n p) (hq n p) (hs n p) U
149  have hz := kernelBadMass_nonneg n (m n) (d+1) p.eigenvalue (δ n) (q n p) (hq n p)
