16def sampleEigenvalues (p : A → ℝ) (hp : Function.Injective p) (Z : Matrix A A ℂ)
17    (L : ℕ) (i : A) : ℝ := p i + sampleScale L * spectralCoordinates (sampleTangent p hp Z L) i

26/-- Every eigenvalue stays strictly positive eventually at a full-rank base. -/
27theorem eventually_sampleEigenvalues_positive (p : A → ℝ) (hp : Function.Injective p)
28    (hp0 : ∀ i, 0 < p i) (Z : Matrix A A ℂ) :
29    ∀ᶠ L : ℕ in atTop, ∀ i, 0 < sampleEigenvalues p hp Z L i := by

32  exact (tendsto_order.mp (sampleEigenvalues_tendsto p hp Z i)).1 0 (hp0 i)

34/-- Every strict spectral gap remains open for all sufficiently large samples. -/
35theorem eventually_sampleEigenvalues_ordered (p : A → ℝ) (hp : Function.Injective p)
36    (hgap : ∀ i j, i < j → p j < p i) (Z : Matrix A A ℂ) :
37    ∀ᶠ L : ℕ in atTop, ∀ i j, i < j →
38      sampleEigenvalues p hp Z L j < sampleEigenvalues p hp Z L i := by
39  apply eventually_all.mpr
40  intro i
41  apply eventually_all.mpr
42  intro j
43  by_cases hij : i < j
44  · have h := ((sampleEigenvalues_tendsto p hp Z i).sub (sampleEigenvalues_tendsto p hp Z j))
45    filter_upwards [(tendsto_order.mp h).1 0 (sub_pos.mpr (hgap i j hij))] with L hL
46    intro _
47    exact sub_pos.mp hL
48  · exact Filter.Eventually.of_forall (fun _ h => (hij h).elim)

50/-- The scaled spectral and orbital parameters, in their fixed finite-dimensional
51ambient space. The spectral trace-zero constraint is proved separately below. -/
52def sampleParameters (p : A → ℝ) (hp : Function.Injective p) (Z : Matrix A A ℂ) (L : ℕ) :
53    (A → ℝ) × (OrbitalPair A → ℂ) :=
54  (spectralCoordinates (sampleTangent p hp Z L), orbitalCoordinates p (sampleTangent p hp Z L))

64/-- One fixed compact ball contains the entire tail of the exact scaled
65parameter sequence. Its radius is determined by the fixed physical tangent. -/
66theorem exists_compact_ball_sampleParameters (p : A → ℝ) (hp : Function.Injective p)
67    (Z : Matrix A A ℂ) :
68    ∃ R : ℝ, 0 < R ∧ IsCompact (Metric.closedBall (0 : (A → ℝ) × (OrbitalPair A → ℂ)) R) ∧
69      ∀ᶠ L : ℕ in atTop, sampleParameters p hp Z L ∈ Metric.closedBall 0 R := by
70  let R := ‖tangentParameters p Z‖ + 1
71  refine ⟨R, by dsimp [R]; positivity, isCompact_closedBall 0 R, ?_⟩
72  have hn := (sampleParameters_tendsto p hp Z).norm
73  have hR : ‖tangentParameters p Z‖ < R := by dsimp [R]; linarith
74  filter_upwards [(tendsto_order.mp hn).2 R hR] with L hL
75  simpa only [Metric.mem_closedBall, dist_zero_right] using hL.le

77/-- A compact subset of the actual trace-zero parameter hyperplane contains
78the tail whenever the tangent purification is normalized and orthogonal. -/
79theorem exists_compact_traceZero_sampleParameters
80    (p : A → ℝ) (hp : Function.Injective p) (hp0 : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1)
81    (Z : Matrix A A ℂ)
82    (horth : ⟪coefficientVector (schmidtCoefficients p), coefficientVector Z⟫_ℂ = 0) :
83    ∃ K : Set ((A → ℝ) × (OrbitalPair A → ℂ)), IsCompact K ∧
84      (∀ θ ∈ K, ∑ i, θ.1 i = 0) ∧ ∀ᶠ L : ℕ in atTop, sampleParameters p hp Z L ∈ K := by
85  obtain ⟨R, _, hR, htail⟩ := exists_compact_ball_sampleParameters p hp Z
86  let H : Set ((A → ℝ) × (OrbitalPair A → ℂ)) := {θ | ∑ i, θ.1 i = 0}
87  have hH : IsClosed H := by
88    apply isClosed_eq _ continuous_const
89    fun_prop
90  refine ⟨Metric.closedBall 0 R ∩ H, hR.inter_right hH, fun θ hθ => hθ.2, ?_⟩
91  filter_upwards [htail, eventually_sample_spectral_sum_zero p hp hp0 hs Z horth] with L hL hzero
92  exact ⟨hL, hzero⟩

94/-- Admissible ordered, normalized, full-rank local eigenvalues and one fixed
95compact parameter window are obtained from the physical tangent assumptions. -/
96theorem eventually_sample_chart_admissible
97    (p : A → ℝ) (hp : Function.Injective p) (hp0 : ∀ i, 0 < p i)
98    (hs : ∑ i, p i = 1) (hgap : ∀ i j, i < j → p j < p i)
99    (Z : Matrix A A ℂ)
100    (horth : ⟪coefficientVector (schmidtCoefficients p), coefficientVector Z⟫_ℂ = 0) :
101    ∃ K : Set ((A → ℝ) × (OrbitalPair A → ℂ)), IsCompact K ∧
102      (∀ θ ∈ K, ∑ i, θ.1 i = 0) ∧
103      ∀ᶠ L : ℕ in atTop, sampleParameters p hp Z L ∈ K ∧
104        (∀ i, 0 < sampleEigenvalues p hp Z L i) ∧
105        (∀ i j, i < j → sampleEigenvalues p hp Z L j < sampleEigenvalues p hp Z L i) ∧
106        ∑ i, sampleEigenvalues p hp Z L i = 1 := by
107  obtain ⟨K, hK, hzero, htail⟩ := exists_compact_traceZero_sampleParameters p hp
108    (fun i => (hp0 i).le) hs Z horth
109  refine ⟨K, hK, hzero, ?_⟩
110  filter_upwards [htail, eventually_sampleEigenvalues_positive p hp hp0 Z,
111    eventually_sampleEigenvalues_ordered p hp hgap Z] with L hL hpos hord
112  refine ⟨hL, hpos, hord, ?_⟩
113  have hz := hzero _ hL
114  change (∑ i, (p i + sampleScale L * spectralCoordinates (sampleTangent p hp Z L) i)) = 1
115  rw [Finset.sum_add_distrib, hs, ← Finset.mul_sum]
116  change 1 + sampleScale L * (∑ i, (sampleParameters p hp Z L).1 i) = 1
117  rw [hz, mul_zero, add_zero]
