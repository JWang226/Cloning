15/-- Commuting with the actual collective generators implies commuting with
16all literal tensor powers, including singular matrices. -/
17theorem tensorOperator_commutes_of_generators
18    (T : TensorRegister n (Fin d) →L[ℂ] TensorRegister n (Fin d))
19    (hT : ∀ a b, T * collectiveGenerator n a b = collectiveGenerator n a b * T)
20    (X : Matrix (Fin d) (Fin d) ℂ) :
21    T * tensorOperator n X = tensorOperator n X * T := by
22  apply Matrix.diagonal_transvection_induction
23    (fun X => T * tensorOperator n X = tensorOperator n X * T) X
24  · intro D _
25    exact tensorOperator_diagonal_commutes T (fun a => hT a a) D
26  · intro t
27    exact tensorOperator_transvection_commutes T t.i t.j t.hij (hT t.i t.j) t.c
28  · intro A B hA hB
29    rw [tensorOperator_mul, ← mul_assoc, hA, mul_assoc, hB, ← mul_assoc]

31/-- Every subspace invariant under the actual collective matrix units is
32invariant under every literal tensor power. -/
33theorem generatorInvariant_tensorOperator_invariant
34    (W : Submodule ℂ (TensorRegister n (Fin d)))
35    (hW : ∀ a b x, x ∈ W → collectiveGenerator n a b x ∈ W)
36    (X : Matrix (Fin d) (Fin d) ℂ) {x : TensorRegister n (Fin d)} (hx : x ∈ W) :
37    tensorOperator n X x ∈ W := by
38  have hcomm := tensorOperator_commutes_of_generators W.starProjection
39    (fun a b => ContinuousLinearMap.ext (invariant_starProjection_commutes W hW a b)) X
40  apply Submodule.starProjection_eq_self_iff.mp
41  have hh := congrArg (fun T : TensorRegister n (Fin d) →L[ℂ] TensorRegister n (Fin d) => T x) hcomm
42  simpa only [ContinuousLinearMap.mul_apply, Submodule.starProjection_eq_self_iff.mpr hx] using hh

58def cyclicTensorOperator (X : Matrix (Fin d) (Fin d) ℂ) :
59    cyclicSector Ω →L[ℂ] cyclicSector Ω :=
60  ((tensorOperator n X).comp (cyclicSector Ω).subtypeL).codRestrict

83/-- Physical unitaries restrict to honest linear isometric equivalences of the
84entire cyclic sector. Both maps are literal tensor powers. -/
85def cyclicUnitary (U : Matrix (Fin d) (Fin d) ℂ) (hU : Uᴴ * U = 1) :
86    cyclicSector Ω ≃ₗᵢ[ℂ] cyclicSector Ω where
87  toLinearEquiv := {
88    toLinearMap := (cyclicTensorOperator Ω mu hweight hraise U).toLinearMap
89    invFun := cyclicTensorOperator Ω mu hweight hraise Uᴴ
90    left_inv := by
91      intro x
92      have hh := congrArg (fun T : cyclicSector Ω →L[ℂ] cyclicSector Ω => T x)
93        (cyclicTensorOperator_mul Ω mu hweight hraise Uᴴ U)
94      simpa only [hU, cyclicTensorOperator_one, ContinuousLinearMap.one_apply,
95        ContinuousLinearMap.mul_apply] using hh.symm
96    right_inv := by
97      intro x
98      have hU' : U * Uᴴ = 1 := mul_eq_one_comm.mp hU
99      have hh := congrArg (fun T : cyclicSector Ω →L[ℂ] cyclicSector Ω => T x)

101      simpa only [hU', cyclicTensorOperator_one, ContinuousLinearMap.one_apply,
102        ContinuousLinearMap.mul_apply] using hh.symm }
103  norm_map' := by
104    intro x
