28theorem internal_flat_fidelity_eventually_le (d s : ℕ) (hs : 1≤s)
29    (hcard : Fintype.card (Fin (d+1) × Fin (d+1))=s+1)
30    (t : ℕ → ℕ) (γ : ℝ) (hγ : 1<γ)
31    (hratio : Tendsto (fun n ↦ ((n+t n : ℕ) : ℝ)/n) atTop (𝓝 γ)) :
32    ∀ ε : ℝ, 0<ε → ∀ᶠ n in atTop,
33      (outputState hcard (flatState d) n (t n)).rootFidelity (tensorState (flatState d) (n+t n)) ≤
34        Cloning.classicalValue (2*γ-1) (d+1)+ε := by
35  obtain ⟨u,hu,hphysical⟩ := Cloning.PCTGlobal.channel_gaussian_product_mixture hs hcard
36    (flatState d) t hγ hratio
37  have hu' : u 0=coefficientVector (schmidtCoefficients (flatSpectrum (d+1))) :=
38    hu.trans (canonicalPurification_diagonal (flatSpectrum (d+1))
39      (fun _ => by unfold flatSpectrum; positivity))
40  let μ := gaussianProductMeasure (fun _ : Fin s ↦ γ-1)
41  have hv : 0<γ-1 := by linarith
42  letI := gaussianProductMeasure_probability (fun _ : Fin s ↦ hv)

45  let A N := outputState hcard (flatState d) N (t N)
46  let B N := tensorState (flatState d) (n N)
47  let M N := ∫ z, (frameState u z (n N)).1 ∂μ
48  let e N := ‖(A N).1-M N‖
49  let a N := ∫ x, Real.sqrt (countMixtureDensity u (γ-1) (n N) x)*Real.sqrt (flatCountDensity d (n N) x)
50  have he : Tendsto e atTop (𝓝 0) := hphysical
51  have ha : Tendsto a atTop (𝓝 (Cloning.classicalValue (2*γ-1) (d+1))) := by
52    have hh := countAffinity_tendsto u hu' (γ-1) hv n hn

67      (countMixtureDensity u (γ-1) (n N)) (flatCountDensity d (n N))
