17lemma idler_witness_moment_le {s : ℕ} (σ : TraceClass (Fock s)) (hσ : 0 ≤ σ.1)
18    {q x : Fin s → ℝ} (hq0 : ∀ i, 0 < q i) (hqx : ∀ i, q i < x i)
19    (hx1 : ∀ i, x i < 1) :
20    (tracePairing ((channel x (fun i => ((hq0 i).trans (hqx i)).le) hx1).toLinearMap σ)
21      (productWitnessOperator (numberBasis s) q x)).re ≤
22      (trace σ.1 σ.2).re * ∏ i, Thermal.fidelity (q i) (x i) := by
23  apply number_law_witness_moment_le (numberBasis s) _ hq0 hqx hx1
24    (a := fun l => (⟪numberBasis s l, σ.1 (numberBasis s l)⟫_ℂ).re)
25  · intro l
26    exact (σ.1.nonneg_iff_isPositive.mp hσ).re_inner_nonneg_right (numberBasis s l)
27  · exact trace_real_hasSum_basis (numberBasis s) σ
28  · intro k
29    exact channel_diagonal_re x (fun i => ((hq0 i).trans (hqx i)).le) hx1 σ k

51/-- The product thermal fidelity upper bound for a genuinely arbitrary joint
52idler density operator; no product, phase-invariance, or diagonality hypothesis. -/
53lemma idler_stateFidelity_le {s : ℕ} (σ : DensityState (Fock s))
54    {q x : Fin s → ℝ} (hq0 : ∀ i, 0 < q i) (hqx : ∀ i, q i < x i)
55    (hx1 : ∀ i, x i < 1) :
56    stateFidelity
57      ((channel x (fun i => ((hq0 i).trans (hqx i)).le) hx1).toPositiveTracePreservingMap.mapState σ)
58      (InfiniteDiagonalFidelity.productGeometricState (numberBasis s) q
59        (fun i => (hq0 i).le) (fun i => (hqx i).trans (hx1 i))) ≤
60      ∏ i, Thermal.fidelity (q i) (x i) := by
61  have h := idler_fidelity_sq_le (TraceClass.ofOperator σ.op σ.traceClass) σ.positive hq0 hqx hx1
62  change stateFidelity
63      ((channel x (fun i => ((hq0 i).trans (hqx i)).le) hx1).toPositiveTracePreservingMap.mapState σ)
64      (InfiniteDiagonalFidelity.productGeometricState (numberBasis s) q
65        (fun i => (hq0 i).le) (fun i => (hqx i).trans (hx1 i))) ^ 2 ≤
66    (trace σ.op σ.traceClass).re * (∏ i, Thermal.fidelity (q i) (x i)) ^ 2 at h
67  rw [σ.trace_one, Complex.one_re, one_mul] at h
68  exact (sq_le_sq₀ (stateFidelity_nonneg _ _)
69    (Finset.prod_nonneg (fun i _ => (Thermal.fidelity_pos (hq0 i).le
70      ((hqx i).trans (hx1 i)) ((hq0 i).trans (hqx i)).le (hx1 i)).le))).mp h

72lemma kraus_vacuum {s : ℕ} (x : Fin s → ℝ)
73    (hx0 : ∀ i, 0 ≤ x i) (hx1 : ∀ i, x i < 1) (m : Occupation s) :
74    kraus x hx0 hx1 m (numberBasis s 0) =
75      (Real.sqrt (productGeometric x m) : ℂ) • numberBasis s m := by
76  ext k
77  rw [kraus_numberBasis_apply]
78  simp only [numberBasis_eq_single, lp.coeFn_smul, Pi.smul_apply,
79    lp.single_apply, Pi.single_apply, zero_add, smul_eq_mul]
80  by_cases h : k = m
81  · subst k
82    simp [BosonicNumberLaw.productLaw, BosonicNumberLaw.seededLaw_zero, productGeometric]
83  · simp [h, Ne.symm h]

85/-- The joint vacuum is sent to the actual product thermal density operator. -/
86lemma channel_vacuum {s : ℕ} (x : Fin s → ℝ)
87    (hx0 : ∀ i, 0 ≤ x i) (hx1 : ∀ i, x i < 1) :
88    (channel x hx0 hx1).toLinearMap (vectorProjector (numberBasis s 0)) =
89      vectorMixture (numberBasis s) (productGeometric x) := by
90  have hs := channel_hasSum x hx0 hx1 (vectorProjector (numberBasis s 0))
91  have he (m : Occupation s) : krausTerm (kraus x hx0 hx1 m) (vectorProjector (numberBasis s 0)) =
92      (productGeometric x m : ℂ) • vectorProjector (numberBasis s m) := by
93    rw [krausTerm_vectorProjector, kraus_vacuum, BosonicAmplifier.vectorProjector_sqrt_smul]
94    exact productGeometric_nonneg hx0 hx1 m
95  simp_rw [he] at hs
96  exact hs.tsum_eq.symm

98lemma vacuum_stateFidelity {s : ℕ} (x q : Fin s → ℝ)
99    (hx0 : ∀ i, 0 ≤ x i) (hx1 : ∀ i, x i < 1)
100    (hq0 : ∀ i, 0 ≤ q i) (hq1 : ∀ i, q i < 1) :
101    stateFidelity
102      ((channel x hx0 hx1).toPositiveTracePreservingMap.mapState
103        (DensityState.pure (numberBasis s 0) ((numberBasis s).orthonormal.norm_eq_one 0)))
104      (InfiniteDiagonalFidelity.productGeometricState (numberBasis s) q hq0 hq1) =
105      ∏ i, Thermal.fidelity (x i) (q i) := by
106  have he :
107      (channel x hx0 hx1).toPositiveTracePreservingMap.mapState
108        (DensityState.pure (numberBasis s 0) ((numberBasis s).orthonormal.norm_eq_one 0)) =
109      InfiniteDiagonalFidelity.productGeometricState (numberBasis s) x hx0 hx1 := by
110    apply ThermalIdlerFidelity.densityState_ext
111    exact congrArg Subtype.val (channel_vacuum x hx0 hx1)
112  rw [he]
113  exact InfiniteDiagonalFidelity.stateFidelity_productGeometric (numberBasis s) x q hx0 hx1 hq0 hq1

117lemma isGreatest_idler_stateFidelity {s : ℕ} {q x : Fin s → ℝ}
118    (hq0 : ∀ i, 0 < q i) (hqx : ∀ i, q i < x i) (hx1 : ∀ i, x i < 1) :
119    IsGreatest (Set.range (fun σ : DensityState (Fock s) =>
120      stateFidelity
121        ((channel x (fun i => ((hq0 i).trans (hqx i)).le) hx1).toPositiveTracePreservingMap.mapState σ)
122        (InfiniteDiagonalFidelity.productGeometricState (numberBasis s) q
123          (fun i => (hq0 i).le) (fun i => (hqx i).trans (hx1 i)))))
124      (∏ i, Thermal.fidelity (q i) (x i)) := by
125  constructor
126  · refine ⟨DensityState.pure (numberBasis s 0) ((numberBasis s).orthonormal.norm_eq_one 0), ?_⟩
127    dsimp only
128    rw [vacuum_stateFidelity]
129    simp only [Thermal.fidelity, mul_comm]
130  · rintro _ ⟨σ, rfl⟩
