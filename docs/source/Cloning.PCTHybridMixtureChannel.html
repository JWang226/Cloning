19/-- Gaussian averaging of the actual joint translation channels. Complete
20positivity and trace preservation hold on the whole hybrid L¹ space. -/
21def tangentMixtureChannel
22    (u : Fin (s + 1) → Register (A × A)) (p : A → ℝ)
23    (b : Fin (k + 1) → EuclideanSpace ℝ A) (e : Fin d ≃ OrbitalPair A)
