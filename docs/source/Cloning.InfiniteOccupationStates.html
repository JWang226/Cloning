4/-!
5# Trace-norm Werner occupation limit
6 
7The actual binomial Werner occupation law defines a convergent series of
8rank-one operators in the trace-class Banach space. Its thermal limit follows
9from the proved binomial asymptotics and normalization, with no assumed
10operator convergence. On an occupation Hilbert basis these are diagonal
11occupation states. Identifying the physical Werner channel output with this
12operator is a separate representation-theoretic obligation.
13-/

31/-- Finite support gives summability even in a finite initial segment where
32the cloning inequality fails. -/
33theorem occupationLaw_summable (n m s : ℕ) : Summable (occupationLaw n m s) := by
34  have hs : HasSum (occupationLaw n m s)
35      (∑ k ∈ occupationSupport s (m - n), occupationLaw n m s k) := by
36    apply hasSum_sum_of_ne_finset_zero
37    intro k hk
38    unfold occupationLaw
39    rw [if_neg]
40    exact fun h => hk (mem_occupationSupport.mpr h)
41  exact hs.summable

43/-- A summable mixture of Hilbert-basis projectors is actually diagonal,
44with the given coefficients as its eigenvalues. -/
45theorem vectorMixture_apply_basis {ι : Type*} (b : HilbertBasis ι ℂ H)
46    (p : ι → ℝ) (hp : Summable p) (i : ι) :
47    (vectorMixture b p).1 (b i) = (p i : ℂ) • b i := by
48  classical
49  let L : TraceClass H →L[ℂ] H :=
50    (ContinuousLinearMap.apply ℂ H (b i)).comp inclusionCLM
51  change L (vectorMixture b p) = _
52  rw [vectorMixture, L.map_tsum (summable_weighted_projectors b b.orthonormal.norm_eq_one p hp)]
53  simp only [map_smul]
54  have heval (j : ι) : L (vectorProjector (b j)) = if j = i then b i else 0 := by
55    change InnerProductSpace.rankOne ℂ (b j) (b j) (b i) = _
56    rw [InnerProductSpace.rankOne_apply, orthonormal_iff_ite.mp b.orthonormal j i]
57    split_ifs with hji
58    · simp [hji]
59    · simp
60  simp_rw [heval]
61  simp

63/-- The finite Werner occupation series as an actual trace-class operator. -/
64def occupationOperator {s : ℕ} (x : (Fin s → ℕ) → H) (n m : ℕ) : TraceClass H :=
65  vectorMixture x (occupationLaw n m s)

67/-- The infinite product-geometric mixture in the trace-class Banach space. -/
68def thermalOperator {s : ℕ} (x : (Fin s → ℕ) → H) (γ : ℝ) : TraceClass H :=

113/-- The quantitative classical-to-quantum bound for the actual occupation
114operators. Orthogonality is not needed for this upper bound. -/
115theorem occupation_norm_sub_le_l1 {s : ℕ} (x : (Fin s → ℕ) → H)

122/-- The actual Werner occupation operators converge in the trace norm.
123Only the cloning-ratio limit is assumed. -/
124theorem occupation_norm_sub_tendsto {s : ℕ} (hs : 1 ≤ s)
125    (x : (Fin s → ℕ) → H) (hx : ∀ k, ‖x k‖ = 1)
126    (m : ℕ → ℕ) {γ : ℝ} (hγ : 1 < γ)
127    (h : Tendsto (fun n ↦ (m n : ℝ) / n) atTop (𝓝 γ)) :
128    Tendsto (fun n ↦ ‖occupationOperator x n (m n) - thermalOperator x γ‖)
129      atTop (𝓝 0) :=
130  squeeze_zero (fun _ ↦ norm_nonneg _)
131    (fun n ↦ occupation_norm_sub_le_l1 x hx n (m n) hγ)

134/-- Convergence in the genuine trace-class Banach space, rather than only
135pointwise convergence of matrix elements or convergence of scalar weights. -/
136theorem occupationOperator_tendsto {s : ℕ} (hs : 1 ≤ s)
137    (x : (Fin s → ℕ) → H) (hx : ∀ k, ‖x k‖ = 1)
