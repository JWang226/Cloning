15/-- A genuine unitary action satisfying the Weyl relations and strong
16continuity. No irreducibility or choice of a vacuum is included. -/
17structure RegularWeyl where
18  toIsometry : (Fin d → ℂ) → H ≃ₗᵢ[ℂ] H
19  zero_apply : ∀ v, toIsometry 0 v = v
20  mul_apply : ∀ a b v, toIsometry a (toIsometry b v) =
21    displacementPhase a b • toIsometry (a + b) v
22  continuous_apply : ∀ v, Continuous (fun a => toIsometry a v)
23 
24variable {d H}

26def RegularWeyl.operator (W : RegularWeyl d H) (a : Fin d → ℂ) : H →L[ℂ] H :=
27  (W.toIsometry a).toContinuousLinearEquiv.toContinuousLinearMap

29@[simp] theorem RegularWeyl.operator_apply (W : RegularWeyl d H)

32@[simp] theorem RegularWeyl.operator_norm (W : RegularWeyl d H)

36theorem RegularWeyl.operator_mul (W : RegularWeyl d H) (a b : Fin d → ℂ) :
37    (W.operator a).comp (W.operator b) = displacementPhase a b • W.operator (a + b) := by
38  ext v
39  exact W.mul_apply a b v

41@[simp] theorem RegularWeyl.operator_zero (W : RegularWeyl d H) :
42    W.operator 0 = ContinuousLinearMap.id ℂ H := by
43  ext v

46/-- The concrete multimode Fock displacement gives a regular representation. -/
47def fockRegularWeyl (d : ℕ) : RegularWeyl d (Fock d) where
48  toIsometry := weylUnitary
49  zero_apply := by intro v; simp
50  mul_apply := by
51    intro a b v
52    exact congrArg (fun T : Fock d →L[ℂ] Fock d => T v) (displacement_comp a b)
53  continuous_apply := continuous_displacement
