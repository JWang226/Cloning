14/-- Global L¹ convergence of the actual physical densities to the limiting
15covariance Gaussian, with moving positive spectra. -/
16theorem tensorYoungDensity_l1_tendsto (d : ℕ) (n : ℕ → ℕ)
17    (hn : Tendsto n atTop atTop) (hn0 : ∀ k, 0 < n k)
18    (p : ℕ → Fin (d + 1) → ℝ) (hp : ∀ k, ∑ i, p k i = 1)
19    (hp0 : ∀ k i, 0 < p k i) (p₀ : Fin (d + 1) → ℝ) (hs₀ : ∑ i, p₀ i = 1)
20    (hp₀ : ∀ i, 0 < p₀ i) (hord : StrictAnti p₀)
21    (hlim : ∀ i, Tendsto (fun k ↦ p k i) atTop (𝓝 (p₀ i))) :
22    Tendsto (fun k ↦ ∫ x, |tensorYoungDensity d (n k) (p k) (fun i ↦ (hp0 k i).le) (hp k) x -
23      covarianceGaussian d p₀ hs₀ x| ∂volume) atTop (𝓝 0) := by
24  apply density_scheffe volume
25    (fun k ↦ tensorYoungDensity d (n k) (p k) (fun i ↦ (hp0 k i).le) (hp k))
26    (covarianceGaussian d p₀ hs₀)
27    (fun k ↦ integrable_tensorYoungDensity d (n k) (hn0 k) (p k) _ (hp k))
28    (integrable_covarianceGaussian d p₀ hs₀ hp₀)
29    (fun k ↦ tensorYoungDensity_nonneg d (n k) (p k) _ (hp k))
30    (covarianceGaussian_nonneg d p₀ hs₀)
31  · intro k
32    rw [integral_tensorYoungDensity d (n k) (hn0 k), integral_covarianceGaussian d p₀ hs₀ hp₀]
33  · exact ae_of_all volume fun x ↦ tensorYoungDensity_tendsto d n hn hn0 p hp hp0 p₀ hs₀ hp₀ hord hlim x

35/-- The reference Gaussian can move with the spectrum at every sample size. -/
36theorem tensorYoungDensity_l1_moving_gaussian (d : ℕ) (n : ℕ → ℕ)
37    (hn : Tendsto n atTop atTop) (hn0 : ∀ k, 0 < n k)
38    (p : ℕ → Fin (d + 1) → ℝ) (hp : ∀ k, ∑ i, p k i = 1)
39    (hp0 : ∀ k i, 0 < p k i) (p₀ : Fin (d + 1) → ℝ) (hs₀ : ∑ i, p₀ i = 1)
40    (hp₀ : ∀ i, 0 < p₀ i) (hord : StrictAnti p₀)
41    (hlim : ∀ i, Tendsto (fun k ↦ p k i) atTop (𝓝 (p₀ i))) :
42    Tendsto (fun k ↦ ∫ x, |tensorYoungDensity d (n k) (p k) (fun i ↦ (hp0 k i).le) (hp k) x -
43      covarianceGaussian d (p k) (hp k) x| ∂volume) atTop (𝓝 0) := by
44  have hf := tensorYoungDensity_l1_tendsto d n hn hn0 p hp hp0 p₀ hs₀ hp₀ hord hlim
45  have hg := covarianceGaussian_l1_tendsto d p p₀ hp hs₀ hp0 hp₀ hlim
46  have ht := hf.add hg
47  simp only [zero_add] at ht
48  apply squeeze_zero (fun k ↦ integral_nonneg (fun _ ↦ abs_nonneg _)) ?_ ht
49  intro k
50  have hfi := integrable_tensorYoungDensity d (n k) (hn0 k) (p k) (fun i ↦ (hp0 k i).le) (hp k)
51  have hgi := integrable_covarianceGaussian d (p k) (hp k) (hp0 k)
52  have hzi := integrable_covarianceGaussian d p₀ hs₀ hp₀
53  erw [← integral_add (hfi.sub hzi).abs (hgi.sub hzi).abs]
54  apply integral_mono (hfi.sub hgi).abs ((hfi.sub hzi).abs.add (hgi.sub hzi).abs)
55  intro x
56  simpa only [abs_sub_comm (covarianceGaussian d p₀ hs₀ x)] using
57    abs_sub_le (tensorYoungDensity d (n k) (p k) (fun i ↦ (hp0 k i).le) (hp k) x)
58      (covarianceGaussian d p₀ hs₀ x) (covarianceGaussian d (p k) (hp k) x)

60/-- The manuscript's uniform Young local limit on every compact subset of
61the strictly ordered positive simplex, for the actual normalized physical
62Schur measurement. No local limit, character limit or physical block law is
63assumed. -/
64theorem uniform_tensorYoungDensity_l1 (d : ℕ) (K : Set (Fin (d + 1) → ℝ))
65    (hK : IsCompact K) (hp : ∀ p ∈ K, ∑ i, p i = 1)
66    (hp0 : ∀ p ∈ K, ∀ i, 0 < p i) (hord : ∀ p ∈ K, StrictAnti p)
67    (ε : ℝ) (hε : 0 < ε) :
68    ∃ N₀ : ℕ, ∀ N ≥ N₀, ∀ (p : Fin (d + 1) → ℝ) (hpK : p ∈ K),
69      (∫ x, |tensorYoungDensity d N p (fun i ↦ (hp0 p hpK i).le) (hp p hpK) x -
70        covarianceGaussian d p (hp p hpK) x| ∂volume) < ε := by
71  classical
72  by_contra h
73  push_neg at h
74  have hbad (k : ℕ) := h (max k 1)
75  choose n hn p hpK hb using hbad
76  obtain ⟨p₀, hp₀K, φ, hφ, hconv⟩ := hK.tendsto_subseq hpK
77  have hnlim : Tendsto n atTop atTop :=
78    tendsto_atTop_mono (fun k ↦ (le_max_left k 1).trans (hn k)) tendsto_id
79  have hn0 : ∀ k, 0 < n (φ k) := fun k ↦ by have h := hn (φ k); omega
80  have hlim i : Tendsto (fun k ↦ p (φ k) i) atTop (𝓝 (p₀ i)) :=
81    (continuous_apply i).continuousAt.tendsto.comp hconv
82  have ht := tensorYoungDensity_l1_moving_gaussian d (fun k ↦ n (φ k))
83    (hnlim.comp hφ.tendsto_atTop) hn0 (fun k ↦ p (φ k))
84    (fun k ↦ hp _ (hpK (φ k))) (fun k ↦ hp0 _ (hpK (φ k)))
85    p₀ (hp p₀ hp₀K) (hp0 p₀ hp₀K) (hord p₀ hp₀K) hlim
86  obtain ⟨k, hk⟩ := (ht.eventually (gt_mem_nhds hε)).exists
87  exact (not_lt_of_ge (hb (φ k))) hk
