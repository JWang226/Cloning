39    (b : OrthonormalBasis (Fin (k+1)) ℝ (EuclideanSpace ℝ (Fin (k+1))))
40    (hb : b 0=sqrtSpectrum p.eigenvalue) (e : Fin s ≃ PairIndex (k+1))

48    (b : OrthonormalBasis (Fin (k+1)) ℝ (EuclideanSpace ℝ (Fin (k+1))))

61/-- The full Gaussian amplification theorem at the actual physical score
62window. No product, covariance, or Gaussian restriction is imposed on the
63channel supremum, and L ranges over all real radii. -/
64theorem scoreGaussianOptimal_tendsto (p : SimpleSpectrum (k+1))
65    (b : OrthonormalBasis (Fin (k+1)) ℝ (EuclideanSpace ℝ (Fin (k+1))))
66    (hb : b 0=sqrtSpectrum p.eigenvalue) (e : Fin s ≃ PairIndex (k+1))
67    (g : ℝ) (hg : 1<g) :
68    Tendsto (scoreGaussianOptimal p b hb e g) atTop (𝓝 (universalValue g p)) := by
69  have h := (scorePhaseBody p.eigenvalue p.positive b hb s).gaussian_optimalPayoff_tendsto
70    (fun _ => 1/2) (fun _ => by norm_num) g hg
71    (fun i => p.ratio (e i)) (fun i => (p.ratio_pos (e i)).le) (fun i => p.ratio_lt_one (e i))
72  have hprod := e.prod_comp (fun ij => Thermal.modeFactor g (p.ratio ij))
73  have hlim : Tendsto ((scorePhaseBody p.eigenvalue p.positive b hb s).optimalPayoff
74      (Real.sqrt g) (reference p e) (reference p e)) atTop (𝓝 (universalValue g p)) := by
75    simpa only [reference,universalValue,classicalValue,orbitalValue,
76      Nat.cast_add,Nat.cast_one,add_sub_cancel_right,hprod] using h
77  apply hlim.congr'
78  filter_upwards [eventually_gt_atTop (0 : ℝ)] with L hL
79  exact (scoreGaussianOptimal_eq p b hb e g hL).symm
