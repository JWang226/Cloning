11/-- Majorization for already decreasing, normalized spectra. -/
12def SpectrumMajorizedBy {d : ℕ} (p' p : SimpleSpectrum d) : Prop :=
13  ∀m : ℕ, (∑i : Fin d with i.val<m,p'.eigenvalue i)≤
14    ∑i : Fin d with i.val<m,p.eigenvalue i

16/-- Increasing every smaller-to-larger spectral ratio reduces every leading
17partial sum. Total sums are already one in `SimpleSpectrum`. -/
18theorem spectrumMajorizedBy_of_ratios {d : ℕ} (p p' : SimpleSpectrum d)
19    (h : ∀ij : PairIndex d,p.ratio ij≤p'.ratio ij) : SpectrumMajorizedBy p' p := by
20  classical
21  intro m
22  let S : Finset (Fin d) := Finset.univ.filter (fun i=>i.val<m)
23  have hcross (i : Fin d) (hi : i∈S) (j : Fin d) (hj : j∈Sᶜ) :
24      p.eigenvalue j*p'.eigenvalue i≤p'.eigenvalue j*p.eigenvalue i := by
25    have hi' : i.val<m := (Finset.mem_filter.mp hi).2
26    have hj' : ¬j.val<m := by simpa [S] using hj
27    have hij : i<j := by simpa using (show i.val<j.val by omega)
28    exact (div_le_div_iff₀ (p.positive i) (p'.positive i)).mp (h ⟨(i,j),hij⟩)
29  have hx : (∑j∈Sᶜ,p.eigenvalue j)*(∑i∈S,p'.eigenvalue i)≤
30      (∑j∈Sᶜ,p'.eigenvalue j)*(∑i∈S,p.eigenvalue i) := by
31    rw [Finset.mul_sum,Finset.mul_sum]
32    apply Finset.sum_le_sum
33    intro i hi
34    rw [Finset.sum_mul,Finset.sum_mul]
35    exact Finset.sum_le_sum (fun j hj=>hcross i hi j hj)
36  have hp : (∑i∈S,p.eigenvalue i)+(∑i∈Sᶜ,p.eigenvalue i)=1 := by
37    rw [Finset.sum_add_sum_compl,p.normalized]
38  have hp' : (∑i∈S,p'.eigenvalue i)+(∑i∈Sᶜ,p'.eigenvalue i)=1 := by
39    rw [Finset.sum_add_sum_compl,p'.normalized]
40  change (∑i∈S,p'.eigenvalue i)≤∑i∈S,p.eigenvalue i
41  have hpC : (∑i∈Sᶜ,p.eigenvalue i)=1-∑i∈S,p.eigenvalue i := by linarith
42  have hpC' : (∑i∈Sᶜ,p'.eigenvalue i)=1-∑i∈S,p'.eigenvalue i := by linarith
43  rw [hpC,hpC'] at hx
44  nlinarith
