14theorem addBox_injective (mu : Fin d → ℕ) : Function.Injective (addBox mu) := by
15  intro r s h
16  by_contra hrs

20/-- The largest nonzero slice forces the exact one-box highest weight. -/
21theorem maximal_lastSlice_weight
22    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
23    (hΩnorm : ‖Ω‖ = 1)
24    (hΩweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
25    (hΩraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
26    (x : TensorRegister (n + 1) (Fin d)) (lambda : Fin d → ℕ)
27    (hx : x ∈ tensorProductSector (cyclicSector Ω) (⊤ : Submodule ℂ (TensorRegister 1 (Fin d))))
28    (hxweight : ∀ a, collectiveGenerator (n + 1) a a x = (lambda a : ℂ) • x)
29    (hxraise : ∀ a b, a < b → collectiveGenerator (n + 1) a b x = 0)
30    (r : Fin d) (hr : lastSlice r x ≠ 0) (htail : ∀ s, r < s → lastSlice s x = 0) :
31    lambda = addBox mu r := by
32  have hline := maximal_lastSlice_highest Ω hΩnorm hΩraise x hx hxraise r htail
33  funext a
34  have hw : collectiveGenerator n a a (lastSlice r x) = (mu a : ℂ) • lastSlice r x := by
35    rw [hline, map_smul, hΩweight]
36    exact smul_comm _ _ _
37  have he := congrArg (lastSlice r) (hxweight a)
38  rw [lastSlice_generator, map_smul, hw] at he
39  have hscalar : ((mu a + if a = r then 1 else 0 : ℕ) : ℂ) • lastSlice r x =
40      (lambda a : ℂ) • lastSlice r x := by
41    by_cases har : a = r
42    · subst a
43      simpa [Nat.cast_add, add_smul] using he
44    · simpa [har, Ne.symm har] using he
45  have heq : ((mu a + if a = r then 1 else 0 : ℕ) : ℂ) = (lambda a : ℂ) :=
46    smul_left_injective ℂ hr hscalar
47  have hn : mu a + (if a = r then 1 else 0) = lambda a := by exact_mod_cast heq
48  exact hn.symm

50/-- Every nonzero physical highest tensor in `S_mu ⊗ C^d` has weight `mu+e_r`.
51Both the row and its nonzero highest slice are constructed, not assumed. -/
52theorem highest_onebox_classification
53    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
54    (hΩnorm : ‖Ω‖ = 1)
55    (hΩweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
56    (hΩraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
57    (x : TensorRegister (n + 1) (Fin d)) (lambda : Fin d → ℕ)
58    (hx : x ∈ tensorProductSector (cyclicSector Ω) (⊤ : Submodule ℂ (TensorRegister 1 (Fin d))))
59    (hxweight : ∀ a, collectiveGenerator (n + 1) a a x = (lambda a : ℂ) • x)
60    (hxraise : ∀ a b, a < b → collectiveGenerator (n + 1) a b x = 0) (hx0 : x ≠ 0) :
61    ∃ r : Fin d, lambda = addBox mu r ∧ lastSlice r x ≠ 0 ∧
62      ∀ s, r < s → lastSlice s x = 0 := by
63  obtain ⟨r, hr, htail⟩ := exists_maximal_lastSlice x hx0
64  exact ⟨r, maximal_lastSlice_weight Ω mu hΩnorm hΩweight hΩraise x lambda
65    hx hxweight hxraise r hr htail, hr, htail⟩

67theorem highest_onebox_lastSlice_vanish_above
68    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
69    (hΩnorm : ‖Ω‖ = 1)
70    (hΩweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
71    (hΩraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
72    (x : TensorRegister (n + 1) (Fin d)) (r : Fin d)
73    (hx : x ∈ tensorProductSector (cyclicSector Ω) (⊤ : Submodule ℂ (TensorRegister 1 (Fin d))))
74    (hxweight : ∀ a, collectiveGenerator (n + 1) a a x = (addBox mu r a : ℂ) • x)
75    (hxraise : ∀ a b, a < b → collectiveGenerator (n + 1) a b x = 0) :
76    ∀ s, r < s → lastSlice s x = 0 := by
77  classical
78  by_cases hx0 : x = 0
79  · simp [hx0]
80  · obtain ⟨r', he, _, ht⟩ := highest_onebox_classification Ω mu hΩnorm hΩweight hΩraise
81      x (addBox mu r) hx hxweight hxraise hx0
82    have hr : r = r' := addBox_injective mu he
83    simpa only [hr] using ht
