24theorem loweringWord_inner_eq_zero_of_height_ne
25    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
26    (hweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
27    (u v : List (PositiveRoot d)) (huv : loweringHeight u ≠ loweringHeight v) :
28    ⟪loweringWord Ω u, loweringWord Ω v⟫_ℂ = 0 := by
29  have hw : loweringWeight u ≠ loweringWeight v := by
30    intro he
31    have hh := loweringWeight_index_sum u
32    rw [he, loweringWeight_index_sum v] at hh
33    exact huv (by exact_mod_cast hh.symm)
34  obtain ⟨a, ha⟩ := Function.ne_iff.mp hw
35  apply inner_eq_zero_of_real_eigenvalues (collectiveGenerator n a a).toLinearMap
36    (cartan_isSymmetric a) _ _ ((mu a : ℝ) + loweringWeight u a)
37      ((mu a : ℝ) + loweringWeight v a)
38  · simpa only [Complex.ofReal_add, Complex.ofReal_natCast, Complex.ofReal_intCast] using
39      cartan_loweringWord Ω (fun a => (mu a : ℂ)) hweight u a
40  · simpa only [Complex.ofReal_add, Complex.ofReal_natCast, Complex.ofReal_intCast] using
41      cartan_loweringWord Ω (fun a => (mu a : ℂ)) hweight v a
42  · intro he
43    apply ha
44    exact_mod_cast (add_left_cancel he)

74theorem iSup_sectorHeightSpace (Ω : TensorRegister n (Fin d)) :
75    (⨆ H : ℕ, sectorHeightSpace Ω H) = ⊤ := by
76  apply le_antisymm le_top
77  intro x _

93theorem tensorWordMultiplier_loweringWord
94    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
95    (hweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
96    (p : Fin d → ℝ) (hp : ∀ a, 0 < p a) (w : List (PositiveRoot d))
97    (v : Fin n → Fin d) (hv : loweringWord Ω w v ≠ 0) :
98    tensorWordMultiplier p v = (∏ a, p a ^ mu a) * wordBoltzmann p w := by

102  apply Complex.ofReal_injective
103  simpa only [tensorWordMultiplier, Complex.ofReal_prod] using mul_right_cancel₀ hv he

105/-- Outside the finite-height Boltzmann bound every computational coefficient
106of the whole height space vanishes, not only those of a preferred word family. -/
107theorem cyclicHeightSpace_coefficient_bound
108    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
109    (hweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
110    (p : Fin d → ℝ) (hp : ∀ a, 0 < p a) (θ : ℝ) (hθ : 0 ≤ θ)
111    (hroot : ∀ r : PositiveRoot d, rootBoltzmann p r ≤ θ ^ r.height)
112    (H : ℕ) {x : TensorRegister n (Fin d)} (hx : x ∈ cyclicHeightSpace Ω H)
113    (v : Fin n → Fin d)
114    (hv : (∏ a, p a ^ mu a) * θ^H < tensorWordMultiplier p v) : x v = 0 := by
115  induction hx using Submodule.span_induction with
116  | mem x hx =>
117    obtain ⟨w, hw, rfl⟩ := hx
118    by_contra hn
119    have he := tensorWordMultiplier_loweringWord Ω mu hweight p hp w v hn
120    have hb : (∏ a, p a ^ mu a) * wordBoltzmann p w ≤
121        (∏ a, p a ^ mu a) * θ ^ loweringHeight w := mul_le_mul_of_nonneg_left
122      (wordBoltzmann_le_pow_height p (fun a => (hp a).le) θ hθ hroot w)
123      (Finset.prod_nonneg (fun a _ => pow_nonneg (hp a).le (mu a)))
124    rw [hw, ← he] at hb
125    exact (not_lt_of_ge hb) hv
126  | zero => rfl
127  | add x y hx hy ihx ihy => simp [lp.coeFn_add, ihx, ihy]
128  | smul c x hx ih => simp [lp.coeFn_smul, ih]

130theorem tensorOperator_diagonal_norm_le_on_height
131    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
132    (hweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
133    (p : Fin d → ℝ) (hp : ∀ a, 0 < p a) (θ : ℝ) (hθ : 0 ≤ θ)
134    (hroot : ∀ r : PositiveRoot d, rootBoltzmann p r ≤ θ ^ r.height)
135    (H : ℕ) {x : TensorRegister n (Fin d)} (hx : x ∈ cyclicHeightSpace Ω H) :
136    ‖tensorOperator n (Matrix.diagonal (fun a => (p a : ℂ))) x‖ ≤
137      ((∏ a, p a ^ mu a) * θ^H) * ‖x‖ := by
138  apply register_norm_le_of_pointwise _ x _
139    (mul_nonneg (Finset.prod_nonneg (fun a _ => pow_nonneg (hp a).le (mu a))) (pow_nonneg hθ H))
140  intro v
141  rw [tensorOperator_diagonal_apply, norm_mul]
142  by_cases hv : x v = 0
143  · simp [hv]
144  · have hb : tensorWordMultiplier p v ≤ (∏ a, p a ^ mu a) * θ^H := by
145      by_contra! hh
146      exact hv (cyclicHeightSpace_coefficient_bound Ω mu hweight p hp θ hθ hroot H hx v hh)
147    have hn : ‖∏ t, (p (v t) : ℂ)‖ = tensorWordMultiplier p v := by
148      rw [← Complex.ofReal_prod, Complex.norm_real, Real.norm_eq_abs,
149        abs_of_nonneg (Finset.prod_nonneg (fun t _ => (hp (v t)).le))]
150      rfl
151    rw [hn]
152    exact mul_le_mul_of_nonneg_right hb (norm_nonneg _)
