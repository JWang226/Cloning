23theorem exists_uniform_model_scale_error (p : SimpleSpectrum (k+1))
24    (b : Fin (k+1) → EuclideanSpace ℝ (Fin (k+1))) (e : Fin s ≃ PairIndex (k+1))
25    (K : Set (Parameters k)) (hK : IsCompact K) (c : ℕ → ℝ) (c₀ : ℝ)
26    (hc : Tendsto c atTop (𝓝 c₀)) :
27    ∃ η : ℕ → ℝ, Tendsto η atTop (𝓝 0) ∧ ∀ n, ∀ θ ∈ K,
28      ‖(model p b e (c n • θ)).1 - (model p b e (c₀ • θ)).1‖ ≤ η n := by
29  letI : CompactSpace K := isCompact_iff_compactSpace.mp hK
30  let F : ℝ → C(K, _) := fun t =>
31    ⟨fun θ => (model p b e (t • θ.val)).1,
32      (continuous_model p b e).comp (continuous_const.smul continuous_subtype_val)⟩
33  have hF : Continuous F := by
34    apply ContinuousMap.continuous_of_continuous_uncurry
35    exact (continuous_model p b e).comp (continuous_fst.smul (continuous_subtype_val.comp continuous_snd))
36  refine ⟨fun n => ‖F (c n)-F c₀‖, ?_, ?_⟩
37  · simpa using (((hF.tendsto c₀).comp hc).sub_const (F c₀)).norm
38  · intro n θ hθ
39    exact ContinuousMap.norm_coe_le_norm (F (c n)-F c₀) ⟨θ,hθ⟩

41/-- The same physical LAN channel sequences supply both sample-size
42comparisons on a fixed window; no additional uniformity in the window is used. -/
43theorem scaled_window_approximations
44    (p : SimpleSpectrum (k+1))
45    (b : Fin (k+1) → EuclideanSpace ℝ (Fin (k+1))) (e : Fin s ≃ PairIndex (k+1))
46    (lan : CompactWindowLAN p b e)
47    (m : ℕ → ℕ) (hm : Tendsto m atTop atTop)
48    (c : ℕ → ℝ) (c₀ : ℝ) (hc : Tendsto c atTop (𝓝 c₀))
49    (K : Set (Parameters k)) (hK : IsCompact K) (hzero : ∀ θ ∈ K, ∑ i, θ.1 i = 0) :
50    ∃ δ η : ℕ → ℝ, Tendsto δ atTop (𝓝 0) ∧ Tendsto η atTop (𝓝 0) ∧
51      ∀ᶠ n in atTop, ∀ θ ∈ K,
52        ‖(lan.reverse n).map (model p b e θ).1 - chartTensor p θ n‖ ≤ δ n ∧
53        ‖(lan.forward (m n)).map (chartTensor p (c n • θ) (m n)) -
54          (model p b e (c₀ • θ)).1‖ ≤ η n := by
55  let C := |c₀|+1
56  let K' : Set (Parameters k) := (fun z : ℝ × Parameters k => z.1 • z.2) '' (Set.Icc (-C) C ×ˢ K)
57  have hK' : IsCompact K' :=
58    (isCompact_Icc.prod hK).image (continuous_fst.smul continuous_snd)
59  have hz' : ∀ θ ∈ K', ∑ i, θ.1 i = 0 := by
60    rintro θ ⟨⟨t,u⟩,hu,rfl⟩
61    simp only [Prod.smul_fst, Pi.smul_apply, smul_eq_mul, ← Finset.mul_sum,
62      hzero u hu.2, mul_zero]
63  obtain ⟨δ,hδ,hδbound⟩ := lan.approximation K hK hzero
64  obtain ⟨η,hη,hηbound⟩ := lan.approximation K' hK' hz'
65  obtain ⟨ω,hω,hωbound⟩ := exists_uniform_model_scale_error p b e K hK c c₀ hc
66  have htail : ∀ᶠ n in atTop, c n ∈ Set.Icc (-C) C := by
67    filter_upwards [hc.norm.eventually_lt_const (show ‖c₀‖ < C by dsimp [C]; linarith)] with n hn
68    simpa only [Set.mem_Icc, Real.norm_eq_abs, abs_le] using hn.le
69  refine ⟨δ,fun n => η (m n)+ω n,hδ,by simpa using (hη.comp hm).add hω,?_⟩
70  filter_upwards [hδbound,hm.eventually hηbound,htail] with n hd he ht θ hθ
71  refine ⟨(hd θ hθ).2,?_⟩
72  have hmem : c n • θ ∈ K' := ⟨(c n,θ),⟨ht,hθ⟩,rfl⟩
73  exact (norm_sub_le_norm_sub_add_norm_sub _ (model p b e (c n • θ)).1 _).trans
74    (add_le_add (he (c n • θ) hmem).1 (hωbound n θ hθ))
