36    ⟪coherentVector z, T.1 (coherentVector z)⟫_ℂ =
37      ∫ a : Fin d → ℂ, (weylGaussianWeight a : ℂ) *

60lemma integrable_coherent_fourier_kernel (μ : Measure (Fin d → ℂ)) [IsFiniteMeasure μ]
61    (T : TraceClass (Fock d)) :
62    Integrable (fun p : (Fin d → ℂ) × (Fin d → ℂ) =>
63      (weylGaussianWeight p.2 : ℂ) *
64        (weylCharacter p.1 p.2 * tracePairing T (displacement p.2))) (μ.prod volume) := by
65  apply ((integrable_weylGaussianWeight.mul_const ‖T‖).comp_snd μ).mono'
66  · exact (((Complex.continuous_ofReal.comp continuous_weylGaussianWeight).comp
67      continuous_snd).mul (continuous_weylCharacter_pair.mul

75/-- Exact Fourier witness identity for every finite coherent mixture.
76The proof justifies the interchange by a product-integrable bound, uniform in
77the coherent amplitude. -/
78theorem tracePairing_coherentMixture_fourier
79    (μ : Measure (Fin d → ℂ)) [IsFiniteMeasure μ] (T : TraceClass (Fock d)) :
80    tracePairing T (∫ z, coherentProjector z ∂μ).1 =
81      ∫ a : Fin d → ℂ, (weylGaussianWeight a : ℂ) *
82        ((∫ z, weylCharacter z a ∂μ) * tracePairing T (displacement a)) := by
83  rw [tracePairing_coherentMixture]
84  simp_rw [coherent_diagonal_characteristic_integral]
85  rw [integral_integral_swap (integrable_coherent_fourier_kernel μ T)]
86  congr 1
87  funext a
