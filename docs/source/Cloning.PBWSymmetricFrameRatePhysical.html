21theorem rootError_le_div_sqrt (r : ℕ) {δ : ℝ} (hδ : 1≤δ) :
22    rootError r δ≤rootErrorRateConstant r/Real.sqrt δ := by
23  have hd : 0<δ := by linarith
24  let t := δ⁻¹
25  have ht0 : 0≤t := inv_nonneg.mpr hd.le
26  have ht1 : t≤1 := inv_le_one_of_one_le₀ hδ
27  have ht2 : t^2≤t := by nlinarith
28  have hts : t≤Real.sqrt t := by
29    have hs := Real.sq_sqrt ht0
30    have hp := Real.sqrt_nonneg t
31    nlinarith
32  have hs : Real.sqrt (((r:ℝ)+1)*t+2*r*((r:ℝ)+1)*t^2)≤
33      Real.sqrt (((r:ℝ)+1)+2*r*((r:ℝ)+1))*Real.sqrt t := by
34    rw [←Real.sqrt_mul (by positivity : 0≤((r:ℝ)+1)+2*r*((r:ℝ)+1))]
35    apply Real.sqrt_le_sqrt
36    have h := mul_le_mul_of_nonneg_left ht2
37      (by positivity : 0≤2*(r:ℝ)*((r:ℝ)+1))
38    nlinarith
39  have hf : 2*(r:ℝ)*t≤2*r*Real.sqrt t :=
40    mul_le_mul_of_nonneg_left hts (by positivity)
41  have hh : max (2*(r:ℝ)*t)
42      (Real.sqrt (((r:ℝ)+1)*t+2*r*((r:ℝ)+1)*t^2))≤
43      rootErrorRateConstant r*Real.sqrt t := by
44    apply max_le
45    · apply hf.trans
46      apply mul_le_mul_of_nonneg_right _ (Real.sqrt_nonneg t)
47      exact le_add_of_nonneg_right (Real.sqrt_nonneg _)
48    · apply hs.trans
49      apply mul_le_mul_of_nonneg_right _ (Real.sqrt_nonneg t)
50      exact le_add_of_nonneg_left (by positivity)
51  simpa only [rootError,div_eq_mul_inv,inv_pow,Real.sqrt_inv,t] using hh
