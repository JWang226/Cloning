15theorem exists_uniform_order_eventually_exponential_near_polynomial
16    (B : ℝ) (hB : 0 ≤ B) (w : List (PositiveRoot d)) (ε : ℝ) (hε : 0 < ε) :
17    ∃ M : ℕ, ∀ m ≥ M, ∀ Q : ℕ, loweringHeight w+m*d ≤ Q →
18      ∀ (mu : ℕ → Fin d → ℕ) (hmu : ∀ N, Antitone (mu N))
19        (δ : ℕ → ℝ), Tendsto δ atTop atTop →
20        (∀ᶠ N in atTop, ∀ a : PositiveRoot d, δ N ≤ rootGap (mu N) a) →
21      ∀ (z : ℕ → PositiveRoot d → ℂ) (z₀ : PositiveRoot d → ℂ),
22        Tendsto z atTop (𝓝 z₀) → (∑ a, ‖z₀ a‖) ≤ B →
23      ∀ᶠ N in atTop,
24        ‖cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q
25            (exp (rootGenerator (n := ∑ j, mu N j) (mu N) (z N))
26              (normalizedLoweringWord (partitionHighestTensor (mu N) (hmu N)) (mu N) w)) -
27          oscillatorTaylor Q z₀ w m‖ < ε := by
28  let R := loweringHeight w
29  obtain ⟨M,hM⟩ := eventually_atTop.mp
30    ((uniformTaylorBound_tendsto (B+1) (by linarith) R d).eventually_lt_const
31      (show 0 < ε/4 by linarith))
32  refine ⟨M, ?_⟩
33  intro m hm Q hQ mu hmu δ hδ hgap z z₀ hz hz₀
34  have hsmall := hM m hm
35  have hzsum : Tendsto (fun N => ∑ a, ‖z N a‖) atTop (𝓝 (∑ a, ‖z₀ a‖)) :=
36    tendsto_finset_sum _ (fun a _ => (tendsto_pi_nhds.mp hz a).norm)
37  have hzB : ∀ᶠ N in atTop, (∑ a, ‖z N a‖) ≤ B+1 :=
38    (hzsum.eventually_lt_const (by linarith)).mono (fun _ h => h.le)
39  have hnorm : ∀ᶠ N in atTop,
40      ‖normalizedLoweringWord (partitionHighestTensor (mu N) (hmu N)) (mu N) w‖ ≤ 2 :=
41    ((partition_normalizedWord_norm_tendsto mu hmu δ hδ hgap w).eventually_lt_const
42      (by norm_num : (1 : ℝ)<2)).mono (fun _ h => h.le)
43  have hpoly := cutoffEmbedding_taylor_tendsto mu hmu δ hδ hgap z z₀ hz R Q m hQ w le_rfl
44  have hpolyerr := (tendsto_iff_norm_sub_tendsto_zero.mp hpoly).eventually_lt_const
45    (show 0 < ε/2 by linarith)
46  have hlarge := hδ.eventually (eventually_ge_atTop (2*(((R+(m+1)*d : ℕ) : ℝ)+1)))
47  filter_upwards [hzB,hnorm,hpolyerr,hgap,hlarge,
48    cutoffEmbedding_eventually_contraction mu hmu δ hδ hgap Q] with N hzN hnN hpN hgN hLN hV
49  let x := normalizedLoweringWord (partitionHighestTensor (mu N) (hmu N)) (mu N) w
50  let G := rootGenerator (n := ∑ j, mu N j) (mu N) (z N)
51  let T := ∑ j ∈ Finset.range (m+1), ((j.factorial : ℝ)⁻¹) • ((G^j) x)
52  have hx : x ∈ cyclicCutoff (partitionHighestTensor (mu N) (hmu N)) (R : ℤ) :=
53    normalizedLoweringWord_mem_cyclicCutoff _ _ _ (by exact_mod_cast (le_rfl : loweringHeight w ≤ R))
54  have hrem : ‖exp G x - T‖ < ε/2 := by
55    have hr := exp_rootGenerator_taylor_remainder
56      (partitionHighestTensor (mu N) (hmu N)) (mu N)
57      (partitionHighestTensor_cartan (mu N) (hmu N))
58      (partitionHighestTensor_raising_zero (mu N) (hmu N)) (z N) R m
59      (fun a => hLN.trans (hgN a)) hx
60    have hb : ‖exp G x - T‖ ≤ uniformTaylorBound (B+1) R d m * 2 := by
61      apply hr.trans
62      unfold uniformTaylorBound
63      rw [div_mul_eq_mul_div]
64      apply div_le_div_of_nonneg_right _ (Nat.cast_nonneg _)
65      apply mul_le_mul _ hnN (norm_nonneg _) (by positivity)
66      apply pow_le_pow_left₀ (by positivity)
67      exact mul_le_mul_of_nonneg_right
68        (mul_le_mul_of_nonneg_left hzN (by norm_num)) (Real.sqrt_nonneg _)
69    linarith
70  let V := cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q
71  have hV' : ∀ y, ‖V y‖ ≤ ‖y‖ := hV
72  change ‖V T - oscillatorTaylor Q z₀ w m‖ < ε/2 at hpN
73  have hVT : ‖V (exp G x) - V T‖ ≤ ‖exp G x - T‖ := by
74    rw [← V.map_sub]
75    exact hV' (exp G x - T)
76  calc
77    ‖V (exp G x) - oscillatorTaylor Q z₀ w m‖ ≤
78        ‖V (exp G x) - V T‖ + ‖V T - oscillatorTaylor Q z₀ w m‖ :=
79      norm_sub_le_norm_sub_add_norm_sub _ _ _
80    _ ≤ ‖exp G x - T‖ + ‖V T - oscillatorTaylor Q z₀ w m‖ := add_le_add hVT le_rfl
81    _ < ε := by linarith

86theorem exists_uniform_cutoff_eventually_word_displacement
87    (B : ℝ) (hB : 0 ≤ B) (w : List (PositiveRoot d)) (ε : ℝ) (hε : 0 < ε) :
88    ∃ Q₀ : ℕ, ∀ Q ≥ Q₀,
89      ∀ (mu : ℕ → Fin d → ℕ) (hmu : ∀ N, Antitone (mu N))
90        (δ : ℕ → ℝ), Tendsto δ atTop atTop →
91        (∀ᶠ N in atTop, ∀ a : PositiveRoot d, δ N ≤ rootGap (mu N) a) →
92      ∀ (z : ℕ → PositiveRoot d → ℂ) (z₀ : PositiveRoot d → ℂ),
93        Tendsto z atTop (𝓝 z₀) → (∑ a, ‖z₀ a‖) ≤ B →
94      ∀ᶠ N in atTop,
95        ‖cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q
96            (exp (rootGenerator (n := ∑ j, mu N j) (mu N) (z N))
97              (normalizedLoweringWord (partitionHighestTensor (mu N) (hmu N)) (mu N) w)) -
98          displacement (rootFockAmplitude z₀) (wordNumberVector w)‖ < ε := by
99  obtain ⟨M,hM⟩ := exists_uniform_order_eventually_exponential_near_polynomial
100    B hB w (ε/2) (by linarith)
101  obtain ⟨K,hK⟩ := exists_uniform_weyl_taylor_order
102    B hB (∑ j, wordFockOccupation w j) (ε/2) (by linarith)
103  let m := max M K
104  refine ⟨loweringHeight w+m*d, ?_⟩
105  intro Q hQ mu hmu δ hδ hgap z z₀ hz hz₀
106  have hsum : (∑ j, ‖rootFockAmplitude z₀ j‖) ≤ B := by
107    rw [← Equiv.sum_comp (Fintype.equivFin (PositiveRoot d))]
108    simpa only [rootFockAmplitude, Equiv.symm_apply_apply] using hz₀
109  have ht := hK m (le_max_right _ _) (rootFockAmplitude z₀) hsum
110    (wordFockOccupation w) (fun j => Finset.single_le_sum
111      (fun _ _ => Nat.zero_le _) (Finset.mem_univ j))
112  rw [norm_sub_rev] at ht
113  filter_upwards [hM m (le_max_left _ _) Q hQ mu hmu δ hδ hgap z z₀ hz hz₀] with N hN
114  rw [oscillatorTaylor_eq_numberTaylor Q m z₀ w hQ] at hN
115  have hh := norm_sub_le_norm_sub_add_norm_sub
116    (cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q
117      (exp (rootGenerator (n := ∑ j, mu N j) (mu N) (z N))
118        (normalizedLoweringWord (partitionHighestTensor (mu N) (hmu N)) (mu N) w)))
119    (numberTaylor (rootFockAmplitude z₀) (wordFockOccupation w) m)
120    (displacement (rootFockAmplitude z₀) (wordNumberVector w))
121  change ‖numberTaylor (rootFockAmplitude z₀) (wordFockOccupation w) m -
122    displacement (rootFockAmplitude z₀) (wordNumberVector w)‖ < ε/2 at ht
123  linarith

194/-- One cutoff handles every retained frame vector and both directions. It is
195chosen from the amplitude bound, retained height and tolerance alone. -/
196theorem exists_uniform_cutoff_eventually_physical_orbital_frames
197    (B : ℝ) (hB : 0 ≤ B) (R : ℕ) (ε : ℝ) (hε : 0 < ε) :
198    ∃ Q₀ : ℕ, R ≤ Q₀ ∧ ∀ Q ≥ Q₀,
199      ∀ (p : Fin d → ℝ), (∀ a : PositiveRoot d, 0 < p a.val.1-p a.val.2) →
200      ∀ (mu : ℕ → Fin d → ℕ) (hmu : ∀ N, Antitone (mu N))
201        (δ : ℕ → ℝ), Tendsto δ atTop atTop →
202        (∀ᶠ N in atTop, ∀ a : PositiveRoot d, δ N ≤ rootGap (mu N) a) →
203        Tendsto (fun N => fun a => (mu N a : ℝ)/((∑ b, mu N b : ℕ) : ℝ)) atTop (𝓝 p) →
204      ∀ (z : ℕ → PositiveRoot d → ℂ) (z₀ : PositiveRoot d → ℂ),
205        Tendsto z atTop (𝓝 z₀) → (∑ a, ‖z₀ a‖) ≤ B →
206      ∀ᶠ N in atTop, ∀ i : CutoffIndex d R,
207        (‖cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q
208            (tensorOperator (∑ j, mu N j)
209              (exp (sampleScale (∑ j, mu N j) • orbitalGenerator p (z N)))
210              (cutoffFrame (partitionHighestTensor (mu N) (hmu N)) (mu N) R i)) -
211          displacement (rootFockAmplitude z₀) (cutoffNumberFrame d R i)‖ < ε) ∧
212        (‖(cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q).adjoint
213            (displacement (rootFockAmplitude z₀) (cutoffNumberFrame d R i)) -
214          tensorOperator (∑ j, mu N j)
215            (exp (sampleScale (∑ j, mu N j) • orbitalGenerator p (z N)))
216            (cutoffFrame (partitionHighestTensor (mu N) (hmu N)) (mu N) R i)‖ < ε) := by
217  classical
218  choose Qf hRf hf using fun i : CutoffIndex d R =>
219    exists_uniform_cutoff_eventually_frame_displacement B hB R i ε hε
220  choose Qr hRr hr using fun i : CutoffIndex d R =>
221    exists_uniform_cutoff_eventually_frame_displacement_reverse B hB R i ε hε
222  let Q₀ := max R (Finset.univ.sup fun i : CutoffIndex d R => max (Qf i) (Qr i))
223  refine ⟨Q₀,le_max_left _ _,?_⟩
224  intro Q hQ p hp mu hmu δ hδ hgap hfreq z z₀ hz hz₀
225  let z' N := scaledRootParameter p (mu N) (sampleScale (∑ j, mu N j)) (z N)
226  have hz' : Tendsto z' atTop (𝓝 z₀) :=
227    scaledRootParameter_tendsto p hp mu hmu (fun N => ∑ j, mu N j) hfreq z z₀ hz
228  have hQi (i : CutoffIndex d R) : max (Qf i) (Qr i) ≤ Q :=
229    (Finset.le_sup (f := fun i : CutoffIndex d R => max (Qf i) (Qr i))
230      (Finset.mem_univ i)).trans ((le_max_right _ _).trans hQ)
231  have hfall := Filter.eventually_all.mpr (fun i =>
232    hf i Q ((le_max_left _ _).trans (hQi i)) mu hmu δ hδ hgap z' z₀ hz' hz₀)
233  have hrall := Filter.eventually_all.mpr (fun i =>
234    hr i Q ((le_max_right _ _).trans (hQi i)) mu hmu δ hδ hgap z' z₀ hz' hz₀)
235  filter_upwards [hfall,hrall,hgap,hδ.eventually (eventually_gt_atTop 0)] with N hNf hNr hg hpos
236  intro i
237  have hg' : ∀ a : PositiveRoot d, 0 < (mu N a.val.1 : ℝ)-mu N a.val.2 :=
238    fun a => hpos.trans_le (hg a)
239  constructor
240  · simpa only [tensorOperator_local_orbital p (mu N) hg', z'] using hNf i
241  · simpa only [tensorOperator_local_orbital p (mu N) hg', z'] using hNr i
