15/-- Hermitian matrices viewed in the actual purification coefficient Hilbert space. -/
16def hermitianSpace (A : Type*) [Fintype A] [DecidableEq A] : Submodule ℝ (Register (A×A)) where
17  carrier := {x | ∀ a b, star (x (b,a))=x (a,b)}
18  zero_mem' := by simp
19  add_mem' := by intro x y hx hy a b; simp [hx a b,hy a b]
20  smul_mem' := by intro c x hx a b; simp [Complex.real_smul,hx a b]

22lemma hermitianSpace_inner_real (x y : hermitianSpace A) :
23    ⟪(x : Register (A×A)),(y : Register (A×A))⟫_ℂ=
24      (⟪x,y⟫_ℝ : ℂ) := by
25  have hs : star ⟪(x : Register (A×A)),(y : Register (A×A))⟫_ℂ=
26      ⟪(x : Register (A×A)),(y : Register (A×A))⟫_ℂ := by
27    rw [lp.inner_eq_tsum]
28    simp only [tsum_fintype,Fintype.sum_prod_type,RCLike.inner_apply,
29      star_sum,star_mul,starRingEnd_apply,star_star]
30    rw [Finset.sum_comm]
31    apply Finset.sum_congr rfl
32    intro a _
33    apply Finset.sum_congr rfl
34    intro b _
35    rw [y.2 a b,← x.2 b a]
36    ring
37  change _=(⟪(x : Register (A×A)),(y : Register (A×A))⟫_ℝ : ℂ)
38  have hr : ⟪(x : Register (A×A)),(y : Register (A×A))⟫_ℝ=
39      (⟪(x : Register (A×A)),(y : Register (A×A))⟫_ℂ).re := by
40    rw [lp.inner_eq_tsum,lp.inner_eq_tsum]

44    exact real_inner_eq_re_inner (𝕜 := ℂ) (x.val i) (y.val i)

80lemma hermitianBasis_complex_span {ι : Type*} [Fintype ι]
81    (b : OrthonormalBasis ι ℝ (hermitianSpace A)) :
82    ⊤≤Submodule.span ℂ (Set.range (fun i => (b i : Register (A×A)))) := by
83  let S := Submodule.span ℂ (Set.range (fun i => (b i : Register (A×A))))
84  have hx (x : hermitianSpace A) : (x : Register (A×A))∈S := by
85    have he := b.sum_repr x
86    have he' := congrArg (fun z : hermitianSpace A => (z : Register (A×A))) he
87    dsimp only at he'
88    simp only [Submodule.coe_sum,Submodule.coe_smul_of_tower] at he'
89    rw [← he']
90    apply Submodule.sum_mem
91    intro i _
92    change ((b.repr x i : ℝ) : ℂ) • (b i : Register (A×A))∈S
93    exact S.smul_mem _ (Submodule.subset_span ⟨i,rfl⟩)
94  intro x _
95  rw [hermitian_decomposition x]
96  exact S.add_mem (hx _) (S.smul_mem _ (hx _))
