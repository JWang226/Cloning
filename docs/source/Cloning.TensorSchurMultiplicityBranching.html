13/-- Each actual irreducible copy branches once at every addable row. -/
14structure FundamentalBranchData (H : PhysicalHighestTensor n d) where
15  copies : List (PhysicalHighestTensor (n + 1) d)
16  rows : Fin copies.length ↪ Fin d
17  labels : ∀ i, (copies.get i).weight = addBox H.weight (rows i)
18  orthogonal : copies.Pairwise (fun A B => A.sector ⟂ B.sector)
19  span_eq : physicalSectorListSpan copies = tensorProductSector H.sector
20    (⊤ : Submodule ℂ (TensorRegister 1 (Fin d)))
21  row_range : ∀ r, Antitone (addBox H.weight r) ↔ ∃ i, rows i = r

23theorem fundamentalBranchData_nonempty (H : PhysicalHighestTensor n d) :
24    Nonempty (FundamentalBranchData H) := by
25  obtain ⟨L, rows, hlabels, horth, hspan⟩ :=
26    exists_fundamental_multiplicityFree_decomposition H.vector H.weight H.norm_one H.cartan H.raising
27  refine ⟨⟨L, rows, hlabels, horth, hspan, ?_⟩⟩
28  intro r
29  apply fundamental_decomposition_row_coverage H.vector H.weight L rows hlabels hspan
30  intro s hs
31  obtain ⟨x, hxmem, hx0, hxweight, hxraise⟩ := exists_fundamental_highest_of_addable
32    H.vector H.weight H.norm_one H.cartan H.raising H.weight_antitone s hs
33  exact ⟨x, ⟨hxmem, hxweight, hxraise⟩, hx0⟩
