19/-- A simultaneous physical Cartan eigenvector has the expected literal
20diagonal tensor eigenvalue, including negative formal weights of zero vectors. -/
21theorem tensorOperator_diagonal_weight
22    (D : Fin d → ℂ) (x : TensorRegister n (Fin d)) (κ : Fin d → ℤ)
23    (hx : ∀ a, collectiveGenerator n a a x = (κ a : ℂ) • x) :
24    tensorOperator n (Matrix.diagonal D) x = (∏ a, D a ^ κ a) • x := by
25  ext w
26  rw [tensorOperator_diagonal_apply]
27  simp only [lp.coeFn_smul, Pi.smul_apply, smul_eq_mul]
28  by_cases hw : x w = 0
29  · simp [hw]
30  · have hc (a : Fin d) : (occupancy w a : ℤ) = κ a := by
31      have he := congrArg (fun y : TensorRegister n (Fin d) => y w) (hx a)
32      simp only [collectiveGenerator_diagonal, lp.coeFn_smul, Pi.smul_apply, smul_eq_mul] at he
33      have hh := mul_right_cancel₀ hw he
34      apply Int.cast_injective (α := ℂ)
35      simpa only [Int.cast_natCast] using hh
36    rw [word_product_eq_occupancy]
37    congr 1
38    apply Finset.prod_congr rfl
39    intro a _
40    rw [← hc a, zpow_natCast]

61theorem sectorGibbsOperator_nonneg (p : Fin d → ℝ) (hp : ∀ a, 0 ≤ p a) :
62    0 ≤ sectorGibbsOperator Ω mu hweight hraise p := by
63  let S : Matrix (Fin d) (Fin d) ℂ := Matrix.diagonal (fun a => (Real.sqrt (p a) : ℂ))
64  have hS : Sᴴ * S = Matrix.diagonal (fun a => (p a : ℂ)) := by
65    simp only [S, Matrix.diagonal_conjTranspose, Complex.conj_ofReal,
66      Matrix.diagonal_mul_diagonal]
67    congr 1
68    funext a
69    simp [Complex.star_def, ← Complex.ofReal_mul, Real.mul_self_sqrt (hp a)]
70  unfold sectorGibbsOperator

95theorem sectorGibbsWeight_ne_zero (hΩ : ‖Ω‖ = 1)
96    (p : Fin d → ℝ) (hp : ∀ a, 0 < p a) :
97    sectorGibbsWeight Ω mu hweight hraise p ≠ 0 := by
98  intro he
99  have hop : sectorGibbsOperator Ω mu hweight hraise p = 0 := congrArg Subtype.val he
100  have h := sectorGibbsOperator_highest Ω mu hweight hraise p
101  rw [hop, ContinuousLinearMap.zero_apply] at h
102  have hc : ((∏ a, p a ^ mu a : ℝ) : ℂ) ≠ 0 := by
103    exact_mod_cast (Finset.prod_pos (fun a _ => pow_pos (hp a) _)).ne'
104  have hz : (⟨Ω, highest_mem_cyclicSector Ω⟩ : cyclicSector Ω) = 0 :=
105    (smul_eq_zero.mp h.symm).resolve_left hc
106  have hz' : Ω = 0 := congrArg Subtype.val hz
