19lemma norm_traceClassRightMultiply_le (T : TraceClass H) :
20    ‖traceClassRightMultiply T‖ ≤ ‖T‖ := by
21  unfold traceClassRightMultiply
22  exact LinearMap.mkContinuous_norm_le _ (norm_nonneg _) _

24lemma norm_tracePairing_le (T : TraceClass H) : ‖tracePairing T‖ ≤ ‖T‖ := by
25  have ht : ‖(traceCLM : TraceClass H →L[ℂ] ℂ)‖ ≤ 1 := by
26    apply ContinuousLinearMap.opNorm_le_bound _ zero_le_one
27    intro A
28    simpa using norm_trace_le_traceNorm A.2
29  calc
30    ‖tracePairing T‖ ≤ ‖(traceCLM : TraceClass H →L[ℂ] ℂ)‖ * ‖traceClassRightMultiply T‖ :=
31      ContinuousLinearMap.opNorm_comp_le _ _
32    _ ≤ 1 * ‖T‖ := mul_le_mul ht (norm_traceClassRightMultiply_le T)
33      (norm_nonneg _) zero_le_one
34    _ = ‖T‖ := one_mul _

36/-- Uniformly bounded linear functionals converge at closure points whenever
37they converge on the generating set. This is an explicit epsilon argument. -/
38lemma tendsto_functionals_at_closure {E : Type*} [NormedAddCommGroup E] [NormedSpace ℂ E]
39    (u : ℕ → E →L[ℂ] ℂ) (f : E →L[ℂ] ℂ) (C : ℝ) (hC : 0 ≤ C)
40    (hu : ∀ n, ‖u n‖ ≤ C) (hf : ‖f‖ ≤ C) {S : Set E}
41    (hpoint : ∀ x ∈ S, Tendsto (fun n => u n x) atTop (𝓝 (f x)))
42    {x : E} (hx : x ∈ closure S) : Tendsto (fun n => u n x) atTop (𝓝 (f x)) := by
43  rw [Metric.tendsto_nhds]
44  intro ε hε
45  let D := 2 * C + 1
46  have hD : 0 < D := by dsimp [D]; linarith
47  obtain ⟨y, hy, hxy⟩ := Metric.mem_closure_iff.mp hx (ε / (3 * D)) (by positivity)
48  have hevent := (Metric.tendsto_nhds.mp (hpoint y hy)) (ε / 3) (by positivity)
49  filter_upwards [hevent] with n hn
50  have hdiff : ‖u n - f‖ ≤ D := by
51    have := norm_sub_le (u n) f
52    dsimp [D]
53    linarith [hu n]
54  have hbound : ‖(u n - f) (x - y)‖ ≤ D * ‖x - y‖ :=
55    ((u n - f).le_opNorm _).trans (mul_le_mul_of_nonneg_right hdiff (norm_nonneg _))
56  have heq : u n x - f x = (u n - f) (x - y) + (u n y - f y) := by
57    simp only [ContinuousLinearMap.sub_apply, map_sub]
58    ring
59  rw [dist_eq_norm, heq]
60  have htri := norm_add_le ((u n - f) (x - y)) (u n y - f y)
61  rw [dist_eq_norm] at hn hxy
62  have hsmall : D * ‖x - y‖ < ε / 3 := by
63    have h := (mul_lt_mul_of_pos_left hxy hD)
64    have hd : D * (ε / (3 * D)) = ε / 3 := by field_simp
65    rwa [hd] at h
66  linarith

68/-- Bounded functionals have a subsequence converging on any countable family
69of observables, with a genuine bounded functional as the limit. -/
70theorem exists_subsequence_functionals_countable {E : Type*}
71    [NormedAddCommGroup E] [NormedSpace ℂ E]
72    (u : ℕ → E →L[ℂ] ℂ) (C : ℝ) (hu : ∀ n, ‖u n‖ ≤ C)
73    (O : ℕ × ℕ → E) :
74    ∃ f : E →L[ℂ] ℂ, ‖f‖ ≤ C ∧ ∃ φ : ℕ → ℕ, StrictMono φ ∧
75      ∀ i, Tendsto (fun n => u (φ n) (O i)) atTop (𝓝 (f (O i))) := by
76  let B : Set (WeakDual ℂ E) := WeakDual.toStrongDual ⁻¹' Metric.closedBall 0 C
77  let ev : WeakDual ℂ E → (ℕ × ℕ → ℂ) := fun f i => f (O i)
78  have hev : Continuous ev := continuous_pi (fun i => WeakDual.eval_continuous (O i))
79  have hc : IsCompact B := WeakDual.isCompact_closedBall ℂ (0 : E →L[ℂ] ℂ) C
80  have hmem (n : ℕ) : ev (StrongDual.toWeakDual (u n)) ∈ ev '' B := by
81    refine ⟨StrongDual.toWeakDual (u n), ?_, rfl⟩
82    simpa [B, Metric.mem_closedBall, dist_zero_right] using hu n
83  obtain ⟨g, hg, φ, hφ, hlim⟩ := (hc.image hev).tendsto_subseq hmem
84  obtain ⟨f, hf, rfl⟩ := hg
85  refine ⟨WeakDual.toStrongDual f, ?_, φ, hφ, ?_⟩
86  · simpa [B, Metric.mem_closedBall, dist_zero_right] using hf
87  · intro i
88    exact (continuous_apply i).tendsto _ |>.comp hlim

90/-- A trace-bounded positive sequence on a separable Hilbert space has an
91actual weak-operator convergent subsequence. Its positive trace-class limit
92can have smaller trace, so trace preservation is not asserted. -/
93theorem exists_subsequence_weak_operator_limit [TopologicalSpace.SeparableSpace H]
94    (A : ℕ → TraceClass H) (hA : ∀ n, 0 ≤ (A n).1) (C : ℝ)
95    (htrace : ∀ n, (trace (A n).1 (A n).2).re ≤ C) :
96    ∃ T : TraceClass H, 0 ≤ T.1 ∧ (trace T.1 T.2).re ≤ C ∧
97      ∃ φ : ℕ → ℕ, StrictMono φ ∧ ∀ x y : H,
98        Tendsto (fun n => ⟪y, (A (φ n)).1 x⟫_ℂ) atTop (𝓝 ⟪y, T.1 x⟫_ℂ) := by
99  let u : ℕ → (H →L[ℂ] H) →L[ℂ] ℂ := fun n => tracePairing (A n)
100  have hnorm (n : ℕ) : ‖A n‖ ≤ C := by
101    rw [TraceClass.norm_eq_trace_re_of_nonneg _ (hA n)]
102    exact htrace n
103  have hu (n : ℕ) : ‖u n‖ ≤ C := (norm_tracePairing_le (A n)).trans (hnorm n)
104  have hC : 0 ≤ C := (norm_nonneg (A 0)).trans (hnorm 0)
105  let d := denseSeq H
106  let O : ℕ × ℕ → H →L[ℂ] H := fun i => InnerProductSpace.rankOne ℂ (d i.1) (d i.2)
107  obtain ⟨f, hf, φ, hφ, hlim⟩ := exists_subsequence_functionals_countable u C hu O
108  have hrank (x y : H) : InnerProductSpace.rankOne ℂ x y ∈ closure (Set.range O) := by
109    have hd : DenseRange (fun i : ℕ × ℕ => (d i.1, d i.2)) :=
110      (denseRange_denseSeq H).prodMap (denseRange_denseSeq H)
111    have hr : Continuous (fun p : H × H => InnerProductSpace.rankOne ℂ p.1 p.2) :=
112      (ContinuousLinearMap.smulRightL ℂ H H).continuous₂.comp
113        (((innerSL ℂ).continuous.comp continuous_snd).prodMk continuous_fst)
114    have h := hr.range_subset_closure_image_dense hd ⟨(x, y), rfl⟩
115    simpa only [← Set.range_comp, Function.comp_def, O] using h
116  have hcoeff (x y : H) : Tendsto (fun n => ⟪y, (A (φ n)).1 x⟫_ℂ) atTop
117      (𝓝 ⟪y, normalPartOp f x⟫_ℂ) := by
118    have h := tendsto_functionals_at_closure (fun n => u (φ n)) f C hC
119      (fun n => hu (φ n)) hf (S := Set.range O)
120      (by rintro _ ⟨i, rfl⟩; exact hlim i) (hrank x y)
121    simpa only [u, tracePairing_rankOne, normalPart_inner] using h
122  have hdiag (x : H) : Tendsto (fun n => ⟪(A (φ n)).1 x, x⟫_ℂ) atTop
123      (𝓝 ⟪normalPartOp f x, x⟫_ℂ) := by
124    convert (hcoeff x x).star using 1
125    · ext n; exact (inner_conj_symm _ _).symm
126    · congr 1; exact (inner_conj_symm _ _).symm
127  obtain ⟨hpos, hTC, hbound⟩ := InfiniteTraceClassWeakLimit.traceClass_of_diagonal_tendsto
128    (fun n => (A (φ n)).1) (normalPartOp f) (fun n => hA (φ n))
129    (fun n => (A (φ n)).2) C (fun n => htrace (φ n)) hdiag
130  exact ⟨TraceClass.ofOperator (normalPartOp f) hTC, hpos, hbound, φ, hφ, hcoeff⟩
