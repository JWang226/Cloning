14lemma rational_curve_difference {E : Type*} [AddCommGroup E] [Module ℝ E]
15    (B A C : E) (t a : ℝ) (hd : 1+t^2*a^2≠0) :
16    (1+t^2*a^2)⁻¹ • (B+t • A+t^2 • C)-B=
17      (1+t^2*a^2)⁻¹ • (t • A+t^2 • (C-a^2 • B)) := by
18  have hc : (1+t^2*a^2)⁻¹-1= -((1+t^2*a^2)⁻¹*t^2*a^2) := by
19    field_simp [hd]
20    ring
21  have hB : (1+t^2*a^2)⁻¹ • B-B= -((1+t^2*a^2)⁻¹*t^2*a^2) • B := by
22    calc
23      _ = ((1+t^2*a^2)⁻¹-1) • B := by rw [sub_smul,one_smul]
24      _ = _ := by rw [hc]
25  calc
26    _ = ((1+t^2*a^2)⁻¹ • B-B)+
27        ((1+t^2*a^2)⁻¹*t) • A+((1+t^2*a^2)⁻¹*t^2) • C := by module
28    _ = _ := by rw [hB]; module

31lemma rational_curve_norm_le {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]
32    (B A C : E) (t a c : ℝ) (ht : 0≤t) (ha : 0≤a) (hc : c≤3/4)
33    (hA : ‖A‖≤2*c*a) (hC : ‖C-a^2 • B‖≤2*a^2) :
34    ‖(1+t^2*a^2)⁻¹ • (B+t • A+t^2 • C)-B‖≤2*t*a := by
35  have hd : 0<1+t^2*a^2 := by positivity
36  rw [rational_curve_difference B A C t a hd.ne']
37  calc
38    _ = (1+t^2*a^2)⁻¹*‖t • A+t^2 • (C-a^2 • B)‖ := by
39      rw [norm_smul,Real.norm_eq_abs,abs_of_pos (inv_pos.2 hd)]
40    _ ≤ (1+t^2*a^2)⁻¹*(t*‖A‖+t^2*‖C-a^2 • B‖) := by
41      apply mul_le_mul_of_nonneg_left _ (inv_nonneg.2 hd.le)
42      have htn : ‖t‖=t := Real.norm_of_nonneg ht
43      have hsn : ‖(t^(2:ℕ):ℝ)‖=t^(2:ℕ) := Real.norm_of_nonneg (sq_nonneg t)
44      simpa only [norm_smul,htn,hsn] using
45        norm_add_le (t • A) ((t^(2:ℕ)) • (C-a^2 • B))
46    _ ≤ (1+t^2*a^2)⁻¹*(t*(2*c*a)+t^2*(2*a^2)) := by
47      exact mul_le_mul_of_nonneg_left
48        (add_le_add (mul_le_mul_of_nonneg_left hA ht)
49          (mul_le_mul_of_nonneg_left hC (sq_nonneg t))) (inv_nonneg.2 hd.le)
50    _ ≤ 2*t*a := by
51      apply (inv_mul_le_iff₀ hd).2
52      have hx : 0≤t*a := mul_nonneg ht ha
53      have hq : c+t*a≤1+(t*a)^2 := by nlinarith [sq_nonneg (t*a-1/2)]
54      have hh := mul_le_mul_of_nonneg_left hq (show 0≤2*(t*a) by positivity)
55      nlinarith

79lemma flat_rational_frobenius_le {r : ℕ} (hr : 2≤r)
80    (Z : Matrix (Fin r) (Fin r) ℂ) (t : ℝ) (ht : 0≤t) :
81    ‖(1+t^2*‖Z‖^2)⁻¹ •
82      (((r:ℝ)⁻¹ • (1 : Matrix (Fin r) (Fin r) ℂ))+
83        t • ((Real.sqrt r)⁻¹ • (Z+Zᴴ))+t^2 • (Z*Zᴴ))-
84        (r:ℝ)⁻¹ • (1 : Matrix (Fin r) (Fin r) ℂ)‖≤2*t*‖Z‖ := by
85  have hr0 : (0:ℝ)<r := by exact_mod_cast (show 0<r by omega)
86  have hc0 : 0≤(Real.sqrt r)⁻¹ := inv_nonneg.2 (Real.sqrt_nonneg _)
87  have hc := inverse_sqrt_le_three_quarters hr
88  have hB : ‖(r:ℝ)⁻¹ • (1 : Matrix (Fin r) (Fin r) ℂ)‖≤1 := by

91    have hs0 : 0<Real.sqrt r := Real.sqrt_pos.2 hr0

93      field_simp
94      nlinarith
95    rw [he]
96    linarith

98  · rw [norm_smul,Real.norm_eq_abs,abs_of_nonneg hc0]
99    have hZ := norm_add_le Z Zᴴ
100    rw [Matrix.frobenius_norm_conjTranspose] at hZ
101    calc
102      _ ≤ (Real.sqrt r)⁻¹*(‖Z‖+‖Z‖) := mul_le_mul_of_nonneg_left hZ hc0
103      _ = _ := by ring

106      _ ≤ ‖Z‖*‖Zᴴ‖+‖Z‖^2*1 := by
