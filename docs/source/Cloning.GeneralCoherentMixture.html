16theorem continuous_oneParticle (s L : ℕ) (a : Fin (s + 1)) :
17    Continuous (fun z : Fin s → ℂ => oneParticle z L a) := by
18  have hd : Continuous (fun z : Fin s → ℂ =>
19      (Real.sqrt (1 + energy z / L) : ℂ)) := by
20    exact Complex.continuous_ofReal.comp
21      (Real.continuous_sqrt.comp (continuous_const.add ((continuous_energy s).div_const _)))
22  have hdn (z : Fin s → ℂ) : (Real.sqrt (1 + energy z / L) : ℂ) ≠ 0 := by
23    exact_mod_cast (Real.sqrt_pos.mpr (by have := energy_nonneg z; positivity)).ne'
24  refine Fin.cases ?_ (fun i => ?_) a
25  · exact continuous_const.div hd hdn
26  · exact ((continuous_apply i).div_const _).div hd hdn

28theorem continuous_productTensor (s L : ℕ) : Continuous (fun z : Fin s → ℂ => productTensor z L) := by
29  classical
30  have he : (fun z : Fin s → ℂ => productTensor z L) =
31      fun z => ∑ w : Word L (s + 1), (∏ i, oneParticle z L (w i)) •
32        (lp.single 2 w 1 : TensorSpace L (s + 1)) := by
33    funext z
34    ext w
35    simp [productTensor_apply, lp.single_apply, Pi.single_apply]
36  rw [he]
37  exact continuous_finset_sum _ (fun w _ =>
38    (continuous_finset_prod _ (fun i _ => continuous_oneParticle s L (w i))).smul continuous_const)

110theorem occupationChannel_mixture_distance_le {s : ℕ} (L : ℕ)
111    (μ : Measure (Fin s → ℂ)) [IsFiniteMeasure μ] :
112    ‖(occupationChannel L s).toLinearMap (coherentMixture μ) - productMixture L μ‖ ≤
113      ∫ z, ‖(occupationChannel L s).toLinearMap (MultimodeCoherent.coherentProjector z) -
114        vectorProjector (productTensor z L)‖ ∂μ := by
115  let C := (occupationChannel L s).toPositiveTracePreservingMap.toContinuousLinearMap
116  change ‖C (∫ z, MultimodeCoherent.coherentProjector z ∂μ) -
117    ∫ z, vectorProjector (productTensor z L) ∂μ‖ ≤ _
