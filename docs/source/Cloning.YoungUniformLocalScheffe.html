13/-- Scheffé's lemma for real nonnegative densities on an arbitrary measure
14space, proved using domination of the minimum by the limiting density. -/
15theorem density_scheffe {X : Type*} [MeasurableSpace X] (ν : Measure X)
16    (f : ℕ → X → ℝ) (g : X → ℝ) (hf : ∀ n, Integrable (f n) ν)
17    (hg : Integrable g ν) (hf0 : ∀ n x, 0 ≤ f n x) (hg0 : ∀ x, 0 ≤ g x)
18    (hmass : ∀ n, (∫ x, f n x ∂ν) = ∫ x, g x ∂ν)
19    (hpoint : ∀ᵐ x ∂ν, Tendsto (fun n ↦ f n x) atTop (𝓝 (g x))) :
20    Tendsto (fun n ↦ ∫ x, |f n x - g x| ∂ν) atTop (𝓝 0) := by
21  have hminint n : Integrable (fun x ↦ min (f n x) (g x)) ν := (hf n).inf hg
22  have hmin : Tendsto (fun n ↦ ∫ x, min (f n x) (g x) ∂ν) atTop (𝓝 (∫ x, g x ∂ν)) := by
23    apply tendsto_integral_of_dominated_convergence g
24      (fun n ↦ (hminint n).aestronglyMeasurable) hg
25    · intro n
26      exact ae_of_all ν fun x ↦ by
27        rw [Real.norm_eq_abs, abs_of_nonneg (le_min (hf0 n x) (hg0 x))]
28        exact min_le_right _ _
29    · filter_upwards [hpoint] with x hx
30      simpa using hx.min (tendsto_const_nhds (x := g x))
31  have heq n : (∫ x, |f n x - g x| ∂ν) =
32      2 * (∫ x, g x ∂ν) - 2 * (∫ x, min (f n x) (g x) ∂ν) := by
33    have hid : (fun x ↦ |f n x - g x|) = fun x ↦ f n x + g x - 2 * min (f n x) (g x) := by
34      funext x
35      by_cases h : f n x ≤ g x
36      · rw [min_eq_left h, abs_of_nonpos (sub_nonpos.mpr h)]; ring
37      · rw [min_eq_right (le_of_not_ge h), abs_of_nonneg (sub_nonneg.mpr (le_of_not_ge h))]; ring
38    rw [hid]
39    erw [integral_sub ((hf n).add hg) ((hminint n).const_mul 2),
40      integral_add (hf n) hg, integral_const_mul, hmass]
41    ring
42  simp_rw [heq]
43  simpa using ((tendsto_const_nhds (x := 2 * (∫ x, g x ∂ν))).sub (hmin.const_mul 2))

45/-- Continuity of the explicit normalized Gaussian along moving positive
46spectra and moving root-space points. -/
47theorem covarianceGaussian_tendsto (d : ℕ) (p : ℕ → Fin (d + 1) → ℝ)
48    (p₀ : Fin (d + 1) → ℝ) (hp : ∀ n, ∑ i, p n i = 1) (hp₀ : ∑ i, p₀ i = 1)
49    (hp0 : ∀ n i, 0 < p n i) (hp₀0 : ∀ i, 0 < p₀ i)
50    (hlim : ∀ i, Tendsto (fun n ↦ p n i) atTop (𝓝 (p₀ i)))
51    (x : ℕ → rootSpace d) (x₀ : rootSpace d)
52    (hx : ∀ i, Tendsto (fun n ↦ (x n).1 i) atTop (𝓝 (x₀.1 i))) :
53    Tendsto (fun n ↦ covarianceGaussian d (p n) (hp n) (x n)) atTop
54      (𝓝 (covarianceGaussian d p₀ hp₀ x₀)) := by
55  have hlog := tendsto_finset_sum Finset.univ (fun i _ ↦
56    (Real.continuousAt_log (hp₀0 i).ne').tendsto.comp (hlim i))
57  have hquad := tendsto_finset_sum Finset.univ (fun i _ ↦
58    ((hx i).pow 2).div (hlim i) (hp₀0 i).ne')
59  simp_rw [covarianceGaussian_eq_exp d _ _ (hp0 _) _, covarianceGaussian_eq_exp d _ _ hp₀0 _]
60  exact Real.continuous_exp.continuousAt.tendsto.comp
61    ((tendsto_const_nhds.sub (hlog.const_mul (1 / 2))).sub (hquad.const_mul (1 / 2)))

63/-- Parameter continuity in L¹ is derived from normalization and pointwise
64continuity, with no common-eigenbasis or density-domination premise. -/
65theorem covarianceGaussian_l1_tendsto (d : ℕ) (p : ℕ → Fin (d + 1) → ℝ)
66    (p₀ : Fin (d + 1) → ℝ) (hp : ∀ n, ∑ i, p n i = 1) (hp₀ : ∑ i, p₀ i = 1)
67    (hp0 : ∀ n i, 0 < p n i) (hp₀0 : ∀ i, 0 < p₀ i)
68    (hlim : ∀ i, Tendsto (fun n ↦ p n i) atTop (𝓝 (p₀ i))) :
69    Tendsto (fun n ↦ ∫ x, |covarianceGaussian d (p n) (hp n) x -
70      covarianceGaussian d p₀ hp₀ x| ∂volume) atTop (𝓝 0) := by
71  apply density_scheffe volume (fun n ↦ covarianceGaussian d (p n) (hp n))
72    (covarianceGaussian d p₀ hp₀)
73    (fun n ↦ integrable_covarianceGaussian d (p n) (hp n) (hp0 n))
74    (integrable_covarianceGaussian d p₀ hp₀ hp₀0)
75    (fun n ↦ covarianceGaussian_nonneg d (p n) (hp n))
76    (covarianceGaussian_nonneg d p₀ hp₀)
77  · intro n
78    rw [integral_covarianceGaussian d _ _ (hp0 n), integral_covarianceGaussian d _ _ hp₀0]
79  · exact ae_of_all volume fun x ↦ covarianceGaussian_tendsto d p p₀ hp hp₀ hp0 hp₀0 hlim
80      (fun _ ↦ x) x (fun _ ↦ tendsto_const_nhds)
