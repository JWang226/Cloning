17theorem tensorOperator_diagonal_generator_weight (D : Fin d → ℂ) (a b : Fin d) :
18    D b • (tensorOperator n (Matrix.diagonal D) * collectiveGenerator n a b) =
19      D a • (collectiveGenerator n a b * tensorOperator n (Matrix.diagonal D)) := by

46/-- Moving one root through the weighted physical trace multiplies by the
47corresponding coordinate ratio, expressed without any division. -/
48theorem tensorWeightTrace_root_cycle
49    (T : TensorRegister n (Fin d) →L[ℂ] TensorRegister n (Fin d)) (a b : Fin d) :
50    X b * tensorWeightTrace (collectiveGenerator n a b * T) =
51      X a * tensorWeightTrace (T * collectiveGenerator n a b) := by
52  apply MvPolynomial.funext
53  intro D
54  simp only [map_mul, eval_X, eval_tensorWeightTrace]
55  let Q := tensorOperator n (Matrix.diagonal D)
56  let E := collectiveGenerator n a b
57  have hw : D b • (Q * E) = D a • (E * Q) := tensorOperator_diagonal_generator_weight D a b
58  calc
59    _ = LinearMap.trace ℂ (TensorRegister n (Fin d)) ((D b • (Q * E)) * T).toLinearMap := by
60      rw [smul_mul_assoc]
61      change D b * LinearMap.trace ℂ _ (Q.toLinearMap * (E.toLinearMap * T.toLinearMap)) =
62        LinearMap.trace ℂ _ (D b • ((Q.toLinearMap * E.toLinearMap) * T.toLinearMap))
63      rw [map_smul, smul_eq_mul, mul_assoc]
64    _ = LinearMap.trace ℂ (TensorRegister n (Fin d)) ((D a • (E * Q)) * T).toLinearMap := by rw [hw]
65    _ = D a * LinearMap.trace ℂ (TensorRegister n (Fin d)) ((E * Q) * T).toLinearMap := by
66      rw [smul_mul_assoc]
67      change LinearMap.trace ℂ _ (D a • (E.toLinearMap * Q.toLinearMap * T.toLinearMap)) = _
68      rw [map_smul, smul_eq_mul]
69      rfl
70    _ = _ := by
71      congr 1
72      change LinearMap.trace ℂ _ (E.toLinearMap * (Q.toLinearMap * T.toLinearMap)) =
73        LinearMap.trace ℂ _ ((Q.toLinearMap * T.toLinearMap) * E.toLinearMap)
74      exact LinearMap.trace_mul_comm ℂ _ _

101/-- A root-pair contribution satisfies the genuine denominator-free radial
102identity; commutation with the projector is the only restriction needed. -/
103theorem tensorWeightTrace_root_pair
104    (T : TensorRegister n (Fin d) →L[ℂ] TensorRegister n (Fin d)) (a b : Fin d)
105    (hT : T * collectiveGenerator n a b = collectiveGenerator n a b * T) :
106    (X a - X b) *
107      (tensorWeightTrace (collectiveGenerator n a b * collectiveGenerator n b a * T) +
108        tensorWeightTrace (collectiveGenerator n b a * collectiveGenerator n a b * T)) =
109      (X a + X b) *
110        (X a * pderiv a (tensorWeightTrace T) - X b * pderiv b (tensorWeightTrace T)) := by
111  have hc := tensorWeightTrace_root_cycle (collectiveGenerator n b a * T) a b
112  rw [← mul_assoc, mul_assoc (collectiveGenerator n b a) T, hT,
113    ← mul_assoc] at hc
114  have hd := tensorWeightTrace_root_difference T a b
115  rw [← hd]
116  linear_combination -2 * hc
