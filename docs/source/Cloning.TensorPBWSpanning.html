18/-- Every bounded physical cyclic cutoff is spanned by the finite canonical
19occupation list. This is an exact equality, valid even when some vectors vanish
20or are linearly dependent. -/
21theorem cyclicCutoff_eq_span_canonical (Ω : TensorRegister n (Fin d)) (R : ℕ) :
22    cyclicCutoff Ω (R : ℤ) = Submodule.span ℂ
23      (Set.range (fun k : HeightOccupation d R => loweringWord Ω (canonicalWord k.val))) := by
24  rw [cyclicCutoff_eq_ordered_span]
25  apply congrArg (Submodule.span ℂ)
26  ext x
27  constructor
28  · rintro ⟨w, hw, hheight, rfl⟩
29    have hk : occupationHeight (fun a => w.count a) ≤ R := by
30      simpa only [occupationHeight, ← loweringHeight_eq_sum_count] using hheight
31    refine ⟨⟨fun a => w.count a, hk⟩, ?_⟩
32    simp only [canonicalWord_counts_of_ordered w hw]
33  · rintro ⟨k, rfl⟩
34    exact ⟨canonicalWord k.val, canonicalWord_ordered _,
35      (canonicalWord_height _).trans_le k.property, rfl⟩

37/-- Exact total-height component of the physical lowering-word span. -/
38def cyclicHeightSpace (Ω : TensorRegister n (Fin d)) (H : ℕ) :
39    Submodule ℂ (TensorRegister n (Fin d)) :=
40  Submodule.span ℂ {x | ∃ w : List (PositiveRoot d), loweringHeight w = H ∧
41    x = loweringWord Ω w}

43theorem cyclicHeightSpace_le_cutoff (Ω : TensorRegister n (Fin d)) (H : ℕ) :
44    cyclicHeightSpace Ω H ≤ cyclicCutoff Ω (H : ℤ) := by
45  apply Submodule.span_le.mpr
46  rintro x ⟨w, hw, rfl⟩
47  exact loweringWord_mem_cyclicCutoff Ω w (by omega)

49theorem cyclicHeightSpace_eq_span_canonical (Ω : TensorRegister n (Fin d)) (H : ℕ) :
50    cyclicHeightSpace Ω H = Submodule.span ℂ
51      {x | ∃ k : HeightOccupation d H, occupationHeight k.val = H ∧
52        x = loweringWord Ω (canonicalWord k.val)} := by
53  apply le_antisymm
54  · apply Submodule.span_le.mpr
55    rintro x ⟨w, hw, rfl⟩
56    apply loweringWord_mem_of_ordered Ω _ w.length H (loweringWeight w)
57      _ w le_rfl hw rfl
58    intro v hv _ hvH _
59    have hk : occupationHeight (fun a => v.count a) = H := by
60      simpa only [occupationHeight, ← loweringHeight_eq_sum_count] using hvH
61    apply Submodule.subset_span
62    exact ⟨⟨fun a => v.count a, hk.le⟩, hk,
63      by rw [canonicalWord_counts_of_ordered v hv]⟩
64  · apply Submodule.span_le.mpr
65    rintro x ⟨k, hk, rfl⟩
66    exact Submodule.subset_span ⟨canonicalWord k.val, (canonicalWord_height _).trans hk, rfl⟩

68/-- The finite canonical occupation spanning family gives an unconditional
69cutoff dimension bound; no asymptotic or positive root-gap premise is needed. -/
70theorem finrank_cyclicCutoff_le_card (Ω : TensorRegister n (Fin d)) (R : ℕ) :
71    Module.finrank ℂ (cyclicCutoff Ω (R : ℤ)) ≤ Fintype.card (HeightOccupation d R) := by
72  rw [cyclicCutoff_eq_span_canonical]
73  exact finrank_range_le_card _

75theorem finrank_cyclicCutoff_le_polynomial (Ω : TensorRegister n (Fin d)) (R : ℕ) :
76    Module.finrank ℂ (cyclicCutoff Ω (R : ℤ)) ≤
77      (R + 1) ^ Fintype.card (PositiveRoot d) :=

80theorem finrank_cyclicHeightSpace_le_polynomial (Ω : TensorRegister n (Fin d)) (H : ℕ) :

86/-- The full cyclic sector is the sum of its exact total-height pieces. -/
87theorem cyclicSector_eq_iSup_cyclicHeightSpace (Ω : TensorRegister n (Fin d)) :
88    cyclicSector Ω = ⨆ H : ℕ, cyclicHeightSpace Ω H := by
89  apply le_antisymm
