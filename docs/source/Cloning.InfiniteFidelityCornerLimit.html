28private theorem projection_star {v : ι → H} (hv : Orthonormal ℂ v) :
29    IsStarProjection (projection v) := by

41theorem norm_corner_eq_mass (v : ι → H) (hv : Orthonormal ℂ v)
42    (A : PositiveTraceClass H) : ‖matrixLift v (matrixOf v A.1.1)‖ = mass v A.1 := by

46/-- Finite matrix fidelity differs from the actual fidelity by a modulus of
47the two discarded trace masses. No diagonal or support hypothesis is used. -/
48theorem abs_rootFidelity_sub_corner_le (v : ι → H) (hv : Orthonormal ℂ v)
49    (A B : PositiveTraceClass H) (hA : ‖A.1‖ = 1) (hB : ‖B.1‖ = 1) :
50    |A.rootFidelity B - MatrixFidelity.fidelity (matrixOf v A.1.1) (matrixOf v B.1.1)| ≤
51      modulus (mass v A.1) (mass v B.1) := by
52  let C : PositiveTraceClass H := ⟨matrixLift v (matrixOf v A.1.1),
53    ofMatrix_nonneg hv (matrixOf_posSemidef v A.2)⟩
54  let D : PositiveTraceClass H := ⟨matrixLift v (matrixOf v B.1.1),
55    ofMatrix_nonneg hv (matrixOf_posSemidef v B.2)⟩
56  have hC : ‖C.1‖ ≤ 1 := by
57    dsimp only [C]
58    rw [corner_eq_sandwich]
59    exact (norm_sandwich_projection_le (projection_star hv) A.1).trans_eq hA
60  have hgA : ‖A.1-C.1‖ ≤ 2 * Real.sqrt (1-mass v A.1) := by
61    have hh := gentle_compression_unit (projection_star hv) A.1 A.2 hA
62    rw [← corner_eq_sandwich, norm_corner_eq_mass v hv A] at hh
63    exact hh
64  have hgB : ‖B.1-D.1‖ ≤ 2 * Real.sqrt (1-mass v B.1) := by
65    have hh := gentle_compression_unit (projection_star hv) B.1 B.2 hB
66    rw [← corner_eq_sandwich, norm_corner_eq_mass v hv B] at hh
67    exact hh
68  have hf := PositiveTraceClass.rootFidelity_continuity A B C D
69  have he : C.rootFidelity D = MatrixFidelity.fidelity (matrixOf v A.1.1) (matrixOf v B.1.1) :=
70    fidelity_ofMatrix hv (matrixOf_posSemidef v A.2) (matrixOf_posSemidef v B.2)
71  rw [he, hB, Real.sqrt_one, mul_one] at hf
72  apply hf.trans
73  unfold modulus
74  exact add_le_add (Real.sqrt_le_sqrt hgA)
75    ((mul_le_mul (Real.sqrt_le_sqrt hgB)
76      (by simpa only [Real.sqrt_one] using Real.sqrt_le_sqrt hC)
77      (Real.sqrt_nonneg _) (Real.sqrt_nonneg _)).trans_eq (mul_one _))

84/-- Varying-Hilbert quantum fidelity convergence from actual finite entries.
85Only limiting corner masses are required to exhaust the two unit states;
86there is no tail or trace-norm convergence premise for the physical states. -/
87theorem tendsto_rootFidelity_of_corner_coefficients
88    (A B : ∀ N, PositiveTraceClass (G N)) (C D : PositiveTraceClass H)
89    (hA : ∀ N, ‖(A N).1‖ = 1) (hB : ∀ N, ‖(B N).1‖ = 1)
90    (hC : ‖C.1‖ = 1) (hD : ‖D.1‖ = 1)
91    (v : ∀ N R, I R → G N) (w : ∀ R, I R → H)
92    (hv : ∀ R, ∀ᶠ N in atTop, Orthonormal ℂ (v N R))
93    (hw : ∀ R, Orthonormal ℂ (w R))
94    (hCA : ∀ R i j, Tendsto (fun N => ⟪v N R i, (A N).1.1 (v N R j)⟫_ℂ)
95      atTop (𝓝 ⟪w R i, C.1.1 (w R j)⟫_ℂ))
96    (hDB : ∀ R i j, Tendsto (fun N => ⟪v N R i, (B N).1.1 (v N R j)⟫_ℂ)
97      atTop (𝓝 ⟪w R i, D.1.1 (w R j)⟫_ℂ))
98    (hCm : Tendsto (fun R => mass (w R) C.1) atTop (𝓝 1))
99    (hDm : Tendsto (fun R => mass (w R) D.1) atTop (𝓝 1)) :
100    Tendsto (fun N => (A N).rootFidelity (B N)) atTop (𝓝 (C.rootFidelity D)) := by
101  have hmod : Tendsto (fun R => modulus (mass (w R) C.1) (mass (w R) D.1)) atTop (𝓝 0) := by
102    simpa only [modulus, sub_self, Real.sqrt_zero, mul_zero, zero_add] using
103      continuous_modulus.continuousAt.tendsto.comp (hCm.prodMk_nhds hDm)
104  apply Metric.tendsto_nhds.mpr
105  intro ε hε
106  obtain ⟨R,hR⟩ := (hmod.eventually (eventually_lt_nhds (show 0 < ε/3 by positivity))).exists
107  have hMA : Tendsto (fun N => matrixOf (v N R) (A N).1.1) atTop
108      (𝓝 (matrixOf (w R) C.1.1)) :=
109    tendsto_pi_nhds.mpr (fun i => tendsto_pi_nhds.mpr (fun j => hCA R i j))
110  have hMB : Tendsto (fun N => matrixOf (v N R) (B N).1.1) atTop
111      (𝓝 (matrixOf (w R) D.1.1)) :=
112    tendsto_pi_nhds.mpr (fun i => tendsto_pi_nhds.mpr (fun j => hDB R i j))
113  have hf := MatrixFidelity.tendsto_fidelity (matrixOf_posSemidef (w R) C.2)
114    (matrixOf_posSemidef (w R) D.2)
115    (fun N => matrixOf_posSemidef (v N R) (A N).2)
116    (fun N => matrixOf_posSemidef (v N R) (B N).2) hMA hMB
117  have hmA : Tendsto (fun N => mass (v N R) (A N).1) atTop (𝓝 (mass (w R) C.1)) := by
118    exact Complex.continuous_re.continuousAt.tendsto.comp
119      (tendsto_finset_sum _ (fun i _ => hCA R i i))
120  have hmB : Tendsto (fun N => mass (v N R) (B N).1) atTop (𝓝 (mass (w R) D.1)) := by
121    exact Complex.continuous_re.continuousAt.tendsto.comp
122      (tendsto_finset_sum _ (fun i _ => hDB R i i))
123  have hbound : Tendsto (fun N => modulus (mass (v N R) (A N).1) (mass (v N R) (B N).1) +
124      |MatrixFidelity.fidelity (matrixOf (v N R) (A N).1.1) (matrixOf (v N R) (B N).1.1) -
125        MatrixFidelity.fidelity (matrixOf (w R) C.1.1) (matrixOf (w R) D.1.1)| +
126      modulus (mass (w R) C.1) (mass (w R) D.1)) atTop
127      (𝓝 (2 * modulus (mass (w R) C.1) (mass (w R) D.1))) := by
128    have hh := continuous_modulus.continuousAt.tendsto.comp (hmA.prodMk_nhds hmB)
129    convert (hh.add (hf.sub_const _).abs).add_const (modulus (mass (w R) C.1) (mass (w R) D.1)) using 1 <;>
130      simp only [sub_self, abs_zero, add_zero, two_mul]
131  filter_upwards [hv R, hbound.eventually (eventually_lt_nhds (show
132      2 * modulus (mass (w R) C.1) (mass (w R) D.1) < ε by linarith))] with N hvN hbN
133  rw [Real.dist_eq]
134  have hsource := abs_rootFidelity_sub_corner_le (v N R) hvN (A N) (B N) (hA N) (hB N)
135  have htarget := abs_rootFidelity_sub_corner_le (w R) (hw R) C D hC hD
136  have ht := abs_sub_le ((A N).rootFidelity (B N))
137    (MatrixFidelity.fidelity (matrixOf (v N R) (A N).1.1) (matrixOf (v N R) (B N).1.1)) (C.rootFidelity D)
138  have ht' := abs_sub_le
139    (MatrixFidelity.fidelity (matrixOf (v N R) (A N).1.1) (matrixOf (v N R) (B N).1.1))
140    (MatrixFidelity.fidelity (matrixOf (w R) C.1.1) (matrixOf (w R) D.1.1)) (C.rootFidelity D)
141  rw [abs_sub_comm (C.rootFidelity D)] at htarget
142  linarith
