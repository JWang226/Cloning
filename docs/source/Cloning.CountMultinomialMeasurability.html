40theorem measurable_density (d N : ℕ) (hN : 0<N) (p : Fin (d+1) → ℝ)

47theorem continuous_density_parameter {Z : Type*} [TopologicalSpace Z]
48    (d N : ℕ) (hN : 0<N) (p : Z → Fin (d+1) → ℝ)
49    (hp : ∀ z i, 0≤p z i) (hs : ∀ z, ∑ i, p z i=1)
50    (hc : ∀ i, Continuous (fun z ↦ p z i)) (x : rootSpace d) :
51    Continuous (fun z ↦ density d N (p z) (hp z) (hs z) x) := by
52  simp_rw [density_eq_word_sum d N hN]
53  apply Continuous.const_mul
54  apply continuous_finset_sum
55  intro w _
56  by_cases hw : sampleLabel d N (flatSpectrum (d+1)) x=shapeLattice d N (countShape w)
57  · simp only [hw, if_true, wordWeight]
58    exact continuous_finset_prod _ (fun i _ ↦ hc (w i))
59  · simp only [hw, if_false]
60    exact continuous_const
