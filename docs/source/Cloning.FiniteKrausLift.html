93/-- Matrix Kraus normalization implies the actual Hilbert-space normalization. -/
94theorem matrixRegister_kraus_complete {ι : Type*} [Fintype ι]
95    (M : ι → Matrix J I ℂ) (hM : ∑ k, (M k).conjTranspose * M k = 1) :
96    RectangularKrausComplete (fun k => matrixRegister (M k)) := by
97  have hop : (∑ k, (matrixRegister (M k)).adjoint.comp (matrixRegister (M k))) =
98      ContinuousLinearMap.id ℂ (Register I) := by
99    simp_rw [← matrixRegister_conjTranspose, ← matrixRegister_mul]
100    rw [← matrixRegister_sum, hM, matrixRegister_one]
101  intro x
102  have hi : ∑ k, ⟪matrixRegister (M k) x, matrixRegister (M k) x⟫_ℂ = ⟪x, x⟫_ℂ := by
103    simp_rw [← ContinuousLinearMap.adjoint_inner_right]
104    rw [← inner_sum]
105    change ⟪x, ∑ k, ((matrixRegister (M k)).adjoint.comp (matrixRegister (M k))) x⟫_ℂ = _
