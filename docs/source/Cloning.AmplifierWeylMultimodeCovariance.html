36theorem channel_weyl_covariant (q : Fin d → ℝ)
37    (hq0 : ∀ i, 0 ≤ q i) (hq1 : ∀ i, q i < 1) (a : Fin d → ℂ) (A : TraceClass (Fock d)) :
38    (channel q hq0 hq1).toLinearMap (displacementTraceMap a A) =
39      displacementTraceMap (fun i => ((((Real.sqrt (1 - q i))⁻¹ : ℝ) : ℂ) * a i))
40        ((channel q hq0 hq1).toLinearMap A) := by
41  let s : Fin d → ℝ := fun i => Real.sqrt (1 - q i)
42  let t : Fin d → ℝ := fun i => Real.sqrt (q i)
43  let r : Fin d → ℝ := fun i => (s i)⁻¹
44  let S : (Fin d → ℂ) → (Fin d → ℂ) := fun z i => (s i : ℂ) * z i
45  let T : (Fin d → ℂ) → (Fin d → ℂ) := fun z i => (t i : ℂ) * z i
46  let R : (Fin d → ℂ) → (Fin d → ℂ) := fun z i => (r i : ℂ) * z i
47  let e : Fin d → ℂ := fun i => -((t i * r i : ℝ) : ℂ) * a i
48  have hs (i : Fin d) : 0 < s i := Real.sqrt_pos.mpr (sub_pos.mpr (hq1 i))
49  have hsr (i : Fin d) : (s i : ℂ) * (r i : ℂ) = 1 := by
50    change (s i : ℂ) * (((s i)⁻¹ : ℝ) : ℂ) = 1
51    rw [← Complex.ofReal_mul, mul_inv_cancel₀ (hs i).ne', Complex.ofReal_one]
52  have hsplit (i : Fin d) : s i + t i ^ 2 * r i = r i := by
53    have hs2 : s i ^ 2 = 1 - q i := Real.sq_sqrt (sub_nonneg.mpr (hq1 i).le)
54    have ht2 : t i ^ 2 = q i := Real.sq_sqrt (hq0 i)
55    dsimp only [r]
56    apply mul_right_cancel₀ (hs i).ne'
57    field_simp [(hs i).ne']
58    nlinarith [hs2, ht2]
59  apply Subtype.ext
60  apply continuousLinearMap_ext_coherent
61  intro w
62  apply sub_eq_zero.mp
63  apply coherentVector_total
64  intro v
65  change ⟪coherentVector v,
66    ((channel q hq0 hq1).toLinearMap (displacementTraceMap a A)).1 (coherentVector w) -
67      (displacementTraceMap (R a) ((channel q hq0 hq1).toLinearMap A)).1 (coherentVector w)⟫_ℂ = 0
68  rw [inner_sub_right, sub_eq_zero]
69  rw [channel_coherent_matrixCoefficient, displacement_coherent_matrixCoefficient,
70    displacement_coherent_matrixCoefficient, channel_coherent_matrixCoefficient]
71  change (((∏ i, (1 - q i) : ℝ) : ℂ) * ⟪coherentVector (T v), coherentVector (T w)⟫_ℂ) *
72      (star (displacementPhase (-a) (S v)) * displacementPhase (-a) (S w) *
73        ⟪coherentVector (-a + S v), A.1 (coherentVector (-a + S w))⟫_ℂ) =
74    star (displacementPhase (-(R a)) v) * displacementPhase (-(R a)) w *
75      ((((∏ i, (1 - q i) : ℝ) : ℂ) *
76        ⟪coherentVector (T (-(R a) + v)), coherentVector (T (-(R a) + w))⟫_ℂ) *
77        ⟪coherentVector (S (-(R a) + v)), A.1 (coherentVector (S (-(R a) + w)))⟫_ℂ)
78  have harg (z : Fin d → ℂ) : S (-(R a) + z) = -a + S z := by
79    funext i
80    change (s i : ℂ) * (-((r i : ℂ) * a i) + z i) = -a i + (s i : ℂ) * z i
81    calc
82      _ = -((s i : ℂ) * (r i : ℂ)) * a i + (s i : ℂ) * z i := by ring
83      _ = _ := by rw [hsr]; ring
84  rw [harg, harg]
85  have hnoise := inner_displacedCoherent e (T v) (T w)
86  simp only [displacedCoherent, inner_smul_left, inner_smul_right] at hnoise
87  have hargt (z : Fin d → ℂ) : e + T z = T (-(R a) + z) := by
88    funext i
89    dsimp only [e, T, R, Pi.add_apply, Pi.neg_apply]
90    push_cast
91    ring
92  rw [hargt, hargt] at hnoise
93  have hp (z : Fin d → ℂ) : displacementPhase (-a) (S z) * displacementPhase e (T z) =
94      displacementPhase (-(R a)) z := amplifier_phase_split hsplit a z
95  rw [← hnoise, ← hp v, ← hp w]
96  simp only [Complex.star_def, map_mul]
97  ring

100theorem channel_weyl_covariant_gain (g : ℝ) (hg : 1 < g)
101    (a : Fin d → ℂ) (A : TraceClass (Fock d)) :
102    (channel (fun _ => 1 - 1 / g) (fun _ => Cloning.BosonicAmplifier.gainNoise_nonneg hg)
103      (fun _ => Cloning.BosonicAmplifier.gainNoise_lt_one hg)).toLinearMap
104        (displacementTraceMap a A) =
105    displacementTraceMap ((Real.sqrt g) • a)
106      ((channel (fun _ => 1 - 1 / g) (fun _ => Cloning.BosonicAmplifier.gainNoise_nonneg hg)
107        (fun _ => Cloning.BosonicAmplifier.gainNoise_lt_one hg)).toLinearMap A) := by
108  have he : (Real.sqrt (1 - (1 - 1 / g)))⁻¹ = Real.sqrt g := by
109    rw [show 1 - (1 - 1 / g) = g⁻¹ by ring, Real.sqrt_inv, inv_inv]
110  simpa only [he, Pi.smul_apply, Complex.real_smul] using channel_weyl_covariant
111    (fun _ : Fin d => 1 - 1 / g) (fun _ => Cloning.BosonicAmplifier.gainNoise_nonneg hg)
