22theorem partitionTransitionChannel_add (mu tau : Fin d → ℕ)
23    (hmu : Antitone mu) (htau : Antitone tau) :
24    partitionTransitionChannel mu (fun i ↦ mu i + tau i) hmu (sumPartition_antitone mu tau hmu htau) =
25      physicalCartanChannel mu tau hmu htau := by
26  have hc : PartitionCompatible mu (fun i ↦ mu i + tau i) :=
27    ⟨fun i ↦ Nat.le_add_right _ _, by simpa only [Nat.add_sub_cancel_left] using htau⟩
28  unfold partitionTransitionChannel
29  rw [dif_pos hc]
30  have hd : (fun i ↦ mu i + tau i - mu i) = tau := funext (fun i ↦ Nat.add_sub_cancel_left _ _)
31  let F (v : {v : Fin d → ℕ // Antitone v}) : Type :=
32    QuantumChannel (cyclicSector (partitionHighestTensor mu hmu))
33      (cyclicSector (partitionHighestTensor (fun i ↦ mu i + v.val i)
34        (sumPartition_antitone mu v.val hmu v.property)))

38      ⟨tau, htau⟩ := Subtype.ext hd

41theorem partitionTransitionChannel_covariant (mu nu : Fin d → ℕ)
42    (hmu : Antitone mu) (hnu : Antitone nu) (hc : PartitionCompatible mu nu)
43    (U : Matrix (Fin d) (Fin d) ℂ) (hU : Uᴴ * U = 1)
44    (A : TraceClass (cyclicSector (partitionHighestTensor mu hmu))) :
45    (partitionTransitionChannel mu nu hmu hnu).toLinearMap
46      (conjugationLinearMap (partitionTensorAction mu hmu U) A) =
47      conjugationLinearMap (partitionTensorAction nu hnu U)
48        ((partitionTransitionChannel mu nu hmu hnu).toLinearMap A) := by
49  have he : (fun i ↦ mu i + (nu i - mu i)) = nu := funext (fun i ↦ Nat.add_sub_of_le (hc.1 i))
50  have hh := physicalCartanChannel_covariant mu (fun i ↦ nu i - mu i) hmu hc.2 U hU A
51  rw [← partitionTransitionChannel_add] at hh
52  let P (v : {v : Fin d → ℕ // Antitone v}) : Prop :=
53    (partitionTransitionChannel mu v.val hmu v.property).toLinearMap
54      (conjugationLinearMap (partitionTensorAction mu hmu U) A) =
55      conjugationLinearMap (partitionTensorAction v.val v.property U)
56        ((partitionTransitionChannel mu v.val hmu v.property).toLinearMap A)
57  have hab : (⟨fun i ↦ mu i + (nu i - mu i), sumPartition_antitone mu _ hmu hc.2⟩ :
58    {v : Fin d → ℕ // Antitone v}) = ⟨nu, hnu⟩ := Subtype.ext he
59  exact Eq.mp (congrArg P hab) hh
