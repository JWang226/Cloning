12/-- Coordinate inclusion of the supported physical one-particle space. -/
13def coordinateInclusion (r k : ℕ) : Matrix (Fin (r+k)) (Fin r) ℂ :=
14  fun i a => if i = Fin.castAdd k a then 1 else 0

16theorem coordinateInclusion_isometry (r k : ℕ) :
17    (coordinateInclusion r k)ᴴ * coordinateInclusion r k = 1 := by
18  ext a b
19  simp [coordinateInclusion, Matrix.mul_apply, Matrix.conjTranspose_apply,
20    Matrix.one_apply, Fin.castAdd_inj, eq_comm]

22theorem coordinateInclusion_range (r k : ℕ) :
23    coordinateInclusion r k * (coordinateInclusion r k)ᴴ = coordinateProjector r k := by
24  have hcross (a : Fin r) (b : Fin k) : Fin.castAdd k a ≠ Fin.natAdd r b := by
25    intro h
26    have hv := congrArg Fin.val h
27    simp only [Fin.val_castAdd, Fin.val_natAdd] at hv
28    omega
29  ext i j
30  induction i using Fin.addCases <;> induction j using Fin.addCases <;>
31    simp [coordinateInclusion, coordinateProjector, Matrix.mul_apply,
32      Matrix.conjTranspose_apply, Matrix.diagonal_apply, hcross, eq_comm]

34/-- Every rank-r orthogonal projector has an actual r-column isometric
35factorization. This is stronger than a supplied support witness. -/
36theorem exists_isometric_factor (r k : ℕ) (P : Projector r k) :
37    ∃ J : Matrix (Fin (r+k)) (Fin r) ℂ, Jᴴ*J=1 ∧ P.val=J*Jᴴ := by
38  obtain ⟨U,hU⟩ := orbitProjector_surjective r k P
39  let V : Matrix (Fin (r+k)) (Fin (r+k)) ℂ := U
40  have hV : Vᴴ*V=1 := Unitary.star_mul_self_of_mem U.property
41  refine ⟨V*coordinateInclusion r k,?_,?_⟩
42  · rw [Matrix.conjTranspose_mul]
43    calc
44      _ = (coordinateInclusion r k)ᴴ * (Vᴴ*V) * coordinateInclusion r k := by
45        simp only [Matrix.mul_assoc]
46      _ = 1 := by rw [hV, Matrix.mul_one, coordinateInclusion_isometry]
47  · have he : P.val=V*coordinateProjector r k*Vᴴ := congrArg Subtype.val hU.symm
48    rw [he, Matrix.conjTranspose_mul, ← coordinateInclusion_range]
49    simp only [Matrix.mul_assoc]

60theorem exists_embedded_flatState (d k : ℕ) (P : Projector (d+1) k) :
61    ∃ J : Matrix (Fin (d+1+k)) (Fin (d+1)) ℂ, ∃ hJ : Jᴴ*J=1,
62      state (by omega : 0<d+1) P = embeddedState J hJ (flatInternalState d) := by
63  obtain ⟨J,hJ,hP⟩ := exists_isometric_factor (d+1) k P
64  refine ⟨J,hJ,?_⟩
65  have hm : (state (by omega : 0<d+1) P).matrix =
66      (embeddedState J hJ (flatInternalState d)).matrix := by
67    change (1/((d+1 : ℕ):ℝ)) • P.val = J*(flatInternalState d).matrix*Jᴴ
68    rw [flatInternalState_matrix, Matrix.mul_smul, Matrix.mul_one, Matrix.smul_mul, hP]

71  cases h₂ : embeddedState J hJ (flatInternalState d)
72  simp only [h₁,h₂,Cloning.MatrixFidelity.State.mk.injEq] at hm ⊢
