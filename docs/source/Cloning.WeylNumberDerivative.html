14/-- The literal finite ladder sum on a number state. -/
15def numberLadder (a : Fin d → ℂ) (k : Fin d → ℕ) : Fock d :=
16  ∑ i, ((a i * (Real.sqrt (k i+1 : ℝ) : ℂ)) •
17      numberBasis d (Function.update k i (k i+1)) -
18    (starRingEnd ℂ (a i) * (Real.sqrt (k i : ℝ) : ℂ)) •
19      numberBasis d (Function.update k i (k i-1)))

23    (∏ j, numberKernel (a j) (z j) (Function.update k i l j)) =
24      (∏ j ∈ Finset.univ.erase i, numberKernel (a j) (z j) (k j)) *
25        numberKernel (a i) (z i) l := by
26  rw [← Finset.prod_erase_mul _ _ (Finset.mem_univ i)]
