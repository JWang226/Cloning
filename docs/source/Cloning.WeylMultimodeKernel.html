13/-- The joint Weyl phase is the product of the individual mode phases. -/
14def displacementPhase {d : ℕ} (a z : Fin d → ℂ) : ℂ :=
15  ∏ i, ComplexCoherent.displacementPhase (a i) (z i)
16 
17/-- The coherent action prescribed by simultaneous displacement of all modes. -/
18def displacedCoherent {d : ℕ} (a z : Fin d → ℂ) : Fock d :=
19  displacementPhase a z • coherentVector (a + z)

21@[simp] theorem displacementPhase_norm {d : ℕ} (a z : Fin d → ℂ) :

34/-- The full complex coherent kernel is preserved jointly in all modes. -/
35theorem inner_displacedCoherent {d : ℕ} (a z w : Fin d → ℂ) :
36    ⟪displacedCoherent a z, displacedCoherent a w⟫_ℂ =
37      ⟪coherentVector z, coherentVector w⟫_ℂ := by
38  have hsingle (i : Fin d) := ComplexCoherent.inner_displacedCoherent (a i) (z i) (w i)
39  simp only [ComplexCoherent.displacedCoherent, inner_smul_left, inner_smul_right] at hsingle
40  simp only [displacedCoherent, inner_smul_left, inner_smul_right, displacementPhase,
41    coherentVector, inner_tensorVector, map_prod]
42  rw [← Finset.prod_mul_distrib, ← Finset.prod_mul_distrib]
43  apply Finset.prod_congr rfl
44  intro i _
45  exact hsingle i

47/-- The finite-mode Weyl cocycle is inherited from the one-mode cocycles. -/
48theorem displacementPhase_cocycle {d : ℕ} (a b z : Fin d → ℂ) :
49    displacementPhase b z * displacementPhase a (b + z) =
50      displacementPhase a b * displacementPhase (a + b) z := by
51  simp only [displacementPhase, Pi.add_apply, ← Finset.prod_mul_distrib,

54/-- Finite multimode coherent superpositions. -/
55def coherentCombination {d : ℕ} : ((Fin d → ℂ) →₀ ℂ) →ₗ[ℂ] Fock d :=
56  Finsupp.linearCombination ℂ coherentVector

58def displacedCombination {d : ℕ} (a : Fin d → ℂ) :

62/-- The prescribed action preserves inner products of arbitrary finite
63superpositions, so it respects all linear relations between coherent vectors. -/
64theorem inner_displacedCombination {d : ℕ} (a : Fin d → ℂ)
65    (c e : (Fin d → ℂ) →₀ ℂ) :
66    ⟪displacedCombination a c, displacedCombination a e⟫_ℂ =
67      ⟪coherentCombination c, coherentCombination e⟫_ℂ := by
68  simp only [displacedCombination, coherentCombination, Finsupp.linearCombination_apply,
69    Finsupp.sum, inner_sum, sum_inner, inner_smul_left, inner_smul_right,
70    inner_displacedCoherent]

72theorem norm_displacedCombination {d : ℕ} (a : Fin d → ℂ)
73    (c : (Fin d → ℂ) →₀ ℂ) :
74    ‖displacedCombination a c‖ = ‖coherentCombination c‖ := by
75  have h := congrArg (fun z : ℂ => z.re) (inner_displacedCombination a c c)
76  change (RCLike.re : ℂ → ℝ) _ = (RCLike.re : ℂ → ℝ) _ at h
77  rw [← InnerProductSpace.norm_sq_eq_re_inner (𝕜 := ℂ),
78    ← InnerProductSpace.norm_sq_eq_re_inner (𝕜 := ℂ)] at h
79  nlinarith [norm_nonneg (displacedCombination a c), norm_nonneg (coherentCombination c)]
