20def copyToFock (n d R : ℕ) (i : SchurCopy n d) :
21    QuantumChannel ((recursivePhysicalDecomposition n d).get i).CanonicalSector (RootFock d) :=
22  let H := (recursivePhysicalDecomposition n d).get i
23  sectorToFockTotal (partitionHighestTensor H.weight H.weight_antitone) H.weight R

25/-- Reverse transport lands in the actual physical copy through its proved

27def fockToCopy (n d R : ℕ) (i : SchurCopy n d) :
28    QuantumChannel (RootFock d) (TensorRegister n (Fin d)) :=
29  let H := (recursivePhysicalDecomposition n d).get i

31    (fockToSectorTotal (partitionHighestTensor H.weight H.weight_antitone) H.weight R
32      (partitionHighestTensor_norm H.weight H.weight_antitone))

36/-- Measure the actual complete physical sector sum, use its sector channel,
37and emit the prescribed probability density for the classical label. All
38quantum and classical trace normalization is derived. -/
39def schurForward (n d R : ℕ) (g : SchurCopy n d → X → ℝ)
40    (hg : ∀ i, Integrable (g i) ν) (hg0 : ∀ i x, 0 ≤ g i x)
41    (hprob : ∀ i, ∫ x, g i x ∂ν = 1) :
42    QuantumToHybrid (TensorRegister n (Fin d)) (RootFock d) ν :=
43  QuantumToHybrid.ofHilbertSum
44    (fun i : SchurCopy n d => ((recursivePhysicalDecomposition n d).get i).canonicalEmbedding)
45    (canonicalSectorHilbertSum (recursivePhysicalDecomposition n d)
46      (recursivePhysicalDecomposition_is_decomposition n d).1
47      (recursivePhysicalDecomposition_is_decomposition n d).2)
48    (copyToFock n d R) g hg hg0 hprob

50theorem schurForward_apply (n d R : ℕ) (g : SchurCopy n d → X → ℝ)
51    (hg : ∀ i, Integrable (g i) ν) (hg0 : ∀ i x, 0 ≤ g i x)
52    (hprob : ∀ i, ∫ x, g i x ∂ν = 1) (A : TraceClass (TensorRegister n (Fin d))) :
53    (schurForward n d R g hg hg0 hprob).map A =
54      ∑ i : SchurCopy n d, prepareL1 (g i) (hg i)
55        ((copyToFock n d R i).toLinearMap
56          (conjugationLinearMap
57            ((recursivePhysicalDecomposition n d).get i).canonicalEmbedding.toContinuousLinearMap.adjoint A)) :=
58  QuantumToHybrid.ofHilbertSum_apply _ _ _ _ _ _ _ A

60/-- The additional outcome is an explicit physical replacement state for

62def copyReverseWithFallback (n d R : ℕ) (ρ : DensityState (TensorRegister n (Fin d))) :
63    Option (SchurCopy n d) → QuantumChannel (RootFock d) (TensorRegister n (Fin d))
64  | none => QuantumChannel.ofContraction 0 (by intro x; simp) ρ
65  | some i => fockToCopy n d R i

67/-- Any proved classical cell selector now yields a genuine reverse mixed

69def schurReverse (n d R : ℕ) (ρ : DensityState (TensorRegister n (Fin d)))
70    (χ : Option (SchurCopy n d) → X → ℝ)
71    (hχ : ∀ i, AEStronglyMeasurable (χ i) ν)
72    (hχ0 : ∀ i x, 0 ≤ χ i x) (hχsum : ∀ x, ∑ i, χ i x = 1) :
73    HybridToQuantum (RootFock d) (TensorRegister n (Fin d)) ν :=
74  HybridToQuantum.finiteSelector χ hχ hχ0 hχsum (copyReverseWithFallback n d R ρ)
