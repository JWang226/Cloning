13/-- The largest and smallest eigenvalues form one actual spectral pair. -/
14def boundaryPair (d : ℕ) (hd : 2≤d) : PairIndex d :=
15  ⟨(⟨0,by omega⟩,⟨d-1,by omega⟩),by change 0<d-1; omega⟩

17theorem one_div_dimension_le_largest {d : ℕ} (hd : 0<d) (p : SimpleSpectrum d) :
18    1/(d:ℝ)≤p.eigenvalue ⟨0,hd⟩ := by
19  have hs : (1:ℝ)≤(d:ℝ)*p.eigenvalue ⟨0,hd⟩ := by
20    calc
21      1=∑i,p.eigenvalue i := p.normalized.symm
22      _≤∑_i : Fin d,p.eigenvalue ⟨0,hd⟩ := Finset.sum_le_sum (fun (i : Fin d) _=>
23        p.strictAnti.antitone (show (⟨0,hd⟩ : Fin d) ≤ i from Nat.zero_le i.val))
24      _=(d:ℝ)*p.eigenvalue ⟨0,hd⟩ := by simp
25  exact (div_le_iff₀ (Nat.cast_pos.mpr hd)).mpr (by nlinarith)

27theorem boundaryRatio_le_dimension_mul_smallest {d : ℕ} (hd : 2≤d)
28    (p : SimpleSpectrum d) :
29    p.ratio (boundaryPair d hd)≤(d:ℝ)*p.eigenvalue ⟨d-1,by omega⟩ := by
30  have hl := one_div_dimension_le_largest (by omega : 0<d) p
31  have hdim : (0:ℝ)<d := Nat.cast_pos.mpr (by omega)
32  have hh : (1:ℝ)≤p.eigenvalue ⟨0,by omega⟩*(d:ℝ) :=
33    (div_le_iff₀ hdim).mp hl
34  unfold SimpleSpectrum.ratio boundaryPair
35  apply (div_le_iff₀ (p.positive _)).mpr
36  have hm := mul_le_mul_of_nonneg_left hh (p.positive ⟨d-1,by omega⟩).le
37  nlinarith

39theorem pairTerm_le_knownCoefficient {d : ℕ} (p : SimpleSpectrum d) (ij : PairIndex d) :
