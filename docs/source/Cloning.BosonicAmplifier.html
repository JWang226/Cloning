4/-!
5# One-mode quantum-limited amplifier coefficients and Stinespring isometry
6 
7For `0 ≤ q < 1`, the gain is `1/(1-q)`. An input number vector `|n⟩`
8is mapped to a coherent superposition of the pairs `|n+k,k⟩` with squared
9amplitudes `choose(n+k,n) (1-q)^(n+1) q^k`. The negative-binomial
10normalization and the orthogonality of these columns are proved here.
11-/

30/-- Squared amplifier amplitude at input number `n` and created-pair number `k`. -/
31def weight (q : ℝ) (n k : ℕ) : ℝ :=
32  ((n + k).choose n : ℝ) * (1 - q) ^ (n + 1) * q ^ k
33 
34theorem weight_nonneg {q : ℝ} (hq0 : 0 ≤ q) (hq1 : q < 1) (n k : ℕ) :
35    0 ≤ weight q n k := by
36  have hq : 0 ≤ 1 - q := sub_nonneg.mpr hq1.le
37  unfold weight
38  positivity

40/-- Exact negative-binomial normalization, including zero gain-noise. -/
41theorem weight_hasSum {q : ℝ} (hq0 : 0 ≤ q) (hq1 : q < 1) (n : ℕ) :
42    HasSum (weight q n) 1 := by
43  have hq : ‖q‖ < 1 := by simpa only [Real.norm_eq_abs, abs_of_nonneg hq0] using hq1
44  have hs := (hasSum_choose_mul_geometric_of_norm_lt_one n hq).mul_left
45    ((1 - q) ^ (n + 1))
46  have hn : (1 - q) ^ (n + 1) ≠ 0 := pow_ne_zero _ (by linarith)
47  convert hs using 1
48  · ext k
49    simp only [weight, Nat.add_comm]
50    ring
51  · field_simp

53theorem weight_tsum {q : ℝ} (hq0 : 0 ≤ q) (hq1 : q < 1) (n : ℕ) :
54    ∑' k, weight q n k = 1 := (weight_hasSum hq0 hq1 n).tsum_eq

56/-- The two-mode joint law of one amplifier column. -/
57def columnWeight (q : ℝ) (n : ℕ) (p : ℕ × ℕ) : ℝ :=
58  if p.1 = n + p.2 then weight q n p.2 else 0

60theorem columnWeight_nonneg {q : ℝ} (hq0 : 0 ≤ q) (hq1 : q < 1)
61    (n : ℕ) (p : ℕ × ℕ) : 0 ≤ columnWeight q n p := by
62  unfold columnWeight
63  split_ifs
64  · exact weight_nonneg hq0 hq1 _ _
65  · exact le_rfl

67theorem columnWeight_hasSum {q : ℝ} (hq0 : 0 ≤ q) (hq1 : q < 1) (n : ℕ) :
68    HasSum (columnWeight q n) 1 := by
69  let e : ℕ → ℕ × ℕ := fun k => (n + k, k)
70  have he : Function.Injective e := fun i j hij => congrArg Prod.snd hij
71  have hzero : ∀ p ∉ Set.range e, columnWeight q n p = 0 := by
72    intro p hp
73    unfold columnWeight
74    split_ifs with h
75    · exact False.elim (hp ⟨p.2, Prod.ext h.symm rfl⟩)
76    · rfl
77  apply (he.hasSum_iff hzero).mp
78  simpa only [Function.comp_def, e, columnWeight, if_pos rfl] using weight_hasSum hq0 hq1 n

80/-- The actual two-mode amplifier output vector for a number input. -/
81def column (q : ℝ) (hq0 : 0 ≤ q) (hq1 : q < 1) (n : ℕ) : TwoModeFock :=
82  Cloning.CoherentCoefficients.amplitudeVector (columnWeight q n)

120/-- A fixed environment-number slice is square summable. -/
121def sliceVector (k : ℕ) (x : TwoModeFock) : Fock := by
122  refine ⟨fun m => x (m, k), memℓp_gen ?_⟩
123  exact ((lp.memℓp x).summable (by norm_num)).comp_injective
124    (fun i j h => congrArg Prod.fst h)
125 
126theorem sliceVector_norm_le (k : ℕ) (x : TwoModeFock) : ‖sliceVector k x‖ ≤ ‖x‖ := by
127  apply lp.norm_le_of_tsum_le (by norm_num) (norm_nonneg x)
128  rw [lp.norm_rpow_eq_tsum (by norm_num) x]
129  exact Summable.tsum_le_tsum_of_inj (fun m : ℕ => (m, k))
130    (fun i j h => congrArg Prod.fst h)
131    (fun p _ => Real.rpow_nonneg (norm_nonneg _) _)
132    (fun _ => le_rfl)
133    ((lp.memℓp (sliceVector k x)).summable (by norm_num))
134    ((lp.memℓp x).summable (by norm_num))

136/-- The bounded slice map extracts the `k`th environment coefficient. -/
137def slice (k : ℕ) : TwoModeFock →L[ℂ] Fock :=
138  LinearMap.mkContinuous
139    { toFun := sliceVector k
140      map_add' := by intro x y; ext m; rfl
141      map_smul' := by intro c x; ext m; rfl }
142    1 (fun x => by simpa only [one_mul] using sliceVector_norm_le k x)

147/-- Summing the slice probabilities is exactly the two-mode Hilbert norm. -/
148theorem slice_norm_sq_hasSum (x : TwoModeFock) :
149    HasSum (fun k => ‖slice k x‖ ^ 2) (‖x‖ ^ 2) := by
150  have hfull : HasSum (fun p : ℕ × ℕ => ‖x p‖ ^ (2 : ℕ)) (‖x‖ ^ 2) := by
151    simpa only [ENNReal.toReal_ofNat, Real.rpow_two] using
152      lp.hasSum_norm (show 0 < (2 : ENNReal).toReal by norm_num) x
153  have hswap : HasSum (fun p : ℕ × ℕ => ‖x (p.2, p.1)‖ ^ (2 : ℕ)) (‖x‖ ^ 2) :=
154    (Equiv.prodComm ℕ ℕ).hasSum_iff.mpr hfull
155  apply hswap.prod_fiberwise
156  intro k
157  simpa only [ENNReal.toReal_ofNat, Real.rpow_two, slice_apply] using
158    lp.hasSum_norm (show 0 < (2 : ENNReal).toReal by norm_num) (slice k x)

160/-- Genuine amplifier Kraus operators, obtained by slicing the isometry. -/
161def kraus (q : ℝ) (hq0 : 0 ≤ q) (hq1 : q < 1) (k : ℕ) : Fock →L[ℂ] Fock :=
162  (slice k).comp (isometry q hq0 hq1).toContinuousLinearMap

164/-- Exact Kraus normalization on every vector, including coherent superpositions. -/
