17/-- Equality with the literal seeded channel follows from the actual
18characteristic law, not from an assumed idler representation. -/
19theorem thermalOutput_exists_seeded_idler
20    (Λ : QuantumChannel (Fock d) (Fock d)) (γ : ℝ) (hγ : 1<γ)
21    (hΛ : ∀ a A, Λ.toLinearMap (displacementTraceMap a A)=
22      displacementTraceMap (Real.sqrt γ • a) (Λ.toLinearMap A))
23    (q : Fin d → ℝ) (hq0 : ∀ i,0≤q i) (hq1 : ∀ i,q i<1) :
24    ∃ σ : TraceClass (Fock d), 0≤σ.1 ∧ traceCLM σ=1 ∧
25      Λ.toLinearMap (vectorMixture (numberBasis d) (productGeometric q))=
26        (MultimodeIdler.channel (fun i => Thermal.amplified γ (q i))
27          (fun i => Thermal.amplified_nonneg hγ (hq0 i) (hq1 i))
28          (fun i => Thermal.amplified_lt_one (lt_trans zero_lt_one hγ) (hq1 i))).toLinearMap σ := by
29  obtain ⟨σ,hσ,ht,hc⟩ := thermalOutput_exists_seeded_characteristic Λ γ hγ hΛ q hq0 hq1
30  refine ⟨σ,hσ,ht,?_⟩
31  apply characteristic_injective
32  funext a
33  exact (hc a).trans (MultimodeIdler.channel_characteristic (fun i => Thermal.amplified γ (q i))
34    (fun i => amplified_pos hγ (hq0 i) (hq1 i))
35    (fun i => Thermal.amplified_lt_one (lt_trans zero_lt_one hγ) (hq1 i)) σ a).symm

37/-- Every bounded observable with a nonnegative antitone product number
38spectrum satisfies the sharp amplified-thermal comparison. -/
39theorem quantumChannel_moment_le_of_diagonal
40    (Λ : QuantumChannel (Fock d) (Fock d)) (γ : ℝ) (hγ : 1<γ)
41    (hΛ : ∀ a A, Λ.toLinearMap (displacementTraceMap a A)=
42      displacementTraceMap (Real.sqrt γ • a) (Λ.toLinearMap A))
43    (q : Fin d → ℝ) (hq0 : ∀ i,0≤q i) (hq1 : ∀ i,q i<1)
44    (w : Fin d → ℕ → ℝ) (hw0 : ∀ i k,0≤w i k) (hw : ∀ i,Antitone (w i))
45    (W : Fock d →L[ℂ] Fock d)
46    (hW : ∀ k, W (numberBasis d k)=((∏ i,w i (k i) : ℝ) : ℂ) • numberBasis d k) :
47    (tracePairing (Λ.toLinearMap (vectorMixture (numberBasis d) (productGeometric q))) W).re≤
48      (tracePairing (vectorMixture (numberBasis d)
49        (productGeometric (fun i => Thermal.amplified γ (q i)))) W).re := by
50  obtain ⟨σ,hσ,ht,he⟩ := thermalOutput_exists_seeded_idler Λ γ hγ hΛ q hq0 hq1
51  rw [he]
52  have ht' : (trace σ.1 σ.2).re=1 := congrArg Complex.re ht
53  simpa only [ht',one_mul] using MultimodeLeastNoise.moment_le_of_diagonal σ hσ
54    (fun i => Thermal.amplified γ (q i))
55    (fun i => Thermal.amplified_nonneg hγ (hq0 i) (hq1 i))
56    (fun i => Thermal.amplified_lt_one (lt_trans zero_lt_one hγ) (hq1 i)) w hw0 hw W hW
