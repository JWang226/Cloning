16theorem isometry_comp_adjoint_eq_projection (V : H →ₗᵢ[ℂ] K)
17    (S : Submodule ℂ K) [S.HasOrthogonalProjection]
18    (hS : V.toLinearMap.range=S) :
19    V.toContinuousLinearMap.comp V.toContinuousLinearMap.adjoint=S.starProjection := by
20  apply ContinuousLinearMap.ext
21  intro y
22  symm
23  apply Submodule.eq_starProjection_of_mem_of_inner_eq_zero

26  · intro w hw
27    rw [← hS] at hw

30    change ⟪y,V x⟫_ℂ-⟪V (V.toContinuousLinearMap.adjoint y),V x⟫_ℂ=0

32    exact sub_self _
