43/-- The exact mass identity for the Jordan decomposition in trace norm. -/
44lemma norm_positivePart_add_norm_negativePart (A : TraceClass H)
45    (hA : IsSelfAdjoint A.1) :
46    ‖positivePart A hA‖ + ‖negativePart A hA‖ = ‖A‖ := by
47  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
48  have hp := summable_inner_abs_of_hilbertBasis (positivePart A hA).2 b
49  have hn := summable_inner_abs_of_hilbertBasis (negativePart A hA).2 b
50  simp only [norm_eq_traceNorm]
51  rw [traceNorm_eq_of_hilbertBasis _ b, traceNorm_eq_of_hilbertBasis _ b,
52    traceNorm_eq_of_hilbertBasis _ b]
53  rw [← hp.tsum_add hn]
54  apply tsum_congr
55  intro i
56  change (⟪b i, CFC.abs A.1⁺ (b i)⟫_ℂ).re +
57    (⟪b i, CFC.abs A.1⁻ (b i)⟫_ℂ).re = _
58  rw [CFC.abs_of_nonneg _ (CFC.posPart_nonneg _),
59    CFC.abs_of_nonneg _ (CFC.negPart_nonneg _), ← Complex.add_re,
60    ← inner_add_right, ← ContinuousLinearMap.add_apply,
61    CFC.posPart_add_negPart A.1 hA]

74/-- The real self-adjoint component of a trace-class operator. -/
75def realComponent (A : TraceClass H) : TraceClass H :=
76  (1 / 2 : ℂ) • (A + adjoint A)

78/-- The imaginary self-adjoint component of a trace-class operator. -/

82lemma realComponent_isSelfAdjoint (A : TraceClass H) :
83    IsSelfAdjoint (realComponent A).1 := by
84  change star ((1 / 2 : ℂ) • (A.1 + star A.1)) = (1 / 2 : ℂ) • (A.1 + star A.1)
85  simp [star_smul, star_add, add_comm]

87lemma imaginaryComponent_isSelfAdjoint (A : TraceClass H) :
88    IsSelfAdjoint (imaginaryComponent A).1 := by
89  change star ((-Complex.I / 2 : ℂ) • (A.1 - star A.1)) =
90    (-Complex.I / 2 : ℂ) • (A.1 - star A.1)
91  simp only [star_smul, star_sub, star_star]
92  rw [show star (-Complex.I / 2 : ℂ) = Complex.I / 2 by simp]
93  rw [show star A.1 - A.1 = -(A.1 - star A.1) by abel, smul_neg]
94  rw [neg_div, neg_smul]

96lemma realComponent_add_I_smul_imaginaryComponent (A : TraceClass H) :
97    realComponent A + Complex.I • imaginaryComponent A = A := by
98  simp only [realComponent, imaginaryComponent, smul_smul]
99  rw [show Complex.I * (-Complex.I / 2) = (1 / 2 : ℂ) by
100    rw [← mul_div_assoc, mul_neg, Complex.I_mul_I, neg_neg]]
101  rw [← smul_add]

104lemma norm_realComponent_le (A : TraceClass H) : ‖realComponent A‖ ≤ ‖A‖ := by
105  calc
106    ‖realComponent A‖ = (1 / 2 : ℝ) * ‖A + adjoint A‖ := by
107      rw [realComponent, norm_smul]
108      norm_num
109    _ ≤ (1 / 2 : ℝ) * (‖A‖ + ‖adjoint A‖) :=
110      mul_le_mul_of_nonneg_left (norm_add_le _ _) (by norm_num)
111    _ = ‖A‖ := by rw [norm_adjoint]; ring
112 
113lemma norm_imaginaryComponent_le (A : TraceClass H) : ‖imaginaryComponent A‖ ≤ ‖A‖ := by
114  calc
115    ‖imaginaryComponent A‖ = (1 / 2 : ℝ) * ‖A - adjoint A‖ := by
116      rw [imaginaryComponent, norm_smul]
117      norm_num

124/-- A positive trace-preserving complex-linear map on the trace-class Banach spaces.
125No contractivity or continuity is assumed in this definition. -/
126structure PositiveTracePreservingMap
127    (H K : Type*) [NormedAddCommGroup H] [InnerProductSpace ℂ H] [CompleteSpace H]
128    [NormedAddCommGroup K] [InnerProductSpace ℂ K] [CompleteSpace K] where
129  toLinearMap : TraceClass H →ₗ[ℂ] TraceClass K
130  map_nonneg : ∀ A, 0 ≤ A.1 → 0 ≤ (toLinearMap A).1
131  trace_preserving : ∀ A,
132    trace (toLinearMap A).1 (toLinearMap A).2 = trace A.1 A.2

142/-- A positive input retains its exact trace norm under a positive TP map. -/
143lemma norm_map_of_nonneg (Φ : PositiveTracePreservingMap H K)
144    (A : TraceClass H) (hA : 0 ≤ A.1) : ‖Φ A‖ = ‖A‖ := by
145  rw [TraceClass.norm_eq_trace_re_of_nonneg _ (Φ.map_nonneg A hA),
146    Φ.trace_preserving, TraceClass.norm_eq_trace_re_of_nonneg _ hA]

148/-- Trace-norm contraction on every self-adjoint trace-class input. -/
149theorem norm_map_le_of_isSelfAdjoint (Φ : PositiveTracePreservingMap H K)
150    (A : TraceClass H) (hA : IsSelfAdjoint A.1) : ‖Φ A‖ ≤ ‖A‖ := by
151  have hd := TraceClass.positivePart_sub_negativePart A hA
152  calc
153    ‖Φ A‖ = ‖Φ (TraceClass.positivePart A hA) -
154        Φ (TraceClass.negativePart A hA)‖ := by rw [← Φ.map_sub, hd]
155    _ ≤ ‖Φ (TraceClass.positivePart A hA)‖ +
156        ‖Φ (TraceClass.negativePart A hA)‖ := norm_sub_le _ _
157    _ = ‖TraceClass.positivePart A hA‖ + ‖TraceClass.negativePart A hA‖ := by
158      rw [Φ.norm_map_of_nonneg _ (TraceClass.positivePart_nonneg A hA),
159        Φ.norm_map_of_nonneg _ (TraceClass.negativePart_nonneg A hA)]
160    _ = ‖A‖ := TraceClass.norm_positivePart_add_norm_negativePart A hA

162/-- In particular, differences of positive states contract in the genuine trace norm. -/
163theorem norm_map_sub_le (Φ : PositiveTracePreservingMap H K)
164    (A B : TraceClass H) (hA : 0 ≤ A.1) (hB : 0 ≤ B.1) :
165    ‖Φ A - Φ B‖ ≤ ‖A - B‖ := by
166  rw [← Φ.map_sub]
167  exact Φ.norm_map_le_of_isSelfAdjoint (A - B)

170/-- Positivity and trace preservation already force boundedness on the whole
171complex trace-class space; no continuity assumption is needed. -/
172theorem norm_map_le_two_mul (Φ : PositiveTracePreservingMap H K) (A : TraceClass H) :
173    ‖Φ A‖ ≤ 2 * ‖A‖ := by
174  have hd := TraceClass.realComponent_add_I_smul_imaginaryComponent A
175  calc
176    ‖Φ A‖ = ‖Φ (TraceClass.realComponent A) +
177        Complex.I • Φ (TraceClass.imaginaryComponent A)‖ := by
178      conv_lhs => rw [← hd]
179      rw [Φ.toLinearMap.map_add, Φ.toLinearMap.map_smul]
180    _ ≤ ‖Φ (TraceClass.realComponent A)‖ +
181        ‖Complex.I • Φ (TraceClass.imaginaryComponent A)‖ := norm_add_le _ _
182    _ = ‖Φ (TraceClass.realComponent A)‖ + ‖Φ (TraceClass.imaginaryComponent A)‖ := by
183      simp [norm_smul]
184    _ ≤ ‖TraceClass.realComponent A‖ + ‖TraceClass.imaginaryComponent A‖ :=
185      add_le_add
186        (Φ.norm_map_le_of_isSelfAdjoint _ (TraceClass.realComponent_isSelfAdjoint A))
187        (Φ.norm_map_le_of_isSelfAdjoint _ (TraceClass.imaginaryComponent_isSelfAdjoint A))
188    _ ≤ ‖A‖ + ‖A‖ := add_le_add (TraceClass.norm_realComponent_le A)
189      (TraceClass.norm_imaginaryComponent_le A)
190    _ = 2 * ‖A‖ := by ring

200/-- The image of a density operator is a density operator. -/
201def mapState (Φ : PositiveTracePreservingMap H K) (ρ : DensityState H) : DensityState K where
202  op := (Φ (TraceClass.ofOperator ρ.op ρ.traceClass)).1
203  positive := Φ.map_nonneg _ ρ.positive
204  traceClass := (Φ (TraceClass.ofOperator ρ.op ρ.traceClass)).2
205  trace_one := (Φ.trace_preserving _).trans ρ.trace_one

207/-- Identity on trace-class operators. -/

213/-- Composition preserves positivity and trace preservation. -/
214def comp (Ψ : PositiveTracePreservingMap K J) (Φ : PositiveTracePreservingMap H K) :
215    PositiveTracePreservingMap H J where
216  toLinearMap := Ψ.toLinearMap.comp Φ.toLinearMap
217  map_nonneg := fun A hA => Ψ.map_nonneg _ (Φ.map_nonneg A hA)
218  trace_preserving := fun A => (Ψ.trace_preserving _).trans (Φ.trace_preserving A)
