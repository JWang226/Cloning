20theorem physicalSector_le_listSpan {L : List (PhysicalHighestTensor n d)}
21    {H : PhysicalHighestTensor n d} (hH : H ∈ L) : H.sector ≤ physicalSectorListSpan L := by
22  induction L with
23  | nil => simp at hH
24  | cons K L ih =>
25      rcases List.mem_cons.mp hH with rfl | hH
26      · exact le_sup_left
27      · exact (ih hH).trans le_sup_right

36/-- Repeatedly extracting an actual highest tensor strictly lowers the
37remaining dimension and exhausts every invariant physical subspace. -/
38theorem exists_physical_cyclic_decomposition
39    (W : Submodule ℂ (TensorRegister n (Fin d)))
40    (hW : ∀ a b x, x ∈ W → collectiveGenerator n a b x ∈ W) :
41    ∃ L : List (PhysicalHighestTensor n d),
42      L.Pairwise (fun H K => H.sector ⟂ K.sector) ∧ physicalSectorListSpan L = W := by
43  classical
44  have aux : ∀ k : ℕ, ∀ W : Submodule ℂ (TensorRegister n (Fin d)),
45      Module.finrank ℂ W = k →
46      (∀ a b x, x ∈ W → collectiveGenerator n a b x ∈ W) →
47      ∃ L : List (PhysicalHighestTensor n d),
48        L.Pairwise (fun H K => H.sector ⟂ K.sector) ∧ physicalSectorListSpan L = W := by
49    intro k
50    induction k using Nat.strong_induction_on with
51    | h k ih =>
52      intro W hdim hW
53      by_cases hz : W = ⊥
54      · exact ⟨[], by simp, hz.symm⟩
55      · obtain ⟨H, hH⟩ := exists_physicalHighestTensor_le W hW hz
56        let R : Submodule ℂ (TensorRegister n (Fin d)) := H.sectorᗮ ⊓ W
57        have hR : ∀ a b x, x ∈ R → collectiveGenerator n a b x ∈ R := by
58          intro a b x hx
59          exact ⟨generatorInvariant_orthogonal H.sector
60            (fun a b y hy => H.generator_invariant a b hy) a b hx.1, hW a b x hx.2⟩
61        have hRne : R ≠ W := by
62          intro he
63          have hm : H.vector ∈ R := by rw [he]; exact hH H.vector_mem
64          have hi := (H.sector.mem_orthogonal H.vector).mp hm.1 H.vector H.vector_mem
65          have hv : H.vector = 0 := inner_self_eq_zero.mp hi
66          have hn := H.norm_one
67          simp only [hv, norm_zero, zero_ne_one] at hn
68        have hlt : Module.finrank ℂ R < k := by
69          rw [← hdim]
70          exact Submodule.finrank_lt_finrank_of_lt
71            (lt_iff_le_and_ne.mpr ⟨inf_le_right, hRne⟩)
72        obtain ⟨L, hLorth, hLspan⟩ := ih (Module.finrank ℂ R) hlt R rfl hR
73        refine ⟨H :: L, List.pairwise_cons.mpr ⟨?_, hLorth⟩, ?_⟩
74        · intro K hK
75          have hKR : K.sector ≤ R := (physicalSector_le_listSpan hK).trans (le_of_eq hLspan)
76          exact (Submodule.isOrtho_orthogonal_right H.sector).mono_right
77            (hKR.trans inf_le_left)
78        · rw [physicalSectorListSpan, hLspan]
79          exact Submodule.sup_orthogonal_inf_of_hasOrthogonalProjection hH
80  exact aux (Module.finrank ℂ W) W rfl hW

82theorem physicalSectorListSpan_eq_iSup (L : List (PhysicalHighestTensor n d)) :
83    physicalSectorListSpan L = ⨆ i : Fin L.length, (L.get i).sector := by

103/-- The full physical tensor register is exhausted by finitely many orthogonal
104actual highest sectors. Labels are partitions, with multiplicities retained. -/
105theorem exists_tensor_cyclic_decomposition (n d : ℕ) :
106    ∃ L : List (PhysicalHighestTensor n d),
107      OrthogonalFamily ℂ (fun i : Fin L.length => (L.get i).sector)
108        (fun i => (L.get i).sector.subtypeₗᵢ) ∧
109      (⨆ i : Fin L.length, (L.get i).sector) = ⊤ := by
110  obtain ⟨L, hLorth, hLspan⟩ := exists_physical_cyclic_decomposition
111    (⊤ : Submodule ℂ (TensorRegister n (Fin d))) (fun _ _ _ _ => Submodule.mem_top)

113    (physicalSectorListSpan_eq_iSup L).symm.trans hLspan⟩
