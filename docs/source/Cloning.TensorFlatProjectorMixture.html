22def coupledCopyRow (p : Fin d → ℝ) (J : SchurCopy n d → SchurCopy m d → ℝ)
23    (i : SchurCopy n d) (j : SchurCopy m d) : ℝ :=
24  if 0 < knownCopyWeight n d p i then J i j / knownCopyWeight n d p i else knownCopyWeight m d p j

26theorem coupledCopyRow_nonneg (p : Fin d → ℝ) (J : SchurCopy n d → SchurCopy m d → ℝ)

34theorem coupledCopyRow_sum (p : Fin d → ℝ) (hp : ∀ a, 0 ≤ p a) (hs : ∑a,p a=1)

44theorem coupledCopyRow_weight (p : Fin d → ℝ) (J : SchurCopy n d → SchurCopy m d → ℝ)
45    (hJ : ∀ i j, 0 ≤ J i j) (hrow : ∀ i, ∑ j, J i j = knownCopyWeight n d p i)
46    (i : SchurCopy n d) (j : SchurCopy m d) :
47    knownCopyWeight n d p i * coupledCopyRow p J i j = J i j := by
48  by_cases hi : 0 < knownCopyWeight n d p i
49  · simp only [coupledCopyRow, if_pos hi]
50    exact mul_div_cancel₀ _ hi.ne'
51  · have hz : knownCopyWeight n d p i=0 := le_antisymm (not_lt.mp hi) (knownCopyWeight_nonneg _ _ _ _)
52    have hj := Finset.single_le_sum (fun j _ => hJ i j) (Finset.mem_univ j)
53    rw [hrow, hz] at hj
54    have hzero : J i j=0 := le_antisymm hj (hJ i j)
55    simp [hz,hzero]

58def coupledCopyChannel (p : Fin d → ℝ) (hp : ∀ a, 0 ≤ p a) (hs : ∑a,p a=1)
59    (J : SchurCopy n d → SchurCopy m d → ℝ) (hJ : ∀ i j, 0 ≤ J i j)
60    (hrow : ∀ i, ∑ j, J i j = knownCopyWeight n d p i) :
61    QuantumChannel (TensorRegister n (Fin d)) (TensorRegister m (Fin d)) :=
62  channel n m d (coupledCopyRow p J) (coupledCopyRow_nonneg p J hJ) (coupledCopyRow_sum p hp hs J hrow)

114/-- Joint concavity gives a genuine finite lower bound for this actual full-register channel. -/
115theorem coupledCopyChannel_payoff_lower [NeZero d]
116    (p : Fin d → ℝ) (hp : ∀ a, 0 ≤ p a) (hs : ∑a,p a=1)
117    (J : SchurCopy n d → SchurCopy m d → ℝ) (hJ : ∀ i j, 0 ≤ J i j)
118    (hrow : ∀ i, ∑ j, J i j = knownCopyWeight n d p i)
119    (hcol : ∀ j, ∑ i, J i j = knownCopyWeight m d p j)
120    (U : unitary (Matrix (Fin d) (Fin d) ℂ)) :
121    (∑ ij : SchurCopy n d × SchurCopy m d, J ij.1 ij.2 * nonnegativeTransitionFidelity p hp U ij.1 ij.2) ≤
122      statePayoff n m (coupledCopyChannel p hp hs J hJ hrow)
123        (conjugatedState (diagonalState p hp hs) U) := by
124  let A (ij : SchurCopy n d × SchurCopy m d) :=
125    (copySectorPositive ((recursivePhysicalDecomposition n d).get ij.1) U p hp).map
126      (copyTransition n m d ij.1 ij.2).toPositiveTracePreservingMap
127  let B (ij : SchurCopy n d × SchurCopy m d) :=
128    (copySectorPositive ((recursivePhysicalDecomposition m d).get ij.2) U p hp).map
129      (QuantumChannel.ofIsometry ((recursivePhysicalDecomposition m d).get ij.2).canonicalEmbedding).toPositiveTracePreservingMap
130  have hmono (ij : SchurCopy n d × SchurCopy m d) :
131      nonnegativeTransitionFidelity p hp U ij.1 ij.2 ≤ (A ij).rootFidelity (B ij) := by
132    let C := (recursivePhysicalDecomposition n d).get ij.1
133    let D := (recursivePhysicalDecomposition m d).get ij.2
134    let S := (copySectorPositive C U p hp).map
135      (partitionTransitionChannel C.weight D.weight C.weight_antitone D.weight_antitone).toPositiveTracePreservingMap
136    let T := copySectorPositive D U p hp
137    exact InfiniteFidelity.fidelity_data_processing (QuantumChannel.ofIsometry D.canonicalEmbedding) S.1 T.1 S.2 T.2
138  have hconc := PositiveTraceClass.finiteMixture_rootFidelity_lower_same
139    (fun ij => J ij.1 ij.2) (fun ij => hJ ij.1 ij.2) A B
140  have hA : (PositiveTraceClass.finiteMixture (fun ij => J ij.1 ij.2) (fun ij => hJ ij.1 ij.2) A).1 =
141      (coupledCopyChannel p hp hs J hJ hrow).toLinearMap
142        (tensorState (conjugatedState (diagonalState p hp hs) U) n).1 := by
143    symm
144    exact coupledCopyChannel_rotated_apply p hp hs J hJ hrow U
145  have hB : (PositiveTraceClass.finiteMixture (fun ij => J ij.1 ij.2) (fun ij => hJ ij.1 ij.2) B).1 =
146      (tensorState (conjugatedState (diagonalState p hp hs) U) m).1 := by
147    symm
148    exact rotatedTensor_eq_coupledCopy_mixture p J hcol U
149  have hA' : PositiveTraceClass.finiteMixture (fun ij => J ij.1 ij.2) (fun ij => hJ ij.1 ij.2) A =
150      (tensorState (conjugatedState (diagonalState p hp hs) U) n).map
151        (coupledCopyChannel p hp hs J hJ hrow).toPositiveTracePreservingMap := Subtype.ext hA
152  have hB' : PositiveTraceClass.finiteMixture (fun ij => J ij.1 ij.2) (fun ij => hJ ij.1 ij.2) B =
153      tensorState (conjugatedState (diagonalState p hp hs) U) m := Subtype.ext hB
154  rw [hA',hB'] at hconc
155  exact (Finset.sum_le_sum (fun ij _ => mul_le_mul_of_nonneg_left (hmono ij) (hJ ij.1 ij.2))).trans hconc
