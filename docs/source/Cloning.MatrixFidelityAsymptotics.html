23/-- The paper's finite `η + 2√ε` estimate for actual positive matrix blocks.
24Retained and discarded contributions may have overlapping support. -/
25theorem lifted_matrix_factorization_bound
26    {ι : Type*} [Fintype ι] [DecidableEq ι]
27    {d s : ι → Type*}
28    [∀ i, Fintype (d i)] [∀ i, DecidableEq (d i)]
29    [∀ i, Fintype (s i)] [∀ i, DecidableEq (s i)]
30    (X G : ∀ i, Matrix (d i) (d i) ℂ)
31    (ρ : ∀ i, State (d i)) (τ : ∀ i, State (s i))
32    (p : ι → ℝ) (c η : ℝ)
33    (hX : ∀ i, (X i).PosSemidef) (hG : ∀ i, (G i).PosSemidef)
34    (hdiff : ∀ i, (X i - G i).PosSemidef)
35    (hp : ∀ i, 0 ≤ p i) (hpsum : ∑ i, p i = 1)
36    (hXmass : ∑ i, (Matrix.trace (X i)).re ≤ 1)
37    (hc0 : 0 ≤ c) (hc1 : c ≤ 1) (hη : 0 ≤ η)
38    (hsector : ∀ i, 0 < (Matrix.trace (G i)).re →
39      |fidelity (normalizedBlock (G i)) (ρ i).matrix - c| ≤ η) :
40    |fidelity (Matrix.blockDiagonal' (fun i => X i ⊗ₖ (τ i).matrix))
41        (Matrix.blockDiagonal' (fun i => (p i • (ρ i).matrix) ⊗ₖ (τ i).matrix)) -
42      c * Cloning.BlockFidelity.classicalAffinity
43        (fun i => (Matrix.trace (X i)).re) p| ≤
44      η + 2 * Real.sqrt (∑ i, ((Matrix.trace (X i)).re - (Matrix.trace (G i)).re)) := by
45  apply Cloning.BlockFidelity.lifted_factorization_bound p
46    (fun i => (Matrix.trace (X i)).re) (fun i => (Matrix.trace (G i)).re)
47    (fun i => fidelity (normalizedBlock (G i)) (ρ i).matrix)
48    _ c η hp (fun i => trace_re_nonneg (hG i))
49    (fun i => trace_re_mono (show G i ≤ X i from hdiff i))
50    hpsum hXmass hc0 hc1 hη hsector
51  exact lifted_deletion_bound X G (fun i => (ρ i).matrix) (fun i => (τ i).matrix) p
52    hX hG (fun i => (ρ i).positive) (fun i => (τ i).positive) hp hdiff
53    (fun i => by rw [(ρ i).trace_one]; rfl)
54    (fun i => by rw [(τ i).trace_one]; rfl) hpsum

56/-- Concrete growing-dimension version of the manuscript's lifted convergence
57argument. There is no assumed quantum continuity or block-fidelity identity. -/
58theorem lifted_matrix_fidelity_converges
59    {I : ℕ → Type*} [∀ n, Fintype (I n)] [∀ n, DecidableEq (I n)]
60    {d s : (n : ℕ) → I n → Type*}
61    [∀ n i, Fintype (d n i)] [∀ n i, DecidableEq (d n i)]
62    [∀ n i, Fintype (s n i)] [∀ n i, DecidableEq (s n i)]
63    (X G : (n : ℕ) → (i : I n) → Matrix (d n i) (d n i) ℂ)
64    (ρ : (n : ℕ) → (i : I n) → State (d n i))
65    (τ : (n : ℕ) → (i : I n) → State (s n i))
66    (p : (n : ℕ) → I n → ℝ) (η : ℕ → ℝ) (c l : ℝ)
67    (hX : ∀ n i, (X n i).PosSemidef) (hG : ∀ n i, (G n i).PosSemidef)
68    (hdiff : ∀ n i, (X n i - G n i).PosSemidef)
69    (hp : ∀ n i, 0 ≤ p n i) (hpsum : ∀ n, ∑ i, p n i = 1)
70    (hXmass : ∀ n, ∑ i, (Matrix.trace (X n i)).re ≤ 1)
71    (hc0 : 0 ≤ c) (hc1 : c ≤ 1) (hη : ∀ n, 0 ≤ η n)
72    (hsector : ∀ n i, 0 < (Matrix.trace (G n i)).re →
73      |fidelity (normalizedBlock (G n i)) (ρ n i).matrix - c| ≤ η n)
74    (hηlimit : Tendsto η atTop (𝓝 0))
75    (hmass : Tendsto (fun n => ∑ i,
76      ((Matrix.trace (X n i)).re - (Matrix.trace (G n i)).re)) atTop (𝓝 0))
77    (hlabel : Tendsto (fun n => Cloning.BlockFidelity.classicalAffinity
78      (fun i => (Matrix.trace (X n i)).re) (p n)) atTop (𝓝 l)) :
79    Tendsto (fun n =>
80      fidelity (Matrix.blockDiagonal' (fun i => X n i ⊗ₖ (τ n i).matrix))
81        (Matrix.blockDiagonal' (fun i => (p n i • (ρ n i).matrix) ⊗ₖ (τ n i).matrix)))
82      atTop (𝓝 (c * l)) := by
83  apply Cloning.lifted_fidelity_converges p
84    (fun n i => (Matrix.trace (X n i)).re) (fun n i => (Matrix.trace (G n i)).re)
85    (fun n i => fidelity (normalizedBlock (G n i)) (ρ n i).matrix)
86    _ η c l hp (fun n i => trace_re_nonneg (hG n i))
87    (fun n i => trace_re_mono (show G n i ≤ X n i from hdiff n i))
88    hpsum hXmass hc0 hc1 hη hsector _ hηlimit hmass hlabel
89  intro n
90  exact lifted_deletion_bound (X n) (G n) (fun i => (ρ n i).matrix)
91    (fun i => (τ n i).matrix) (p n) (hX n) (hG n)
92    (fun i => (ρ n i).positive) (fun i => (τ n i).positive) (hp n) (hdiff n)
93    (fun i => by rw [(ρ n i).trace_one]; rfl)
94    (fun i => by rw [(τ n i).trace_one]; rfl) (hpsum n)

96/-- Achievability needs only lower sector bounds. This conclusion deliberately
97does not assert an upper bound or exact convergence of mixture fidelities. -/
98theorem lifted_matrix_eventual_achievability
99    {I : ℕ → Type*} [∀ n, Fintype (I n)] [∀ n, DecidableEq (I n)]
100    {d s : (n : ℕ) → I n → Type*}
101    [∀ n i, Fintype (d n i)] [∀ n i, DecidableEq (d n i)]
102    [∀ n i, Fintype (s n i)] [∀ n i, DecidableEq (s n i)]
103    (X G : (n : ℕ) → (i : I n) → Matrix (d n i) (d n i) ℂ)
104    (ρ : (n : ℕ) → (i : I n) → State (d n i))
105    (τ : (n : ℕ) → (i : I n) → State (s n i))
106    (p : (n : ℕ) → I n → ℝ) (η : ℕ → ℝ) (c l : ℝ)
107    (hX : ∀ n i, (X n i).PosSemidef) (hG : ∀ n i, (G n i).PosSemidef)
108    (hdiff : ∀ n i, (X n i - G n i).PosSemidef)
109    (hp : ∀ n i, 0 ≤ p n i) (hpsum : ∀ n, ∑ i, p n i = 1)
110    (hXmass : ∀ n, ∑ i, (Matrix.trace (X n i)).re ≤ 1)
111    (hc0 : 0 ≤ c) (hc1 : c ≤ 1) (hη : ∀ n, 0 ≤ η n)
112    (hsectorLower : ∀ n i, 0 < (Matrix.trace (G n i)).re →
113      c - η n ≤ fidelity (normalizedBlock (G n i)) (ρ n i).matrix)
114    (hηlimit : Tendsto η atTop (𝓝 0))
115    (hmass : Tendsto (fun n => ∑ i,
116      ((Matrix.trace (X n i)).re - (Matrix.trace (G n i)).re)) atTop (𝓝 0))
117    (hlabel : Tendsto (fun n => Cloning.BlockFidelity.classicalAffinity
118      (fun i => (Matrix.trace (X n i)).re) (p n)) atTop (𝓝 l)) :
119    ∀ ε : ℝ, 0 < ε → ∀ᶠ n in atTop, c * l - ε <
120      fidelity (Matrix.blockDiagonal' (fun i => X n i ⊗ₖ (τ n i).matrix))
121        (Matrix.blockDiagonal' (fun i => (p n i • (ρ n i).matrix) ⊗ₖ (τ n i).matrix)) := by
122  have hsqrt : Tendsto (fun n => Real.sqrt (∑ i,
123      ((Matrix.trace (X n i)).re - (Matrix.trace (G n i)).re))) atTop (𝓝 0) := by
124    simpa using Real.continuous_sqrt.continuousAt.tendsto.comp hmass
125  have hlower : Tendsto (fun n => c * Cloning.BlockFidelity.classicalAffinity
126      (fun i => (Matrix.trace (X n i)).re) (p n) - η n -
127      Real.sqrt (∑ i, ((Matrix.trace (X n i)).re - (Matrix.trace (G n i)).re)))
128      atTop (𝓝 (c * l)) := by
129    simpa using ((hlabel.const_mul c).sub hηlimit).sub hsqrt
130  intro ε hε
131  have hgap : c * l - ε < c * l := sub_lt_self _ hε
132  filter_upwards [hlower.eventually_const_lt hgap] with n hn
133  apply hn.trans_le
134  exact fidelity_lifted_achievability (X n) (G n) (fun i => (ρ n i).matrix)
135    (fun i => (τ n i).matrix) (p n) c (η n) (hX n) (hG n)
136    (fun i => (ρ n i).positive) (fun i => (τ n i).positive) (hp n)
137    (fun i => by rw [(τ n i).trace_one]; rfl) (hpsum n) (hXmass n)
138    (fun i => show G n i ≤ X n i from hdiff n i) hc0 hc1 (hη n) (hsectorLower n)
