25/-- An arbitrary covariant CPTP amplifier has at most the thermal-test
26moment of the quantum-limited amplifier, in every finite mode count. -/
27theorem covariant_thermal_test_moment_le
28    (Φ : QuantumChannel (Fock d) (Fock d)) (r : ℝ)
29    (hcov : ∀ a T, Φ.toLinearMap (displacementTraceMap a T) =
30      displacementTraceMap (r • a) (Φ.toLinearMap T))
31    (hr : 1 < r ^ 2) {q y : Fin d → ℝ}
32    (hq0 : ∀ i, 0 < q i) (hq1 : ∀ i, q i < 1)
33    (hy0 : ∀ i, 0 < y i) (hy1 : ∀ i, y i < 1) :
34    (tracePairing (Φ.toLinearMap (vectorMixture (numberBasis d) (productGeometric q)))
35      (vectorMixture (numberBasis d) (productGeometric y)).1).re ≤
36    (tracePairing (vectorMixture (numberBasis d)
37      (productGeometric (fun i => Cloning.Thermal.amplified (r ^ 2) (q i))))
38      (vectorMixture (numberBasis d) (productGeometric y)).1).re := by
39  let t : Fin d → ℝ := fun i => geometricFourierWidth (y i) + r ^ 2 * geometricFourierWidth (q i)
40  have ht (i : Fin d) : (r ^ 2 - 1) / 2 ≤ t i :=
41    thermal_moment_width_ge_noise r (hq0 i) (hq1 i) (hy0 i) (hy1 i)
42  have hbound := quantumChannel_weyl_gaussian_bound Φ r hcov hr ht
43  have hfourier := covariant_productThermal_moment_fourier
44    Φ.toPositiveTracePreservingMap.toContinuousLinearMap r hcov hq0 hq1 hy0 hy1
45  change tracePairing (Φ.toLinearMap (vectorMixture (numberBasis d) (productGeometric q)))
46      (vectorMixture (numberBasis d) (productGeometric y)).1 =
47      ((Real.pi ^ d : ℝ) : ℂ)⁻¹ * ∫ a : Fin d → ℂ,
48        (gaussianWeight t a : ℂ) * weylMultiplier Φ.heisenberg r a at hfourier
49  have hx0 (i : Fin d) : 0 < Cloning.Thermal.amplified (r ^ 2) (q i) :=
50    (hq0 i).trans (Cloning.Thermal.lt_amplified hr (hq1 i))
51  have hx1 (i : Fin d) : Cloning.Thermal.amplified (r ^ 2) (q i) < 1 :=
52    Cloning.Thermal.amplified_lt_one (by linarith) (hq1 i)
53  have hoverlap := productThermal_overlap_fourier hx0 hx1 hy0 hy1
54  rw [hfourier, hoverlap]
55  simp only [← Complex.ofReal_inv, Complex.mul_re, Complex.ofReal_re,
56    Complex.ofReal_im, zero_mul, sub_zero]
57  have hc : 0 ≤ (Real.pi ^ d)⁻¹ := inv_nonneg.mpr (pow_pos Real.pi_pos _).le
58  apply (mul_le_mul_of_nonneg_left ((Complex.re_le_norm _).trans hbound) hc).trans_eq
59  congr 1
60  apply Finset.prod_congr rfl
61  intro i _
62  rw [geometricFourierWidth_amplified (by linarith : r ^ 2 ≠ 0) (hq1 i).ne]
63  congr 1
64  dsimp only [t]
