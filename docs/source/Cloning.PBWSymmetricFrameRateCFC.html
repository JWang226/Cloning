13lemma abs_sqrt_sub_one_le (x : ℝ) : |Real.sqrt x-1|≤|x-1| := by
14  by_cases hx : 0≤x
15  · have hs := Real.sq_sqrt hx
16    have hp := Real.sqrt_nonneg x
17    by_cases h : 1≤x
18    · have hroot : 1≤Real.sqrt x := by nlinarith
19      rw [abs_of_nonneg (by linarith : 0≤Real.sqrt x-1),
20        abs_of_nonneg (by linarith : 0≤x-1)]
21      nlinarith
22    · have hroot : Real.sqrt x≤1 := by nlinarith
23      rw [abs_of_nonpos (by linarith : Real.sqrt x-1≤0),
24        abs_of_nonpos (by linarith : x-1≤0)]
25      nlinarith
26  · rw [Real.sqrt_eq_zero_of_nonpos (le_of_not_ge hx)]
27    rw [abs_of_nonpos (by linarith : x-1≤0)]
28    norm_num
29    linarith

34theorem norm_sqrt_sub_one_le (G : E →L[ℂ] E) (hG : 0≤G) :
35    ‖CFC.sqrt G-1‖≤‖G-1‖ := by
36  have hs : CFC.sqrt G=cfc Real.sqrt G := by
37    rw [CFC.sqrt_eq_real_sqrt G hG, cfcₙ_eq_cfc]
38  have hself : IsSelfAdjoint G := IsSelfAdjoint.of_nonneg hG
39  have hsub : cfc (fun x : ℝ=>Real.sqrt x-1) G=CFC.sqrt G-1 := by
40    rw [cfc_sub Real.sqrt (fun _ : ℝ=>1) G, ←hs, cfc_const_one ℝ G]
41  have hid : cfc (fun x : ℝ=>x-1) G=G-1 := by
42    rw [cfc_sub (fun x : ℝ=>x) (fun _ : ℝ=>1) G, cfc_id' ℝ G, cfc_const_one ℝ G]
43  rw [←hsub]
44  apply norm_cfc_le (norm_nonneg _)
45  intro x hx
46  have h := norm_apply_le_norm_cfc (fun x : ℝ=>x-1) G hx
47  rw [hid] at h
48  exact (abs_sqrt_sub_one_le x).trans h

68theorem polar_adjoint_comp_self (V : E →L[ℂ] H) (hV : IsUnit (gram V)) :
69    (polar V).adjoint.comp (polar V)=1 := by
70  let S := CFC.sqrt (gram V)
71  have hS : star S=S := (IsSelfAdjoint.of_nonneg (CFC.sqrt_nonneg (gram V))).star_eq
72  have hI : star (Ring.inverse S)=Ring.inverse S := by rw [←Ring.inverse_star,hS]
73  have hSS : S*S=gram V := CFC.sqrt_mul_sqrt_self _ (gram_nonneg V)
74  have hunit : IsUnit S := sqrt_isUnit V hV
75  change ((V.comp (Ring.inverse S)).adjoint).comp (V.comp (Ring.inverse S))=1
76  rw [ContinuousLinearMap.adjoint_comp]
77  change (star (Ring.inverse S)).comp (V.adjoint.comp (V.comp (Ring.inverse S)))=1
78  rw [hI, ←ContinuousLinearMap.comp_assoc V.adjoint V]
79  change Ring.inverse S*(gram V*Ring.inverse S)=1
80  rw [←hSS]
81  calc
82    Ring.inverse S*(S*S*Ring.inverse S)=(Ring.inverse S*S)*(S*Ring.inverse S) := by simp only [mul_assoc]
83    _ = 1 := by rw [Ring.inverse_mul_cancel _ hunit, Ring.mul_inverse_cancel _ hunit, one_mul]

91/-- Symmetric normalization changes each vector by at most the Gram operator
92error times the norm of its coefficient vector. -/
93theorem polar_sub_apply_le (V : E →L[ℂ] H) (hV : IsUnit (gram V)) (x : E) :
94    ‖polar V x-V x‖≤‖gram V-1‖*‖x‖ := by
95  have he : polar V x-V x=polar V ((1-CFC.sqrt (gram V)) x) := by
96    rw [ContinuousLinearMap.sub_apply, one_apply, map_sub]
97    congr 1
98    exact (congrArg (fun T : E →L[ℂ] H=>T x) (polar_mul_sqrt V hV)).symm
99  rw [he,polar_norm V hV]
100  apply ((1-CFC.sqrt (gram V)).le_opNorm x).trans
101  apply mul_le_mul_of_nonneg_right _ (norm_nonneg x)
102  rw [norm_sub_rev]
103  exact norm_sqrt_sub_one_le _ (gram_nonneg V)
