31theorem sectorGibbsOperator_height_invariant (p : Fin d → ℝ)
32    (hp : ∀ a, 0 < p a) (H : ℕ) (x : cyclicSector Ω)
33    (hx : x ∈ sectorHeightSpace Ω H) :
34    sectorGibbsOperator Ω mu hweight hraise p x ∈ sectorHeightSpace Ω H := by
35  change tensorOperator n (Matrix.diagonal (fun a => (p a : ℂ))) x.val ∈ cyclicHeightSpace Ω H
36  change x.val ∈ cyclicHeightSpace Ω H at hx
37  suffices hh : ∀ y : TensorRegister n (Fin d), y ∈ cyclicHeightSpace Ω H →
38      tensorOperator n (Matrix.diagonal (fun a => (p a : ℂ))) y ∈ cyclicHeightSpace Ω H from hh x.val hx
39  intro y hy
40  induction hy using Submodule.span_induction with
41  | mem x hx =>
42    obtain ⟨w, hw, rfl⟩ := hx
43    rw [tensorOperator_diagonal_loweringWord Ω mu hweight p hp]
44    exact Submodule.smul_mem _ _ (Submodule.subset_span ⟨w,hw,rfl⟩)
45  | zero => simp
46  | add x y hx hy ihx ihy => simpa only [map_add] using Submodule.add_mem _ ihx ihy
47  | smul c x hx ih => simpa only [map_smul] using Submodule.smul_mem _ c ih
