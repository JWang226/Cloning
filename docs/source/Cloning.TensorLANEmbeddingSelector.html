18def labelMultiplicity (labels : ι → Λ) (a : Λ) : ℕ :=
19  (Finset.univ.filter (fun i => labels i = a)).card

21def copyCellWeight (labels : ι → Λ) (q : X → Λ) (i : ι) (x : X) : ℝ :=
22  if q x = labels i then 1 / (labelMultiplicity labels (labels i) : ℝ) else 0

24def uniformLabelSelector (labels : ι → Λ) (q : X → Λ) : Option ι → X → ℝ
25  | none, x => 1 - ∑ i, copyCellWeight labels q i x
26  | some i, x => copyCellWeight labels q i x

28theorem copyCellWeight_nonneg (labels : ι → Λ) (q : X → Λ) (i : ι) (x : X) :
29    0 ≤ copyCellWeight labels q i x := by
30  unfold copyCellWeight
31  split_ifs <;> positivity
32 
33theorem copyCellWeight_sum (labels : ι → Λ) (q : X → Λ) (x : X) :
34    (∑ i, copyCellWeight labels q i x) =
35      if labelMultiplicity labels (q x) = 0 then 0 else 1 := by
36  have he : (∑ i, copyCellWeight labels q i x) =
37      (∑ i : ι, if labels i = q x then (1 : ℝ) else 0) / labelMultiplicity labels (q x) := by
38    rw [Finset.sum_div]
39    apply Finset.sum_congr rfl
40    intro i _
41    by_cases hi : labels i = q x <;> simp [copyCellWeight, hi, eq_comm]
42  rw [he]
43  have hc : (∑ i : ι, if labels i = q x then (1 : ℝ) else 0) =
44      (labelMultiplicity labels (q x) : ℝ) := by
45    simp [labelMultiplicity]
46  rw [hc]
47  by_cases hz : labelMultiplicity labels (q x) = 0
48  · simp [hz]
49  · simp [hz, Nat.cast_ne_zero.mpr hz]

51theorem uniformLabelSelector_nonneg (labels : ι → Λ) (q : X → Λ) (i : Option ι) (x : X) :
52    0 ≤ uniformLabelSelector labels q i x := by
53  cases i with
54  | none => simp only [uniformLabelSelector, copyCellWeight_sum]; split_ifs <;> norm_num
55  | some i => exact copyCellWeight_nonneg labels q i x

90/-- Read a measurable classical label, choose uniformly among its actual
91physical copies, and use the explicit physical fallback outside the label set. -/
92def schurReverseFromLabels (n d R : ℕ) (ρ : DensityState (TensorRegister n (Fin d)))
93    (q : X → Fin d → ℕ) (hq : Measurable q) :
94    HybridToQuantum (RootFock d) (TensorRegister n (Fin d)) ν :=
95  schurReverse n d R ρ (uniformLabelSelector (schurCopyLabel n d) q)
96    (fun i => (uniformLabelSelector_measurable _ _ hq i).aestronglyMeasurable)
97    (uniformLabelSelector_nonneg _ _) (uniformLabelSelector_sum _ _)
