4/-!
5# Literal matrix-unit generators on finite tensor registers
6 
7Each one-slot generator changes the letter in that slot. The construction is
8an actual bounded operator on the computational-word Hilbert space, rather
9than an abstract representation satisfying postulated Lie relations.
10-/

23def slotMatrix (t : Fin n) (a b : A) : Matrix (Fin n → A) (Fin n → A) ℂ :=
24  fun w v => if w t = a ∧ v = Function.update w t b then 1 else 0

26def slotGenerator (t : Fin n) (a b : A) : TensorRegister n A →L[ℂ] TensorRegister n A :=
27  ofMatrix (registerBasis (Fin n → A)) (slotMatrix t a b)

29/-- The operator acts by the literal tensor matrix-unit formula. -/
30@[simp] theorem slotGenerator_apply (t : Fin n) (a b : A)
31    (x : TensorRegister n A) (w : Fin n → A) :
32    slotGenerator t a b x w = if w t = a then x (Function.update w t b) else 0 := by
33  rw [← register_inner_single, ← registerBasis_apply]
34  rw [slotGenerator, inner_ofMatrix_apply (registerBasis _).orthonormal]
35  simp only [registerBasis_apply, register_inner_single]
36  by_cases h : w t = a <;> simp [slotMatrix, h]

39theorem update_relation (t : Fin n) (a b : A) (w v : Fin n → A) :
40    (w t = a ∧ v = Function.update w t b) ↔
41      (v t = b ∧ w = Function.update v t a) := by
42  constructor
43  · rintro ⟨h, rfl⟩
44    refine ⟨by simp, ?_⟩
45    rw [Function.update_idem, ← h, Function.update_eq_self]
46  · rintro ⟨h, rfl⟩
47    refine ⟨by simp, ?_⟩
48    rw [Function.update_idem, ← h, Function.update_eq_self]

51theorem slotMatrix_adjoint (t : Fin n) (a b : A) :
52    (slotMatrix t a b)ᴴ = slotMatrix t b a := by
53  ext w v
54  have he := update_relation t a b v w
55  by_cases h : v t = a ∧ w = Function.update v t b
56  · simp only [Matrix.conjTranspose_apply, slotMatrix, if_pos h, if_pos (he.mp h), star_one]
57  · have h' : ¬(w t = b ∧ v = Function.update w t a) := fun hh => h (he.mpr hh)
58    simp only [Matrix.conjTranspose_apply, slotMatrix, if_neg h, if_neg h', star_zero]

60theorem slotGenerator_adjoint (t : Fin n) (a b : A) :
61    (slotGenerator t a b).adjoint = slotGenerator t b a := by
62  rw [← ContinuousLinearMap.star_eq_adjoint, slotGenerator,
63    ← ofMatrix_conjTranspose, slotMatrix_adjoint]
64  rfl

66theorem slotGenerator_mul_same (t : Fin n) (a b c e : A) :
67    slotGenerator t a b * slotGenerator t c e =
68      if b = c then slotGenerator t a e else 0 := by
69  ext x w
70  simp only [ContinuousLinearMap.mul_apply, slotGenerator_apply,
71    Function.update_self, Function.update_idem]
72  by_cases hbc : b = c <;> by_cases hwa : w t = a <;> simp [hbc, hwa]

74theorem slotGenerator_commute {t u : Fin n} (htu : t ≠ u) (a b c e : A) :
75    Commute (slotGenerator t a b) (slotGenerator u c e) := by
76  apply ContinuousLinearMap.ext
77  intro x
78  ext w
79  simp only [ContinuousLinearMap.mul_apply, slotGenerator_apply,
80    Function.update_of_ne htu, Function.update_of_ne htu.symm]
81  rw [Function.update_comm htu]
82  by_cases hwa : w t = a <;> by_cases hwc : w u = c <;> simp [hwa, hwc]

84theorem slotGenerator_norm_le_one (t : Fin n) (a b : A) :
85    ‖slotGenerator t a b‖ ≤ 1 := by
86  have hp := CStarRing.norm_star_mul_self (x := slotGenerator t b b)
87  simp only [ContinuousLinearMap.star_eq_adjoint, slotGenerator_adjoint,
88    slotGenerator_mul_same, if_true] at hp
89  have hpbound : ‖slotGenerator t b b‖ ≤ 1 := by
90    have := norm_nonneg (slotGenerator t b b)
91    nlinarith
92  have he := CStarRing.norm_star_mul_self (x := slotGenerator t a b)
93  simp only [ContinuousLinearMap.star_eq_adjoint, slotGenerator_adjoint,
94    slotGenerator_mul_same, if_true] at he
95  have := norm_nonneg (slotGenerator t a b)
96  nlinarith

98def collectiveGenerator (n : ℕ) (a b : A) :
99    TensorRegister n A →L[ℂ] TensorRegister n A := ∑ t : Fin n, slotGenerator t a b

112theorem collectiveGenerator_norm_le (a b : A) :
