117theorem eventually_flat_sampleShape_compatible (d : ℕ) (m : ℕ → ℕ)
118    (hm : Tendsto m atTop atTop) (γ : ℝ) (hγ : 1<γ)
119    (hratio : Tendsto (fun N : ℕ ↦ (m N : ℝ)/(N : ℝ)) atTop (𝓝 γ))
120    (x : rootSpace d) (hx : StrictAnti (fun i ↦ x.1 i)) :
121    ∀ᶠ N in atTop, PartitionCompatible
122      (fun i ↦ (sampleShape d N (flatSpectrum (d+1)) x i).val)
123      (fun i ↦ (sampleShape d (m N) (flatSpectrum (d+1)) x i).val) := by
124  have hsqrt : 1<Real.sqrt γ := by
125    simpa using Real.sqrt_lt_sqrt (by norm_num : (0 : ℝ)≤1) hγ
126  have he : ∀ᶠ N in atTop, ∀ r : PositiveRoot (d+1),
127      0 < ((((sampleShape d (m N) (flatSpectrum (d+1)) x r.val.1).val : ℝ)-
128          ((sampleShape d (m N) (flatSpectrum (d+1)) x r.val.2).val : ℝ))-
129        (((sampleShape d N (flatSpectrum (d+1)) x r.val.1).val : ℝ)-
130          ((sampleShape d N (flatSpectrum (d+1)) x r.val.2).val : ℝ)))/Real.sqrt (N : ℝ) :=
131    eventually_all.mpr fun r ↦ (sampleShape_flat_cross_gap_tendsto d m hm γ hratio x r.val.1 r.val.2).eventually
132      (eventually_gt_nhds (mul_pos (sub_pos.mpr hsqrt) (sub_pos.mpr (hx r.property))))
133  filter_upwards [he, eventually_flat_sampleShape_dominates d m hm γ hγ hratio x,
134    eventually_gt_atTop 0] with N hgap hdom hN
135  refine ⟨hdom, ?_⟩
136  intro i j hij
137  rcases hij.eq_or_lt with rfl | hij
138  · exact le_rfl
139  have hh := hgap ⟨(i,j),hij⟩
140  have hn : 0<Real.sqrt (N : ℝ) := by positivity
141  have hreal := (div_pos_iff_of_pos_right hn).mp hh
142  have ht :
143      (((sampleShape d (m N) (flatSpectrum (d+1)) x j).val-
144        (sampleShape d N (flatSpectrum (d+1)) x j).val : ℕ) : ℝ) ≤
145      (((sampleShape d (m N) (flatSpectrum (d+1)) x i).val-
146        (sampleShape d N (flatSpectrum (d+1)) x i).val : ℕ) : ℝ) := by
147    rw [Nat.cast_sub (hdom i), Nat.cast_sub (hdom j)]
