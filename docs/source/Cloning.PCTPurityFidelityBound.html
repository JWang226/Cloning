29/-- The exact physical particle, for all sample sizes (including the harmless
30zero convention), has Frobenius deviation at most twice its tangent scale. -/
31theorem particle_frobenius_deviation_le (hr : 2≤r)
32    (u : OrthonormalBasis (Fin (s+1)) ℂ (Register (Fin r×Fin r)))
33    (hu : u 0=coefficientVector (schmidtCoefficients (flatSpectrum r)))
34    (z : Fin s→ℂ) (L : ℕ) :
35    ‖particle u z L-flatBase r‖≤2*Real.sqrt (GeneralCoherent.energy z/L) := by
36  let Z := frameTangentMatrix u z
37  have hn : ‖Z‖=Real.sqrt (GeneralCoherent.energy z) := by
38    rw [frobenius_norm_eq_coefficientVector]
39    have he := frameTangentMatrix_energy u u.orthonormal z
40    rw [←he,Real.sqrt_sq (norm_nonneg _)]
41  have ht : 0≤sampleScale L := by unfold sampleScale; positivity
42  have hf := flat_rational_frobenius_le hr Z (sampleScale L) ht
43  rw [particle_exact u hu,flatBase_eq_smul_one]
44  have hz : ‖coefficientVector Z‖=‖Z‖ := (frobenius_norm_eq_coefficientVector Z).symm
45  change ‖(1+sampleScale L^2*‖coefficientVector Z‖^2)⁻¹ •
46      ((r:ℝ)⁻¹ • (1 : Matrix (Fin r) (Fin r) ℂ)+sampleScale L •
47        partialTraceDifferential (flatSpectrum r) Z+sampleScale L^2 • (Z*Zᴴ))-
48      (r:ℝ)⁻¹ • (1 : Matrix (Fin r) (Fin r) ℂ)‖≤_
49  rw [hz,flat_tangent_eq]
50  calc
51    _ ≤ 2*sampleScale L*‖Z‖ := hf
52    _ = _ := by
53      rw [hn,Real.sqrt_div (GeneralCoherent.energy_nonneg z),sampleScale]
54      ring
