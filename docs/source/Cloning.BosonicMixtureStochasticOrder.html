19/-- Exact factorization of the seeded expectation of a product test. -/
20theorem productLaw_antitone_weighted_hasSum {s : ℕ} {r : Fin s → ℝ}
21    (hr0 : ∀ i, 0 ≤ r i) (hr1 : ∀ i, r i < 1) {w : Fin s → ℕ → ℝ}
22    (hw0 : ∀ i k, 0 ≤ w i k) (hw : ∀ i, Antitone (w i)) (l : Fin s → ℕ) :
23    HasSum (fun k ↦ productLaw r l k * ∏ i, w i (k i))
24      (∏ i, ∑' k, seededLaw (r i) (l i) k * w i k) := by
25  simpa only [productLaw, Finset.prod_mul_distrib] using
26    Thermal.hasSum_fin_product s (fun i k ↦ seededLaw (r i) (l i) k * w i k)
27      (fun i ↦ ∑' k, seededLaw (r i) (l i) k * w i k)
28      (fun i k ↦ mul_nonneg (seededLaw_nonneg (hr0 i) (hr1 i) _ _) (hw0 i k))
29      (fun i ↦ (seededLaw_weighted_summable (hr0 i) (hr1 i) (hw0 i) (hw i) (l i)).hasSum)

31/-- Full fixed-seed multimode least-noise bound. -/
32theorem productLaw_antitone_moment_le {s : ℕ} {r : Fin s → ℝ}
33    (hr0 : ∀ i, 0 ≤ r i) (hr1 : ∀ i, r i < 1) {w : Fin s → ℕ → ℝ}
34    (hw0 : ∀ i k, 0 ≤ w i k) (hw : ∀ i, Antitone (w i)) (l : Fin s → ℕ) :
35    (∑' k, productLaw r l k * ∏ i, w i (k i)) ≤
36      ∏ i, ∑' k, Thermal.geometric (r i) k * w i k := by
37  rw [(productLaw_antitone_weighted_hasSum hr0 hr1 hw0 hw l).tsum_eq]
38  apply Finset.prod_le_prod
39  · intro i hi
40    exact tsum_nonneg (fun k ↦ mul_nonneg (seededLaw_nonneg (hr0 i) (hr1 i) _ _) (hw0 i k))
41  · intro i hi
42    exact seededLaw_antitone_moment_le (hr0 i) (hr1 i) (hw0 i) (hw i) (l i)

44/-- The actual correlated output has an absolutely convergent product-test
45expectation; boundedness is derived from antitonicity. -/
46theorem mixtureLaw_antitone_weighted_summable {s : ℕ} {r : Fin s → ℝ}
47    {a : (Fin s → ℕ) → ℝ} (hr0 : ∀ i, 0 ≤ r i) (hr1 : ∀ i, r i < 1)
48    {w : Fin s → ℕ → ℝ} (hw0 : ∀ i k, 0 ≤ w i k) (hw : ∀ i, Antitone (w i))
49    (ha0 : ∀ l, 0 ≤ a l) (ha : Summable a) :
50    Summable (fun k ↦ mixtureLaw r a k * ∏ i, w i (k i)) := by
51  have hs := (mixtureLaw_hasSum hr0 hr1 ha0 ha.hasSum).summable
52  apply (hs.mul_right (∏ i, w i 0)).of_norm_bounded
53  intro k
54  have hm := mixtureLaw_nonneg hr0 hr1 ha0 k
55  have hwk : 0 ≤ ∏ i, w i (k i) := Finset.prod_nonneg (fun i _ ↦ hw0 i _)
56  rw [Real.norm_eq_abs, abs_of_nonneg (mul_nonneg hm hwk)]
57  exact mul_le_mul_of_nonneg_left
58    (Finset.prod_le_prod (fun i _ ↦ hw0 i _) (fun i _ ↦ hw i (Nat.zero_le _))) hm

60/-- Fubini for the full product-test expectation of the correlated output law. -/
61theorem mixtureLaw_antitone_moment_eq {s : ℕ} {r : Fin s → ℝ}
62    {a : (Fin s → ℕ) → ℝ} (hr0 : ∀ i, 0 ≤ r i) (hr1 : ∀ i, r i < 1)
63    {w : Fin s → ℕ → ℝ} (hw0 : ∀ i k, 0 ≤ w i k) (hw : ∀ i, Antitone (w i))
64    (ha0 : ∀ l, 0 ≤ a l) (ha : Summable a) :
65    (∑' k, mixtureLaw r a k * ∏ i, w i (k i)) =
66      ∑' l, a l * ∏ i, ∑' k, seededLaw (r i) (l i) k * w i k := by
67  have hs := mixtureJoint_summable hr0 hr1 ha0 ha
68  have hW0 (k : Fin s → ℕ) : 0 ≤ ∏ i, w i (k i) :=
69    Finset.prod_nonneg (fun i _ ↦ hw0 i _)
70  have hW (k : Fin s → ℕ) : (∏ i, w i (k i)) ≤ ∏ i, w i 0 := by
71    exact Finset.prod_le_prod (fun i _ ↦ hw0 i _) (fun i _ ↦ hw i (Nat.zero_le _))
72  have hsW : Summable (fun lk : (Fin s → ℕ) × (Fin s → ℕ) ↦
73      a lk.1 * productLaw r lk.1 lk.2 * ∏ i, w i (lk.2 i)) := by
74    apply (hs.mul_right (∏ i, w i 0)).of_norm_bounded
75    intro lk
76    have hnonneg := mul_nonneg (ha0 lk.1) (productLaw_nonneg hr0 hr1 lk.1 lk.2)
77    rw [Real.norm_eq_abs, abs_of_nonneg (mul_nonneg hnonneg (hW0 _))]
78    exact mul_le_mul_of_nonneg_left (hW _) hnonneg
79  unfold mixtureLaw
80  simp_rw [← tsum_mul_right]
81  rw [hsW.tsum_comm (f := fun l k ↦ a l * productLaw r l k * ∏ i, w i (k i))]
82  simp_rw [mul_assoc, tsum_mul_left,
83    (productLaw_antitone_weighted_hasSum hr0 hr1 hw0 hw _).tsum_eq]

85/-- The manuscript's complete scalar least-noise test class: every product of
86bounded nonnegative nonincreasing number functions, under arbitrary correlated
87idler occupation weights, with the exact total-mass factor. -/
88theorem mixtureLaw_antitone_moment_le {s : ℕ} {r : Fin s → ℝ}
89    {a : (Fin s → ℕ) → ℝ} {c : ℝ}
90    (hr0 : ∀ i, 0 ≤ r i) (hr1 : ∀ i, r i < 1) {w : Fin s → ℕ → ℝ}
91    (hw0 : ∀ i k, 0 ≤ w i k) (hw : ∀ i, Antitone (w i))
92    (ha0 : ∀ l, 0 ≤ a l) (ha : HasSum a c) :
93    (∑' k, mixtureLaw r a k * ∏ i, w i (k i)) ≤
94      c * ∏ i, ∑' k, Thermal.geometric (r i) k * w i k := by
95  rw [mixtureLaw_antitone_moment_eq hr0 hr1 hw0 hw ha0 ha.summable]
96  have hM0 (l : Fin s → ℕ) :
97      0 ≤ ∏ i, ∑' k, seededLaw (r i) (l i) k * w i k :=
98    Finset.prod_nonneg (fun i _ ↦ tsum_nonneg
99      (fun k ↦ mul_nonneg (seededLaw_nonneg (hr0 i) (hr1 i) _ _) (hw0 i k)))
100  have hM (l : Fin s → ℕ) :
101      (∏ i, ∑' k, seededLaw (r i) (l i) k * w i k) ≤
102        ∏ i, ∑' k, Thermal.geometric (r i) k * w i k := by
103    rw [← (productLaw_antitone_weighted_hasSum hr0 hr1 hw0 hw l).tsum_eq]
104    exact productLaw_antitone_moment_le hr0 hr1 hw0 hw l
105  have hmajor := ha.summable.mul_right (∏ i, ∑' k, Thermal.geometric (r i) k * w i k)
106  have hminor : Summable (fun l ↦ a l * ∏ i, ∑' k, seededLaw (r i) (l i) k * w i k) := by
107    apply hmajor.of_norm_bounded
108    intro l
109    rw [Real.norm_eq_abs, abs_of_nonneg (mul_nonneg (ha0 l) (hM0 l))]
110    exact mul_le_mul_of_nonneg_left (hM l) (ha0 l)
111  calc
112    _ ≤ ∑' l, a l * ∏ i, ∑' k, Thermal.geometric (r i) k * w i k :=
113      hminor.tsum_le_tsum (fun l ↦ mul_le_mul_of_nonneg_left (hM l) (ha0 l)) hmajor
114    _ = _ := by rw [tsum_mul_right, ha.tsum_eq]

116/-- The same result written directly as comparison with the product geometric
117vacuum-output law, matching the two expectations in the least-noise proposition. -/
118theorem mixtureLaw_antitone_moment_le_geometric {s : ℕ} {r : Fin s → ℝ}
119    {a : (Fin s → ℕ) → ℝ} {c : ℝ}
120    (hr0 : ∀ i, 0 ≤ r i) (hr1 : ∀ i, r i < 1) {w : Fin s → ℕ → ℝ}
121    (hw0 : ∀ i k, 0 ≤ w i k) (hw : ∀ i, Antitone (w i))
122    (ha0 : ∀ l, 0 ≤ a l) (ha : HasSum a c) :
123    (∑' k, mixtureLaw r a k * ∏ i, w i (k i)) ≤
124      c * ∑' k : Fin s → ℕ, (∏ i, Thermal.geometric (r i) (k i)) * ∏ i, w i (k i) := by
125  have hs := productLaw_antitone_weighted_hasSum hr0 hr1 hw0 hw (fun _ ↦ 0)
126  simp only [productLaw, seededLaw_zero] at hs
127  rw [hs.tsum_eq]
128  exact mixtureLaw_antitone_moment_le hr0 hr1 hw0 hw ha0 ha

130/-- One-mode correlated-seed notation, convenient for the actual idler channel. -/
131theorem single_mixture_antitone_moment_le {r c : ℝ} {a w : ℕ → ℝ}
132    (hr0 : 0 ≤ r) (hr1 : r < 1) (hw0 : ∀ k, 0 ≤ w k) (hw : Antitone w)
133    (ha0 : ∀ l, 0 ≤ a l) (ha : HasSum a c) :
134    (∑' k, (∑' l, a l * seededLaw r l k) * w k) ≤
135      c * ∑' k, Thermal.geometric r k * w k := by
136  let e : (Fin 1 → ℕ) ≃ ℕ := Equiv.funUnique (Fin 1) ℕ
137  have he (l : Fin 1 → ℕ) : e l = l 0 := rfl
138  have ha' : HasSum (fun l : Fin 1 → ℕ ↦ a (l 0)) c := by
139    simpa only [Function.comp_apply, he] using e.hasSum_iff.mpr ha
140  have h := mixtureLaw_antitone_moment_le (r := fun _ : Fin 1 ↦ r)
141    (w := fun _ ↦ w) (a := fun l ↦ a (l 0))
142    (fun _ ↦ hr0) (fun _ ↦ hr1) (fun _ ↦ hw0) (fun _ ↦ hw)
143    (fun l ↦ ha0 (l 0)) ha'
144  simp only [mixtureLaw, productLaw, Fin.prod_univ_one] at h
145  have hinner (k : ℕ) : (∑' l : Fin 1 → ℕ, a (l 0) * seededLaw r (l 0) k) =
146      ∑' l, a l * seededLaw r l k := e.tsum_eq (fun l ↦ a l * seededLaw r l k)
147  simp_rw [hinner] at h
148  have houter := e.tsum_eq (fun k ↦ (∑' l, a l * seededLaw r l k) * w k)
149  rw [show (∑' k : Fin 1 → ℕ, (∑' l, a l * seededLaw r l (k 0)) * w (k 0)) =
150      (∑' k, (∑' l, a l * seededLaw r l k) * w k) from houter] at h
151  exact h
