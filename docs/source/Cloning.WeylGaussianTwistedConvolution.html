18theorem integral_gaussian_twisted_single {t : ℝ} (ht : 0 < t) (r : ℝ) (z : ℂ) :
19    (∫ a : ℂ, (Real.exp (-t * ‖a‖ ^ 2) : ℂ) *
20      (Real.exp (-t * ‖a + z‖ ^ 2) : ℂ) *
21      (displacementPhase (-a) (a + z) / displacementPhase (-(r • a)) (r • (a + z)))) =
22      ((Real.pi / (2 * t) : ℝ) : ℂ) *
23        (Real.exp (-(t / 2 + (r ^ 2 - 1) ^ 2 / (8 * t)) * ‖z‖ ^ 2) : ℂ) := by
24  let u : ℂ := (((r : ℂ) ^ 2 - 1) / 2 - (t : ℂ)) * starRingEnd ℂ z
25  let v : ℂ := (-((r : ℂ) ^ 2 - 1) / 2 - (t : ℂ)) * z
26  have he (a : ℂ) : (Real.exp (-t * ‖a‖ ^ 2) : ℂ) *
27      (Real.exp (-t * ‖a + z‖ ^ 2) : ℂ) *
28      (displacementPhase (-a) (a + z) / displacementPhase (-(r • a)) (r • (a + z))) =
29      Complex.exp (-(t : ℂ) * ((‖z‖ ^ 2 : ℝ) : ℂ)) *
30        Complex.exp (-((2 * t : ℝ) : ℂ) * ((‖a‖ ^ 2 : ℝ) : ℂ) +
31          u * a + v * starRingEnd ℂ a) := by
32    simp only [Complex.ofReal_exp, Complex.ofReal_mul, Complex.ofReal_neg,
33      displacementPhase, ← Complex.exp_sub, ← Complex.exp_add]
34    congr 1
35    simp only [norm_sq_cast', map_add, map_neg, Complex.real_smul, map_mul,
36      Complex.conj_ofReal, Complex.ofReal_mul, Complex.ofReal_ofNat, u, v]
37    ring
38  simp_rw [he]
39  rw [integral_const_mul, integral_complex_gaussian_linear_scaled (by positivity : 0 < 2 * t)]
40  rw [mul_left_comm]
41  congr 1
42  rw [← Complex.exp_add, Complex.ofReal_exp]
43  congr 1
44  have htC : (t : ℂ) ≠ 0 := Complex.ofReal_ne_zero.mpr ht.ne'
45  simp only [u, v, norm_sq_cast', Complex.ofReal_mul, Complex.ofReal_add,
46    Complex.ofReal_neg, Complex.ofReal_div, Complex.ofReal_pow, Complex.ofReal_sub,
47    Complex.ofReal_one, Complex.ofReal_ofNat]
48  simp only [norm_sq_cast]
49  field_simp
50  ring

151/-- The exact double-integral identity feeding the quantum-positive Gaussian
152recursion. The shear and Fubini steps are justified in the full Bochner
153integral by a proved integrable dominating Gaussian. -/
154theorem integral_integral_gaussian_quantumKernel {t : Fin d → ℝ} (ht : ∀ i, 0 < t i)
155    {f : (Fin d → ℂ) → ℂ} (hf : Continuous f) (hfbound : ∀ a, ‖f a‖ ≤ 1) (r : ℝ) :
156    (∫ a, ∫ b, (gaussianWeight t a : ℂ) * (gaussianWeight t b : ℂ) *
157      quantumPositiveKernel f r a b) =
158      ((∏ i, Real.pi / (2 * t i) : ℝ) : ℂ) *
159        ∫ z, (gaussianWeight (fun i => t i / 2 + (r ^ 2 - 1) ^ 2 / (8 * t i)) z : ℂ) * f z := by
160  let K := fun a b => (gaussianWeight t a : ℂ) * (gaussianWeight t b : ℂ) *
161    quantumPositiveKernel f r a b
162  have hi := (measurePreserving_prod_add volume volume).integrable_comp_of_integrable
163    (integrable_gaussian_kernel ht hf hfbound r)
164  have hshift : (∫ a, ∫ b, K a b) = ∫ a, ∫ z, K a (a + z) := by
165    apply integral_congr_ae
166    exact Eventually.of_forall fun a => (integral_add_left_eq_self (K a) a).symm
167  change (∫ a, ∫ b, K a b) = _
168  have hi' : Integrable (Function.uncurry (fun a z => K a (a + z))) (volume.prod volume) := hi
169  rw [hshift, integral_integral_swap (f := fun a z => K a (a + z)) hi']
170  have he (a z : Fin d → ℂ) : K a (a + z) =
171      ((gaussianWeight t a : ℂ) * (gaussianWeight t (a + z) : ℂ) *
172        (displacementPhase (-a) (a + z) / displacementPhase (-(r • a)) (r • (a + z)))) * f z := by
173    simp only [K, quantumPositiveKernel, add_sub_cancel_left, mul_assoc]
174  simp_rw [he, integral_mul_const, integral_gaussian_twisted_product ht r]
175  rw [← integral_const_mul]
176  apply integral_congr_ae
177  exact Eventually.of_forall fun z => mul_assoc _ _ _
