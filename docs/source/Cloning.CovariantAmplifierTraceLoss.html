13/-- Normalize the actual covariant CP map, apply a normalized-channel moment
14bound, and restore its exact success trace. No success-probability parameter or
15normalization premise on the competing map is supplied. -/
16theorem covariant_tni_trace_scaled_moment_le
17    (r : ℝ) (A : TraceClass (Fock d)) (hA : traceCLM A=1)
18    (W : Fock d →L[ℂ] Fock d) (C : ℝ)
19    (hmoment : ∀ Φ : QuantumChannel (Fock d) (Fock d),
20      (∀ a T, Φ.toLinearMap (displacementTraceMap a T)=
21        displacementTraceMap (r • a) (Φ.toLinearMap T)) →
22      (tracePairing (Φ.toLinearMap A) W).re≤C)
23    (Γ : TraceClass (Fock d) →ₗ[ℂ] TraceClass (Fock d))
24    (hCP : IsCompletelyPositive Γ)
25    (htrace : ∀ T, 0≤T.1 → (traceCLM (Γ T)).re≤(traceCLM T).re)
26    (hcov : ∀ a T, Γ (displacementTraceMap a T)=displacementTraceMap (r • a) (Γ T)) :
27    (tracePairing (Γ A) W).re≤(traceCLM (Γ A)).re*C := by
28  rcases covariant_cpTNI_zero_or_scaled_channel Γ hCP.map_nonneg hCP htrace r hcov with
29    hz | ⟨c,Φ,hc,_,heq,hΦ⟩
30  · rw [hz]
31    simp only [LinearMap.zero_apply, ← tracePairingCLM_apply, map_zero,
32      ContinuousLinearMap.zero_apply, Complex.zero_re, zero_mul, le_refl]
33  · have hs : (tracePairing (Γ A) W).re=c*(tracePairing (Φ.toLinearMap A) W).re := by
34      rw [heq]
35      change (tracePairingCLM ((c : ℂ) • Φ.toLinearMap A) W).re=_
36      simp only [map_smul, ContinuousLinearMap.smul_apply, smul_eq_mul,
37        Complex.mul_re, Complex.ofReal_re, Complex.ofReal_im, zero_mul, sub_zero,
38        tracePairingCLM_apply]
39    have ht : traceCLM (Φ.toLinearMap A)=traceCLM A :=
40      Φ.toPositiveTracePreservingMap.trace_preserving A
41    have hmass : (traceCLM (Γ A)).re=c := by
42      rw [heq]
43      change (traceCLM ((c : ℂ) • Φ.toLinearMap A)).re=c
44      rw [map_smul, ht, hA]
45      simp
46    rw [hs, hmass]
47    exact mul_le_mul_of_nonneg_left (hmoment Φ hΦ) hc.le
