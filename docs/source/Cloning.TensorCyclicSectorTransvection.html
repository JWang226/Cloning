15private theorem slotMatrix_product (t : Fin n) (a b : Fin d)
16    (w v : Fin n → Fin d) :
17    (∏ j ∈ Finset.univ.erase t, (1 : Matrix (Fin d) (Fin d) ℂ) (w j) (v j)) *
18        Matrix.single a b (1 : ℂ) (w t) (v t) = slotMatrix t a b w v := by
19  by_cases h : w t = a ∧ v = Function.update w t b
20  · rcases h with ⟨ha, rfl⟩
21    rw [slotMatrix, if_pos ⟨ha, rfl⟩]
22    have hp : (∏ j ∈ Finset.univ.erase t,
23        (1 : Matrix (Fin d) (Fin d) ℂ) (w j) (Function.update w t b j)) = 1 := by
24      apply Finset.prod_eq_one
25      intro j hj
26      rw [Function.update_of_ne (Finset.mem_erase.mp hj).1]
27      simp
28    simp [hp, ha]
29  · rw [slotMatrix, if_neg h]
30    by_cases ha : w t = a
31    · by_cases hb : v t = b
32      · have hv : ∃ j, j ≠ t ∧ w j ≠ v j := by
33          by_contra hn
34          push_neg at hn
35          apply h
36          refine ⟨ha, funext (fun j => ?_)⟩
37          by_cases hj : j = t
38          · subst j
39            simpa using hb
40          · simpa [Function.update_of_ne hj] using (hn j hj).symm
41        obtain ⟨j, hj, hwv⟩ := hv
42        have hp : (∏ k ∈ Finset.univ.erase t,
43            (1 : Matrix (Fin d) (Fin d) ℂ) (w k) (v k)) = 0 :=
44          Finset.prod_eq_zero (i := j) (by simp [hj]) (by simp [Matrix.one_apply, hwv])
45        rw [hp, zero_mul]
46      · simp [Matrix.single, ha, hb, Ne.symm hb]

55      (∑ t : Fin n, slotMatrix t a b) 0 := by
56    apply hasDerivAt_pi.mpr

58    apply hasDerivAt_pi.mpr

61        (fun z : ℂ => Matrix.transvection a b z (w j) (v j))

97/-- A matrix-unit commutant commutes with the corresponding literal tensor
98transvection, with no representation-integration premise. -/
99theorem tensorOperator_transvection_commutes
100    (T : TensorRegister n (Fin d) →L[ℂ] TensorRegister n (Fin d))
101    (a b : Fin d) (hab : a ≠ b)
102    (hT : T * collectiveGenerator n a b = collectiveGenerator n a b * T) (z : ℂ) :
103    T * tensorOperator n (Matrix.transvection a b z) =
104      tensorOperator n (Matrix.transvection a b z) * T := by
105  let F (s : ℂ) := tensorOperator n (Matrix.transvection a b (-s)) * T *
106    tensorOperator n (Matrix.transvection a b s)
107  have hd (s : ℂ) : HasDerivAt F 0 s := by
108    have hbase := tensorOperator_transvection_hasDerivAt_right (n := n) a b hab (-s)
109    have hminus := (show HasDerivAt _ _ (0 - s) by simpa using hbase).comp_const_sub 0 s
110    simp only [zero_sub] at hminus
111    have hplus := tensorOperator_transvection_hasDerivAt (n := n) a b hab s
112    have hh := (hminus.mul_const T).mul hplus
113    convert hh using 1
114    simp only [neg_mul]
115    rw [mul_assoc (tensorOperator n (Matrix.transvection a b (-s)))
116      (collectiveGenerator n a b) T, ← hT]
117    noncomm_ring
118  have hc : F z = F 0 := is_const_of_deriv_eq_zero
119    (fun s => (hd s).differentiableAt) (fun s => (hd s).deriv) z 0
120  have hcancel : tensorOperator n (Matrix.transvection a b z) *
121      tensorOperator n (Matrix.transvection a b (-z)) = 1 := by
122    rw [← tensorOperator_mul, Matrix.transvection_mul_transvection_same a b hab,
123      add_neg_cancel, Matrix.transvection_zero, tensorOperator_one]
124  have hh := congrArg (fun S => tensorOperator n (Matrix.transvection a b z) * S) hc
125  simpa only [F, neg_zero, Matrix.transvection_zero, tensorOperator_one, one_mul,
126    mul_one, ← mul_assoc, hcancel] using hh
