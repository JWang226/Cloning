44/-- The flat orbit-root-fidelity average at an arbitrary real radius.
45For positive radii this is exactly the normalized Lebesgue box integral. -/
46def realFlatBoxPayoff (gain : ℝ) (Φ : QuantumChannel (Fock d) (Fock d))
47    (A B : PositiveTraceClass (Fock d)) (L : ℝ) : ℝ :=
48  (volume.real (phaseSpaceUnitBox d))⁻¹ * ∫ z in phaseSpaceUnitBox d,
49    orbitPayoff gain Φ A B (L • z)

65lemma realFlatBoxPayoff_le_sqrt (gain : ℝ)
66    (Φ : QuantumChannel (Fock d) (Fock d)) (A B : PositiveTraceClass (Fock d)) (L : ℝ) :
67    realFlatBoxPayoff gain Φ A B L ≤ Real.sqrt ‖A.1‖ * Real.sqrt ‖B.1‖ := by
68  have hint : IntegrableOn (fun z => orbitPayoff gain Φ A B (L • z))
69      (phaseSpaceUnitBox d) :=
70    ((continuous_orbitPayoff gain Φ A B).comp
71      (continuous_const.smul continuous_id)).continuousOn.integrableOn_compact
72        (phaseSpaceUnitBox_compact d)
73  have hp (z : Fin d → ℂ) : orbitPayoff gain Φ A B (L • z) ≤
74      Real.sqrt ‖A.1‖ * Real.sqrt ‖B.1‖ := by
75    rw [orbitPayoff_eq_translated]
76    simpa only [PositiveTraceClass.norm_map] using
77      PositiveTraceClass.rootFidelity_le_sqrt
78        (PositiveTraceClass.map (translatedChannel gain Φ (L • z)).toPositiveTracePreservingMap A) B
79  have hi := integral_mono hint (integrableOn_const (hs :=
80    (phaseSpaceUnitBox_compact d).measure_ne_top)) hp
81  change (volume.real (phaseSpaceUnitBox d))⁻¹ * _ ≤ _
82  calc
83    _ ≤ (volume.real (phaseSpaceUnitBox d))⁻¹ *
84        ∫ _z in phaseSpaceUnitBox d, Real.sqrt ‖A.1‖ * Real.sqrt ‖B.1‖ :=
85      mul_le_mul_of_nonneg_left hi (inv_nonneg.mpr measureReal_nonneg)
86    _ = _ := by
87      simp only [integral_const, measureReal_def, Measure.restrict_apply_univ, smul_eq_mul]
88      change (volume.real (phaseSpaceUnitBox d))⁻¹ *
89        (volume.real (phaseSpaceUnitBox d) * (Real.sqrt ‖A.1‖ * Real.sqrt ‖B.1‖)) = _
90      rw [← mul_assoc, inv_mul_cancel₀ (phaseSpaceUnitBox_volume_real_pos d).ne', one_mul]

93lemma realFlatBoxPayoff_le_integer (gain : ℝ)
94    (Φ : QuantumChannel (Fock d) (Fock d)) (A B : PositiveTraceClass (Fock d))
95    {L : ℝ} (hL : 0 < L) (n : ℕ) (hLn : L ≤ (n : ℝ) + 1) :
96    realFlatBoxPayoff gain Φ A B L ≤
97      (((n : ℝ) + 1) / L) ^ Module.finrank ℝ (Fin d → ℂ) *
98        flatBoxPayoff gain Φ A B n := by
99  have hN : 0 < (n : ℝ) + 1 := by positivity
100  have hi : (∫ a in phaseSpaceBox d L, orbitPayoff gain Φ A B a) ≤
101      ∫ a in phaseSpaceBox d ((n : ℝ) + 1), orbitPayoff gain Φ A B a :=
102    setIntegral_mono_set
103      ((continuous_orbitPayoff gain Φ A B).continuousOn.integrableOn_compact
104        (phaseSpaceBox_compact d _))
105      (Eventually.of_forall fun _ => PositiveTraceClass.rootFidelity_nonneg _ _)
106      (phaseSpaceBox_mono hLn).eventuallyLE
107  rw [realFlatBoxPayoff_eq_normalized_box gain Φ A B hL,
108    flatBoxPayoff_eq_normalized_box, phaseSpaceBox_volume_real hL,
109    phaseSpaceBox_volume_real hN]
110  calc
111    _ ≤ (L ^ Module.finrank ℝ (Fin d → ℂ) * volume.real (phaseSpaceUnitBox d))⁻¹ *
112        ∫ a in phaseSpaceBox d ((n : ℝ) + 1), orbitPayoff gain Φ A B a :=
113      mul_le_mul_of_nonneg_left hi (by positivity)
114    _ = _ := by
115      rw [div_pow]
116      field_simp [hL.ne', hN.ne', (phaseSpaceUnitBox_volume_real_pos d).ne']

126/-- An integer-radius bound uniform over channels remains uniform at every
127sufficiently large real radius. No continuity in the choice of channel is needed. -/
128lemma eventually_forall_realFlatBox_le_of_integer (gain : ℝ)
129    (A B : PositiveTraceClass (Fock d)) {C : ℝ}
130    (h : ∀ ε > 0, ∀ᶠ n in atTop, ∀ Φ : QuantumChannel (Fock d) (Fock d),
131      flatBoxPayoff gain Φ A B n ≤ C + ε) :
132    ∀ ε > 0, ∀ᶠ L : ℝ in atTop, ∀ Φ : QuantumChannel (Fock d) (Fock d),
133      realFlatBoxPayoff gain Φ A B L ≤ C + ε := by
134  intro ε hε
135  have hfloor : Tendsto (Nat.floor : ℝ → ℕ) atTop atTop :=
136    Nat.floor_mono.tendsto_atTop_atTop (fun n => ⟨(n : ℝ), by simp⟩)
137  have hint := hfloor.eventually (h (ε / 2) (by linarith))
138  have ht : Tendsto (fun L : ℝ =>
139      (1 + L⁻¹) ^ Module.finrank ℝ (Fin d → ℂ) * (C + ε / 2)) atTop
140      (𝓝 (C + ε / 2)) := by
141    simpa using (realFlatBox_overhead_tendsto_one d).mul_const (C + ε / 2)
142  have hnear := ht.eventually_lt_const (by linarith : C + ε / 2 < C + ε)
143  filter_upwards [hint, hnear, eventually_gt_atTop (0 : ℝ)] with L hLi hnear hL
144  intro Φ
145  have hc := realFlatBoxPayoff_le_integer gain Φ A B hL (Nat.floor L)
146    (Nat.lt_floor_add_one L).le
147  have hr : (((Nat.floor L : ℕ) : ℝ) + 1) / L ≤ 1 + L⁻¹ := by
148    apply (div_le_iff₀ hL).mpr
149    have hf := Nat.floor_le hL.le
150    rw [add_mul, one_mul, inv_mul_cancel₀ hL.ne']
151    linarith
152  calc
153    realFlatBoxPayoff gain Φ A B L ≤
154        ((((Nat.floor L : ℕ) : ℝ) + 1) / L) ^ Module.finrank ℝ (Fin d → ℂ) *
155          flatBoxPayoff gain Φ A B (Nat.floor L) := hc
156    _ ≤ (1 + L⁻¹) ^ Module.finrank ℝ (Fin d → ℂ) *
157          flatBoxPayoff gain Φ A B (Nat.floor L) := by
158      apply mul_le_mul_of_nonneg_right _ (flatBoxPayoff_nonneg gain Φ A B _)
159      exact pow_le_pow_left₀ (by positivity) hr _
160    _ ≤ (1 + L⁻¹) ^ Module.finrank ℝ (Fin d → ℂ) * (C + ε / 2) :=
161      mul_le_mul_of_nonneg_left (hLi Φ) (by positivity)
162    _ ≤ C + ε := hnear.le

215/-- The manuscript orbital factor at physical amplitude gain `sqrt γ`, for
216all real radii tending to infinity. The payoff is average root fidelity. -/
217theorem limsup_realFlatBox_thermal_le_modeFactor
218    (γ : ℝ) (hγ : 1 < γ) {q : Fin d → ℝ}
219    (hq0 : ∀ i, 0 < q i) (hq1 : ∀ i, q i < 1) :
220    Filter.limsup (realFlatBoxOptimalPayoff (Real.sqrt γ)
221      (thermalPositive q (fun i => (hq0 i).le) hq1)
222      (thermalPositive q (fun i => (hq0 i).le) hq1)) atTop ≤
223        ∏ i, Thermal.modeFactor γ (q i) := by
224  have hs : Real.sqrt γ ^ 2 = γ := Real.sq_sqrt (by linarith)
225  have h := limsup_realFlatBox_thermal_le (Real.sqrt γ) (by rwa [hs]) hq0 hq1
226  simp_rw [hs, Thermal.fidelity_amplified_eq_modeFactor hγ (hq0 _).le (hq1 _)] at h
