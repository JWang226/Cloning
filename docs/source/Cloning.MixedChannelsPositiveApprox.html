18/-- A sequence of everywhere positive, integrable simple fields converges to any
19positive L¹ input in the actual trace norm. -/
20theorem PositiveL1.exists_simple_approx (A : PositiveL1 H μ) :
21    ∃ (f : ℕ → SimpleFunc Ω (TraceClass H)) (hf : ∀ n, Integrable (f n) μ),
22      (∀ n y, 0 ≤ (f n y).1) ∧
23      Tendsto (fun n => (hf n).toL1 (f n)) atTop (𝓝 A.1) := by
24  let a : {g : Lp (TraceClass H) 1 μ // 0 ≤ g} :=
25    ⟨A.1, (Lp.coeFn_nonneg A.1).mp A.2⟩
26  obtain ⟨x, hx, ht⟩ := mem_closure_iff_seq_limit.mp
27    (Lp.simpleFunc.denseRange_coeSimpleFuncNonnegToLpNonneg 1 μ (TraceClass H)
28      (by simp) a)
29  choose g hg using hx
30  have hgt : Tendsto (fun n => (g n).1.1) atTop (𝓝 A.1) := by
31    have hv := continuous_subtype_val.tendsto a |>.comp ht
32    change Tendsto (fun n => (x n).1) atTop (𝓝 A.1) at hv
33    convert hv using 1
34    funext n
35    exact congrArg Subtype.val (hg n)
36  choose f hf he using fun n =>
37    Lp.simpleFunc.exists_simpleFunc_nonneg_ae_eq (g n).2
38  have hfi (n) : Integrable (f n) μ :=
39    (L1.integrable_coeFn (g n).1.1).congr (he n)
40  refine ⟨f, hfi, hf, ?_⟩
41  have heq (n) : (hfi n).toL1 (f n) = (g n).1.1 := by
42    apply Lp.ext
43    exact ((hfi n).coeFn_toL1).trans (he n).symm
44  simpa only [heq] using hgt
