55theorem pairIndex_card (d : ℕ) : Fintype.card (PairIndex d)*2=d*(d-1) := by
56  let e : PairIndex d ≃ Σj : Fin d, Fin j.val :=
57    { toFun := fun ij => ⟨ij.val.2,⟨ij.val.1.val,ij.property⟩⟩
58      invFun := fun ji => ⟨(⟨ji.2.val,lt_trans ji.2.isLt ji.1.isLt⟩,ji.1),ji.2.isLt⟩
59      left_inv := by intro ij; rfl
60      right_inv := by intro ji; rfl }
61  rw [Fintype.card_congr e,Fintype.card_sigma]
62  simp only [Fintype.card_fin]
63  rw [Fin.sum_univ_eq_sum_range (fun i : ℕ => i)]
64  exact Finset.sum_range_id_mul_two d
