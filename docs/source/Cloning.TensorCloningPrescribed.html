13def prescribedChannel (n m d : ℕ)
14    (Φ : QuantumChannel (TensorRegister n (Fin d)) (TensorRegister m (Fin d))) :
15    QuantumChannel (TensorRegister n (Fin d)) (TensorRegister m (Fin d)) :=
16  if h : m≤n then restrictionChannel n m d h else Φ

18theorem prescribedChannel_of_le (n m d : ℕ)
19    (Φ : QuantumChannel (TensorRegister n (Fin d)) (TensorRegister m (Fin d))) (h : m≤n) :
20    prescribedChannel n m d Φ=restrictionChannel n m d h := by simp only [prescribedChannel,dif_pos h]

38theorem prescribedChannel_covariant (n m d : ℕ)
39    (Φ : QuantumChannel (TensorRegister n (Fin d)) (TensorRegister m (Fin d)))
40    (U : Matrix (Fin d) (Fin d) ℂ) (hU : Uᴴ*U=1)
41    (hΦ : ∀ X,Φ.toLinearMap (conjugationLinearMap (tensorOperator n U) X)=
42      conjugationLinearMap (tensorOperator m U) (Φ.toLinearMap X))
43    (X : TraceClass (TensorRegister n (Fin d))) :
44    (prescribedChannel n m d Φ).toLinearMap (conjugationLinearMap (tensorOperator n U) X)=
45      conjugationLinearMap (tensorOperator m U) ((prescribedChannel n m d Φ).toLinearMap X) := by
46  by_cases h : m≤n
47  · rw [prescribedChannel_of_le n m d Φ h]
48    exact restrictionChannel_covariant n m d h U hU X
49  · rw [prescribedChannel_of_lt n m d Φ (not_le.mp h)]
50    exact hΦ X
