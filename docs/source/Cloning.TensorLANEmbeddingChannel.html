15/-- A genuine Hilbert-space contraction gives a trace-nonincreasing conjugation.
16The proof is the actual positive trace-class rank-one expansion. -/
17theorem conjugationLinearMap_contraction_trace (V : H →L[ℂ] K)
18    (hV : ∀ x, ‖V x‖ ≤ ‖x‖) (A : TraceClass H) (hA : 0 ≤ A.1) :
19    (traceCLM (conjugationLinearMap V A)).re ≤ (traceCLM A).re := by
20  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
21  have hl : HasSum (fun i : w => ‖V (CFC.sqrt A.1 (b i))‖ ^ 2)
22      (traceCLM (conjugationLinearMap V A)).re := by
23    have h := (conjugationLinearMap_positive_hasSum V A hA b).mapL traceCLM
24    simpa only [traceCLM_vectorProjector, Complex.ofReal_re] using Complex.hasSum_re h
25  have hr : HasSum (fun i : w => ‖CFC.sqrt A.1 (b i)‖ ^ 2) (traceCLM A).re := by
26    simpa only [norm_vectorProjector] using positive_rankOne_mass hA A.2 b
27  exact hasSum_le (fun i => (sq_le_sq₀ (norm_nonneg _) (norm_nonneg _)).mpr (hV _)) hl hr

29/-- Apply the rectangular compression, then put exactly the missing trace in
30an explicit fixed density state. This is CPTP on every complex trace-class input. -/
31def QuantumChannel.ofContraction (V : H →L[ℂ] K) (hV : ∀ x, ‖V x‖ ≤ ‖x‖)
32    (σ : DensityState K) : QuantumChannel H K :=
33  QuantumChannel.traceRepair (conjugationLinearMap V) (conjugationLinearMap_completelyPositive V)
34    (conjugationLinearMap_contraction_trace V hV) σ

42/-- Exact action on pure inputs, including the trace of their discarded component. -/
43theorem QuantumChannel.ofContraction_vectorProjector (V : H →L[ℂ] K)
44    (hV : ∀ x, ‖V x‖ ≤ ‖x‖) (σ : DensityState K) (x : H) :
45    (QuantumChannel.ofContraction V hV σ).toLinearMap (vectorProjector x) =
46      vectorProjector (V x) + ((‖x‖ ^ 2 - ‖V x‖ ^ 2 : ℝ) : ℂ) •
47        TraceClass.ofOperator σ.op σ.traceClass := by
48  rw [QuantumChannel.ofContraction_apply, conjugationLinearMap_vectorProjector,
49    traceCLM_vectorProjector, traceCLM_vectorProjector]
50  simp only [Complex.ofReal_sub, Complex.ofReal_pow]

52/-- Compression is exact on every retained pure vector. -/
53theorem QuantumChannel.ofContraction_vectorProjector_of_norm_eq (V : H →L[ℂ] K)

59/-- The trace-norm cost of the fallback is exactly the lost probability. -/
60theorem QuantumChannel.ofContraction_repair_error (V : H →L[ℂ] K)
61    (hV : ∀ x, ‖V x‖ ≤ ‖x‖) (σ : DensityState K) (A : TraceClass H) (hA : 0 ≤ A.1) :
62    ‖(QuantumChannel.ofContraction V hV σ).toLinearMap A - conjugationLinearMap V A‖ =
63      (traceCLM A).re - (traceCLM (conjugationLinearMap V A)).re :=
64  norm_traceRepair_sub (conjugationLinearMap_completelyPositive V)
