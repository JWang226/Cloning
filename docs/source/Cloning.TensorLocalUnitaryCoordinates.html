32/-- A positive-root sum in literal one-particle matrix units. -/
33def rootSkewMatrix (c : PositiveRoot d → ℂ) : Matrix (Fin d) (Fin d) ℂ :=
34  ∑ a, (c a • Matrix.single a.val.2 a.val.1 1 -
35    star (c a) • Matrix.single a.val.1 a.val.2 1)

37private theorem upperRootSum (c : PositiveRoot d → ℂ) (i j : Fin d) :
38    (∑ a : PositiveRoot d, c a • Matrix.single a.val.1 a.val.2 (1 : ℂ)) i j =
39      if hij : i < j then c ⟨(i,j),hij⟩ else 0 := by
40  simp only [Matrix.sum_apply, Matrix.smul_apply, smul_eq_mul]
41  by_cases hij : i < j
42  · rw [dif_pos hij, Finset.sum_eq_single (⟨(i,j),hij⟩ : PositiveRoot d)]
43    · simp
44    · intro a _ ha
45      have hne : ¬ (a.val.1 = i ∧ a.val.2 = j) := by
46        rintro ⟨h1,h2⟩
47        apply ha
48        exact Subtype.ext (Prod.ext h1 h2)
49      simp [Matrix.single, Matrix.of_apply, hne]
50    · simp
51  · rw [dif_neg hij]
52    apply Finset.sum_eq_zero
53    intro a _
54    have hne : ¬ (a.val.1 = i ∧ a.val.2 = j) := by
55      rintro ⟨h1,h2⟩
56      apply hij
57      simpa only [← h1, ← h2] using a.property
58    simp [Matrix.single, Matrix.of_apply, hne]

119/-- At the physical sample scale, the coefficient is the square root of the
120empirical spectral gap divided by the base spectral gap. -/
121theorem scaledRootParameter_sampleScale (p : Fin d → ℝ) (mu : Fin d → ℕ)
122    (hmu : Antitone mu) (L : ℕ) (z : PositiveRoot d → ℂ) (a : PositiveRoot d) :
123    scaledRootParameter p mu (sampleScale L) z a =

132/-- Convergence of the actual local-chart orbital coefficients. The partition
133size is the physical sample size `L`; no cloning-ratio factor is introduced. -/
134theorem scaledRootParameter_tendsto (p : Fin d → ℝ)
135    (hp : ∀ a : PositiveRoot d, 0 < p a.val.1-p a.val.2)
136    (mu : ℕ → Fin d → ℕ) (hmu : ∀ N, Antitone (mu N)) (L : ℕ → ℕ)
137    (hfreq : Tendsto (fun N => fun a => (mu N a : ℝ)/(L N : ℝ)) atTop (𝓝 p))
138    (z : ℕ → PositiveRoot d → ℂ) (z₀ : PositiveRoot d → ℂ)
139    (hz : Tendsto z atTop (𝓝 z₀)) :
140    Tendsto (fun N => scaledRootParameter p (mu N) (sampleScale (L N)) (z N))
141      atTop (𝓝 z₀) := by
142  apply tendsto_pi_nhds.mpr
143  intro a
144  have hf := (tendsto_pi_nhds.mp hfreq a.val.1).sub
145    (tendsto_pi_nhds.mp hfreq a.val.2)
146  simp only [← sub_div] at hf
147  have hcoef := (hf.sqrt).div_const (Real.sqrt (p a.val.1-p a.val.2))
148  rw [div_self (ne_of_gt (Real.sqrt_pos.mpr (hp a)))] at hcoef
149  have hc := (Complex.continuous_ofReal.tendsto 1).comp hcoef
150  have h := hc.mul (tendsto_pi_nhds.mp hz a)
151  simpa only [Complex.ofReal_one, one_mul, scaledRootParameter_sampleScale p _ (hmu _) ] using h
