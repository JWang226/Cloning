20# The trace-class Banach space
21 
22## i. Overview
23 
24The trace-class operators form a Banach space `𝒮₁(H)` under the trace norm. The trace norm dominates
25the operator norm, so an absolutely convergent series of trace-class operators converges in operator
26norm. Its limit is trace class because the trace norm is lower semicontinuous along such limits,
27which gives completeness.
28 
29## ii. Key results
30 
31- `TraceClass` : the trace-class operators with the trace norm.
32- `opNorm_le_traceNorm` : the operator norm is at most the trace norm.
33- `isTraceClass_of_tendsto_of_traceNorm_bounded` : an operator-norm limit of trace-class operators
34  with bounded trace norms is trace class.
35- `TraceClass.instCompleteSpace` : `𝒮₁(H)` is complete.
36

57lemma isTraceClass_zero : IsTraceClass (0 : H →L[ℂ] H) := by
58  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
59  exact ⟨w, b, by simp [CFC.abs_zero]⟩

61/-- Trace-class operators are closed under scalar multiplication. -/
62lemma isTraceClass_smul (c : ℂ) {T : H →L[ℂ] H} (hT : IsTraceClass T) :
63    IsTraceClass (c • T) := by
64  obtain ⟨w, b, hb⟩ := hT
65  refine ⟨w, b, ?_⟩
66  have habs : CFC.abs (c • T) = ‖c‖ • CFC.abs T := CFC.abs_smul c T
67  have heq : ∀ i : w, (⟪b i, CFC.abs (c • T) (b i)⟫_ℂ).re
68      = ‖c‖ * (⟪b i, CFC.abs T (b i)⟫_ℂ).re := by
69    intro i
70    rw [habs, ContinuousLinearMap.smul_apply,
71      RCLike.real_smul_eq_coe_smul (K := ℂ), inner_smul_right]
72    simp [Complex.mul_re]
73  simpa only [heq] using hb.mul_left ‖c‖

81lemma isTraceClass_neg {T : H →L[ℂ] H} (hT : IsTraceClass T) : IsTraceClass (-T) := by
82  have h := isTraceClass_smul (-1 : ℂ) hT
83  rwa [neg_one_smul] at h

85lemma traceNorm_zero : traceNorm (0 : H →L[ℂ] H) isTraceClass_zero = 0 := by

90/-- The trace norm is absolutely homogeneous: `‖c • T‖₁ = ‖c‖ ‖T‖₁`. -/
91lemma traceNorm_smul (c : ℂ) {T : H →L[ℂ] H} (hT : IsTraceClass T) :
92    traceNorm (c • T) (isTraceClass_smul c hT) = ‖c‖ * traceNorm T hT := by
93  set w₀ : Set H := hT.choose with hw₀
94  set b₀ : HilbertBasis w₀ ℂ H := hT.choose_spec.choose with hb₀def
95  have habs : CFC.abs (c • T) = ‖c‖ • CFC.abs T := CFC.abs_smul c T
96  have heq : ∀ i : w₀, (⟪b₀ i, CFC.abs (c • T) (b₀ i)⟫_ℂ).re
97      = ‖c‖ * (⟪b₀ i, CFC.abs T (b₀ i)⟫_ℂ).re := by
98    intro i
99    rw [habs, ContinuousLinearMap.smul_apply,
100      RCLike.real_smul_eq_coe_smul (K := ℂ), inner_smul_right]
101    simp [Complex.mul_re]
102  rw [traceNorm_eq_of_hilbertBasis (isTraceClass_smul c hT) b₀]
103  calc ∑' i : w₀, (⟪b₀ i, CFC.abs (c • T) (b₀ i)⟫_ℂ).re
104      = ∑' i : w₀, ‖c‖ * (⟪b₀ i, CFC.abs T (b₀ i)⟫_ℂ).re := tsum_congr heq
105    _ = ‖c‖ * ∑' i : w₀, (⟪b₀ i, CFC.abs T (b₀ i)⟫_ℂ).re := tsum_mul_left
106    _ = ‖c‖ * traceNorm T hT := rfl

108lemma traceNorm_neg {T : H →L[ℂ] H} (hT : IsTraceClass T) (hnegT : IsTraceClass (-T)) :
109    traceNorm (-T) hnegT = traceNorm T hT := by
110  have h1 := traceNorm_smul (-1 : ℂ) hT
111  have h2 := traceNorm_transport (neg_one_smul ℂ T) (isTraceClass_smul (-1) hT)
112  calc
113    traceNorm (-T) hnegT =
114        traceNorm (-T) (neg_one_smul ℂ T ▸ isTraceClass_smul (-1) hT) := traceNorm_congr
115    _ = traceNorm ((-1 : ℂ) • T) (isTraceClass_smul (-1) hT) := h2.symm
116    _ = ‖(-1 : ℂ)‖ * traceNorm T hT := h1
117    _ = traceNorm T hT := by norm_num

119-- The trace norm is subadditive, `‖T + T'‖₁ ≤ ‖T‖₁ + ‖T'‖₁`: `traceNorm_add_le`, ported into
120-- `IdealNorm.lean` from the duality-bound argument (see module docstring), already has exactly
121-- this signature, so it is used directly (e.g. in `traceClassAddGroupNorm` below) rather than
122-- restated here.

124/-- **The operator norm is at most the trace norm.** With `S = √|T|`, Parseval and Cauchy–Schwarz
125give `‖S x‖² ≤ ‖T‖₁ ‖x‖²`, and `‖T‖ = ‖S‖²`. -/
126lemma opNorm_le_traceNorm {T : H →L[ℂ] H} (hT : IsTraceClass T) : ‖T‖ ≤ traceNorm T hT := by
127  set A : H →L[ℂ] H := CFC.abs T with hAdef
128  have hAnonneg : 0 ≤ A := CFC.abs_nonneg T
129  have hAself : IsSelfAdjoint A := .of_nonneg hAnonneg
130  set S : H →L[ℂ] H := CFC.sqrt A with hSdef
131  have hSself : IsSelfAdjoint S := .of_nonneg (CFC.sqrt_nonneg A)
132  have hSS : S * S = A := CFC.sqrt_mul_sqrt_self A hAnonneg
133  have hTA : ‖T‖ = ‖A‖ := (CFC.norm_abs).symm
134  have hAeq : ‖A‖ = ‖S‖ ^ 2 := by rw [← hSS]; exact hSself.norm_mul_self
135  set w₀ : Set H := hT.choose with hw₀
136  set b₀ : HilbertBasis w₀ ℂ H := hT.choose_spec.choose with hb₀def
137  have hb₀ : Summable (fun i : w₀ => (⟪b₀ i, A (b₀ i)⟫_ℂ).re) := hT.choose_spec.choose_spec
138  have hSstar : ContinuousLinearMap.adjoint S = S :=
139    (ContinuousLinearMap.star_eq_adjoint S).symm.trans hSself
140  have hpt : ∀ i : w₀, (⟪b₀ i, A (b₀ i)⟫_ℂ).re = ‖S (b₀ i)‖ ^ 2 := by
141    intro i
142    have hinner : ⟪b₀ i, A (b₀ i)⟫_ℂ = ⟪S (b₀ i), S (b₀ i)⟫_ℂ := by
143      rw [← hSS]
144      show ⟪b₀ i, (S * S) (b₀ i)⟫_ℂ = _
145      rw [ContinuousLinearMap.mul_def, ContinuousLinearMap.comp_apply]
146      rw [← ContinuousLinearMap.adjoint_inner_left S (S (b₀ i)) (b₀ i), hSstar]
147    rw [hinner, inner_self_eq_norm_sq_to_K]; norm_cast
148  have hSum : Summable (fun i : w₀ => ‖S (b₀ i)‖ ^ 2) := hb₀.congr hpt
149  have htrEq : traceNorm T hT = ∑' i : w₀, ‖S (b₀ i)‖ ^ 2 := tsum_congr hpt
150  have htrNonneg : 0 ≤ traceNorm T hT := traceNorm_nonneg T hT
151  have hbound : ∀ x : H, ‖S x‖ ^ 2 ≤ traceNorm T hT * ‖x‖ ^ 2 := by
152    intro x
153    have hpar : HasSum (fun i : w₀ => ‖⟪b₀ i, S x⟫_ℂ‖ ^ 2) (‖S x‖ ^ 2) :=
154      hasSum_norm_sq_inner_basis b₀ (S x)
155    have heq2 : ∀ i : w₀, ⟪b₀ i, S x⟫_ℂ = ⟪S (b₀ i), x⟫_ℂ := fun i => by
156      have h := ContinuousLinearMap.adjoint_inner_left S x (b₀ i)
157      rw [hSstar] at h
158      exact h.symm
159    have hpar' : HasSum (fun i : w₀ => ‖⟪S (b₀ i), x⟫_ℂ‖ ^ 2) (‖S x‖ ^ 2) := by
160      simpa [heq2] using hpar
161    have hCS : ∀ i : w₀, ‖⟪S (b₀ i), x⟫_ℂ‖ ^ 2 ≤ ‖S (b₀ i)‖ ^ 2 * ‖x‖ ^ 2 := fun i => by
162      have h : ‖⟪S (b₀ i), x⟫_ℂ‖ ≤ ‖S (b₀ i)‖ * ‖x‖ := norm_inner_le_norm _ _
163      calc ‖⟪S (b₀ i), x⟫_ℂ‖ ^ 2 ≤ (‖S (b₀ i)‖ * ‖x‖) ^ 2 :=
164            pow_le_pow_left₀ (norm_nonneg _) h 2
165        _ = ‖S (b₀ i)‖ ^ 2 * ‖x‖ ^ 2 := by ring
166    have hdom : HasSum (fun i : w₀ => ‖S (b₀ i)‖ ^ 2 * ‖x‖ ^ 2) (traceNorm T hT * ‖x‖ ^ 2) := by
167      rw [htrEq]; exact hSum.hasSum.mul_right (‖x‖ ^ 2)
168    exact hasSum_le hCS hpar' hdom
169  have hboundNorm : ∀ x : H, ‖S x‖ ≤ Real.sqrt (traceNorm T hT) * ‖x‖ := by
170    intro x
171    have h1 : ‖S x‖ ^ 2 ≤ traceNorm T hT * ‖x‖ ^ 2 := hbound x
172    have h2 : Real.sqrt (‖S x‖ ^ 2) ≤ Real.sqrt (traceNorm T hT * ‖x‖ ^ 2) :=
173      Real.sqrt_le_sqrt h1
174    rwa [Real.sqrt_sq (norm_nonneg _), Real.sqrt_mul htrNonneg,
175      Real.sqrt_sq (norm_nonneg _)] at h2
176  have hSnorm_le : ‖S‖ ≤ Real.sqrt (traceNorm T hT) :=
177    S.opNorm_le_bound (Real.sqrt_nonneg _) hboundNorm
178  have hSsq_le : ‖S‖ ^ 2 ≤ traceNorm T hT := by
179    have h := pow_le_pow_left₀ (norm_nonneg S) hSnorm_le 2
180    rwa [Real.sq_sqrt htrNonneg] at h
181  rw [hTA, hAeq]
182  exact hSsq_le

184/-! ## B. The trace-class submodule and Banach space -/
185 
186/-- The trace-class operators, as a submodule of the bounded operators. -/
187def traceClassSubmodule (H : Type*) [NormedAddCommGroup H] [InnerProductSpace ℂ H]
188    [CompleteSpace H] : Submodule ℂ (H →L[ℂ] H) where
189  carrier := {T | IsTraceClass T}
190  zero_mem' := isTraceClass_zero
191  add_mem' ha hb := isTraceClass_add ha hb
192  smul_mem' c _ ha := isTraceClass_smul c ha
193 
194/-- **The trace-class Banach space `𝒮₁(H)`**, normed by the trace norm. It is a `def` so that the
195operator norm inherited from the ambient submodule is not found as an instance. -/
196def TraceClass (H : Type*) [NormedAddCommGroup H] [InnerProductSpace ℂ H] [CompleteSpace H] :
197    Type _ :=
198  traceClassSubmodule H
