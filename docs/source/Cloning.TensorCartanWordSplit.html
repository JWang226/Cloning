14/-- All ways to send the successive roots to the left or the right tensor
15factor, retaining the original order separately within each factor. -/
16noncomputable def loweringSplits : List (PositiveRoot d) → List (List (PositiveRoot d) × List (PositiveRoot d))
17  | [] => [([], [])]
18  | a :: w => (loweringSplits w).map (fun uv => (a :: uv.1, uv.2)) ++
19      (loweringSplits w).map (fun uv => (uv.1, a :: uv.2))

21theorem loweringSplits_sublist (w : List (PositiveRoot d))
22    (u v : List (PositiveRoot d)) (h : (u, v) ∈ loweringSplits w) :
23    u.Sublist w ∧ v.Sublist w := by
24  induction w generalizing u v with
25  | nil =>
26    simp only [loweringSplits, List.mem_singleton, Prod.mk.injEq] at h
27    rcases h with ⟨rfl, rfl⟩
28    exact ⟨List.Sublist.refl _, List.Sublist.refl _⟩
29  | cons a w ih =>
30    simp only [loweringSplits, List.mem_append, List.mem_map] at h
31    rcases h with ⟨⟨s, t⟩, hst, he⟩ | ⟨⟨s, t⟩, hst, he⟩
32    · cases he
33      exact ⟨(ih s t hst).1.cons_cons a, (ih s t hst).2.cons a⟩
34    · cases he
35      exact ⟨(ih s t hst).1.cons a, (ih s t hst).2.cons_cons a⟩

69/-- The exact lowering coproduct follows directly from the literal tensor
70Leibniz rule. It holds for arbitrary physical tensors. -/
71theorem loweringWord_tensorJoin_split (Ω : TensorRegister n (Fin d))
72    (Ψ : TensorRegister m (Fin d)) (w : List (PositiveRoot d)) :
73    loweringWord (tensorJoin Ω Ψ) w =
74      ((loweringSplits w).map (fun uv => tensorJoin (loweringWord Ω uv.1)
75        (loweringWord Ψ uv.2))).sum := by
76  induction w with
77  | nil => simp [loweringWord, loweringSplits]
78  | cons a w ih =>
79    rw [loweringWord, ih, map_list_sum]
80    simp only [loweringSplits, List.map_append, List.sum_append, List.map_map,
81      Function.comp_def, loweringWord, collectiveGenerator_tensorJoin]
82    exact List.sum_map_add

84/-- Exact coefficient of one root assignment after the oscillator word
85normalizations on source and both target factors. -/
86def normalizedSplitCoefficient (mu nu : Fin d → ℕ)
87    (w u v : List (PositiveRoot d)) : ℂ :=
88  normalizedLoweringScale (fun a => mu a + nu a) w /
89    (normalizedLoweringScale mu u * normalizedLoweringScale nu v)

91theorem normalizedLoweringWord_tensorJoin_split
92    (Ω : TensorRegister n (Fin d)) (Ψ : TensorRegister m (Fin d))
93    (mu nu : Fin d → ℕ)
94    (hmu : ∀ a : PositiveRoot d, 0 < (mu a.val.1 : ℝ) - mu a.val.2)
95    (hnu : ∀ a : PositiveRoot d, 0 < (nu a.val.1 : ℝ) - nu a.val.2)
96    (w : List (PositiveRoot d)) :
97    normalizedLoweringWord (tensorJoin Ω Ψ) (fun a => mu a + nu a) w =
98      ((loweringSplits w).map (fun uv => normalizedSplitCoefficient mu nu w uv.1 uv.2 •
99        tensorJoin (normalizedLoweringWord Ω mu uv.1)
100          (normalizedLoweringWord Ψ nu uv.2))).sum := by
101  rw [normalizedLoweringWord_eq_scale, loweringWord_tensorJoin_split, List.smul_sum]
102  simp only [List.map_map, Function.comp_def]
103  apply congrArg List.sum
104  apply List.map_congr_left
105  intro uv _
106  simp only [normalizedLoweringWord_eq_scale, tensorJoin_smul_left, tensorJoin_smul_right,
107    smul_smul, normalizedSplitCoefficient]
108  congr 1
109  field_simp [normalizedLoweringScale_ne_zero mu hmu uv.1,
110    normalizedLoweringScale_ne_zero nu hnu uv.2]
