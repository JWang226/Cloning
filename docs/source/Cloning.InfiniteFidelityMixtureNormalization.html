13theorem norm_finiteMixture (q : I → ℝ) (hq : ∀ i, 0 ≤ q i)
14    (A : I → PositiveTraceClass H) :
15    ‖(finiteMixture q hq A).1‖ = ∑ i, q i * ‖(A i).1‖ := by
16  rw [positive_norm_eq_trace]

23def normalized (A fallback : PositiveTraceClass H) : PositiveTraceClass H :=
24  if h : 0 < ‖A.1‖ then scale ⟨‖A.1‖⁻¹, inv_nonneg.mpr (norm_nonneg _)⟩ A else fallback

26theorem norm_normalized (A fallback : PositiveTraceClass H) (hf : ‖fallback.1‖ = 1) :
27    ‖(normalized A fallback).1‖ = 1 := by
28  by_cases h : 0 < ‖A.1‖
29  · rw [normalized, dif_pos h, norm_scale]
30    exact inv_mul_cancel₀ (ne_of_gt h)
31  · simpa only [normalized, dif_neg h] using hf

33theorem scale_normalized (A fallback : PositiveTraceClass H) :
34    scale ⟨‖A.1‖, norm_nonneg _⟩ (normalized A fallback) = A := by
35  apply Subtype.ext
36  by_cases h : 0 < ‖A.1‖
37  · simp only [normalized, dif_pos h, scale, smul_smul]
38    change ((‖A.1‖ : ℂ) * (‖A.1‖⁻¹ : ℝ)) • A.1 = A.1
39    rw [← Complex.ofReal_mul, mul_inv_cancel₀ (ne_of_gt h), Complex.ofReal_one, one_smul]
40  · have hz : ‖A.1‖ = 0 := le_antisymm (le_of_not_gt h) (norm_nonneg _)
41    have hA : A.1 = 0 := norm_eq_zero.mp hz
42    change (‖A.1‖ : ℂ) • (normalized A fallback).1 = A.1
43    rw [hz, Complex.ofReal_zero, zero_smul, hA]

45theorem normalized_finiteMixture_of_pos
46    (q : I → ℝ) (hq : ∀ i, 0 ≤ q i) (A : I → PositiveTraceClass H)
47    (hA : ∀ i, ‖(A i).1‖ = 1) (fallback : PositiveTraceClass H)
48    (hs : 0 < ∑ i, q i) :
49    normalized (finiteMixture q hq A) fallback =
50      finiteMixture (fun i ↦ q i / ∑ a, q a)
51        (fun i ↦ div_nonneg (hq i) hs.le) A := by
52  have hn : ‖(finiteMixture q hq A).1‖ = ∑ i, q i := by
53    simp only [norm_finiteMixture, hA, mul_one]
54  rw [normalized, dif_pos (hn ▸ hs)]
55  apply Subtype.ext
56  change ((‖(finiteMixture q hq A).1‖⁻¹ : ℝ) : ℂ) • (finiteMixture q hq A).1 = _
57  rw [hn]
58  simp only [finiteMixture_val, Finset.smul_sum, smul_smul]
59  apply Finset.sum_congr rfl
60  intro i hi
