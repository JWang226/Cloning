17/-- Transfer concrete scalar target brackets to a finite-sample fidelity
18whose fixed-gain limit is known. -/
19theorem physical_target_brackets_of_tendsto {F : ℝ → ℕ → ℝ} {f : ℝ → ℝ}
20    {a η : ℝ} (ha : 0 < a) (hη : 0 < η) (hη1 : η < 1)
21    (hlim : Tendsto (fun δ => (1-f δ)/δ^2) (𝓝[>] (0:ℝ)) (𝓝 a))
22    (hphysical : ∀ δ, 0 < δ → Tendsto (F δ) atTop (𝓝 (f δ))) :
23    ∀ᶠ ε in 𝓝[>] (0:ℝ), ∀ᶠ n in atTop,
24      1-ε < F ((1-η)*Real.sqrt (ε/a)) n ∧
25      F ((1+η)*Real.sqrt (ε/a)) n < 1-ε := by
26  filter_upwards [fidelity_target_brackets ha hη hη1 hlim,
27    self_mem_nhdsWithin] with ε hε hεpos
28  have hroot : 0 < Real.sqrt (ε/a) := Real.sqrt_pos.mpr (div_pos hεpos ha)
29  exact ((hphysical _ (mul_pos (sub_pos.mpr hη1) hroot)).eventually_const_lt hε.1).and
30    ((hphysical _ (mul_pos (by linarith) hroot)).eventually_lt_const hε.2)

32/-- Actual optimum over all physical CPTP maps at a fixed simple spectrum:
33the predicted gain straddles the squared-fidelity target. -/
34theorem known_physical_target_brackets {k : ℕ} (hk : 1 ≤ k)
35    (p : SimpleSpectrum (k+1)) (m : ℝ → ℕ → ℕ)
36    (hgain : ∀ δ, 0 < δ →
37      Tendsto (fun n => (m δ n : ℝ)/n) atTop (𝓝 (1+δ)))
38    {η : ℝ} (hη : 0 < η) (hη1 : η < 1) :
39    ∀ᶠ ε in 𝓝[>] (0:ℝ), ∀ᶠ n in atTop,
40      1-ε < TensorCloning.knownSpectrumValue n
41        (m ((1-η)*Real.sqrt (ε/knownCoefficient p)) n) p ^ 2 ∧
42      TensorCloning.knownSpectrumValue n
43        (m ((1+η)*Real.sqrt (ε/knownCoefficient p)) n) p ^ 2 < 1-ε := by
44  apply physical_target_brackets_of_tendsto
45    (F := fun δ n => TensorCloning.knownSpectrumValue n (m δ n) p ^ 2)
46    (knownCoefficient_pos (by omega) p)
47    hη hη1 (known_infidelity_div_sq_tendsto p)
48  intro δ hδ
49  exact (TensorCloning.knownSpectrumValue_tendsto p (m δ) (1+δ)
50    (by linarith) (hgain δ hδ)).pow 2

52/-- The actual state-independent PCT channel has the larger small-error
53coefficient at every full-rank density matrix with distinct eigenvalues. -/
54theorem pct_physical_target_brackets {k : ℕ} (hk : 1 ≤ k)
55    (ρ : MatrixFidelity.State (Fin (k+1))) (hpos : ρ.matrix.PosDef)
56    (hsimple : Function.Injective ρ.positive.isHermitian.eigenvalues)
57    (r : ℝ → ℕ → ℕ)
58    (hgain : ∀ δ, 0 < δ →
59      Tendsto (fun n => ((n+r δ n : ℕ):ℝ)/n) atTop (𝓝 (1+δ)))
60    {η : ℝ} (hη : 0 < η) (hη1 : η < 1) :
61    let p := PCTPhysicalState.densitySpectrum ρ hpos hsimple
62    ∀ᶠ ε in 𝓝[>] (0:ℝ), ∀ᶠ n in atTop,
63      1-ε < ((PCTPhysicalState.physicalPCTOutput ρ n
64        (r ((1-η)*Real.sqrt (ε/pctCoefficient p)) n)).rootFidelity
65        (PCTPhysicalState.tensorState ρ
66          (n+r ((1-η)*Real.sqrt (ε/pctCoefficient p)) n))) ^ 2 ∧
67      ((PCTPhysicalState.physicalPCTOutput ρ n
68        (r ((1+η)*Real.sqrt (ε/pctCoefficient p)) n)).rootFidelity
69        (PCTPhysicalState.tensorState ρ
70          (n+r ((1+η)*Real.sqrt (ε/pctCoefficient p)) n))) ^ 2 < 1-ε := by
71  dsimp only
72  apply physical_target_brackets_of_tendsto
73    (F := fun δ n => ((PCTPhysicalState.physicalPCTOutput ρ n (r δ n)).rootFidelity
74      (PCTPhysicalState.tensorState ρ (n+r δ n))) ^ 2)
75    (pctCoefficient_pos (by omega) (PCTPhysicalState.densitySpectrum ρ hpos hsimple))
76    hη hη1 (pct_infidelity_div_sq_tendsto _)
77  intro δ hδ
78  exact (PCTPhysicalState.physical_pct_fidelity hk ρ hpos hsimple (r δ)
79    (1+δ) (by linarith) (hgain δ hδ)).pow 2

81/-- The explicit state-independent universal channel attains the lower
82small-error bracket uniformly over every eigenbasis. -/
83theorem universal_physical_target_achievable {k : ℕ} (hk : 1 ≤ k)
84    (p : SimpleSpectrum (k+1)) (m : ℝ → ℕ → ℕ)
85    (hgain : ∀ δ, 0 < δ →
86      Tendsto (fun n => (m δ n : ℝ)/n) atTop (𝓝 (1+δ)))
87    {η : ℝ} (hη : 0 < η) (hη1 : η < 1) :
88    ∀ᶠ ε in 𝓝[>] (0:ℝ), ∀ᶠ n in atTop,
89      ∀ U : unitary (Matrix (Fin (k+1)) (Fin (k+1)) ℂ),
90      let M := m ((1-η)*Real.sqrt (ε/universalCoefficient p)) n
91      1-ε < TensorCloning.spectrumPayoff n M
92        (TensorCloning.universalChannel n M k) p U ^ 2 := by
93  have hid : Tendsto (fun ε:ℝ => ε) (𝓝[>] 0) (𝓝 0) := nhdsWithin_le_nhds
94  have heps : ∀ᶠ ε in 𝓝[>] (0:ℝ), ε < 1 := hid.eventually_lt_const (by norm_num)
95  filter_upwards [universal_target_brackets (by omega) p hη hη1,
96    self_mem_nhdsWithin, heps] with ε hbracket hε hε1
97  let δ := (1-η)*Real.sqrt (ε/universalCoefficient p)
98  have hδ : 0 < δ := mul_pos (sub_pos.mpr hη1)
99    (Real.sqrt_pos.mpr (div_pos hε (universalCoefficient_pos (by omega) p)))
100  have hpos := universalValue_pos (by linarith : 1 < 1+δ) p
101  have hs : (Real.sqrt (1-ε))^2 = 1-ε := Real.sq_sqrt (by linarith)
102  have hmargin : 0 < universalValue (1+δ) p-Real.sqrt (1-ε) := by
103    have h := hbracket.1
104    change 1-ε < universalValue (1+δ) p^2 at h
105    nlinarith [Real.sqrt_nonneg (1-ε)]
106  have h := TensorCloning.eventually_universalChannel_payoff_lower hk {p}
107    (isCompact_singleton) (m δ) (1+δ) (by linarith) (hgain δ hδ)
108    (universalValue (1+δ) p-Real.sqrt (1-ε)) hmargin
109  filter_upwards [h] with n hn U
110  have hh := hn p (Set.mem_singleton p) U
111  dsimp only
112  change 1-ε < TensorCloning.spectrumPayoff n (m δ n)
113    (TensorCloning.universalChannel n (m δ n) k) p U ^ 2
114  have hnonneg := TensorCloning.spectrumPayoff_nonneg n (m δ n)

118/-- Two-sided small-error brackets for the exact prescribed universal
119protocol, uniformly over the entire unitary orbit. -/
120theorem universal_physical_target_brackets {k : ℕ} (hk : 1 ≤ k)
121    (p : SimpleSpectrum (k+1)) (m : ℝ → ℕ → ℕ)
122    (hgain : ∀ δ, 0 < δ →
123      Tendsto (fun n => (m δ n : ℝ)/n) atTop (𝓝 (1+δ)))
124    {η : ℝ} (hη : 0 < η) (hη1 : η < 1) :
125    ∀ᶠ ε in 𝓝[>] (0:ℝ), ∀ᶠ n in atTop,
126      ∀ U : unitary (Matrix (Fin (k+1)) (Fin (k+1)) ℂ),
127      let L := m ((1-η)*Real.sqrt (ε/universalCoefficient p)) n
128      let H := m ((1+η)*Real.sqrt (ε/universalCoefficient p)) n
129      1-ε < TensorCloning.spectrumPayoff n L
130        (TensorCloning.prescribedUniversalChannel n L k) p U ^ 2 ∧
131      TensorCloning.spectrumPayoff n H
132        (TensorCloning.prescribedUniversalChannel n H k) p U ^ 2 < 1-ε := by
133  have hlim (δ : ℝ) (hδ : 0 < δ) :
134      Tendsto (fun n => TensorCloning.spectrumPayoff n (m δ n)
135        (TensorCloning.prescribedUniversalChannel n (m δ n) k) p 1)
136        atTop (𝓝 (universalValue (1+δ) p)) := by
137    apply Metric.tendsto_nhds.mpr
138    intro ε hε
139    filter_upwards [TensorCloning.eventually_uniform_prescribedUniversalChannel_payoff
140      hk {p} isCompact_singleton (m δ) (1+δ) (by linarith) (hgain δ hδ) ε hε] with n hn
141    simpa only [Real.dist_eq] using hn p (Set.mem_singleton p) 1
142  have h := physical_target_brackets_of_tendsto
143    (F := fun δ n => TensorCloning.spectrumPayoff n (m δ n)
144      (TensorCloning.prescribedUniversalChannel n (m δ n) k) p 1 ^ 2)
145    (universalCoefficient_pos (by omega) p) hη hη1
146    (universal_infidelity_div_sq_tendsto p) (fun δ hδ => (hlim δ hδ).pow 2)
147  filter_upwards [h] with ε hε
148  filter_upwards [hε] with n hn U
149  have hc (M : ℕ) : TensorCloning.spectrumPayoff n M
150      (TensorCloning.prescribedUniversalChannel n M k) p U =
151      TensorCloning.spectrumPayoff n M (TensorCloning.prescribedUniversalChannel n M k) p 1 :=
152    TensorCloning.spectrumPayoff_unitary_of_covariant n M _ p U
153      (TensorCloning.prescribedUniversalChannel_covariant n M k U
154        (Unitary.star_mul_self_of_mem U.property))
155  dsimp only
156  simpa only [hc] using hn
