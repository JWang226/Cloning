14theorem wernerScale_eq_vacuum (n m s : ℕ) (hnm : n ≤ m) :
15    wernerScale n m s = Cloning.WernerAsymptotics.coefficient n m s 0 := by
16  have h := Cloning.Occupation.werner_dimension_identity n m s m hnm
17  have hc : (Nat.choose m n : ℝ) ≠ 0 := by exact_mod_cast (Nat.choose_pos hnm).ne'
18  simpa only [wernerScale, Cloning.WernerAsymptotics.coefficient, Nat.sub_zero,
19    div_self hc, mul_one] using h

21theorem wernerScale_tendsto (m : ℕ → ℕ) {γ : ℝ} (hγ : 1 < γ)
22    (h : Tendsto (fun n => (m n : ℝ)/n) atTop (𝓝 γ)) (s : ℕ) :
23    Tendsto (fun n => wernerScale n (m n) s) atTop (𝓝 ((1/γ)^s)) := by
24  have ht := Cloning.WernerAsymptotics.coefficient_tendsto m hγ h s 0
25  simp only [pow_zero, mul_one] at ht
26  apply ht.congr'
27  filter_upwards [Cloning.WernerAsymptotics.eventually_add_le m hγ h 0] with n hn
28  exact (wernerScale_eq_vacuum n (m n) s (by simpa using hn)).symm

30theorem supportFactor_tendsto (m : ℕ → ℕ) {γ : ℝ} (hγ : 1 < γ)
31    (h : Tendsto (fun n => (m n : ℝ)/n) atTop (𝓝 γ)) (small large : ℕ) :
32    Tendsto (fun n => supportFactor n (m n) small large) atTop
33      (𝓝 ((1/γ)^large / (1/γ)^small)) :=
34  (wernerScale_tendsto m hγ h large).div (wernerScale_tendsto m hγ h small)
35    (pow_ne_zero _ (one_div_ne_zero (by linarith)))

37theorem sqrt_supportFactor_tendsto (m : ℕ → ℕ) {γ : ℝ} (hγ : 1 < γ)
38    (h : Tendsto (fun n => (m n : ℝ)/n) atTop (𝓝 γ)) (small large : ℕ) :
39    Tendsto (fun n => Real.sqrt (supportFactor n (m n) small large)) atTop
40      (𝓝 (γ ^ (-((large : ℝ)-(small : ℝ))/2))) := by
41  have hg : 0 < γ := by linarith
42  have ht := (supportFactor_tendsto m hγ h small large).sqrt
43  have he : Real.sqrt ((1/γ)^large / (1/γ)^small) =
44      γ ^ (-((large : ℝ)-(small : ℝ))/2) := by
45    rw [Real.sqrt_eq_rpow, ← Real.rpow_natCast, ← Real.rpow_natCast,
46      ← Real.rpow_sub (one_div_pos.mpr hg), ← Real.rpow_mul (one_div_pos.mpr hg).le]
47    rw [one_div, Real.inv_rpow hg.le, ← Real.rpow_neg hg.le]
48    congr 1
49    ring
50  exact he ▸ ht
