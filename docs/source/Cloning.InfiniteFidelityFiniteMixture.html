23theorem sum_rootFidelity_le (s : Finset ι) (A B : ι → PositiveTraceClass H) :
24    (∑ i ∈ s, (A i).rootFidelity (B i)) ≤ (finsetSum s A).rootFidelity (finsetSum s B) := by
25  induction s using Finset.induction_on with
26  | empty => simpa only [Finset.sum_empty] using rootFidelity_nonneg (finsetSum ∅ A) (finsetSum ∅ B)

28    rw [Finset.sum_insert hi]

31      (finsetSum s A).1 (finsetSum s B).1 (A i).2 (B i).2 (finsetSum s A).2 (finsetSum s B).2
