12def weylDenominator (d : ℕ) : MvPolynomial (Fin d) ℂ :=

18theorem weylDenominator_factor (r : PositiveRoot d) :

24/-- Product rule in a form retaining the exact missing-factor product. -/
25theorem pderiv_finset_prod {ι : Type*} [DecidableEq ι] (s : Finset ι)
26    (f : ι → MvPolynomial (Fin d) ℂ) (a : Fin d) :
27    pderiv a (∏ i ∈ s, f i) = ∑ i ∈ s, pderiv a (f i) * ∏ j ∈ s.erase i, f j := by
28  induction s using Finset.induction_on with
29  | empty => simp
30  | @insert i s hi ih =>
31    rw [Finset.prod_insert hi, pderiv_mul, ih, Finset.sum_insert hi,
32      Finset.erase_insert hi]
33    congr 1
34    rw [Finset.mul_sum]
35    apply Finset.sum_congr rfl
36    intro j hj
37    have hji : j ≠ i := ne_of_mem_of_not_mem hj hi
38    rw [Finset.erase_insert_of_ne hji.symm, Finset.prod_insert (by simp [hi])]
39    ring

49theorem weylDenominator_euler_cross (f : MvPolynomial (Fin d) ℂ) :
50    (∑ a : Fin d, (X a * pderiv a (weylDenominator d)) * (X a * pderiv a f)) =

52        (X r.val.1 * (X r.val.1 * pderiv r.val.1 f) -
