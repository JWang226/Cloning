14theorem finiteMixture_rootFidelity_lower_trimmed (q r : ι → ℝ)
15    (hq : ∀ i, 0 ≤ q i) (hr : ∀ i, 0 ≤ r i) (hrsum : ∑ i, r i ≤ 1)
16    (A B : ι → PositiveTraceClass H) (keep : ι → Prop) (c : ℝ)
17    (hc0 : 0 ≤ c) (hc1 : c ≤ 1)
18    (hgood : ∀ i, keep i → c ≤ (A i).rootFidelity (B i)) :
19    c * (∑ i, Real.sqrt (q i) * Real.sqrt (r i)) -
20        Real.sqrt (∑ i, if keep i then 0 else q i) ≤
21      (finiteMixture q hq A).rootFidelity (finiteMixture r hr B) := by
22  let w : ι → ℝ := fun i ↦ Real.sqrt (q i) * Real.sqrt (r i)
23  have hw (i : ι) : 0 ≤ w i := mul_nonneg (Real.sqrt_nonneg _) (Real.sqrt_nonneg _)
24  have hpoint (i : ι) : c * w i ≤ w i * (A i).rootFidelity (B i) +
25      (if keep i then 0 else w i) := by
26    by_cases hi : keep i
27    · simp only [if_pos hi, add_zero]
28      simpa only [mul_comm c] using mul_le_mul_of_nonneg_left (hgood i hi) (hw i)
29    · simp only [if_neg hi]
30      have hnon := mul_nonneg (hw i) (rootFidelity_nonneg (A i) (B i))
31      have hle := mul_le_mul_of_nonneg_right hc1 (hw i)
32      nlinarith
33  have hsum := Finset.sum_le_sum (fun i (_ : i ∈ Finset.univ) ↦ hpoint i)
34  rw [← Finset.mul_sum, Finset.sum_add_distrib] at hsum
35  have hbad : (∑ i, if keep i then 0 else w i) ≤
36      Real.sqrt (∑ i, if keep i then 0 else q i) := by
37    calc
38      _ = ∑ i, Real.sqrt (if keep i then 0 else q i) * Real.sqrt (r i) := by
39        apply Finset.sum_congr rfl
40        intro i hi
41        by_cases hk : keep i <;> simp [hk, w]
42      _ ≤ Real.sqrt (∑ i, if keep i then 0 else q i) * Real.sqrt (∑ i, r i) :=
43        Real.sum_sqrt_mul_sqrt_le Finset.univ (fun i ↦ by split_ifs; exact le_rfl; exact hq i) hr
44      _ ≤ Real.sqrt (∑ i, if keep i then 0 else q i) * 1 :=
45        mul_le_mul_of_nonneg_left (by simpa only [Real.sqrt_one] using Real.sqrt_le_sqrt hrsum)
46          (Real.sqrt_nonneg _)
47      _ = _ := mul_one _
48  have hconc := finiteMixture_rootFidelity_lower q r hq hr A B
49  change (∑ i, w i * (A i).rootFidelity (B i)) ≤ _ at hconc
50  change c * (∑ i, w i) - _ ≤ _
