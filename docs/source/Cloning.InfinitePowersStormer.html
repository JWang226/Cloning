6/-!
7# Powers–Størmer inequality in the analytic trace-class setting
8 
9The square roots of positive trace-class operators are compared in Hilbert–Schmidt
10norm. The proof uses the positive common majorant `B + (A - B)⁺` and positivity
11of Hilbert–Schmidt trace pairings, without assuming square-root continuity.
12-/

24/-- Analytic trace of a product of two Hilbert–Schmidt operators. -/
25def productTrace (R S : H →L[ℂ] H) (hR : IsHilbertSchmidt R)
26    (hS : IsHilbertSchmidt S) : ℂ :=
27  trace (R * S) (isTraceClass_mul_of_isHilbertSchmidt hR hS)

29lemma productTrace_comm {R S : H →L[ℂ] H} (hR : IsHilbertSchmidt R)
30    (hS : IsHilbertSchmidt S) : productTrace R S hR hS = productTrace S R hS hR := by
31  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
32  unfold productTrace
33  rw [trace_eq_of_hilbertBasis _ b, trace_eq_of_hilbertBasis _ b]
34  exact tsum_diagonal_mul_eq_tsum_diagonal_swap b b hR hS

36lemma productTrace_sub_left {R S T : H →L[ℂ] H} (hR : IsHilbertSchmidt R)
37    (hS : IsHilbertSchmidt S) (hT : IsHilbertSchmidt T) :
38    productTrace (R - S) T (isHilbertSchmidt_sub hR hS) hT =
39      productTrace R T hR hT - productTrace S T hS hT := by
40  have heq : TraceClass.ofOperator ((R - S) * T)
41      (isTraceClass_mul_of_isHilbertSchmidt (isHilbertSchmidt_sub hR hS) hT) =
42      TraceClass.ofOperator (R * T) (isTraceClass_mul_of_isHilbertSchmidt hR hT) -
43      TraceClass.ofOperator (S * T) (isTraceClass_mul_of_isHilbertSchmidt hS hT) := by
44    apply Subtype.ext
45    exact sub_mul R S T
46  have htrace := congrArg (traceCLM (H := H)) heq
47  simpa only [map_sub, traceCLM_apply, productTrace] using htrace

49lemma productTrace_sub_right {R S T : H →L[ℂ] H} (hR : IsHilbertSchmidt R)
50    (hS : IsHilbertSchmidt S) (hT : IsHilbertSchmidt T) :
51    productTrace R (S - T) hR (isHilbertSchmidt_sub hS hT) =
52      productTrace R S hR hS - productTrace R T hR hT := by
53  rw [productTrace_comm hR (isHilbertSchmidt_sub hS hT),
54    productTrace_sub_left hS hT hR, productTrace_comm hS hR,
55    productTrace_comm hT hR]

63lemma productTrace_self_eq_hilbertSchmidt_square {R : H →L[ℂ] H}
64    (hR : IsHilbertSchmidt R) (hRstar : IsSelfAdjoint R)
65    {w : Set H} (b : HilbertBasis w ℂ H) :

70/-- The common positive majorant whose trace excess is exactly the trace norm
71of the difference. This is built from the actual Jordan decomposition. -/
72lemma exists_positive_traceClass_majorant {A B : H →L[ℂ] H}
73    (hA : 0 ≤ A) (hB : 0 ≤ B) (hTA : IsTraceClass A) (hTB : IsTraceClass B) :
74    ∃ C : TraceClass H, 0 ≤ C.1 ∧ A ≤ C.1 ∧ B ≤ C.1 ∧
75      2 * (trace C.1 C.2).re - (trace A hTA).re - (trace B hTB).re =
76        ‖TraceClass.ofOperator A hTA - TraceClass.ofOperator B hTB‖ := by
77  let a := TraceClass.ofOperator A hTA
78  let b := TraceClass.ofOperator B hTB
79  let d : TraceClass H := a - b
80  have hd : IsSelfAdjoint d.1 := (IsSelfAdjoint.of_nonneg hA).sub
81    (IsSelfAdjoint.of_nonneg hB)
82  let p := TraceClass.positivePart d hd
83  let q := TraceClass.negativePart d hd
84  have hp : 0 ≤ p.1 := TraceClass.positivePart_nonneg d hd
85  have hq : 0 ≤ q.1 := TraceClass.negativePart_nonneg d hd
86  refine ⟨b + p, add_nonneg hB hp, ?_, ?_, ?_⟩
87  · change A ≤ B + (A - B)⁺
88    calc
89      A = B + (A - B) := by abel
90      _ ≤ B + (A - B)⁺ := add_le_add (le_refl B) (CFC.le_posPart hd)
91  · exact le_add_of_nonneg_right hp
92  · have hdecomp : p - q = a - b := TraceClass.positivePart_sub_negativePart d hd
93    have ht := congrArg (fun X : TraceClass H => (traceCLM X).re) hdecomp
94    simp only [map_sub, Complex.sub_re] at ht
95    have hm : ‖p‖ + ‖q‖ = ‖d‖ := TraceClass.norm_positivePart_add_norm_negativePart d hd
96    rw [TraceClass.norm_eq_trace_re_of_nonneg p hp,
97      TraceClass.norm_eq_trace_re_of_nonneg q hq] at hm
98    change (traceCLM p).re + (traceCLM q).re = ‖a - b‖ at hm
99    change 2 * (traceCLM (b + p)).re - (traceCLM a).re - (traceCLM b).re = ‖a - b‖
100    rw [map_add, Complex.add_re]
101    linarith

103/-- Positivity of three trace pairings bounds the square-root overlap below
104by the traces and the trace of any common positive majorant. -/
105lemma sqrt_overlap_ge_of_majorant {A B C : H →L[ℂ] H}
106    (hA : 0 ≤ A) (hB : 0 ≤ B) (hC : 0 ≤ C)
107    (hTA : IsTraceClass A) (hTB : IsTraceClass B) (hTC : IsTraceClass C)
108    (hAC : A ≤ C) (hBC : B ≤ C) :
109    (trace A hTA).re + (trace B hTB).re - (trace C hTC).re ≤
110      (productTrace (CFC.sqrt A) (CFC.sqrt B)
111        (isHilbertSchmidt_sqrt hA hTA) (isHilbertSchmidt_sqrt hB hTB)).re := by
112  have ha := isHilbertSchmidt_sqrt hA hTA
113  have hb := isHilbertSchmidt_sqrt hB hTB
114  have hc := isHilbertSchmidt_sqrt hC hTC
115  have hca := isHilbertSchmidt_sub hc ha
116  have hcb := isHilbertSchmidt_sub hc hb
117  have hca0 : 0 ≤ CFC.sqrt C - CFC.sqrt A := sub_nonneg.mpr (CFC.sqrt_le_sqrt A C hAC)
118  have hcb0 : 0 ≤ CFC.sqrt C - CFC.sqrt B := sub_nonneg.mpr (CFC.sqrt_le_sqrt B C hBC)
119  have hmain := trace_re_mul_nonneg_of_isHilbertSchmidt hca hcb hca0 hcb0
120  have hleft := trace_re_mul_nonneg_of_isHilbertSchmidt hca ha hca0 (CFC.sqrt_nonneg A)
121  have hright := trace_re_mul_nonneg_of_isHilbertSchmidt hcb hb hcb0 (CFC.sqrt_nonneg B)
122  change 0 ≤ (productTrace _ _ hca hcb).re at hmain
123  change 0 ≤ (productTrace _ _ hca ha).re at hleft
124  change 0 ≤ (productTrace _ _ hcb hb).re at hright
125  rw [productTrace_sub_left hc ha hcb, productTrace_sub_right hc hc hb,
126    productTrace_sub_right ha hc hb, productTrace_comm ha hc] at hmain
127  rw [productTrace_sub_left hc ha ha] at hleft
128  rw [productTrace_sub_left hc hb hb] at hright
129  rw [productTrace_sqrt_self hC hTC] at hmain
130  rw [productTrace_sqrt_self hA hTA] at hleft
131  rw [productTrace_sqrt_self hB hTB] at hright
132  simp only [Complex.sub_re] at hmain hleft hright
133  linarith

135/-- Trace-class membership of a difference, derived from the Banach-space operations. -/
136lemma isTraceClass_sub {A B : H →L[ℂ] H} (hA : IsTraceClass A) (hB : IsTraceClass B) :
137    IsTraceClass (A - B) :=
138  (TraceClass.ofOperator A hA - TraceClass.ofOperator B hB).2

140/-- Powers–Størmer: the squared Hilbert–Schmidt distance between positive
141square roots is bounded by the actual analytic trace norm of the difference. -/
142theorem sqrt_hilbertSchmidt_square_le_traceNorm {A B : H →L[ℂ] H}
143    (hA : 0 ≤ A) (hB : 0 ≤ B) (hTA : IsTraceClass A) (hTB : IsTraceClass B)
144    {w : Set H} (b : HilbertBasis w ℂ H) :
145    (∑' i : w, ‖(CFC.sqrt A - CFC.sqrt B) (b i)‖ ^ 2) ≤
146      traceNorm (A - B) (isTraceClass_sub hTA hTB) := by
147  obtain ⟨C, hC, hAC, hBC, hmass⟩ := exists_positive_traceClass_majorant hA hB hTA hTB
148  have hoverlap := sqrt_overlap_ge_of_majorant hA hB hC hTA hTB C.2 hAC hBC
149  have ha := isHilbertSchmidt_sqrt hA hTA
150  have hb := isHilbertSchmidt_sqrt hB hTB
151  have hd := isHilbertSchmidt_sub ha hb
152  have hdstar : IsSelfAdjoint (CFC.sqrt A - CFC.sqrt B) :=
153    (IsSelfAdjoint.of_nonneg (CFC.sqrt_nonneg A)).sub
154      (IsSelfAdjoint.of_nonneg (CFC.sqrt_nonneg B))
155  rw [← productTrace_self_eq_hilbertSchmidt_square hd hdstar b,
156    productTrace_sub_left ha hb hd, productTrace_sub_right ha ha hb,
157    productTrace_sub_right hb ha hb, productTrace_comm hb ha]
158  rw [productTrace_sqrt_self hA hTA, productTrace_sqrt_self hB hTB]
159  simp only [Complex.sub_re]
160  change 2 * (trace C.1 C.2).re - (trace A hTA).re - (trace B hTB).re =
161    traceNorm (A - B) (isTraceClass_sub hTA hTB) at hmass
162  linarith
