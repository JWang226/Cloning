5/-!
6# Binomial-to-Poisson convergence of coherent-product number coefficients
7 
8This is the concrete probability calculation in `coherent-product-limit`.
9Both laws are defined explicitly; positivity, exact normalization, pointwise
10convergence, full `ℓ¹` convergence, and square-root amplitude convergence are
11proved. No probability normalization or convergence is assumed.
12 
13The occupation basis, the quantum embedding, and the identification of these
14coefficients with coherent/product vectors remain separate constructions.
15-/

24/-- The number distribution of the length-`L` coherent-product approximation.
25Natural binomial coefficients automatically pad this distribution by zero
26beyond occupation `L`. The convention at `L = 0` is the vacuum law. -/
27def binomialWeight (t : ℝ) (L j : ℕ) : ℝ :=
28  (L.choose j : ℝ) * (t / L) ^ j / (1 + t / L) ^ L
29 
30/-- The Poisson number distribution of a coherent vector with squared
31amplitude `t`. -/
32def poissonWeight (t : ℝ) (j : ℕ) : ℝ :=
33  Real.exp (-t) * t ^ j / j.factorial

35theorem binomialWeight_nonneg {t : ℝ} (ht : 0 ≤ t) (L j : ℕ) :
36    0 ≤ binomialWeight t L j := by

40theorem poissonWeight_nonneg {t : ℝ} (ht : 0 ≤ t) (j : ℕ) :

45theorem binomialWeight_eq_zero_of_lt (t : ℝ) {L j : ℕ} (hLj : L < j) :
46    binomialWeight t L j = 0 := by
47  simp [binomialWeight, Nat.choose_eq_zero_of_lt hLj]

49/-- Exact normalization, including the zero-length and zero-amplitude cases. -/
50theorem binomialWeight_sum {t : ℝ} (ht : 0 ≤ t) (L : ℕ) :
51    ∑ j ∈ Finset.range (L + 1), binomialWeight t L j = 1 := by
52  have hden : (1 + t / L : ℝ) ^ L ≠ 0 := by positivity
53  have hbin : (∑ j ∈ Finset.range (L + 1), (L.choose j : ℝ) * (t / L) ^ j) =
54      (1 + t / L) ^ L := by
55    simpa only [one_pow, mul_one, mul_comm, add_comm] using
56      (add_pow (t / L) (1 : ℝ) L).symm
57  simp only [binomialWeight, ← Finset.sum_div, hbin, div_self hden]
58 
59theorem binomialWeight_hasSum {t : ℝ} (ht : 0 ≤ t) (L : ℕ) :
60    HasSum (binomialWeight t L) 1 := by
61  rw [← binomialWeight_sum ht L]
62  apply hasSum_sum_of_ne_finset_zero
63  intro j hj
64  exact binomialWeight_eq_zero_of_lt t
65    (by simpa only [Finset.mem_range, not_lt, Nat.succ_le_iff] using hj)

71/-- The scaled binomial coefficient converges to the corresponding exponential
72series coefficient. -/
73theorem tendsto_binomial_numerator (t : ℝ) (j : ℕ) :
74    Tendsto (fun L : ℕ => (L.choose j : ℝ) * (t / L) ^ j) atTop
75      (𝓝 (t ^ j / j.factorial)) := by
76  apply ProbabilityTheory.tendsto_choose_mul_pow_atTop
77  change Tendsto (fun L : ℕ => (L : ℝ) * (t / L)) atTop (𝓝 t)
78  apply tendsto_const_nhds.congr'
79  filter_upwards [eventually_ge_atTop 1] with L hL
80  have hL0 : (L : ℝ) ≠ 0 := by exact_mod_cast (Nat.ne_of_gt hL)
81  field_simp

83/-- The actual binomial probabilities converge pointwise to the Poisson law. -/
84theorem binomialWeight_tendsto (t : ℝ) (j : ℕ) :
85    Tendsto (fun L => binomialWeight t L j) atTop (𝓝 (poissonWeight t j)) := by
86  have h := (tendsto_binomial_numerator t j).div (Real.tendsto_one_add_div_pow_exp t)
87    (Real.exp_ne_zero t)
88  change Tendsto (fun L => binomialWeight t L j) atTop
89    (𝓝 (t ^ j / j.factorial / Real.exp t)) at h
90  have he : t ^ j / j.factorial / Real.exp t = poissonWeight t j := by
91    unfold poissonWeight
92    rw [Real.exp_neg]
93    ring
94  rwa [he] at h

96/-- Full `ℓ¹` convergence of the concrete number distributions. -/
97theorem binomialWeight_l1_tendsto {t : ℝ} (ht : 0 ≤ t) :
98    Tendsto (fun L => CountableScheffe.l1Distance (binomialWeight t L) (poissonWeight t))
99      atTop (𝓝 0) :=
100  CountableScheffe.discrete_scheffe _ _
101    (binomialWeight_nonneg ht) (poissonWeight_nonneg ht)
102    (binomialWeight_hasSum ht) (poissonWeight_hasSum ht) (binomialWeight_tendsto t)

104/-- Square-root amplitude convergence in the coherent-product calculation. -/
105theorem sqrt_binomialWeight_tendsto {t : ℝ} (ht : 0 ≤ t) :

107      Real.sqrt (poissonWeight t j)) ^ 2) atTop (𝓝 0) :=
