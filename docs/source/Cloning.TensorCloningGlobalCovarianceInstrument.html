4/-! Covariance of a finite rectangular instrument, with all dependent Hilbert
5spaces kept abstract during the finite-sum algebra. -/
6noncomputable section

8namespace Cloning.TensorCloning

14  [∀ i, NormedAddCommGroup (E i)] [∀ i, InnerProductSpace ℂ (E i)] [∀ i, CompleteSpace (E i)]

17theorem finite_instrument_covariant
18    (V : ∀ i, H →L[ℂ] E i) (Φ : ∀ i, κ → TraceClass (E i) →ₗ[ℂ] TraceClass K)
19    (q : ι → κ → ℂ) (S : H →L[ℂ] H) (T : K →L[ℂ] K) (W : ∀ i, E i →L[ℂ] E i)
20    (hin : ∀ i (A : TraceClass H), conjugationLinearMap (V i) (conjugationLinearMap S A) =
21      conjugationLinearMap (W i) (conjugationLinearMap (V i) A))
22    (hout : ∀ i j (A : TraceClass (E i)), Φ i j (conjugationLinearMap (W i) A) =
23      conjugationLinearMap T (Φ i j A)) (A : TraceClass H) :
24    (∑ i, ∑ j, q i j • Φ i j (conjugationLinearMap (V i) (conjugationLinearMap S A))) =
25      conjugationLinearMap T (∑ i, ∑ j, q i j • Φ i j (conjugationLinearMap (V i) A)) := by
26  simp only [map_sum, map_smul]
27  apply Finset.sum_congr rfl
28  intro i _
29  apply Finset.sum_congr rfl
30  intro j _
31  rw [hin i A, hout]

33end Cloning.TensorCloning
