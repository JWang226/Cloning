23/-- Arbitrary positive trace-class idlers obey the exact least-noise moment
24bound, including idlers of deficient trace and idlers with coherences. -/
25lemma idler_witness_moment_le (σ : TraceClass Fock) (hσ : 0 ≤ σ.1)
26    {q x : ℝ} (hq0 : 0 < q) (hqx : q < x) (hx1 : x < 1) :
27    (tracePairing ((idlerChannel x (hq0.trans hqx).le hx1).toLinearMap σ)
28      (witnessOperator numberBasis q x)).re ≤
29      (trace σ.1 σ.2).re * Thermal.fidelity q x := by
30  apply single_number_law_witness_moment_le numberBasis _ hq0 hqx hx1
31    (a := fun l => (⟪numberBasis l, σ.1 (numberBasis l)⟫_ℂ).re)
32  · intro l
33    exact (σ.1.nonneg_iff_isPositive.mp hσ).re_inner_nonneg_right (numberBasis l)
34  · exact trace_real_hasSum_basis numberBasis σ
35  · intro k
36    exact idlerChannel_diagonal_re x (hq0.trans hqx).le hx1 σ k

38/-- Fidelity upper bound for the actual seeded bosonic output. -/
39lemma idler_fidelity_sq_le (σ : TraceClass Fock) (hσ : 0 ≤ σ.1)
40    {q x : ℝ} (hq0 : 0 < q) (hqx : q < x) (hx1 : x < 1) :
41    fidelity ((idlerChannel x (hq0.trans hqx).le hx1).toLinearMap σ).1
42      (vectorMixture numberBasis (Thermal.geometric q)).1
43      ((idlerChannel x (hq0.trans hqx).le hx1).map_nonneg σ hσ)
44      (vectorMixture_nonneg numberBasis numberBasis.orthonormal.norm_eq_one _
45        (Thermal.geometric_hasSum hq0.le (hqx.trans hx1)).summable
46        (Thermal.geometric_nonneg hq0.le (hqx.trans hx1).le))
47      ((idlerChannel x (hq0.trans hqx).le hx1).toLinearMap σ).2
48      (vectorMixture numberBasis (Thermal.geometric q)).2 ^ 2 ≤
49      (trace σ.1 σ.2).re * Thermal.fidelity q x ^ 2 := by
50  apply fidelity_sq_le_of_single_number_law numberBasis _ _ hq0 hqx hx1

55  · intro k

58/-- Every genuine idler density operator, with arbitrary coherences, has
59fidelity at most the quantum-limited thermal affinity. -/
60lemma idler_stateFidelity_le (σ : DensityState Fock)
61    {q x : ℝ} (hq0 : 0 < q) (hqx : q < x) (hx1 : x < 1) :
62    stateFidelity
63      ((idlerChannel x (hq0.trans hqx).le hx1).toPositiveTracePreservingMap.mapState σ)
64      (InfiniteDiagonalFidelity.geometricState numberBasis q hq0.le (hqx.trans hx1)) ≤
65      Thermal.fidelity q x := by
66  have h := idler_fidelity_sq_le (TraceClass.ofOperator σ.op σ.traceClass) σ.positive hq0 hqx hx1
67  change stateFidelity
68      ((idlerChannel x (hq0.trans hqx).le hx1).toPositiveTracePreservingMap.mapState σ)
69      (InfiniteDiagonalFidelity.geometricState numberBasis q hq0.le (hqx.trans hx1)) ^ 2 ≤
70    (trace σ.op σ.traceClass).re * Thermal.fidelity q x ^ 2 at h
71  rw [σ.trace_one, Complex.one_re, one_mul] at h
72  exact (sq_le_sq₀ (stateFidelity_nonneg _ _)
73    (Thermal.fidelity_pos hq0.le (hqx.trans hx1) (hq0.trans hqx).le hx1).le).mp h

87/-- Vacuum idler attains the quantum-limited thermal output as an operator
88identity in the actual trace-class Banach space. -/
89lemma idlerChannel_vacuum (x : ℝ) (hx0 : 0 ≤ x) (hx1 : x < 1) :
90    (idlerChannel x hx0 hx1).toLinearMap (vectorProjector (numberBasis 0)) =
91      vectorMixture numberBasis (Thermal.geometric x) := by
92  have hs := idlerChannel_hasSum x hx0 hx1 (vectorProjector (numberBasis 0))
93  have he (m : ℕ) : krausTerm (idlerKraus x hx0 hx1 m) (vectorProjector (numberBasis 0)) =
94      (Thermal.geometric x m : ℂ) • vectorProjector (numberBasis m) := by
95    rw [krausTerm_vectorProjector, idlerKraus_vacuum, vectorProjector_sqrt_smul]
96    exact Thermal.geometric_nonneg hx0 hx1.le m
97  simp_rw [he] at hs
98  exact hs.tsum_eq.symm

100/-- The universal seed-state upper bound is attained by the actual vacuum
101idler, whose channel output is the amplified thermal state. -/
102lemma idler_vacuum_stateFidelity (x q : ℝ) (hx0 : 0 ≤ x) (hx1 : x < 1)
103    (hq0 : 0 ≤ q) (hq1 : q < 1) :
104    stateFidelity
105      ((idlerChannel x hx0 hx1).toPositiveTracePreservingMap.mapState
106        (DensityState.pure (numberBasis 0) (numberBasis.orthonormal.norm_eq_one 0)))
107      (InfiniteDiagonalFidelity.geometricState numberBasis q hq0 hq1) =
108      Thermal.fidelity x q := by
109  have he :
110      (idlerChannel x hx0 hx1).toPositiveTracePreservingMap.mapState
111        (DensityState.pure (numberBasis 0) (numberBasis.orthonormal.norm_eq_one 0)) =
112      InfiniteDiagonalFidelity.geometricState numberBasis x hx0 hx1 := by
113    apply densityState_ext
114    exact congrArg Subtype.val (idlerChannel_vacuum x hx0 hx1)
115  rw [he]
116  exact InfiniteDiagonalFidelity.stateFidelity_geometric numberBasis x q hx0 hx1 hq0 hq1

118/-- Exact optimum over all actual idler density operators. This theorem is
119about the explicitly constructed seeded channel family; it does not assume
120or claim that every displacement-covariant channel has this representation. -/
121lemma isGreatest_idler_stateFidelity {q x : ℝ}
122    (hq0 : 0 < q) (hqx : q < x) (hx1 : x < 1) :
123    IsGreatest (Set.range (fun σ : DensityState Fock =>
124      stateFidelity
125        ((idlerChannel x (hq0.trans hqx).le hx1).toPositiveTracePreservingMap.mapState σ)
126        (InfiniteDiagonalFidelity.geometricState numberBasis q hq0.le (hqx.trans hx1))))
127      (Thermal.fidelity q x) := by
128  constructor
129  · refine ⟨DensityState.pure (numberBasis 0) (numberBasis.orthonormal.norm_eq_one 0), ?_⟩
130    dsimp only
131    rw [idler_vacuum_stateFidelity]
132    simp only [Thermal.fidelity, mul_comm]
133  · rintro _ ⟨σ, rfl⟩
134    exact idler_stateFidelity_le σ hq0 hqx hx1
