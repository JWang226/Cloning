15theorem canonicalEmbedding_adjoint_tensorOperator {n d : ℕ} (H : PhysicalHighestTensor n d)
16    (X : Matrix (Fin d) (Fin d) ℂ) :
17    H.canonicalEmbedding.toContinuousLinearMap.adjoint.comp (tensorOperator n X) =
18      (H.canonicalTensorOperator X).comp H.canonicalEmbedding.toContinuousLinearMap.adjoint := by
19  have he : (tensorOperator n Xᴴ).comp H.canonicalEmbedding.toContinuousLinearMap =
20      H.canonicalEmbedding.toContinuousLinearMap.comp (H.canonicalTensorOperator Xᴴ) := by
21    apply ContinuousLinearMap.ext
22    intro x
23    exact (H.canonicalEmbedding_tensorOperator Xᴴ x).symm
24  have ha := congrArg ContinuousLinearMap.adjoint he
25  simpa only [ContinuousLinearMap.adjoint_comp, tensorOperator_star,
26    PhysicalHighestTensor.canonicalTensorOperator, cyclicTensorOperator_star,
27    ContinuousLinearMap.adjoint_adjoint] using ha

29theorem embedded_copyIdentity_commutes {n d : ℕ} (H : PhysicalHighestTensor n d)
30    (X : Matrix (Fin d) (Fin d) ℂ) :
31    tensorOperator n X *
32        (conjugationLinearMap H.canonicalEmbedding.toContinuousLinearMap (copyIdentity H)).1 =
33      (conjugationLinearMap H.canonicalEmbedding.toContinuousLinearMap (copyIdentity H)).1 *
34        tensorOperator n X := by
35  have he : (tensorOperator n X).comp H.canonicalEmbedding.toContinuousLinearMap =
36      H.canonicalEmbedding.toContinuousLinearMap.comp (H.canonicalTensorOperator X) := by

47theorem inflatedTensor_commutes (n d : ℕ) (p : Fin d → ℝ) (X : Matrix (Fin d) (Fin d) ℂ) :
48    tensorOperator n X * (inflatedTensor n d p).1 = (inflatedTensor n d p).1 * tensorOperator n X := by
49  change tensorOperator n X * inclusionCLM (∑ i : SchurCopy n d, _) =
50    inclusionCLM (∑ i : SchurCopy n d, _) * tensorOperator n X
51  rw [map_sum, Finset.mul_sum, Finset.sum_mul]
52  apply Finset.sum_congr rfl
53  intro i _
54  rw [map_smul, mul_smul_comm, smul_mul_assoc, inclusionCLM_apply, embedded_copyIdentity_commutes]

56theorem inflatedTensor_unitary (n d : ℕ) (p : Fin d → ℝ)
57    (U : Matrix (Fin d) (Fin d) ℂ) (hU : Uᴴ * U = 1) :
58    conjugationLinearMap (tensorOperator n U) (inflatedTensor n d p) = inflatedTensor n d p := by
