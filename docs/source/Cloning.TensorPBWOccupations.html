14/-- The weighted height of a positive-root occupation function. -/
15def occupationHeight (k : PositiveRoot d → ℕ) : ℕ := ∑ a, a.height * k a

17/-- Occupations whose total root height is at most the fixed cutoff. -/
18def HeightOccupation (d R : ℕ) := {k : PositiveRoot d → ℕ // occupationHeight k ≤ R}

20theorem HeightOccupation.apply_le {R : ℕ} (k : HeightOccupation d R)
21    (a : PositiveRoot d) : k.val a ≤ R := by
22  have hpos := a.height_pos
23  have hsum : a.height * k.val a ≤ occupationHeight k.val := by
24    unfold occupationHeight
25    exact Finset.single_le_sum (f := fun r : PositiveRoot d => r.height * k.val r)
26      (fun _ _ => Nat.zero_le _) (Finset.mem_univ a)
27  exact (Nat.le_mul_of_pos_left _ hpos).trans (hsum.trans k.property)

29def HeightOccupation.toBox {R : ℕ} (k : HeightOccupation d R) :
30    PositiveRoot d → Fin (R + 1) := fun a => ⟨k.val a, Nat.lt_succ_of_le (k.apply_le a)⟩

32theorem HeightOccupation.toBox_injective {R : ℕ} :
33    Function.Injective (@HeightOccupation.toBox d R) := by
34  intro k l h

55/-- The roots occur in ascending `rootKey` order. In the operator convention
56used by `loweringWord`, the leftmost root acts last. -/
57def canonicalWord (k : PositiveRoot d → ℕ) : List (PositiveRoot d) :=
58  (occupationMultiset k).toList.mergeSort (fun a b => rootKey a ≤ rootKey b)

60@[simp] theorem canonicalWord_count (k : PositiveRoot d → ℕ) (a : PositiveRoot d) :
61    (canonicalWord k).count a = k a := by
62  rw [canonicalWord, (List.mergeSort_perm _ _).count_eq]
63  have h := Multiset.coe_count a (occupationMultiset k).toList
64  rw [Multiset.coe_toList] at h
65  convert h.symm.trans (occupationMultiset_count k a) using 1
66  simp only [List.count, beq_eq_decide]

68theorem canonicalWord_ordered (k : PositiveRoot d → ℕ) : RootOrdered (canonicalWord k) := by
69  have ht : ∀ a b c : PositiveRoot d, decide (rootKey a ≤ rootKey b) = true →
70      decide (rootKey b ≤ rootKey c) = true → decide (rootKey a ≤ rootKey c) = true := by
71    intro a b c hab hbc
72    exact decide_eq_true (le_trans (of_decide_eq_true hab) (of_decide_eq_true hbc))
73  have htotal : ∀ a b : PositiveRoot d,
74      (decide (rootKey a ≤ rootKey b) || decide (rootKey b ≤ rootKey a)) = true := by
75    intro a b
76    simpa only [Bool.or_eq_true, decide_eq_true_eq] using le_total (rootKey a) (rootKey b)
77  simpa only [RootOrdered, canonicalWord, decide_eq_true_eq] using
78    List.pairwise_mergeSort ht htotal (occupationMultiset k).toList

80theorem canonicalWord_perm_iff (k l : PositiveRoot d → ℕ) :

93theorem loweringHeight_eq_sum_count (w : List (PositiveRoot d)) :
