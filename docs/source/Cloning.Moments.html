10/-!
11# The order and moment steps of the Gaussian converse
12 
13This file proves the stochastic-coupling implication in
14`prop:least-noise-moment`, a finite diagonal version of the weighted fidelity
15bound, and the last limit argument of `prop:flat-prior-converse`.
16It does NOT construct the bosonic amplifier, identify its number distribution,
17or prove the compact-CP-map subsequence theorem. In particular, the coupling
18theorem is applicable to arbitrarily correlated idlers: no independence
19assumption is used once a coupling with coordinatewise domination is given.
20-/

27/-- A nonnegative antitone product decreases under coordinatewise domination. -/
28theorem antitone_product_le {ι : Type*} [Fintype ι]
29    (w : ι → ℕ → ℝ) (hw : ∀ i n, 0 ≤ w i n)
30    (hanti : ∀ i, Antitone (w i)) (x y : ι → ℕ)
31    (hxy : ∀ i, x i ≤ y i) :
32    (∏ i, w i (y i)) ≤ ∏ i, w i (x i) := by
33  exact Finset.prod_le_prod (fun i _ => hw i (y i))
34    (fun i _ => hanti i (hxy i))
35 
36/-- Adding any number of nonnegative occupation counts dominates the first count. -/
37theorem first_count_le_sum (x : ℕ → ℕ) (l : ℕ) :
38    x 0 ≤ ∑ j ∈ Finset.range (l + 1), x j := by
39  exact Finset.single_le_sum (fun _ _ => Nat.zero_le _)
40    (Finset.mem_range.mpr (Nat.zero_lt_succ l))

42/-- The pathwise part of the negative-binomial coupling used in the manuscript. -/
43theorem occupation_product_le {ι : Type*} [Fintype ι]
44    (w : ι → ℕ → ℝ) (hw : ∀ i n, 0 ≤ w i n)
45    (hanti : ∀ i, Antitone (w i)) (x : ι → ℕ → ℕ) (l : ι → ℕ) :
46    (∏ i, w i (∑ j ∈ Finset.range (l i + 1), x i j)) ≤
47      ∏ i, w i (x i 0) := by
48  exact antitone_product_le w hw hanti (fun i => x i 0)
49    (fun i => ∑ j ∈ Finset.range (l i + 1), x i j)
50    (fun i => first_count_le_sum (x i) (l i))

52/-- An actual Bochner-integral moment inequality, on an arbitrary probability
53space (indeed an arbitrary measure). Integrability is explicit. -/
54theorem coupled_moment_le {Ω ι : Type*} [MeasurableSpace Ω] [Fintype ι]
55    (μ : Measure Ω) (w : ι → ℕ → ℝ)
56    (hw : ∀ i n, 0 ≤ w i n) (hanti : ∀ i, Antitone (w i))
57    (x y : Ω → ι → ℕ) (hxy : ∀ᵐ ω ∂μ, ∀ i, x ω i ≤ y ω i)
58    (hx : Integrable (fun ω => ∏ i, w i (x ω i)) μ)
59    (hy : Integrable (fun ω => ∏ i, w i (y ω i)) μ) :
60    (∫ ω, ∏ i, w i (y ω i) ∂μ) ≤ ∫ ω, ∏ i, w i (x ω i) ∂μ := by
61  apply integral_mono_ae hy hx
62  filter_upwards [hxy] with ω hω
63  exact antitone_product_le w hw hanti (x ω) (y ω) hω

65/-- Correlated random idler occupations require no product assumption. -/
66theorem random_idler_moment_le {Ω ι : Type*} [MeasurableSpace Ω] [Fintype ι]
67    (μ : Measure Ω) (w : ι → ℕ → ℝ)
68    (hw : ∀ i n, 0 ≤ w i n) (hanti : ∀ i, Antitone (w i))
69    (x : Ω → ι → ℕ → ℕ) (l : Ω → ι → ℕ)
70    (hx : Integrable (fun ω => ∏ i, w i (x ω i 0)) μ)
71    (hy : Integrable
72      (fun ω => ∏ i, w i (∑ j ∈ Finset.range (l ω i + 1), x ω i j)) μ) :
73    (∫ ω, ∏ i, w i (∑ j ∈ Finset.range (l ω i + 1), x ω i j) ∂μ) ≤
74      ∫ ω, ∏ i, w i (x ω i 0) ∂μ := by
75  exact integral_mono hy hx (fun ω => occupation_product_le w hw hanti (x ω) (l ω))

77/-- Classical/diagonal weighted root-fidelity bound, with no normalization
78assumption. This is not the noncommutative trace-class version. -/
79theorem weighted_affinity_sq_le {ι : Type*} [Fintype ι]
80    (r t w : ι → ℝ) (hr : ∀ i, 0 ≤ r i) (ht : ∀ i, 0 ≤ t i)
81    (hw : ∀ i, 0 < w i) :
82    (∑ i, Real.sqrt (r i * t i)) ^ 2 ≤
83      (∑ i, r i * w i) * ∑ i, t i / w i := by
84  apply Finset.sum_sq_le_sum_mul_sum_of_sq_eq_mul
85  · intro i _
86    exact mul_nonneg (hr i) (hw i).le
87  · intro i _
88    exact div_nonneg (ht i) (hw i).le
89  · intro i _
90    rw [Real.sq_sqrt (mul_nonneg (hr i) (ht i))]
91    field_simp [ne_of_gt (hw i)]

93/-- Fidelity-from-moment conclusion, retaining the possibly deficient trace.
94The weighted inequality and least-noise moment are premises, not axioms. -/
95theorem fidelity_le_of_moment {f m c o t : ℝ}
96    (hc : 0 ≤ c) (ho : 0 ≤ o) (htrace : t ≤ c)
97    (hmoment : m ≤ t * o) (hweighted : f ^ 2 ≤ m * c * o) :
98    f ≤ c * o := by
99  have hmo : m ≤ c * o := hmoment.trans (mul_le_mul_of_nonneg_right htrace ho)
100  have hco : 0 ≤ c * o := mul_nonneg hc ho
101  have hsq : f ^ 2 ≤ (c * o) ^ 2 := by
102    calc
103      f ^ 2 ≤ m * c * o := hweighted
104      _ = m * (c * o) := by ring
105      _ ≤ (c * o) * (c * o) := mul_le_mul_of_nonneg_right hmo hco
106      _ = (c * o) ^ 2 := by ring
107  nlinarith [sq_nonneg (f - c * o)]

109/-- Last subsequence argument of the flat-prior converse. The analytic task of
110constructing convergent moment subsequences is explicitly outside this lemma. -/
111theorem flat_prior_limit_bound {f m : ℕ → ℝ} {F M c o t : ℝ}
112    (hf : Tendsto f atTop (nhds F)) (hm : Tendsto m atTop (nhds M))
113    (hc : 0 ≤ c) (ho : 0 ≤ o) (htrace : t ≤ c)
114    (hmoment : M ≤ t * o)
115    (hweighted : ∀ᶠ n in atTop, f n ^ 2 ≤ m n * c * o) : F ≤ c * o := by
116  have hlimit : F ^ 2 ≤ M * c * o :=
117    le_of_tendsto_of_tendsto (hf.pow 2) ((hm.mul_const c).mul_const o) hweighted
118  exact fidelity_le_of_moment hc ho htrace hmoment hlimit
