15lemma fidelity_continuity_left {A C B : H →L[ℂ] H}
16    (hA : 0 ≤ A) (hC : 0 ≤ C) (hB : 0 ≤ B)
17    (hTA : IsTraceClass A) (hTC : IsTraceClass C) (hTB : IsTraceClass B) :
18    |fidelity A B hA hB hTA hTB - fidelity C B hC hB hTC hTB| ≤
19      Real.sqrt (traceNorm (A - C) (InfinitePowersStormer.isTraceClass_sub hTA hTC)) *
20        Real.sqrt (trace B hTB).re := by
21  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
22  exact (fidelity_continuity_left_roots hA hC hB hTA hTC hTB b).trans
23    (mul_le_mul_of_nonneg_right (Real.sqrt_le_sqrt
24      (InfinitePowersStormer.sqrt_hilbertSchmidt_square_le_traceNorm hA hC hTA hTC b))
25      (Real.sqrt_nonneg _))

27lemma fidelity_continuity_right {A B D : H →L[ℂ] H}
28    (hA : 0 ≤ A) (hB : 0 ≤ B) (hD : 0 ≤ D)
29    (hTA : IsTraceClass A) (hTB : IsTraceClass B) (hTD : IsTraceClass D) :
30    |fidelity A B hA hB hTA hTB - fidelity A D hA hD hTA hTD| ≤
31      Real.sqrt (traceNorm (B - D) (InfinitePowersStormer.isTraceClass_sub hTB hTD)) *
32        Real.sqrt (trace A hTA).re := by
33  rw [fidelity_comm hA hB hTA hTB, fidelity_comm hA hD hTA hTD]
34  exact fidelity_continuity_left hB hD hA hTB hTD hTA

36lemma fidelity_continuity {A B C D : H →L[ℂ] H}
37    (hA : 0 ≤ A) (hB : 0 ≤ B) (hC : 0 ≤ C) (hD : 0 ≤ D)
38    (hTA : IsTraceClass A) (hTB : IsTraceClass B)
39    (hTC : IsTraceClass C) (hTD : IsTraceClass D) :
40    |fidelity A B hA hB hTA hTB - fidelity C D hC hD hTC hTD| ≤
41      Real.sqrt (traceNorm (A - C) (InfinitePowersStormer.isTraceClass_sub hTA hTC)) *
42        Real.sqrt (trace B hTB).re +
43      Real.sqrt (traceNorm (B - D) (InfinitePowersStormer.isTraceClass_sub hTB hTD)) *
44        Real.sqrt (trace C hTC).re := by
45  exact (abs_sub_le _ (fidelity C B hC hB hTC hTB) _).trans
46    (add_le_add (fidelity_continuity_left hA hC hB hTA hTC hTB)
47      (fidelity_continuity_right hC hB hD hTC hTB hTD))

49/-- The analytic trace distance of two actual density states. -/
50def stateTraceDistance (ρ σ : DensityState H) : ℝ :=
51  traceNorm (ρ.op - σ.op) (InfinitePowersStormer.isTraceClass_sub ρ.traceClass σ.traceClass)

53lemma stateTraceDistance_nonneg (ρ σ : DensityState H) : 0 ≤ stateTraceDistance ρ σ :=

73lemma stateFidelity_continuity (ρ σ ρ' σ' : DensityState H) :
74    |stateFidelity ρ σ - stateFidelity ρ' σ'| ≤
75      Real.sqrt (stateTraceDistance ρ ρ') + Real.sqrt (stateTraceDistance σ σ') := by
76  have h := fidelity_continuity ρ.positive σ.positive ρ'.positive σ'.positive
77    ρ.traceClass σ.traceClass ρ'.traceClass σ'.traceClass
78  simpa only [σ.trace_one, ρ'.trace_one, Complex.one_re, Real.sqrt_one, mul_one] using h

80lemma stateFidelity_continuity_of_traceNorm_le (ρ σ ρ' σ' : DensityState H)
81    {ε δ : ℝ} (hε : stateTraceDistance ρ ρ' ≤ ε) (hδ : stateTraceDistance σ σ' ≤ δ) :
82    |stateFidelity ρ σ - stateFidelity ρ' σ'| ≤ Real.sqrt ε + Real.sqrt δ :=

86lemma traceNorm_sub_eq_trace_sub {A G : H →L[ℂ] H}
87    (hTA : IsTraceClass A) (hTG : IsTraceClass G) (hGA : G ≤ A) :
88    traceNorm (A - G) (InfinitePowersStormer.isTraceClass_sub hTA hTG) =
89      (trace A hTA).re - (trace G hTG).re := by
90  have htrace : trace (A - G) (InfinitePowersStormer.isTraceClass_sub hTA hTG) =
91      trace A hTA - trace G hTG := by
92    have h := trace_add hTA (isTraceClass_smul (-1 : ℂ) hTG)
93    rw [trace_smul] at h
94    simpa only [neg_one_smul, neg_one_mul, sub_eq_add_neg] using h
95  rw [← trace_re_eq_traceNorm (sub_nonneg.mpr hGA), htrace, Complex.sub_re]

97/-- Deleting a positive contribution costs at most the square root of its actual trace mass,
98times the square root of the target trace. The retained operator need not be normalized. -/
99lemma fidelity_deletion_bound {A G B : H →L[ℂ] H}
100    (hA : 0 ≤ A) (hG : 0 ≤ G) (hB : 0 ≤ B)
101    (hTA : IsTraceClass A) (hTG : IsTraceClass G) (hTB : IsTraceClass B) (hGA : G ≤ A) :
102    |fidelity A B hA hB hTA hTB - fidelity G B hG hB hTG hTB| ≤
103      Real.sqrt ((trace A hTA).re - (trace G hTG).re) * Real.sqrt (trace B hTB).re := by
104  simpa only [traceNorm_sub_eq_trace_sub hTA hTG hGA] using
105    fidelity_continuity_left hA hG hB hTA hTG hTB
