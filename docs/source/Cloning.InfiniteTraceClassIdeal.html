16# Products of Hilbert–Schmidt operators
17 
18## i. Overview
19 
20A product of two Hilbert–Schmidt operators is trace class. Every trace-class operator `T` factors as
21`(U √|T|) √|T|` with both factors Hilbert–Schmidt, so trace-class operators form a two-sided ideal
22closed under sums and adjoints.
23 
24## ii. Key results
25 
26- `isTraceClass_mul_of_isHilbertSchmidt` : a product of two Hilbert–Schmidt operators is trace
27  class.
28- `isTraceClass_add` : trace-class operators are closed under addition.
29- `isTraceClass_mul_mul` : trace-class operators form a two-sided ideal.
30- `isTraceClass_star` : trace-class operators are closed under the adjoint.
31 
32-/

45/-- The diagonal of `star W * (R * S)` (for any contraction `W`) unwinds to a Hilbert–Schmidt
46Cauchy–Schwarz pairing. This is the pointwise identity feeding both the master product theorem and
47the general additive/ideal closure results below. -/
48lemma diagonal_star_mul_eq_inner {R S W : H →L[ℂ] H} {w : Set H} (b : HilbertBasis w ℂ H)
49    (i : w) :
50    ⟪b i, (star W * (R * S)) (b i)⟫_ℂ = ⟪(ContinuousLinearMap.adjoint R * W) (b i), S (b i)⟫_ℂ := by
51  rw [ContinuousLinearMap.star_eq_adjoint, ContinuousLinearMap.mul_def,
52    ContinuousLinearMap.comp_apply, ContinuousLinearMap.mul_def, ContinuousLinearMap.comp_apply,
53    ContinuousLinearMap.adjoint_inner_right W (b i) (R (S (b i))),
54    ← ContinuousLinearMap.adjoint_inner_left R (S (b i)) (W (b i)),
55    ContinuousLinearMap.mul_def, ContinuousLinearMap.comp_apply]

57/-- **A product of two Hilbert–Schmidt operators is trace class.** -/
58lemma isTraceClass_mul_of_isHilbertSchmidt {R S : H →L[ℂ] H}
59    (hR : IsHilbertSchmidt R) (hS : IsHilbertSchmidt S) : IsTraceClass (R * S) := by
60  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
61  refine ⟨w, b, ?_⟩
62  set W : H →L[ℂ] H := Polar.polarFactor (R * S) with hWdef
63  have hWnorm : ‖W‖ ≤ 1 := Polar.polarFactor_opNorm_le (R * S)
64  have hRstar : IsHilbertSchmidt (ContinuousLinearMap.adjoint R) := by
65    rcases hR with ⟨w₀, b₀, hb₀⟩
66    exact ⟨w₀, b₀, summable_norm_sq_adjoint_of_summable_norm_sq b₀ hb₀⟩
67  have hRW : IsHilbertSchmidt (ContinuousLinearMap.adjoint R * W) :=
68    isHilbertSchmidt_mul_right_of_opNorm_le_one hWnorm hRstar
69  have hRWb : Summable (fun i : w => ‖(ContinuousLinearMap.adjoint R * W) (b i)‖ ^ 2) :=
70    summable_norm_sq_apply_of_hilbertBasis w b hRW
71  have hSb : Summable (fun i : w => ‖S (b i)‖ ^ 2) := summable_norm_sq_apply_of_hilbertBasis w b hS
72  have hprod : Summable (fun i : w =>
73      ‖(ContinuousLinearMap.adjoint R * W) (b i)‖ * ‖S (b i)‖) :=
74    summable_norm_mul_of_square_sums _ _ hRWb hSb (fun i => norm_nonneg _) (fun i => norm_nonneg _)
75  have habs : Polar.absOperator (R * S) = star W * (R * S) := (Polar.star_polarFactor_mul_self
76      (R * S)).symm
77  have hpoint : ∀ i : w,
78      ⟪b i, CFC.abs (R * S) (b i)⟫_ℂ = ⟪(ContinuousLinearMap.adjoint R * W) (b i), S (b i)⟫_ℂ := by
79    intro i
80    show ⟪b i, Polar.absOperator (R * S) (b i)⟫_ℂ = _
81    rw [habs]
82    exact diagonal_star_mul_eq_inner b i
83  apply Summable.of_norm_bounded hprod
84  intro i
85  calc
86    ‖(⟪b i, CFC.abs (R * S) (b i)⟫_ℂ).re‖ ≤ ‖⟪b i, CFC.abs (R * S) (b i)⟫_ℂ‖ :=
87      Complex.abs_re_le_norm _
88    _ = ‖⟪(ContinuousLinearMap.adjoint R * W) (b i), S (b i)⟫_ℂ‖ := by rw [hpoint i]
89    _ ≤ ‖(ContinuousLinearMap.adjoint R * W) (b i)‖ * ‖S (b i)‖ := norm_inner_le_norm _ _

91/-- Every trace-class operator factors as a product of two Hilbert–Schmidt operators: the polar
92factor composed with the Hilbert–Schmidt square root of `|T|`, and that same square root again. -/
93lemma isHilbertSchmidt_polarFactor_mul_sqrt_abs_and_sqrt_abs {T : H →L[ℂ] H}
94    (hT : IsTraceClass T) :
95    IsHilbertSchmidt (Polar.polarFactor T * CFC.sqrt (CFC.abs T)) ∧
96      IsHilbertSchmidt (CFC.sqrt (CFC.abs T)) ∧
97      Polar.polarFactor T * CFC.sqrt (CFC.abs T) * CFC.sqrt (CFC.abs T) = T := by
98  have hS : IsHilbertSchmidt (CFC.sqrt (CFC.abs T)) := isHilbertSchmidt_sqrt_abs_of_isTraceClass hT
99  have hUS : IsHilbertSchmidt (Polar.polarFactor T * CFC.sqrt (CFC.abs T)) :=
100    isHilbertSchmidt_mul_left_of_opNorm_le_one (Polar.polarFactor_opNorm_le T) hS
101  refine ⟨hUS, hS, ?_⟩
102  rw [mul_assoc, CFC.sqrt_mul_sqrt_self (CFC.abs T) (CFC.abs_nonneg T)]
103  exact Polar.polarFactor_mul_absOperator T

105/-- Trace-class operators are closed under addition. -/
106lemma isTraceClass_add {T T' : H →L[ℂ] H} (hT : IsTraceClass T) (hT' : IsTraceClass T') :
107    IsTraceClass (T + T') := by
108  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
109  refine ⟨w, b, ?_⟩
110  set W : H →L[ℂ] H := Polar.polarFactor (T + T') with hWdef
111  have hWnorm : ‖W‖ ≤ 1 := Polar.polarFactor_opNorm_le (T + T')
112  have hsummable_conj : ∀ {X : H →L[ℂ] H}, IsTraceClass X →
113      Summable (fun i : w => ⟪(b i : H), (star W * X) (b i)⟫_ℂ) := by
114    intro X hX
115    obtain ⟨hUS, hS, hfactor⟩ := isHilbertSchmidt_polarFactor_mul_sqrt_abs_and_sqrt_abs hX
116    set R : H →L[ℂ] H := Polar.polarFactor X * CFC.sqrt (CFC.abs X) with hRdef
117    set S : H →L[ℂ] H := CFC.sqrt (CFC.abs X) with hSdef
118    have hRstar : IsHilbertSchmidt (ContinuousLinearMap.adjoint R) := by
119      rcases hUS with ⟨w₀, b₀, hb₀⟩
120      exact ⟨w₀, b₀, summable_norm_sq_adjoint_of_summable_norm_sq b₀ hb₀⟩
121    have hRW : IsHilbertSchmidt (ContinuousLinearMap.adjoint R * W) :=
122      isHilbertSchmidt_mul_right_of_opNorm_le_one hWnorm hRstar
123    have hRWb : Summable (fun i : w => ‖(ContinuousLinearMap.adjoint R * W) (b i)‖ ^ 2) :=
124      summable_norm_sq_apply_of_hilbertBasis w b hRW
125    have hSb : Summable (fun i : w => ‖S (b i)‖ ^ 2) :=
126      summable_norm_sq_apply_of_hilbertBasis w b hS
127    have hprod : Summable (fun i : w =>
128        ‖(ContinuousLinearMap.adjoint R * W) (b i)‖ * ‖S (b i)‖) :=
129      summable_norm_mul_of_square_sums _ _ hRWb hSb
130        (fun i => norm_nonneg _) (fun i => norm_nonneg _)
131    have hpoint : ∀ i : w, ⟪b i, (star W * X) (b i)⟫_ℂ =
132        ⟪(ContinuousLinearMap.adjoint R * W) (b i), S (b i)⟫_ℂ := by
133      intro i
134      rw [show X = R * S from hfactor.symm]
135      exact diagonal_star_mul_eq_inner b i
136    apply Summable.of_norm_bounded hprod
137    intro i
138    rw [hpoint i]
139    exact norm_inner_le_norm _ _
140  have hTsum := hsummable_conj hT
141  have hT'sum := hsummable_conj hT'
142  have habs : CFC.abs (T + T') = star W * (T + T') := (Polar.star_polarFactor_mul_self (T +
143      T')).symm
144  have hsplit : (fun i : w => ⟪b i, CFC.abs (T + T') (b i)⟫_ℂ) =
145      (fun i : w => ⟪b i, (star W * T) (b i)⟫_ℂ + ⟪b i, (star W * T') (b i)⟫_ℂ) := by
146    funext i
147    rw [habs]
148    show ⟪b i, (star W * (T + T')) (b i)⟫_ℂ = _
149    have hstep : (star W * (T + T')) (b i) = (star W * T) (b i) + (star W * T') (b i) := by
150      rw [ContinuousLinearMap.mul_def, ContinuousLinearMap.comp_apply,
151        ContinuousLinearMap.add_apply, map_add, ContinuousLinearMap.mul_def,
152        ContinuousLinearMap.comp_apply, ContinuousLinearMap.mul_def,
153        ContinuousLinearMap.comp_apply]
154    rw [hstep, inner_add_right]
155  have hsum : Summable (fun i : w =>
156      ⟪b i, (star W * T) (b i)⟫_ℂ + ⟪b i, (star W * T') (b i)⟫_ℂ) := hTsum.add hT'sum
157  have hsum0 : Summable (fun i : w => ⟪b i, CFC.abs (T + T') (b i)⟫_ℂ) := by
158    rw [hsplit]
159    exact hsum
160  exact hsum0.map Complex.reCLM.toAddMonoidHom Complex.reCLM.continuous

162/-- **Trace-class operators form a two-sided ideal**: `A T B` is trace class for trace-class `T` and
163bounded `A`, `B`. -/
164lemma isTraceClass_mul_mul {A B T : H →L[ℂ] H} (hT : IsTraceClass T) :
165    IsTraceClass (A * T * B) := by
166  obtain ⟨_, hS, hfactor⟩ := isHilbertSchmidt_polarFactor_mul_sqrt_abs_and_sqrt_abs hT
167  set R : H →L[ℂ] H := Polar.polarFactor T * CFC.sqrt (CFC.abs T) with hRdef
168  set S : H →L[ℂ] H := CFC.sqrt (CFC.abs T) with hSdef
169  have hUS : IsHilbertSchmidt R :=
170    isHilbertSchmidt_mul_left_of_opNorm_le_one (Polar.polarFactor_opNorm_le T) hS
171  have hAR : IsHilbertSchmidt (A * R) := isHilbertSchmidt_mul_left hUS
172  have hSB : IsHilbertSchmidt (S * B) := isHilbertSchmidt_mul_right hS
173  have heq : A * T * B = (A * R) * (S * B) := by
174    rw [show T = R * S from hfactor.symm]; simp only [mul_assoc]
175  rw [heq]
176  exact isTraceClass_mul_of_isHilbertSchmidt hAR hSB

178/-- Taking the adjoint preserves trace class. -/
179lemma isTraceClass_star {T : H →L[ℂ] H} (hT : IsTraceClass T) : IsTraceClass (star T) := by
180  obtain ⟨hR, hS, hfactor⟩ := isHilbertSchmidt_polarFactor_mul_sqrt_abs_and_sqrt_abs hT
181  have hstar : IsTraceClass (star (CFC.sqrt (CFC.abs T)) *
182      star (Polar.polarFactor T * CFC.sqrt (CFC.abs T))) :=
183    isTraceClass_mul_of_isHilbertSchmidt (isHilbertSchmidt_star hS) (isHilbertSchmidt_star hR)
184  rw [← hfactor]
185  simpa only [star_mul] using hstar
186 
187end
