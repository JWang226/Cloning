14/-- Strictly positive interior spectra approaching every nonnegative thermal
15spectrum, including any combination of exact vacuum modes. -/
16def positiveApprox (q : Fin d → ℝ) (n : ℕ) (i : Fin d) : ℝ :=
17  (q i + ((n : ℝ) + 1)⁻¹) / (1 + ((n : ℝ) + 1)⁻¹)
18 
19lemma positiveApprox_pos (q : Fin d → ℝ) (hq0 : ∀ i, 0 ≤ q i) (n : ℕ) (i : Fin d) :
20    0 < positiveApprox q n i := by
21  unfold positiveApprox
22  have h : 0 < ((n : ℝ) + 1)⁻¹ := by positivity
23  exact div_pos (add_pos_of_nonneg_of_pos (hq0 i) h) (by linarith)
24 
25lemma positiveApprox_lt_one (q : Fin d → ℝ) (hq1 : ∀ i, q i < 1) (n : ℕ) (i : Fin d) :
26    positiveApprox q n i < 1 := by
27  unfold positiveApprox
28  apply (div_lt_one (by positivity : 0 < 1 + ((n : ℝ) + 1)⁻¹)).mpr
29  linarith [hq1 i]
30 
31lemma positiveApprox_tendsto (q : Fin d → ℝ) (i : Fin d) :
32    Tendsto (fun n => positiveApprox q n i) atTop (𝓝 (q i)) := by
33  have h : Tendsto (fun n : ℕ => ((n : ℝ) + 1)⁻¹) atTop (𝓝 (0 : ℝ)) := by
34    simpa only [one_div] using (tendsto_one_div_add_atTop_nhds_zero_nat (𝕜 := ℝ))
35  simpa [positiveApprox] using (tendsto_const_nhds.add h).div (tendsto_const_nhds.add h)
36    (by norm_num : (1 : ℝ) + 0 ≠ 0)

42/-- Product thermal density operators vary continuously in trace norm all
43the way to zero mode parameters. -/
44theorem productThermal_tendsto {q : ℕ → Fin d → ℝ} {x : Fin d → ℝ}
45    (hq0 : ∀ n i, 0 ≤ q n i) (hq1 : ∀ n i, q n i < 1)
46    (hx0 : ∀ i, 0 ≤ x i) (hx1 : ∀ i, x i < 1)
47    (hlim : ∀ i, Tendsto (fun n => q n i) atTop (𝓝 (x i))) :
48    Tendsto (fun n => vectorMixture (numberBasis d) (productGeometric (q n))) atTop
49      (𝓝 (vectorMixture (numberBasis d) (productGeometric x))) := by
50  apply vectorMixture_tendsto_of_pointwise (numberBasis d) (numberBasis d).orthonormal.norm_eq_one
51    (fun n => productGeometric (q n)) (productGeometric x)
52    (fun n => productGeometric_nonneg (hq0 n) (hq1 n)) (productGeometric_nonneg hx0 hx1)
53    (fun n => productGeometric_hasSum (hq0 n) (hq1 n)) (productGeometric_hasSum hx0 hx1)
54  intro k
55  apply tendsto_finset_prod
56  intro i _
57  exact (tendsto_const_nhds.sub (hlim i)).mul ((hlim i).pow (k i))

84/-- Exact physical thermal amplification also covers vacuum modes; it follows
85by trace-norm continuity of the actual channel, not a formal Gaussian label. -/
86theorem gainChannel_productThermal_nonneg (g : ℝ) (hg : 1 < g) {q : Fin d → ℝ}
87    (hq0 : ∀ i, 0 ≤ q i) (hq1 : ∀ i, q i < 1) :
88    (gainChannel g hg).toLinearMap (vectorMixture (numberBasis d) (productGeometric q)) =
89      vectorMixture (numberBasis d) (productGeometric (fun i => Thermal.amplified g (q i))) := by
90  let qn := Thermal.positiveApprox q
91  have hn0 (n : ℕ) (i : Fin d) : 0 < qn n i := Thermal.positiveApprox_pos q hq0 n i
92  have hn1 (n : ℕ) (i : Fin d) : qn n i < 1 := Thermal.positiveApprox_lt_one q hq1 n i
93  have hX := productThermal_tendsto (fun n i => (hn0 n i).le) hn1 hq0 hq1
94    (Thermal.positiveApprox_tendsto q)
95  have hleft := (gainChannel g hg).toPositiveTracePreservingMap.continuous.tendsto
96    (vectorMixture (numberBasis d) (productGeometric q)) |>.comp hX
97  have hx0 (i : Fin d) : 0 ≤ Thermal.amplified g (q i) :=
98    (hq0 i).trans (Thermal.lt_amplified hg (hq1 i)).le
99  have hx1 (i : Fin d) : Thermal.amplified g (q i) < 1 :=
100    Thermal.amplified_lt_one (by linarith) (hq1 i)
101  have hright := productThermal_tendsto
102    (fun n i => ((hn0 n i).le.trans (Thermal.lt_amplified hg (hn1 n i)).le))
103    (fun n i => Thermal.amplified_lt_one (by linarith) (hn1 n i)) hx0 hx1
104    (fun i => by
105      unfold Thermal.amplified
106      exact tendsto_const_nhds.sub ((tendsto_const_nhds.sub (Thermal.positiveApprox_tendsto q i)).div_const g))
107  apply tendsto_nhds_unique hleft
108  convert hright using 1
109  funext n
110  exact gainChannel_productThermal g hg (hn0 n) (hn1 n)
