24theorem cutoffWord_perm_iff (R : ℕ) (i j : CutoffIndex d R) :

52theorem cutoffFrame_mem (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
53    (R : ℕ) (i : CutoffIndex d R) : cutoffFrame Ω mu R i ∈ cyclicCutoff Ω (R : ℤ) := by
54  have hspan : Submodule.span ℂ (Set.range (cutoffFrame Ω mu R)) ≤ cyclicCutoff Ω (R : ℤ) := by
55    rw [span_cutoffFrame_eq_raw]
56    apply Submodule.span_le.mpr
57    rintro x ⟨j, rfl⟩
58    exact cutoffRawFrame_mem Ω mu R j
59  exact hspan (Submodule.subset_span (Set.mem_range_self i))

61theorem span_cutoffFrame_eq_cutoff (Ω : TensorRegister n (Fin d))
62    (mu : Fin d → ℕ) (R : ℕ)
63    (hgap : ∀ a : PositiveRoot d, 0 < rootGap mu a) :
64    Submodule.span ℂ (Set.range (cutoffFrame Ω mu R)) = cyclicCutoff Ω (R : ℤ) := by
65  rw [span_cutoffFrame_eq_raw]
66  have hr : Set.range (cutoffRawFrame Ω mu R) =
67      Set.range (fun k : HeightOccupation d R => normalizedLoweringWord Ω mu (canonicalWord k.val)) := by
68    ext x
69    constructor
70    · rintro ⟨i, rfl⟩
71      exact ⟨cutoffOccupation d R i, rfl⟩
72    · rintro ⟨k, rfl⟩
73      obtain ⟨i, rfl⟩ := (cutoffOccupation d R).surjective k
74      exact ⟨i, rfl⟩
75  rw [hr, span_normalizedLoweringWord_eq Ω mu hgap, cyclicCutoff_eq_span_canonical]

96def cutoffOrthonormalBasis (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ) (R : ℕ)
97    (hgap : ∀ a : PositiveRoot d, 0 < rootGap mu a)
98    (hli : LinearIndependent ℂ (cutoffRawFrame Ω mu R)) :
99    OrthonormalBasis (CutoffIndex d R) ℂ (cyclicCutoff Ω (R : ℤ)) := by
100  have hon : Orthonormal ℂ (cutoffFrameVector Ω mu R) := by
101    rw [orthonormal_iff_ite]
102    intro i j
103    exact orthonormal_iff_ite.mp (gramSchmidtNormed_orthonormal (𝕜 := ℂ) hli) i j
104  have hspan : Submodule.span ℂ (Set.range (cutoffFrameVector Ω mu R)) = ⊤ :=
105    (Submodule.span_range_subtype_eq_top_iff (cyclicCutoff Ω (R : ℤ))

138      ∀ i, (b i : TensorRegister (∑ a, mu N a) (Fin d)) =
