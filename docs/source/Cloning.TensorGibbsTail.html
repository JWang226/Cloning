41theorem sectorHeightMass_le_envelope (p : Fin d → ℝ) (hp : ∀ a, 0 < p a)
42    (θ : ℝ) (hθ : 0 ≤ θ)
43    (hroot : ∀ r : PositiveRoot d, rootBoltzmann p r ≤ θ ^ r.height) (H : ℕ) :
44    sectorHeightMass Ω mu hweight hraise p H ≤ heightEnvelope d θ H := by
45  have hm : 0 < ∏ a, p a ^ mu a := Finset.prod_pos (fun a _ => pow_pos (hp a) _)
46  apply (div_le_iff₀ hm).mpr
47  have hh := sectorHeightTrace_le Ω mu hweight hraise p hp θ hθ hroot H
48  simpa only [heightEnvelope, mul_assoc, mul_left_comm, mul_comm] using hh

57theorem sectorHeightMass_tail_le (p : Fin d → ℝ) (hp : ∀ a, 0 < p a)

59    (hroot : ∀ r : PositiveRoot d, rootBoltzmann p r ≤ θ ^ r.height) (R : ℕ) :
