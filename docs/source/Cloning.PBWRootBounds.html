3/-!
4# Deriving local root bounds without assuming PBW independence
5 
6The fixed-height PBW argument must bound normalized root operators before it
7can use Gram matrices. Here that bound follows from adjointness, strict lowering
8of the annihilator through complete cutoff spaces, and the diagonal commutator
9estimate. In particular it does not assume a PBW basis or a Gram bound.
10-/

19/-- Adjointness transfers a local creator bound to an annihilator whose image
20lies in that local subspace. No bounded inverse or basis is required. -/
21theorem annihilator_norm_le_from_local_creator
22    (F A : H →L[ℂ] H) (S : Submodule ℂ H) (C : ℝ) (hC : 0 ≤ C)
23    (hadj : ∀ x y, ⟪F x, y⟫_ℂ = ⟪x, A y⟫_ℂ)
24    (hF : ∀ y ∈ S, ‖F y‖ ≤ C * ‖y‖) (x : H) (hx : A x ∈ S) :
25    ‖A x‖ ≤ C * ‖x‖ := by
26  have hsq : ‖A x‖ ^ 2 ≤ C * ‖A x‖ * ‖x‖ := by
27    calc
28      ‖A x‖ ^ 2 = (⟪F (A x), x⟫_ℂ).re := by
29        rw [hadj]
30        exact norm_sq_eq_re_inner (𝕜 := ℂ) (A x)
31      _ ≤ ‖⟪F (A x), x⟫_ℂ‖ := Complex.re_le_norm _
32      _ ≤ ‖F (A x)‖ * ‖x‖ := norm_inner_le_norm _ _
33      _ ≤ (C * ‖A x‖) * ‖x‖ := mul_le_mul_of_nonneg_right (hF _ hx) (norm_nonneg _)
34  have hnonneg : 0 ≤ C * ‖x‖ := mul_nonneg hC (norm_nonneg _)
35  nlinarith [norm_nonneg (A x)]

37/-- The diagonal canonical-commutator estimate gives the basic norm recurrence. -/
38theorem creator_norm_sq_le_of_diagonal
39    (F A : H →L[ℂ] H)
40    (hadj : ∀ x y, ⟪F x, y⟫_ℂ = ⟪x, A y⟫_ℂ)
41    (δ : ℝ) (x : H) (hdef : ‖A (F x) - F (A x) - x‖ ≤ δ * ‖x‖) :
42    ‖F x‖ ^ 2 ≤ ‖A x‖ ^ 2 + (1 + δ) * ‖x‖ ^ 2 := by
43  have hF : (⟪x, A (F x)⟫_ℂ).re = ‖F x‖ ^ 2 := by
44    rw [← hadj]
45    exact (norm_sq_eq_re_inner (𝕜 := ℂ) (F x)).symm
46  have hA : (⟪x, F (A x)⟫_ℂ).re = ‖A x‖ ^ 2 := by
47    calc
48      _ = (⟪F (A x), x⟫_ℂ).re := inner_re_symm (𝕜 := ℂ) x (F (A x))
49      _ = _ := by
50        rw [hadj]
51        exact (norm_sq_eq_re_inner (𝕜 := ℂ) (A x)).symm
52  have hinner : (⟪x, A (F x) - F (A x) - x⟫_ℂ).re =
53      ‖F x‖ ^ 2 - ‖A x‖ ^ 2 - ‖x‖ ^ 2 := by
54    have hxnorm : (⟪x, x⟫_ℂ).re = ‖x‖ ^ 2 := (norm_sq_eq_re_inner (𝕜 := ℂ) x).symm
55    simp only [inner_sub_right, Complex.sub_re, hF, hA, hxnorm]
56  have hbound : (⟪x, A (F x) - F (A x) - x⟫_ℂ).re ≤ δ * ‖x‖ ^ 2 := by
57    calc
58      _ ≤ ‖⟪x, A (F x) - F (A x) - x⟫_ℂ‖ := Complex.re_le_norm _
59      _ ≤ ‖x‖ * ‖A (F x) - F (A x) - x‖ := norm_inner_le_norm _ _
60      _ ≤ ‖x‖ * (δ * ‖x‖) := mul_le_mul_of_nonneg_left hdef (norm_nonneg _)
61      _ = _ := by ring
62  rw [hinner] at hbound
63  linarith

65/-- The local normalized creator bound is derived by induction through complete
66cutoff spaces: its squared norm on cutoff `r` is at most `(r+1)(1+δ)`.
67The annihilator must vanish on cutoff zero and lower every higher cutoff.
68This is the noncircular operator estimate used before PBW Gram convergence. -/
69theorem creator_norm_sq_le_cutoff
70    (F A : H →L[ℂ] H) (K : ℕ → Submodule ℂ H)
71    (hadj : ∀ x y, ⟪F x, y⟫_ℂ = ⟪x, A y⟫_ℂ)
72    (hzero : ∀ x ∈ K 0, A x = 0)
73    (hlower : ∀ r x, x ∈ K (r + 1) → A x ∈ K r)
74    (L : ℕ) (δ : ℝ) (hδ : 0 ≤ δ)
75    (hdiag : ∀ r, r ≤ L → ∀ x ∈ K r,
76      ‖A (F x) - F (A x) - x‖ ≤ δ * ‖x‖)
77    (r : ℕ) (hr : r ≤ L) (x : H) (hx : x ∈ K r) :
78    ‖F x‖ ^ 2 ≤ ((r : ℝ) + 1) * (1 + δ) * ‖x‖ ^ 2 := by
79  induction r generalizing x with
80  | zero =>
81      have hh := creator_norm_sq_le_of_diagonal F A hadj δ x (hdiag 0 hr x hx)
82      simpa [hzero x hx] using hh
83  | succ r ih =>
84      have hr' : r ≤ L := by omega
85      let C : ℝ := Real.sqrt (((r : ℝ) + 1) * (1 + δ))
86      have hC : 0 ≤ C := Real.sqrt_nonneg _
87      have hCsq : C ^ 2 = ((r : ℝ) + 1) * (1 + δ) := Real.sq_sqrt (by positivity)
88      have hlocal : ∀ y ∈ K r, ‖F y‖ ≤ C * ‖y‖ := by
89        intro y hy
90        have hybound := ih hr' y hy
91        have hpos : 0 ≤ C * ‖y‖ := mul_nonneg hC (norm_nonneg _)
92        have heq : (C * ‖y‖) ^ 2 = ((r : ℝ) + 1) * (1 + δ) * ‖y‖ ^ 2 := by
93          rw [mul_pow, hCsq]
94        nlinarith [norm_nonneg (F y)]
95      have hAbound := annihilator_norm_le_from_local_creator F A (K r) C hC hadj hlocal
96        x (hlower r x hx)
97      have hAsq : ‖A x‖ ^ 2 ≤ ((r : ℝ) + 1) * (1 + δ) * ‖x‖ ^ 2 := by
98        have hh := sq_le_sq₀ (norm_nonneg (A x)) (mul_nonneg hC (norm_nonneg x)) |>.mpr hAbound
99        simpa [mul_pow, hCsq] using hh
100      have hh := creator_norm_sq_le_of_diagonal F A hadj δ x (hdiag (r + 1) hr x hx)
101      simp only [Nat.cast_add, Nat.cast_one]
102      nlinarith
