21theorem rankFidelity_limsup_le (d k : ℕ) (P : Projector (d+1) k)
22    (m : ℕ → ℕ) (γ : ℝ) (hγ : 1<γ)
23    (hgain : Tendsto (fun n => (m n : ℝ)/n) atTop (𝓝 γ)) :
24    limsup (fun n => rankFidelity d k P n (m n)) atTop ≤
25      γ ^ (-((((d+1)*k : ℕ) : ℝ)/2)) *
26        (Real.sqrt (2*γ-1)/γ) ^ ((((d+1 : ℕ) : ℝ)-1)/2) := by
27  rw [← limsup_congr (eventually_rankFidelity d k P m hγ hgain)]
28  exact PhysicalFlatPCT.fidelity_limsup_le d k P (fun n => m n-n) γ hγ (addedCopies_ratio m hγ hgain)

38/-- One sample threshold works for all unknown projectors. -/
39theorem rankFidelity_uniform_upper (d k : ℕ) (m : ℕ → ℕ) (γ : ℝ) (hγ : 1<γ)
40    (hgain : Tendsto (fun n => (m n : ℝ)/n) atTop (𝓝 γ)) (ε : ℝ) (hε : 0<ε) :
41    ∀ᶠ n in atTop, ∀ P : Projector (d+1) k,
42      rankFidelity d k P n (m n) <
43        γ ^ (-((((d+1)*k : ℕ) : ℝ)/2)) *
44          (Real.sqrt (2*γ-1)/γ) ^ ((((d+1 : ℕ) : ℝ)-1)/2)+ε := by
45  have h := PhysicalFlatPCT.fidelity_uniform_upper d k (fun n => m n-n) γ hγ
46    (addedCopies_ratio m hγ hgain) ε hε
47  filter_upwards [h,eventually_output_gt_input m hγ hgain] with n hn hnm P
48  have he := extendChannel_payoff_of_gt (PhysicalFlatPCT.channel d k) n (m n) hnm
49    (state (by omega : 0<d+1) P)
50  change rankFidelity d k P n (m n)=PhysicalFlatPCT.fidelity d k P n (m n-n) at he
51  rw [he]
52  exact hn P
