79theorem trace_tensorOperator_eq_sum_blocks (X : Matrix (Fin d) (Fin d) ℂ) :
80    LinearMap.trace ℂ (TensorRegister n (Fin d)) (tensorOperator n X).toLinearMap =
81      ∑ i : Fin L.length, LinearMap.trace ℂ (L.get i).CanonicalSector
82        ((L.get i).canonicalTensorOperator X).toLinearMap := by
83  rw [LinearMap.trace_eq_sum_inner _ (physicalSchurBasis L hL hspan)]

95/-- The full physical diagonal tensor trace, including the zero-fold tensor. -/
96theorem trace_tensorOperator_diagonal (D : Fin d → ℂ) :
97    LinearMap.trace ℂ (TensorRegister n (Fin d))
98      (tensorOperator n (Matrix.diagonal D)).toLinearMap = (∑ a, D a) ^ n := by
99  rw [LinearMap.trace_eq_sum_inner _ (registerBasis (Fin n → Fin d)).toOrthonormalBasis]
100  have he : ∀ w : Fin n → Fin d,
101      ⟪(registerBasis (Fin n → Fin d)).toOrthonormalBasis w,
102        (tensorOperator n (Matrix.diagonal D)).toLinearMap
103          ((registerBasis (Fin n → Fin d)).toOrthonormalBasis w)⟫_ℂ = ∏ t, D (w t) := by
104    intro w
105    simp only [HilbertBasis.coe_toOrthonormalBasis]
106    change ⟪registerBasis (Fin n → Fin d) w,
107      tensorOperator n (Matrix.diagonal D) (registerBasis (Fin n → Fin d) w)⟫_ℂ = _
108    rw [tensorOperator_diagonal_basis, inner_smul_right]
109    have hh := orthonormal_iff_ite.mp (registerBasis (Fin n → Fin d)).orthonormal w w
110    simp only [ite_true] at hh
111    rw [hh, mul_one]
112  simp only [he]
113  rw [← Fintype.prod_sum (fun (_ : Fin n) a => D a)]
114  simp
