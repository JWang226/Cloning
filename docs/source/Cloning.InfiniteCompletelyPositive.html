5# Complete positivity of actual trace-class maps

24/-- Positivity of the operator on the finite Hilbert direct sum. -/
25def BlockPositive (A : ι → ι → TraceClass H) : Prop :=
26  ∀ x : ι → H, 0 ≤ ∑ i, ∑ j, ⟪x i, (A i j).1 (x j)⟫_ℂ

28/-- Complete positivity at every finite ancilla dimension. -/
29def IsCompletelyPositive (Φ : TraceClass H →ₗ[ℂ] TraceClass K) : Prop :=
30  ∀ n : ℕ, ∀ A : Fin n → Fin n → TraceClass H,
31    BlockPositive A → BlockPositive (fun i j => Φ (A i j))

34theorem nonneg_of_inner_nonneg (T : H →L[ℂ] H)
35    (h : ∀ x, 0 ≤ ⟪x, T x⟫_ℂ) : 0 ≤ T := by
36  apply T.nonneg_iff_isPositive.mpr
37  apply (ContinuousLinearMap.isPositive_iff_complex T).mpr
38  intro x
39  have hx := Complex.nonneg_iff.mp (h x)

51    0 ≤ (∑ i, ∑ j, (star (c i) * c j) • A i j : TraceClass H).1 := by

67theorem BlockPositive.traceMatrix {A : ι → ι → TraceClass H}
68    (hA : BlockPositive A) (c : ι → ℂ) :
69    0 ≤ ∑ i, ∑ j, star (c i) * c j * traceCLM (A i j) := by
70  let T : TraceClass H := ∑ i, ∑ j, (star (c i) * c j) • A i j
71  have hp := hA.scalarCompression c
72  have ht := trace_eq_traceNorm_of_nonneg hp T.2

166/-- A genuine infinite-dimensional CPTP map on the trace-class Banach space. -/
167structure QuantumChannel (H K : Type*)
168    [NormedAddCommGroup H] [InnerProductSpace ℂ H] [CompleteSpace H]
169    [NormedAddCommGroup K] [InnerProductSpace ℂ K] [CompleteSpace K]
170    extends PositiveTracePreservingMap H K where
171  completelyPositive : IsCompletelyPositive toLinearMap

175  toPositiveTracePreservingMap := Cloning.InfiniteTraceClass.projectionReplacement P hP σ
