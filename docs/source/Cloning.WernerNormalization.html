5/-!
6# Normalization of Werner's occupation law on the full Fock index set
7 
8We count fixed-total occupation vectors by stars and bars and pass from the
9finite shell identity to a normalized probability law on `Fin s → ℕ`.
10-/

18/-- Werner's eigenvalue, padded by zero beyond its finite occupation support. -/
19def occupationLaw (n m s : ℕ) (k : Fin s → ℕ) : ℝ :=
20  if (∑ i, k i) ≤ m - n then
21    Occupation.wernerWeight n m (s + 1) (∑ i, k i) else 0

23theorem occupationLaw_nonneg (n m s : ℕ) (k : Fin s → ℕ) :
24    0 ≤ occupationLaw n m s k := by
25  unfold occupationLaw Occupation.wernerWeight
26  split_ifs <;> positivity

28/-- Stars and bars for the actual occupation vectors, rather than grouped weights. -/
29theorem antidiagonalTuple_card (s t : ℕ) :
30    (Finset.Nat.antidiagonalTuple (s + 1) t).card = (t + s).choose s := by
31  have hlen : ∀ s t, (List.Nat.antidiagonalTuple (s + 1) t).length =
32      (t + s).choose s := by
33    intro s
34    induction s with
35    | zero => intro t; simp
36    | succ s ih =>
37      intro t
38      rw [List.Nat.antidiagonalTuple]
39      simp only [List.length_flatMap, List.length_map]
40      simp only [List.Nat.antidiagonal, List.map_map, Function.comp_def, ih]
41      rw [← List.sum_toFinset _ List.nodup_range]
42      simp only [List.toFinset_range]
43      rw [← Finset.sum_flip]
44      calc
45        _ = ∑ r ∈ range (t + 1), (r + s).choose s := by
46          apply sum_congr rfl
47          intro r hr
48          rw [Nat.sub_sub_self (Nat.lt_succ_iff.mp (mem_range.mp hr))]
49        _ = _ := Nat.sum_range_add_choose t s
50  exact hlen s t

52/-- The finite set containing exactly all padded occupation indices. -/
53def occupationSupport (s b : ℕ) : Finset (Fin s → ℕ) :=
54  (range (b + 1)).biUnion (Finset.Nat.antidiagonalTuple s)

56theorem mem_occupationSupport {s b : ℕ} {k : Fin s → ℕ} :
57    k ∈ occupationSupport s b ↔ (∑ i, k i) ≤ b := by
58  simp only [occupationSupport, mem_biUnion, mem_range,
59    Finset.Nat.mem_antidiagonalTuple]
60  constructor
61  · rintro ⟨t, ht, hkt⟩
62    omega
63  · intro hk
64    exact ⟨∑ i, k i, by omega, rfl⟩

66theorem occupationLaw_sum (n m s : ℕ) (hnm : n ≤ m) (hs : 1 ≤ s) :
67    (∑ k ∈ occupationSupport s (m - n), occupationLaw n m s k) = 1 := by
68  have hd : Set.PairwiseDisjoint (↑(range (m - n + 1)))
69      (Finset.Nat.antidiagonalTuple s) := by
70    intro a ha b hb hab
71    apply disjoint_left.mpr
72    intro k hka hkb
73    apply hab
74    exact (Finset.Nat.mem_antidiagonalTuple.mp hka).symm.trans
75      (Finset.Nat.mem_antidiagonalTuple.mp hkb)
76  rw [occupationSupport, sum_biUnion hd]
77  have hinner : ∀ t ∈ range (m - n + 1),
78      (∑ k ∈ Finset.Nat.antidiagonalTuple s t, occupationLaw n m s k) =
79        ((t + (s + 1) - 2).choose ((s + 1) - 2) : ℝ) *
80          Occupation.wernerWeight n m (s + 1) t := by
81    intro t ht
82    have ht' : t ≤ m - n := Nat.lt_succ_iff.mp (mem_range.mp ht)
83    have heq : (∑ k ∈ Finset.Nat.antidiagonalTuple s t, occupationLaw n m s k) =
84        (Finset.Nat.antidiagonalTuple s t).card *
85          Occupation.wernerWeight n m (s + 1) t := by
86      calc
87        _ = ∑ _k ∈ Finset.Nat.antidiagonalTuple s t,
88            Occupation.wernerWeight n m (s + 1) t := by
89          apply sum_congr rfl
90          intro k hk
91          simp only [occupationLaw, Finset.Nat.mem_antidiagonalTuple.mp hk,
92            if_pos ht']
93        _ = _ := by simp
94    rw [heq]
95    have hs' : s = (s - 1) + 1 := by omega
96    rw [hs', antidiagonalTuple_card]
97    congr 2
98  rw [sum_congr rfl hinner]
99  exact Occupation.werner_occupation_normalization n m (s + 1) hnm (by omega)

101theorem occupationLaw_hasSum (n m s : ℕ) (hnm : n ≤ m) (hs : 1 ≤ s) :
102    HasSum (occupationLaw n m s) 1 := by
