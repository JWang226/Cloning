18/-- Each Cartan coordinate occurs in precisely d-1 unordered root pairs. -/
19theorem sum_positiveRoot_endpoints (f : Fin d → MvPolynomial (Fin d) ℂ) :
20    (∑ r : PositiveRoot d, (f r.val.1 + f r.val.2)) =
21      ((d : ℂ) - 1) • (∑ a, f a) := by
22  have hs := sum_matrix_eq_diagonal_add_roots (fun a (_ : Fin d) => f a)
23  simp only [Finset.sum_const, Finset.card_univ, Fintype.card_fin, ← Finset.smul_sum] at hs
24  have hc : (d : ℕ) • (∑ a, f a) = (d : ℂ) • (∑ a, f a) := by
25    simp [Algebra.smul_def]
26  rw [hc] at hs
27  rw [sub_smul, one_smul]
28  exact eq_sub_of_add_eq' hs.symm
