17/-- The Weyl product is the dimension of the actual physical cyclic sector. -/
18theorem physicalSector_finrank_eq_dimensionProduct
19    (Ω : TensorRegister n (Fin d)) (μ : Fin d → ℕ) (hΩ : ‖Ω‖ = 1)
20    (hweight : ∀ a, collectiveGenerator n a a Ω = (μ a : ℂ) • Ω)
21    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0) :
22    (Module.finrank ℂ (cyclicSector Ω) : ℝ) = dimensionProduct μ := by
23  have hμ := highest_weight_antitone Ω μ hweight hraise hΩ
24  have hχ : physicalCharacterPolynomial Ω * denominator d =
25      alternant (fun i ↦ μ i + staircase i) := by
26    rw [mul_comm]
27    exact physicalCharacterPolynomial_weyl Ω μ hΩ hweight hraise
28  have he := eval_one_eq_dimensionProduct (physicalCharacterPolynomial Ω) μ hμ hχ
29  rw [physicalCharacterPolynomial_eval_one Ω (fun a ↦ (μ a : ℂ)) hweight hraise] at he
30  exact_mod_cast he
