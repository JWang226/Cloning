111/-- Actual height trace, bounded by the derived dimension and Boltzmann
112operator bound. The estimate is uniform in the highest weight and tensor size. -/
113theorem sectorHeightTrace_le (p : Fin d → ℝ) (hp : ∀ a, 0 < p a)
114    (θ : ℝ) (hθ : 0 ≤ θ)
115    (hroot : ∀ r : PositiveRoot d, rootBoltzmann p r ≤ θ ^ r.height) (H : ℕ) :
116    sectorHeightTrace Ω mu hweight hraise p H ≤
117      ((H+1 : ℕ) : ℝ)^Fintype.card (PositiveRoot d) * ((∏ a, p a ^ mu a) * θ^H) := by
118  have hC : 0 ≤ (∏ a, p a ^ mu a) * θ^H :=
119    mul_nonneg (Finset.prod_nonneg (fun a _ => pow_nonneg (hp a).le _)) (pow_nonneg hθ _)
120  have hi (i : Fin (Module.finrank ℂ (sectorHeightSpace Ω H))) :
121      (⟪sectorHeightFrame Ω ⟨H,i⟩,
122        sectorGibbsOperator Ω mu hweight hraise p (sectorHeightFrame Ω ⟨H,i⟩)⟫_ℂ).re ≤
123          (∏ a, p a ^ mu a) * θ^H := by
124    have hn : ‖sectorHeightFrame Ω ⟨H,i⟩‖ = 1 :=
125      (stdOrthonormalBasis ℂ (sectorHeightSpace Ω H)).norm_eq_one i
126    have hb := tensorOperator_diagonal_norm_le_on_height Ω mu hweight p hp θ hθ hroot H
127      (stdOrthonormalBasis ℂ (sectorHeightSpace Ω H) i).property
128    change ‖sectorGibbsOperator Ω mu hweight hraise p (sectorHeightFrame Ω ⟨H,i⟩)‖ ≤
129      ((∏ a, p a ^ mu a) * θ^H) * ‖sectorHeightFrame Ω ⟨H,i⟩‖ at hb
130    rw [hn, mul_one] at hb
131    exact (Complex.re_le_norm _).trans ((norm_inner_le_norm _ _).trans (by simpa [hn] using hb))
132  calc
133    _ ≤ ∑ i : Fin (Module.finrank ℂ (sectorHeightSpace Ω H)),
134        ((∏ a, p a ^ mu a) * θ^H) := Finset.sum_le_sum (fun i _ => hi i)
135    _ = (Module.finrank ℂ (sectorHeightSpace Ω H) : ℝ) * ((∏ a, p a ^ mu a) * θ^H) := by simp
136    _ ≤ _ := mul_le_mul_of_nonneg_right (by exact_mod_cast sectorHeightSpace_finrank_le Ω H) hC
