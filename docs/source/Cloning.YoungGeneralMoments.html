6# General-rank crossing-dimension moment limits
7 
8The expectations are under genuine PMFs identified with the explicit tableau
9formula at the flat spectrum.  Their tails are proved in the preceding files,
10and the dimension observable is the literal crossing-root Weyl product.
11No moment or concentration premise is assumed.  Representation-space
12dimension identification and Schur measurement identification remain separate.
13-/

105theorem mean_error_le {r N : ℕ} (P : PMF (Shape r N)) (p : Fin r → ℝ)
106    (ε δ B : ℝ) (hδ : 0 ≤ δ) (F : Shape r N → ℝ)
107    (hglobal : ∀ μ, |F μ - 1| ≤ B)
108    (hgood : ∀ μ, (¬∃ i, (N : ℝ) * ε ≤ |((μ i).val : ℝ) - N * p i|) → |F μ - 1| ≤ δ) :
109    (∑ μ, (P μ).toReal * |F μ - 1|) ≤ δ + B * tailProbability P p ε := by
110  have hpoint (μ : Shape r N) : (P μ).toReal * |F μ - 1| ≤ (P μ).toReal * δ +
111      B * (if ∃ i, (N : ℝ) * ε ≤ |((μ i).val : ℝ) - N * p i| then (P μ).toReal else 0) := by
112    split_ifs with ht
113    · have hb := mul_le_mul_of_nonneg_left (hglobal μ) (show 0 ≤ (P μ).toReal from ENNReal.toReal_nonneg)
114      have hz := mul_nonneg (show 0 ≤ (P μ).toReal from ENNReal.toReal_nonneg) hδ
115      nlinarith
116    · simp only [mul_zero, add_zero]
117      exact mul_le_mul_of_nonneg_left (hgood μ ht) ENNReal.toReal_nonneg
118  have hs : ∑ μ, (P μ).toReal = 1 := by
119    simpa only [tsum_fintype, YoungCompatibility.probability] using YoungCompatibility.tsum_probability P
120  have hsum := Finset.sum_le_sum (fun μ (_ : μ ∈ (Finset.univ : Finset (Shape r N))) ↦ hpoint μ)
121  simpa only [Finset.sum_add_distrib, ← Finset.sum_mul, ← Finset.mul_sum, hs, one_mul,
122    tailProbability] using hsum

166private theorem inverse_error_of_error {R D δ : ℝ} (hR : 0 < R) (hD : 0 < D)
167    (hδ : δ ≤ 1 / 2) (herr : |R / D - 1| ≤ δ) : |D / R - 1| ≤ 2 * δ := by
168  have hl : (1 / 2 : ℝ) ≤ R / D := by have := (abs_le.mp herr).1; linarith
169  have hl' : D ≤ 2 * R := by have := (le_div_iff₀ hD).mp hl; linarith
170  have hinv : D / R ≤ 2 := (div_le_iff₀ hR).mpr hl'
171  have heq : D / R - 1 = -(D / R) * (R / D - 1) := by field_simp; ring
172  rw [heq, abs_mul, abs_neg, abs_of_pos (div_pos hD hR)]
173  exact (mul_le_mul_of_nonneg_left herr (div_pos hD hR).le).trans
174    (mul_le_mul_of_nonneg_right hinv (le_trans (abs_nonneg _) herr))

185/-- General-rank reciprocal dimension moment.  Polynomial growth on extreme
186shapes is controlled by the proved stretched-exponential Young tail. -/
187theorem mean_inverseNormalizedDimension_error_tendsto_zero (r k : ℕ) (hr : 0 < r)
188    (P : ∀ N, PMF (Shape r N))
189    (hformula : ∀ N μ, (P N μ).toReal = youngWeight N (flatSpectrum r) (fun i ↦ (μ i).val)) :
190    Tendsto (fun N ↦ ∑ μ, (P N μ).toReal * |inverseNormalizedDimension r k N μ - 1|)
191      atTop (𝓝 0) := by
192  let c := leadingConstant r k
193  let a := r * k
194  have hc : 0 ≤ c := (leadingConstant_pos r k hr).le
195  have htail := (polynomial_concentrationEnvelope_tendsto_zero r a).const_mul (c + 1)
196  have hbound := ((dimensionErrorEnvelope_tendsto_zero r k).const_mul 2).add htail
197  simp only [mul_zero, add_zero] at hbound
198  apply squeeze_zero' (Eventually.of_forall (fun N ↦
199    Finset.sum_nonneg (fun μ _ ↦ mul_nonneg ENNReal.toReal_nonneg (abs_nonneg _)))) ?_ hbound
200  have hsmall : ∀ᶠ N in atTop, dimensionErrorEnvelope r k N ≤ 1 / 2 :=
201    (dimensionErrorEnvelope_tendsto_zero r k).eventually (eventually_le_nhds (by norm_num))
202  filter_upwards [Filter.eventually_ge_atTop 1, hsmall] with N hN heN
203  have hmean := mean_error_le (P N) (flatSpectrum r) (shrinkingRadius N)
204    (2 * dimensionErrorEnvelope r k N) (c * (N : ℝ) ^ a + 1)
205    (mul_nonneg (by norm_num) (dimensionErrorEnvelope_nonneg r k N))
206    (inverseNormalizedDimension r k N)
207    (inverseNormalizedDimension_global_error r k N hr)
208    (inverseNormalizedDimension_typical_error r k N hr hN heN)
209  have htailN : tailProbability (P N) (flatSpectrum r) (shrinkingRadius N) ≤ concentrationEnvelope r N := by
210    rw [tailProbability_eq_atypicalMass _ _ (hformula N)]
211    exact atypicalMass_shrinking_le N hN (flatSpectrum r)
212      (fun _ ↦ by dsimp [flatSpectrum]; positivity) (flatSpectrum_sum r hr) antitone_const
213  have hpw : (N : ℝ) ^ a ≤ ((N : ℝ) + 1) ^ a := pow_le_pow_left₀ (by positivity) (by linarith) _
214  have hpw1 : (1 : ℝ) ≤ ((N : ℝ) + 1) ^ a := one_le_pow₀ (by have := Nat.cast_nonneg (α := ℝ) N; linarith)
215  have hcoeff : c * (N : ℝ) ^ a + 1 ≤ (c + 1) * ((N : ℝ) + 1) ^ a := by
216    nlinarith [mul_le_mul_of_nonneg_left hpw hc]
217  have ht0 : 0 ≤ tailProbability (P N) (flatSpectrum r) (shrinkingRadius N) := by
218    unfold tailProbability
219    exact Finset.sum_nonneg (fun μ _ ↦ by split_ifs <;> positivity)
220  have htailbound : (c * (N : ℝ) ^ a + 1) * tailProbability (P N) (flatSpectrum r) (shrinkingRadius N) ≤
221      (c + 1) * (((N : ℝ) + 1) ^ a * concentrationEnvelope r N) := by
222    have hh := mul_le_mul hcoeff htailN ht0 (by positivity : 0 ≤ (c + 1) * ((N : ℝ) + 1) ^ a)
223    convert hh using 1 <;> ring
224  exact hmean.trans (by linarith)
