5# Concrete classical Gaussian witnesses

37theorem witness_closed {a b : ℝ} (ha : 0 < a) (hb : 0 < b) (x : ℝ) :
38    witness a b x = Real.sqrt (Real.sqrt a / Real.sqrt b) *
39      Real.exp (-((a - b) / 2) * x ^ 2) := by
40  unfold witness
41  rw [sqrt_density, sqrt_density, mul_div_mul_comm,
42    ← Real.sqrt_div (by positivity : 0 ≤ Real.sqrt a / Real.sqrt Real.pi),
43    ← Real.exp_sub]
44  have hp : Real.sqrt Real.pi ≠ 0 := (Real.sqrt_pos.mpr Real.pi_pos).ne'
45  have hratio : (Real.sqrt a / Real.sqrt Real.pi) /
46      (Real.sqrt b / Real.sqrt Real.pi) = Real.sqrt a / Real.sqrt b := by
47    field_simp
48  rw [hratio]
49  congr 2
50  ring

91theorem integral_gaussian_translate {a c r v : ℝ} (ha : 0 < a) (hc : 0 ≤ c) :
92    (∫ h : ℝ, Real.exp (-a * h ^ 2 - c * (v + r * h) ^ 2)) =

94        Real.sqrt (Real.pi / (a + c * r ^ 2)) := by

115theorem dilation_prefactor {a g : ℝ} (ha : 0 < a) (hg : 1 ≤ g) :
116    Real.sqrt (Real.sqrt g) * (Real.sqrt a / Real.sqrt Real.pi) *
117      Real.sqrt (Real.pi / (a + (a - a / g) / 2 * (Real.sqrt g) ^ 2)) =
118      Real.sqrt (Cloning.Thermal.classicalBase g) := by
119  have hg0 : 0 < g := lt_of_lt_of_le zero_lt_one hg
120  have hc : 0 ≤ (a - a / g) / 2 := by
121    have : a / g ≤ a := (div_le_iff₀ hg0).mpr (by nlinarith)
122    linarith
123  have hA : 0 < a + (a - a / g) / 2 * (Real.sqrt g) ^ 2 := by positivity
124  have hpi : Real.sqrt Real.pi ≠ 0 := (Real.sqrt_pos.mpr Real.pi_pos).ne'
125  have heq : a + (a - a / g) / 2 * (Real.sqrt g) ^ 2 = a * (1 + g) / 2 := by

242theorem integral_product_translated_witness_le (a : ι → ℝ) (ha : ∀ i, 0 < a i)
243    {g : ℝ} (hg : 1 ≤ g) (v : ι → ℝ) :
244    (∫ h : ι → ℝ, productDensity a h *
245      productWitness a (fun i => a i / g) (fun i => v i + Real.sqrt g * h i)) ≤
246      Cloning.Thermal.classicalBase g ^ ((Fintype.card ι : ℝ) / 2) := by
247  simp only [productDensity, productWitness, ← Finset.prod_mul_distrib]
248  rw [integral_fintype_prod_volume_eq_prod
249    (fun i t => density (a i) t * witness (a i) (a i / g) (v i + Real.sqrt g * t))]
250  calc
251    _ ≤ ∏ _i : ι, Real.sqrt (Cloning.Thermal.classicalBase g) := by
252      apply Finset.prod_le_prod
253      · intro i _
254        exact integral_nonneg fun h => mul_nonneg (density_nonneg _ _)
