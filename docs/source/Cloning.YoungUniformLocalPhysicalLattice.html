15/-- Complete the head coordinates of any finite shape into the full affine
16lattice; the physical law gives zero mass to inconsistent finite shapes. -/
17def shapeLattice (d N : ℕ) (μ : Shape (d + 1) N) : Lattice d (N : ℤ) :=
18  latticeCoordinates d N (fun i ↦ (μ i.castSucc).val)

20theorem shapeLattice_apply (d N : ℕ) (μ : Shape (d + 1) N)
21    (hμ : ∑ i, (μ i).val = N) (i : Fin (d + 1)) :
22    (shapeLattice d N μ).1 i = ((μ i).val : ℤ) := by
23  refine Fin.lastCases ?_ (fun j ↦ ?_) i
24  · have ht := (shapeLattice d N μ).2
25    change (∑ i, (shapeLattice d N μ).1 i) = (N : ℤ) at ht
26    rw [Fin.sum_univ_castSucc] at ht hμ
27    have heq : ∀ j : Fin d, (shapeLattice d N μ).1 j.castSucc = ((μ j.castSucc).val : ℤ) :=
28      fun j ↦ latticeCoordinates_head d N _ j
29    simp only [heq] at ht
30    have hcast : (∑ j : Fin d, ((μ j.castSucc).val : ℤ)) + ((μ (Fin.last d)).val : ℤ) = N := by
31      exact_mod_cast hμ
32    omega
33  · exact latticeCoordinates_head d N _ j

35theorem shapeLattice_injective_of_sum (d N : ℕ) (μ ν : Shape (d + 1) N)
36    (hμ : ∑ i, (μ i).val = N) (hν : ∑ i, (ν i).val = N)
37    (he : shapeLattice d N μ = shapeLattice d N ν) : μ = ν := by
38  funext i
39  apply Fin.ext
40  have h := congrArg (fun z : Lattice d (N : ℤ) ↦ z.1 i) he
41  dsimp only at h
42  rw [shapeLattice_apply d N μ hμ, shapeLattice_apply d N ν hν] at h
43  exact_mod_cast h

45/-- The normalized, genuinely physical law on every affine lattice label. -/
46def tensorYoungLatticePMF (d N : ℕ) (p : Fin (d + 1) → ℝ)
47    (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) : PMF (Lattice d (N : ℤ)) :=
48  (tensorYoungPMF N (d + 1) p hp hs).map (shapeLattice d N)

50theorem tensorYoungLatticePMF_apply_shape (d N : ℕ) (p : Fin (d + 1) → ℝ)
51    (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) (μ : Shape (d + 1) N)
52    (hμ : ∑ i, (μ i).val = N) :
53    tensorYoungLatticePMF d N p hp hs (shapeLattice d N μ) =
54      tensorYoungPMF N (d + 1) p hp hs μ := by
55  rw [tensorYoungLatticePMF, PMF.map_apply, tsum_eq_single μ]
56  · simp
57  · intro ν hν
58    by_cases ht : ∑ i, (ν i).val = N
59    · have hn : shapeLattice d N μ ≠ shapeLattice d N ν := by
60        intro he
61        exact hν (shapeLattice_injective_of_sum d N ν μ ht hμ he.symm)
62      simp [hn]
63    · have hz : tensorYoungPMF N (d + 1) p hp hs ν = 0 :=
64        physicalYoungPMF_eq_zero_of_not_partition _ _ _ _ _ _ ν (Or.inr ht)
65      simp [hz]
