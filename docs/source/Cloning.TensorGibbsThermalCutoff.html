13def bosonicOccupationWeight (p : Fin d → ℝ) (k : PositiveRoot d → ℕ) : ℝ :=
14  (∏ r : PositiveRoot d, (1 - rootBoltzmann p r)) * wordBoltzmann p (canonicalWord k)

16theorem bosonicOccupationWeight_hasSum (p : Fin d → ℝ) (hp : ∀ a, 0 < p a)
17    (hord : StrictAnti p) : HasSum (bosonicOccupationWeight p) 1 := by

97theorem rootThermalCutoff_error (p : Fin d → ℝ) (hp : ∀ a, 0 < p a) (hord : StrictAnti p) (R : ℕ) :
98    ‖rootThermalState p - rootThermalCutoff p R‖ =
99      1 - ∑ k : HeightOccupation d R, bosonicOccupationWeight p k.val := by
100  rw [rootThermalCutoff, spectralFrame_residual_norm _ (rootThermalState_nonneg p hp hord) _
