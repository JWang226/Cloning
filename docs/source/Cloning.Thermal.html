21/-- Number probabilities of a one-mode thermal state. -/
22def geometric (q : ℝ) (n : ℕ) : ℝ := (1 - q) * q ^ n

24/-- Root fidelity of two geometric distributions in closed form. -/
25def fidelity (q x : ℝ) : ℝ :=
26  Real.sqrt ((1 - q) * (1 - x)) / (1 - Real.sqrt (q * x))

28/-- Thermal parameter after quantum-limited amplification. -/
29def amplified (g q : ℝ) : ℝ := 1 - (1 - q) / g

31/-- Thermal parameter in the purify-clone-trace limit. -/
32def pct (g q : ℝ) : ℝ := (g - 1 + g * q) / (g + (g - 1) * q)

93theorem fidelity_le_one {q x : ℝ} (hq0 : 0 ≤ q) (hq1 : q < 1)
94    (hx0 : 0 ≤ x) (hx1 : x < 1) : fidelity q x ≤ 1 := by
95  have hprod : q * x < 1 := mul_lt_one_of_nonneg_of_lt_one_right hq1.le hx0 hx1

100    rw [mul_pow, Real.sq_sqrt hq0, Real.sq_sqrt hx0]

109/-- Algebra behind strict decrease of thermal fidelity beyond its maximum. -/
110theorem fidelity_sq_strict_aux {a b c : ℝ} (ha : 0 ≤ a) (hab : a < b)
111    (hbc : b < c) (hc : c < 1) :
112    (1 - a ^ 2) * (1 - c ^ 2) / (1 - a * c) ^ 2 <
113      (1 - a ^ 2) * (1 - b ^ 2) / (1 - a * b) ^ 2 := by
114  have hb : 0 ≤ b := le_trans ha hab.le
115  have hc0 : 0 ≤ c := le_trans hb hbc.le
116  have ha1 : a < 1 := lt_trans hab (lt_trans hbc hc)
117  have hab1 : a * b < 1 := mul_lt_one_of_nonneg_of_lt_one_right ha1.le hb (lt_trans hbc hc)
118  have hac1 : a * c < 1 := mul_lt_one_of_nonneg_of_lt_one_right ha1.le hc0 hc
119  have ha2 : 0 < 1 - a ^ 2 := by nlinarith
120  apply (div_lt_div_iff₀ (sq_pos_of_pos (sub_pos.mpr hac1))
121    (sq_pos_of_pos (sub_pos.mpr hab1))).mpr
122  have hid :
123      (1 - a ^ 2) * (1 - b ^ 2) * (1 - a * c) ^ 2 -
124        (1 - a ^ 2) * (1 - c ^ 2) * (1 - a * b) ^ 2 =
125      (1 - a ^ 2) * (c - b) *
126        ((b - a) * (1 - a * c) + (c - a) * (1 - a * b)) := by ring
127  have hp : 0 < (1 - a ^ 2) * (c - b) *
128      ((b - a) * (1 - a * c) + (c - a) * (1 - a * b)) := by
129    apply mul_pos (mul_pos ha2 (sub_pos.mpr hbc))
130    exact add_pos (mul_pos (sub_pos.mpr hab) (sub_pos.mpr hac1))
131      (mul_pos (sub_pos.mpr (lt_trans hab hbc)) (sub_pos.mpr hab1))
132  linarith

134theorem fidelity_strictAnti_right {q x y : ℝ} (hq0 : 0 ≤ q)
135    (hqx : q < x) (hxy : x < y) (hy1 : y < 1) :
136    fidelity q y < fidelity q x := by
137  have hx0 : 0 ≤ x := le_trans hq0 hqx.le
138  have hy0 : 0 ≤ y := le_trans hx0 hxy.le

147  have hp' := fidelity_pos hq0 hq1 hx0 hx1

417/-- An absolutely convergent product-distribution identity, including zero modes. -/
418theorem hasSum_fin_product (s : ℕ) (f : Fin s → ℕ → ℝ) (a : Fin s → ℝ)
419    (hn : ∀ i n, 0 ≤ f i n) (hs : ∀ i, HasSum (f i) (a i)) :
420    HasSum (fun k : Fin s → ℕ => ∏ i, f i (k i)) (∏ i, a i) := by
421  induction s with
422  | zero =>
423    simpa using hasSum_fintype (fun k : Fin 0 → ℕ => ∏ i, f i (k i))
424  | succ s ih =>
425    have ht : HasSum (fun k : Fin s → ℕ => ∏ i : Fin s, f i.succ (k i))
426        (∏ i : Fin s, a i.succ) := ih (fun i => f i.succ) (fun i => a i.succ)
427      (fun i n => hn i.succ n) (fun i => hs i.succ)
428    have hn' : ∀ k : Fin s → ℕ, 0 ≤ ∏ i : Fin s, f i.succ (k i) :=
429      fun k => Finset.prod_nonneg (fun i _ => hn i.succ (k i))
430    have hsum : Summable (fun k : ℕ × (Fin s → ℕ) =>
431        f 0 k.1 * ∏ i : Fin s, f i.succ (k.2 i)) := by
432      apply (summable_prod_of_nonneg (fun k => mul_nonneg (hn 0 k.1) (hn' k.2))).mpr
433      constructor
434      · intro n
435        exact ht.summable.mul_left (f 0 n)
436      · have he : (fun n : ℕ => ∑' k : Fin s → ℕ, f 0 n * ∏ i : Fin s, f i.succ (k i)) =
437            fun n => f 0 n * ∏ i : Fin s, a i.succ := by
438          funext n
439          rw [tsum_mul_left, ht.tsum_eq]
440        rw [he]
441        exact (hs 0).summable.mul_right (∏ i : Fin s, a i.succ)
442    have hm : HasSum (fun k : ℕ × (Fin s → ℕ) =>
443        f 0 k.1 * ∏ i : Fin s, f i.succ (k.2 i)) (a 0 * ∏ i : Fin s, a i.succ) :=
444      HasSum.mul (f := f 0) (g := fun k : Fin s → ℕ => ∏ i : Fin s, f i.succ (k i))
445        (s := a 0) (t := ∏ i : Fin s, a i.succ) (hs 0) ht hsum
446    rw [← (Fin.consEquiv (fun _ : Fin (s + 1) => ℕ)).hasSum_iff]
447    simpa [Fin.prod_univ_succ, Fin.consEquiv, Function.comp_def] using hm
