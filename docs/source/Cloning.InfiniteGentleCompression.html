14theorem norm_projection_apply_le {P : H →L[ℂ] H} (hP : IsStarProjection P) (x : H) :
15    ‖P x‖ ≤ ‖x‖ := (P.le_opNorm x).trans (by nlinarith [hP.norm_le, norm_nonneg x])

17theorem norm_vectorProjector_sub_projection_le {P : H →L[ℂ] H}
18    (hP : IsStarProjection P) (x : H) :
19    ‖vectorProjector x - vectorProjector (P x)‖ ≤ 2 * (‖x‖ * ‖(1-P) x‖) := by
20  have h := norm_vectorProjector_sub_le x (P x)
21  have hPx := norm_projection_apply_le hP x
22  have he : (1-P) x = x - P x := by simp
23  rw [he]
24  exact h.trans (by nlinarith [norm_nonneg (x-P x)])

26/-- Compression costs at most twice the square root of total mass times
27mass outside the projection, for arbitrary positive trace-class operators. -/
28theorem norm_sub_sandwich_projection_le {P : H →L[ℂ] H} (hP : IsStarProjection P)
29    (A : TraceClass H) (hA : 0 ≤ A.1) :
30    ‖A - sandwichCLM P P A‖ ≤
31      2 * Real.sqrt ‖A‖ * Real.sqrt ‖sandwichCLM (1-P) (1-P) A‖ := by
32  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
33  let v : w → H := fun i ↦ CFC.sqrt A.1 (b i)
34  have hs : HasSum (fun i ↦ vectorProjector (v i)) A := positive_rankOne_series hA A.2 b
35  have hc : HasSum (fun i ↦ vectorProjector (P (v i))) (sandwichCLM P P A) := by
36    simpa only [sandwichCLM_vectorProjector P hP.isSelfAdjoint] using
37      hs.mapL (sandwichCLM P P)
38  have hd : HasSum (fun i ↦ vectorProjector ((1-P) (v i))) (sandwichCLM (1-P) (1-P) A) := by
39    have he : (fun i : w ↦ sandwichCLM (1-P) (1-P) (vectorProjector (v i))) =
40        (fun i : w ↦ vectorProjector ((1-P) (v i))) :=
41      funext fun i ↦ sandwichCLM_vectorProjector (1-P) hP.one_sub.isSelfAdjoint (v i)
42    rw [← he]
43    exact hs.mapL (sandwichCLM (1-P) (1-P))
44  have hm : HasSum (fun i ↦ ‖v i‖^2) ‖A‖ := by
45    simpa only [v, norm_vectorProjector, TraceClass.norm_eq_trace_re_of_nonneg A hA] using
46      positive_rankOne_mass hA A.2 b
47  have hmd : HasSum (fun i ↦ ‖(1-P) (v i)‖^2) ‖sandwichCLM (1-P) (1-P) A‖ := by
48    have h := Complex.hasSum_re (hd.mapL traceCLM)
49    simpa only [traceCLM_vectorProjector, Complex.ofReal_re,
50      TraceClass.norm_eq_trace_re_of_nonneg _ (sandwichCLM_nonneg hP.one_sub.isSelfAdjoint A hA)] using h
51  have hprod := summable_norm_mul_of_square_sums (fun i ↦ ‖v i‖) (fun i ↦ ‖(1-P) (v i)‖)
52    hm.summable hmd.summable (fun _ ↦ norm_nonneg _) (fun _ ↦ norm_nonneg _)
53  have hnorm : Summable (fun i ↦ ‖vectorProjector (v i) - vectorProjector (P (v i))‖) :=
54    (hprod.mul_left 2).of_nonneg_of_le (fun _ ↦ norm_nonneg _)
55      (fun i ↦ norm_vectorProjector_sub_projection_le hP (v i))
56  have hcs := tsum_mul_le_sqrt_mul_sqrt hm.summable hmd.summable
57    (fun _ ↦ norm_nonneg _) (fun _ ↦ norm_nonneg _)
58  rw [hm.tsum_eq, hmd.tsum_eq] at hcs
59  calc
60    _ = ‖(∑' i, (vectorProjector (v i) - vectorProjector (P (v i))))‖ := by rw [(hs.sub hc).tsum_eq]
61    _ ≤ ∑' i, ‖vectorProjector (v i) - vectorProjector (P (v i))‖ := norm_tsum_le_tsum_norm hnorm
62    _ ≤ ∑' i, 2 * (‖v i‖ * ‖(1-P) (v i)‖) :=
63      hnorm.tsum_le_tsum (fun i ↦ norm_vectorProjector_sub_projection_le hP (v i)) (hprod.mul_left 2)
64    _ = 2 * ∑' i, ‖v i‖ * ‖(1-P) (v i)‖ := tsum_mul_left
65    _ ≤ _ := by nlinarith

67theorem norm_sandwich_complement_eq {P : H →L[ℂ] H} (hP : IsStarProjection P)
68    (A : TraceClass H) (hA : 0 ≤ A.1) :
69    ‖sandwichCLM (1-P) (1-P) A‖ = ‖A‖ - ‖sandwichCLM P P A‖ := by
70  rw [TraceClass.norm_eq_trace_re_of_nonneg _ (sandwichCLM_nonneg hP.one_sub.isSelfAdjoint A hA),
71    TraceClass.norm_eq_trace_re_of_nonneg _ hA,
72    TraceClass.norm_eq_trace_re_of_nonneg _ (sandwichCLM_nonneg hP.isSelfAdjoint A hA)]
73  have h := congrArg Complex.re (traceCLM_projection_split hP A)
74  simp only [Complex.add_re, traceCLM_apply] at h
75  linarith

77/-- The standard quantitative gentle bound with discarded trace written as
78the difference of positive trace norms. -/
79theorem gentle_compression {P : H →L[ℂ] H} (hP : IsStarProjection P)
80    (A : TraceClass H) (hA : 0 ≤ A.1) :
81    ‖A - sandwichCLM P P A‖ ≤
82      2 * Real.sqrt (‖A‖ * (‖A‖ - ‖sandwichCLM P P A‖)) := by
83  have h := norm_sub_sandwich_projection_le hP A hA
84  rw [norm_sandwich_complement_eq hP A hA] at h
85  rw [Real.sqrt_mul (norm_nonneg A)]
86  simpa only [mul_assoc] using h

88theorem gentle_compression_unit {P : H →L[ℂ] H} (hP : IsStarProjection P)
89    (A : TraceClass H) (hA : 0 ≤ A.1) (hunit : ‖A‖ = 1) :
90    ‖A - sandwichCLM P P A‖ ≤ 2 * Real.sqrt (1 - ‖sandwichCLM P P A‖) := by
91  simpa only [hunit, one_mul] using gentle_compression hP A hA
