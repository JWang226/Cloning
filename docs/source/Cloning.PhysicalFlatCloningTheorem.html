11/-- The sharp physical minimax value for rank-r projectors in dimension r+k.
12The upper bound applies to all channels; the lower bound uses an explicitly

14theorem minimaxValue_tendsto (r k : ℕ) (hr : 0<r)
15    (m : ℕ → ℕ) (γ : ℝ) (hγ : 1<γ)
16    (hratio : Tendsto (fun n => (m n : ℝ)/(n : ℝ)) atTop (𝓝 γ)) :
17    Tendsto (fun n => minimaxValue r k hr n (m n)) atTop
18      (𝓝 (γ^(-(((r*k : ℕ) : ℝ)/2)))) := by
19  have hm : Tendsto m atTop atTop := by
20    apply (tendsto_natCast_atTop_iff (R := ℝ)).mp
21    have h := hratio.pos_mul_atTop (lt_trans zero_lt_one hγ)
22      (tendsto_natCast_atTop_atTop (R := ℝ))
23    apply h.congr'
24    filter_upwards [eventually_gt_atTop (0:ℕ)] with n hn
25    exact div_mul_cancel₀ (m n : ℝ) (Nat.cast_ne_zero.mpr hn.ne')
26  apply Metric.tendsto_atTop.mpr
27  intro ε hε
28  have hl := TensorCloning.eventually_rankFlat_minimaxValue_lower r k hr m hm γ hγ hratio ε hε
29  have hu := eventually_minimaxValue_upper r k hr m hm γ (lt_trans zero_lt_one hγ) hratio ε hε
30  obtain ⟨N,hN⟩ := eventually_atTop.mp (hl.and hu)
31  refine ⟨N,fun n hn => ?_⟩
32  rw [Real.dist_eq,abs_lt]
33  have h := hN n hn
34  constructor <;> linarith [h.1,h.2]
