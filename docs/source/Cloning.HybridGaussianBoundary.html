17/-- Fixed Gaussian preparation is continuous on the actual trace-class

19theorem continuous_gaussian_prepareL1 (a : Fin k → ℝ) (ha : ∀ i, 0 < a i) :
20    Continuous (prepareL1 (H := Fock s) (GaussianAffinity.productDensity a)
21      (GaussianAffinity.integrable_productDensity a ha)) := by
22  let P := prepareL1 (H := Fock s) (GaussianAffinity.productDensity a)
23    (GaussianAffinity.integrable_productDensity a ha)
24  have hp (X : TraceClass (Fock s)) : ‖P X‖ ≤ 1 * ‖X‖ := by
25    rw [norm_prepareL1 _ _ (GaussianAffinity.productDensity_nonneg a)
26      (GaussianAffinity.integral_productDensity a ha), one_mul]
27  exact (P.mkContinuous 1 hp).continuous

48/-- Reference perturbation has one modulus for every channel and every
49phase-space displacement. The factor two comes from the proved whole-space
50operator norm bound for physical hybrid channels. -/
51theorem orbitPayoff_self_le_add_reference_error (r : ℝ) (Λ : HybridChannel k s)
52    (A C : HybridPositive k s) (ξ : PhaseSpace k s) :
53    orbitPayoff r Λ A A ξ ≤ orbitPayoff r Λ C C ξ +
54      (Real.sqrt (2 * ‖A.1 - C.1‖) * Real.sqrt ‖A.1‖ +
55        Real.sqrt ‖A.1 - C.1‖ * Real.sqrt ‖C.1‖) := by
56  rw [orbitPayoff_eq_translated, orbitPayoff_eq_translated]
57  let Γ := translatedChannel r Λ ξ
58  have h := PositiveL1.rootFidelity_continuity (A.map Γ) A (C.map Γ) C
59  rw [PositiveL1.norm_map] at h
60  have hd : ‖(A.map Γ).1 - (C.map Γ).1‖ ≤ 2 * ‖A.1 - C.1‖ := by
61    change ‖Γ.map A.1 - Γ.map C.1‖ ≤ _
62    rw [← map_sub]
63    exact Γ.norm_le_two _
64  have hb := h.trans (add_le_add
65    (mul_le_mul_of_nonneg_right (Real.sqrt_le_sqrt hd) (Real.sqrt_nonneg _)) le_rfl)
66  have hle := (le_abs_self _).trans hb
67  dsimp only [Γ] at hle
68  linarith

70/-- Averaging over any probability prior preserves the same uniform
71reference-state modulus. -/
72theorem integral_orbitPayoff_self_le_add_reference_error
73    (ν : Measure (PhaseSpace k s)) [IsProbabilityMeasure ν]
74    (r : ℝ) (Λ : HybridChannel k s) (A C : HybridPositive k s) :
75    (∫ ξ, orbitPayoff r Λ A A ξ ∂ν) ≤ (∫ ξ, orbitPayoff r Λ C C ξ ∂ν) +
76      (Real.sqrt (2 * ‖A.1 - C.1‖) * Real.sqrt ‖A.1‖ +
77        Real.sqrt ‖A.1 - C.1‖ * Real.sqrt ‖C.1‖) := by
78  have h := integral_mono (integrable_orbitPayoff ν r Λ A A)
79    ((integrable_orbitPayoff ν r Λ C C).add (integrable_const
80      (Real.sqrt (2 * ‖A.1 - C.1‖) * Real.sqrt ‖A.1‖ +
81        Real.sqrt ‖A.1 - C.1‖ * Real.sqrt ‖C.1‖)))
82    (orbitPayoff_self_le_add_reference_error r Λ A C)
83  simpa only [Pi.add_apply, integral_add (integrable_orbitPayoff ν r Λ C C)
84    (integrable_const _), integral_const, probReal_univ, one_smul] using h

89/-- Uniform channel bounds pass to L1 limits of reference states. The
90approximation index is chosen before both radius and arbitrary competitor. -/
91theorem eventually_forall_orbitPayoff_le_of_reference_tendsto (r : ℝ)
92    (A : ℕ → HybridPositive k s) (B : HybridPositive k s)
93    (hA : Tendsto A atTop (𝓝 B)) (c : ℕ → ℝ) {C : ℝ}
94    (hc : Tendsto c atTop (𝓝 C))
95    (hbound : ∀ m, ∀ ε > 0, ∀ᶠ n in atTop, ∀ Λ : HybridChannel k s,
96      (∫ ξ, orbitPayoff r Λ (A m) (A m) ξ ∂p.expandingPrior n) ≤ c m + ε) :
97    ∀ ε > 0, ∀ᶠ n in atTop, ∀ Λ : HybridChannel k s,
98      (∫ ξ, orbitPayoff r Λ B B ξ ∂p.expandingPrior n) ≤ C + ε := by
99  intro ε hε
100  have hv : Tendsto (fun m => (A m).1) atTop (𝓝 B.1) :=
101    continuous_subtype_val.tendsto _ |>.comp hA
102  let E := fun m => Real.sqrt (2 * ‖B.1 - (A m).1‖) * Real.sqrt ‖B.1‖ +
103    Real.sqrt ‖B.1 - (A m).1‖ * Real.sqrt ‖(A m).1‖
104  have he : Tendsto E atTop (𝓝 0) := by
105    have hd := ((tendsto_const_nhds (x := B.1)).sub hv).norm
106    convert (((hd.const_mul 2).sqrt).mul tendsto_const_nhds).add
107      (hd.sqrt.mul hv.norm.sqrt) using 1
108    simp
109  have hsum : Tendsto (fun m => c m + E m) atTop (𝓝 C) := by
110    simpa using hc.add he
111  obtain ⟨m, hm⟩ := (hsum.eventually_lt_const (by linarith : C < C + ε / 2)).exists
112  filter_upwards [hbound m (ε / 2) (by linarith)] with n hn
113  intro Λ
114  have hp := integral_orbitPayoff_self_le_add_reference_error
115    (p.expandingPrior n) r Λ B (A m)
116  change (∫ ξ, orbitPayoff r Λ B B ξ ∂p.expandingPrior n) ≤
117    (∫ ξ, orbitPayoff r Λ (A m) (A m) ξ ∂p.expandingPrior n) + E m at hp
118  linarith [hn Λ]
