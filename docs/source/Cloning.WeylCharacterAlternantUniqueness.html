11theorem support_on_orbit_of_eigen_dominance {d : ℕ}
12    (P : MvPolynomial (Fin d) ℂ) (a : Fin d → ℕ) (ha : StrictAnti a)
13    (heigen : eulerSquares P = (∑ i, (a i : ℂ)^2) • P)
14    (hdom : ∀ m, coeff m P ≠ 0 → SubsetDominatedBy (fun i ↦ (m i : ℝ)) (fun i ↦ (a i : ℝ)))
15    (m : Fin d →₀ ℕ) (hm : coeff m P ≠ 0) :
16    ∃ σ : Equiv.Perm (Fin d), m = exponentOf (a ∘ σ) := by
17  let σ : Equiv.Perm (Fin d) := Tuple.sort (α := OrderDual ℕ) (fun i : Fin d ↦ (m i : OrderDual ℕ))
18  have hb : Antitone (fun i ↦ m (σ i)) := by
19    intro i j hij
20    exact (Tuple.monotone_sort (α := OrderDual ℕ) (fun i : Fin d ↦ (m i : OrderDual ℕ))) hij
21  have hbr : Antitone (fun i ↦ (m (σ i) : ℝ)) := by
22    intro i j hij
23    change (m (σ j) : ℝ) ≤ (m (σ i) : ℝ)
24    exact_mod_cast hb hij
25  have har : StrictAnti (fun i ↦ (a i : ℝ)) := by
26    intro i j hij
27    change (a j : ℝ) < (a i : ℝ)
28    exact_mod_cast ha hij
29  have hsq := support_eulerSquares_eigenvalue P _ heigen m hm
30  have hsqN : (∑ i, m i ^ 2) = ∑ i, a i ^ 2 := by exact_mod_cast hsq
31  have hsqR : (∑ i, (m i : ℝ)^2) = ∑ i, (a i : ℝ)^2 := by exact_mod_cast hsqN
32  have hperm : (∑ i, (m (σ i) : ℝ)^2) = ∑ i, (m i : ℝ)^2 :=
33    Equiv.sum_comp σ (fun i ↦ (m i : ℝ)^2)
34  have heq := fin_dominant_eq_of_sum_squares
35    (fun i ↦ (a i : ℝ)) (fun i ↦ (m (σ i) : ℝ)) har hbr
36    ((subsetDominatedBy_iff_permutations _ _).mp (hdom m hm) σ)
37    (by rw [hperm, hsqR])
38  refine ⟨σ.symm, ?_⟩
39  ext i
40  have hi := congrFun heq (σ.symm i)
41  simpa only [Equiv.apply_symm_apply, exponentOf_apply, Function.comp_apply,
42    Nat.cast_inj] using hi.symm

63theorem alternating_eq_of_orbit_and_top {d : ℕ} (P Q : MvPolynomial (Fin d) ℂ)
64    (a : Fin d → ℕ) (hP : Alternating P) (hQ : Alternating Q)
65    (hsP : ∀ m, coeff m P ≠ 0 → ∃ σ : Equiv.Perm (Fin d), m = exponentOf (a ∘ σ))
66    (hsQ : ∀ m, coeff m Q ≠ 0 → ∃ σ : Equiv.Perm (Fin d), m = exponentOf (a ∘ σ))
67    (htop : coeff (exponentOf a) P = coeff (exponentOf a) Q) : P = Q := by
68  apply MvPolynomial.ext
69  intro m
70  by_cases hs : ∃ σ : Equiv.Perm (Fin d), m = exponentOf (a ∘ σ)
71  · obtain ⟨σ, rfl⟩ := hs
72    rw [exponentOf_comp_eq_mapDomain]
73    apply mul_left_cancel₀ (sign_complex_ne_zero σ.symm)
74    rw [← alternating_coeff_permuted P hP, ← alternating_coeff_permuted Q hQ, htop]
75  · have hp : coeff m P = 0 := by by_contra hp; exact hs (hsP m hp)
76    have hq : coeff m Q = 0 := by by_contra hq; exact hs (hsQ m hq)
77    rw [hp, hq]

121theorem eq_alternant_of_spectral_data {d : ℕ} (P : MvPolynomial (Fin d) ℂ)
122    (a : Fin d → ℕ) (ha : StrictAnti a) (hP : Alternating P)
123    (heigen : eulerSquares P = (∑ i, (a i : ℂ)^2) • P)
124    (hdom : ∀ m, coeff m P ≠ 0 → SubsetDominatedBy (fun i ↦ (m i : ℝ)) (fun i ↦ (a i : ℝ)))
125    (htop : coeff (exponentOf a) P = 1) : P = alternant a := by
126  apply alternating_eq_of_orbit_and_top P (alternant a) a hP (alternant_alternating a)
127    (support_on_orbit_of_eigen_dominance P a ha heigen hdom) (alternant_support_orbit a)
128  rw [htop, alternant_top_coeff a ha.injective]
