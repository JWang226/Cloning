37lemma normalPart_nonneg (f : (H →L[ℂ] H) →L[ℂ] ℂ)
38    (hf : ∀ A, 0 ≤ A → 0 ≤ f A) : 0 ≤ normalPartOp f := by
39  apply (normalPartOp f).nonneg_iff_isPositive.mpr
40  apply (ContinuousLinearMap.isPositive_iff_complex _).mpr

55lemma normalPart_finite_diagonal_le (f : (H →L[ℂ] H) →L[ℂ] ℂ)
56    {w : Set H} (b : HilbertBasis w ℂ H) (s : Finset w) :
57    (∑ i ∈ s, (⟪b i, normalPartOp f (b i)⟫_ℂ).re) ≤ ‖f‖ := by
58  have heq : (∑ i ∈ s, (⟪b i, normalPartOp f (b i)⟫_ℂ).re) =
59      (f (basisProjection b s)).re := by
60    simp [basisProjection, normalPart_inner, map_sum, Complex.re_sum]
61  rw [heq]
62  calc
63    (f (basisProjection b s)).re ≤ ‖f (basisProjection b s)‖ := Complex.re_le_norm _
64    _ ≤ ‖f‖ * ‖basisProjection b s‖ := f.le_opNorm _
65    _ ≤ ‖f‖ * 1 := mul_le_mul_of_nonneg_left
66      (basisProjection_isStarProjection b s).norm_le (norm_nonneg _)
67    _ = ‖f‖ := mul_one _

91/-- Multiplication by a fixed trace-class operator, with the bounded operator
92as the variable, is continuous into the trace-class Banach space. -/
93def traceClassRightMultiply (T : TraceClass H) : (H →L[ℂ] H) →L[ℂ] TraceClass H :=
94  LinearMap.mkContinuous
95    { toFun := fun O => TraceClass.ofOperator (T.1 * O)
96        (by simpa using (isTraceClass_mul_mul (A := 1) (B := O) T.2))
97      map_add' := by
98        intro A B
99        apply Subtype.ext
100        change T.1 * (A + B) = T.1 * A + T.1 * B
101        exact mul_add _ _ _
102      map_smul' := by
103        intro c A
104        apply Subtype.ext
105        change T.1 * (c • A) = c • (T.1 * A)
106        simp }
107    ‖T‖ (fun O => by
108      change traceNorm (T.1 * O) _ ≤ traceNorm T.1 T.2 * ‖O‖
109      have h := traceNorm_mul_mul_le (A := (1 : H →L[ℂ] H)) (B := O) T.2
110        (isTraceClass_mul_mul T.2)
111      simp only [one_mul] at h
112      apply h.trans
113      have h1 : ‖(1 : H →L[ℂ] H)‖ ≤ 1 :=
114        ContinuousLinearMap.opNorm_le_bound _ zero_le_one (by intro x; simp)
115      calc
116        ‖(1 : H →L[ℂ] H)‖ * traceNorm T.1 T.2 * ‖O‖ ≤
117            1 * traceNorm T.1 T.2 * ‖O‖ :=
118          mul_le_mul_of_nonneg_right
119            (mul_le_mul_of_nonneg_right h1 (traceNorm_nonneg _ _)) (norm_nonneg _)
120        _ = _ := by ring)

152/-- Strong basis projections converge uniformly on every compact set of vectors. -/
153lemma basisProjection_tendstoUniformlyOn {ι : Type*} (b : HilbertBasis ι ℂ H)
154    {S : Set H} (hS : IsCompact S) :
155    TendstoUniformlyOn (fun s : Finset ι => basisProjection b s) (fun x => x) atTop S := by
156  classical
157  rw [Metric.tendstoUniformlyOn_iff]
158  intro ε hε
159  obtain ⟨t, htS, ht, hcover⟩ := hS.finite_cover_balls (show 0 < ε / 3 by positivity)
160  have hevent : ∀ᶠ s : Finset ι in atTop, ∀ y ∈ t, dist (basisProjection b s y) y < ε / 3 := by
161    rw [ht.eventually_all]
162    intro y _
163    exact (Metric.tendsto_nhds.mp (basisProjection_tendsto b y)) _ (by positivity)
164  filter_upwards [hevent] with s hs
165  intro x hx
166  obtain ⟨y, hy, hxy⟩ : ∃ y ∈ t, dist x y < ε / 3 := by
167    simpa only [Set.mem_iUnion, Metric.mem_ball, exists_prop] using hcover hx
168  have hcon : dist (basisProjection b s y) (basisProjection b s x) ≤ dist y x := by
169    rw [dist_eq_norm, ← map_sub, dist_eq_norm]
170    calc
171      ‖basisProjection b s (y - x)‖ ≤ ‖basisProjection b s‖ * ‖y - x‖ :=
172        (basisProjection b s).le_opNorm _
173      _ ≤ 1 * ‖y - x‖ := mul_le_mul_of_nonneg_right
174        (basisProjection_isStarProjection b s).norm_le (norm_nonneg _)
175      _ = ‖y - x‖ := one_mul _
176  have hnet := hs y hy
177  have htri := dist_triangle x y (basisProjection b s x)
178  have htri' := dist_triangle y (basisProjection b s y) (basisProjection b s x)
179  rw [dist_comm y x] at hcon
180  rw [dist_comm (basisProjection b s y) y] at hnet
181  linarith

183/-- Every compact observable is approximated in operator norm by finite-basis
184range truncations. Compactness, rather than a finite-rank density premise, is used. -/
185lemma basisProjection_mul_tendsto {ι : Type*} (b : HilbertBasis ι ℂ H)
186    (K : H →L[ℂ] H) (hK : IsCompactOperator K) :
187    Tendsto (fun s : Finset ι => basisProjection b s * K) atTop (𝓝 K) := by
188  have hc : IsCompact (closure (K '' Metric.closedBall (0 : H) 1)) :=
189    hK.isCompact_closure_image_closedBall 1
190  have hu := Metric.tendstoUniformlyOn_iff.mp (basisProjection_tendstoUniformlyOn b hc)
191  rw [Metric.tendsto_nhds]
192  intro ε hε
193  filter_upwards [hu (ε / 2) (by positivity)] with s hs
194  rw [dist_eq_norm]
195  apply lt_of_le_of_lt (b := ε / 2)
196  · apply ContinuousLinearMap.opNorm_le_of_unit_norm (by positivity)
197    intro x hx
198    have hmem : K x ∈ closure (K '' Metric.closedBall (0 : H) 1) :=
199      subset_closure ⟨x, by simp [Metric.mem_closedBall, dist_zero_right, hx], rfl⟩
200    have h := hs (K x) hmem
201    change ‖basisProjection b s (K x) - K x‖ ≤ ε / 2
202    rw [dist_eq_norm, norm_sub_rev] at h
203    exact h.le
204  · linarith

212/-- The reconstructed trace-class operator agrees with the original bounded
213positive functional on every compact observable. -/
214theorem normalPart_tracePairing_compact (f : (H →L[ℂ] H) →L[ℂ] ℂ)
215    (hf : ∀ A, 0 ≤ A → 0 ≤ f A) (K : H →L[ℂ] H) (hK : IsCompactOperator K) :
216    tracePairing (normalPart f hf) K = f K := by
217  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
218  have heq (s : Finset w) :
219      tracePairing (normalPart f hf) (basisProjection b s * K) = f (basisProjection b s * K) := by
220    rw [basisProjection_mul_eq_sum_rankOne]
221    simp only [map_sum, normalPart_tracePairing_rankOne]
222  have hl := (tracePairing (normalPart f hf)).continuous.tendsto K |>.comp
223    (basisProjection_mul_tendsto b K hK)
224  have hr := f.continuous.tendsto K |>.comp (basisProjection_mul_tendsto b K hK)
225  simp only [Function.comp_def, heq] at hl
226  exact tendsto_nhds_unique hl hr
