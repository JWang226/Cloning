187/-- The actual Cartan inclusion commutes with the physical rank embeddings
188on the entire cyclic sector, with the highest-vector phases fixed. -/
189theorem rankCartanInclusion_restriction (mu nu : Fin r→ℕ)
190    (hmu : Antitone mu) (hnu : Antitone nu) (k : ℕ) :
191    (cartanInclusion _ _ (padPartition_antitone mu hmu k) (padPartition_antitone nu hnu k)).toLinearMap.comp
192      (rankSumSectorEmbedding mu nu hmu hnu k).toLinearMap =
193      (rankProductSectorEmbedding mu nu hmu hnu k).toLinearMap.comp
194        (cartanInclusion mu nu hmu hnu).toLinearMap := by
195  let f := (cartanInclusion _ _ (padPartition_antitone mu hmu k) (padPartition_antitone nu hnu k)).toLinearMap.comp
196      (rankSumSectorEmbedding mu nu hmu hnu k).toLinearMap
197  let g := (rankProductSectorEmbedding mu nu hmu hnu k).toLinearMap.comp
198        (cartanInclusion mu nu hmu hnu).toLinearMap
199  change f=g
200  apply LinearMap.ext
201  rintro ⟨x,hx⟩
202  induction hx using Submodule.span_induction with
203  | mem x hx =>
204    obtain ⟨w,rfl⟩ := hx
205    change cartanInclusion _ _ _ _ (rankSumSectorEmbedding mu nu hmu hnu k _) =
206      rankProductSectorEmbedding mu nu hmu hnu k (cartanInclusion mu nu hmu hnu _)
207    rw [rankSumSectorEmbedding_loweringWord]
208    exact (rankProductSectorEmbedding_loweringWord mu nu hmu hnu k w).symm
209  | zero => change f 0=g 0; simp
210  | add x y hx hy ihx ihy =>
211    change f (⟨x,hx⟩+⟨y,hy⟩) = g (⟨x,hx⟩+⟨y,hy⟩)
212    simp only [map_add,ihx,ihy]
213  | smul c x hx ih =>
214    change f (c • (⟨x,hx⟩ : cyclicSector _)) = g (c • (⟨x,hx⟩ : cyclicSector _))
215    simp only [map_smul,ih]
