5/-!
6# Physical Gaussian attainment of the orbital fidelity bound
7 
8The constructed quantum-limited amplifier sends every displaced product
9thermal state to the corresponding displaced amplified thermal state.
10Its root fidelity with the target orbit is the same exact mode-factor product
11at every displacement. Consequently it attains every probability average,
12and the optimal expanding real-box payoff converges to that product.
13-/

34theorem orbitPayoff_eq_rootFidelity_of_covariant
35    (gain : ℝ) (Φ : QuantumChannel (Fock d) (Fock d))
36    (hcov : ∀ a A, Φ.toLinearMap (displacementTraceMap a A) =
37      displacementTraceMap (gain • a) (Φ.toLinearMap A))

47/-- The physical amplifier achieves the exact orbital root fidelity at
48every phase-space displacement, with no averaging or limit. -/
49theorem gainChannel_orbitPayoff (g : ℝ) (hg : 1 < g) {q : Fin d → ℝ}
50    (hq0 : ∀ i, 0 < q i) (hq1 : ∀ i, q i < 1) (a : Fin d → ℂ) :
51    orbitPayoff (Real.sqrt g) (gainChannel g hg)
52      (thermalPositive q (fun i => (hq0 i).le) hq1)
53      (thermalPositive q (fun i => (hq0 i).le) hq1) a =
54      ∏ i, Thermal.modeFactor g (q i) := by
55  let x : Fin d → ℝ := fun i => Thermal.amplified g (q i)
56  have hx0 (i : Fin d) : 0 ≤ x i :=
57    ((hq0 i).trans (Thermal.lt_amplified hg (hq1 i))).le
58  have hx1 (i : Fin d) : x i < 1 := Thermal.amplified_lt_one (by linarith) (hq1 i)
59  have hmap : PositiveTraceClass.map (gainChannel g hg).toPositiveTracePreservingMap
60      (thermalPositive q (fun i => (hq0 i).le) hq1) = thermalPositive x hx0 hx1 := by
61    apply Subtype.ext
62    exact gainChannel_productThermal g hg hq0 hq1
63  rw [orbitPayoff_eq_rootFidelity_of_covariant (Real.sqrt g) (gainChannel g hg)
64    (gainChannel_covariant g hg), hmap, PositiveTraceClass.rootFidelity_comm,
65    thermalPositive_rootFidelity]
66  apply Finset.prod_congr rfl
67  intro i _
68  exact Thermal.fidelity_amplified_eq_modeFactor hg (hq0 i).le (hq1 i)

116/-- A single physical amplifier supplies a lower bound for every competing
117channel supremum, independently of the box radius. -/
118theorem modeFactor_le_realFlatBoxOptimalPayoff (g : ℝ) (hg : 1 < g) {q : Fin d → ℝ}
119    (hq0 : ∀ i, 0 < q i) (hq1 : ∀ i, q i < 1) (L : ℝ) :
120    (∏ i, Thermal.modeFactor g (q i)) ≤ realFlatBoxOptimalPayoff (Real.sqrt g)
121      (thermalPositive q (fun i => (hq0 i).le) hq1)
122      (thermalPositive q (fun i => (hq0 i).le) hq1) L := by
123  let A := thermalPositive q (fun i => (hq0 i).le) hq1
124  have hb : BddAbove (Set.range (fun Φ : QuantumChannel (Fock d) (Fock d) =>
125      realFlatBoxPayoff (Real.sqrt g) Φ A A L)) := by
126    refine ⟨Real.sqrt ‖A.1‖ * Real.sqrt ‖A.1‖, ?_⟩
127    rintro _ ⟨Φ, rfl⟩
128    exact realFlatBoxPayoff_le_sqrt _ Φ A A L
129  rw [← gainChannel_realFlatBoxPayoff g hg hq0 hq1 L]
130  exact le_csSup hb (Set.mem_range_self (gainChannel g hg))

132/-- The optimal average root fidelity has an actual limit over arbitrary
133real expanding boxes, with the channel supremum taken before the limit. -/
134theorem realFlatBoxOptimalPayoff_tendsto_modeFactor
135    (g : ℝ) (hg : 1 < g) {q : Fin d → ℝ}
136    (hq0 : ∀ i, 0 < q i) (hq1 : ∀ i, q i < 1) :
137    Tendsto (realFlatBoxOptimalPayoff (Real.sqrt g)
138      (thermalPositive q (fun i => (hq0 i).le) hq1)
139      (thermalPositive q (fun i => (hq0 i).le) hq1)) atTop
140      (𝓝 (∏ i, Thermal.modeFactor g (q i))) := by
141  let A := thermalPositive q (fun i => (hq0 i).le) hq1
142  let F : ℝ := ∏ i, Thermal.modeFactor g (q i)
143  have hs : Real.sqrt g ^ 2 = g := Real.sq_sqrt (by linarith)
144  have hupper := eventually_forall_realFlatBox_thermal_le (Real.sqrt g) (by rwa [hs]) hq0 hq1
145  simp_rw [hs, Thermal.fidelity_amplified_eq_modeFactor hg (hq0 _).le (hq1 _)] at hupper
146  have hopt : ∀ ε > 0, ∀ᶠ L : ℝ in atTop,
147      realFlatBoxOptimalPayoff (Real.sqrt g) A A L ≤ F + ε := by
148    intro ε hε
149    filter_upwards [hupper ε hε] with L hL
150    apply csSup_le ⟨_, Set.mem_range_self (gainChannel g hg)⟩
151    rintro _ ⟨Φ, rfl⟩
152    exact hL Φ
153  apply tendsto_order.2
154  constructor
155  · intro c hc
156    exact Eventually.of_forall fun L => hc.trans_le (modeFactor_le_realFlatBoxOptimalPayoff g hg hq0 hq1 L)
157  · intro c hc
158    filter_upwards [hopt ((c - F) / 2) (by change F < c at hc; linarith)] with L hL
159    change F < c at hc
160    change realFlatBoxOptimalPayoff (Real.sqrt g) A A L < c
161    linarith
