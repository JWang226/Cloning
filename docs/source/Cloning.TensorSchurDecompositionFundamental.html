16/-- Orthogonal unit highest vectors cannot occupy the same one-box highest line. -/
17theorem fundamental_highest_not_orthogonal
18    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
19    (hΩnorm : ‖Ω‖ = 1)
20    (hΩweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
21    (hΩraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
22    (r : Fin d) (x y : TensorRegister (n + 1) (Fin d))
23    (hx : x ∈ fundamentalHighestSpace Ω (addBox mu r))
24    (hy : y ∈ fundamentalHighestSpace Ω (addBox mu r))
25    (hxnorm : ‖x‖ = 1) (hynorm : ‖y‖ = 1) : ⟪x, y⟫_ℂ ≠ 0 := by
26  intro horth
27  let cx := ⟪Ω, lastSlice r x⟫_ℂ
28  let cy := ⟪Ω, lastSlice r y⟫_ℂ
29  have hcy : cy ≠ 0 := by
30    intro hz
31    have hy0 := highest_onebox_zero_of_coefficient_zero Ω mu hΩnorm hΩweight hΩraise r y hy hz
32    simp only [hy0, norm_zero, zero_ne_one] at hynorm
33  let z := cy • x - cx • y
34  have hzmem : z ∈ fundamentalHighestSpace Ω (addBox mu r) :=
35    (fundamentalHighestSpace Ω (addBox mu r)).sub_mem
36      ((fundamentalHighestSpace Ω (addBox mu r)).smul_mem cy hx)
37      ((fundamentalHighestSpace Ω (addBox mu r)).smul_mem cx hy)
38  have hz : z = 0 := by
39    apply highest_onebox_zero_of_coefficient_zero Ω mu hΩnorm hΩweight hΩraise r z hzmem
40    simp only [z, map_sub, map_smul, inner_sub_right, inner_smul_right, cx, cy]
41    ring
42  have he := congrArg (fun v : TensorRegister (n + 1) (Fin d) => ⟪x, v⟫_ℂ) hz
43  simp only [z, inner_sub_right, inner_smul_right, horth, mul_zero, sub_zero,
44    inner_zero_right, inner_self_eq_norm_sq_to_K, hxnorm] at he
45  norm_num at he
46  exact hcy he

48/-- The already constructed exhaustive physical decomposition of `S_mu ⊗ C^d`
49has distinct one-box labels. Branch existence is not assumed here. -/
50theorem exists_fundamental_multiplicityFree_decomposition
51    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
52    (hΩnorm : ‖Ω‖ = 1)
53    (hΩweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
54    (hΩraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0) :
55    ∃ L : List (PhysicalHighestTensor (n + 1) d), ∃ rows : Fin L.length ↪ Fin d,
56      (∀ i, (L.get i).weight = addBox mu (rows i)) ∧
57      L.Pairwise (fun H K => H.sector ⟂ K.sector) ∧
58      physicalSectorListSpan L = tensorProductSector (cyclicSector Ω)
59        (⊤ : Submodule ℂ (TensorRegister 1 (Fin d))) := by
60  classical
61  let W := tensorProductSector (cyclicSector Ω) (⊤ : Submodule ℂ (TensorRegister 1 (Fin d)))
62  have hW : ∀ a b x, x ∈ W → collectiveGenerator (n + 1) a b x ∈ W := by
63    intro a b x hx
64    exact tensorProductSector_generator_invariant _ _
65      (fun a b y hy => cyclicSector_generator_invariant Ω (fun a => (mu a : ℂ)) hΩweight hΩraise a b hy)
66      (fun _ _ _ _ => Submodule.mem_top) a b hx
67  obtain ⟨L, hLorth, hLspan⟩ := exists_physical_cyclic_decomposition W hW
68  have hmem (i : Fin L.length) : (L.get i).vector ∈ W :=
69    (physicalSector_le_listSpan (List.get_mem L i)).trans (le_of_eq hLspan) (L.get i).vector_mem
70  have hrow (i : Fin L.length) : ∃ r : Fin d, (L.get i).weight = addBox mu r := by
71    have hn : (L.get i).vector ≠ 0 := by
72      intro hz
73      have hh := (L.get i).norm_one
74      simp only [hz, norm_zero, zero_ne_one] at hh
75    obtain ⟨r, hr, _, _⟩ := highest_onebox_classification Ω mu hΩnorm hΩweight hΩraise
76      (L.get i).vector (L.get i).weight (hmem i) (L.get i).cartan (L.get i).raising hn
77    exact ⟨r, hr⟩
78  choose rows hrows using hrow
79  have hinj : Function.Injective rows := by
80    intro i j hij
81    by_contra hne
82    have hOrth : (L.get i).sector ⟂ (L.get j).sector := by
83      rcases lt_or_gt_of_ne hne with hlt | hgt
84      · exact hLorth.rel_get_of_lt hlt
85      · exact (hLorth.rel_get_of_lt hgt).symm
86    have hxi : (L.get i).vector ∈ fundamentalHighestSpace Ω (addBox mu (rows i)) := by
87      refine ⟨hmem i, ?_, (L.get i).raising⟩
88      intro a
89      rw [← hrows i]
90      exact (L.get i).cartan a
91    have hxj : (L.get j).vector ∈ fundamentalHighestSpace Ω (addBox mu (rows i)) := by
92      refine ⟨hmem j, ?_, (L.get j).raising⟩
93      intro a
94      rw [hij, ← hrows j]
95      exact (L.get j).cartan a
96    exact fundamental_highest_not_orthogonal Ω mu hΩnorm hΩweight hΩraise (rows i)
97      (L.get i).vector (L.get j).vector hxi hxj (L.get i).norm_one (L.get j).norm_one
98      (hOrth.inner_eq (L.get i).vector_mem (L.get j).vector_mem)
99  exact ⟨L, ⟨rows, hinj⟩, hrows, hLorth, hLspan⟩
