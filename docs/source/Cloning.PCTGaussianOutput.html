24/-- The random-displacement channel, constructed on the entire trace class. -/
25def displacementAverageChannel (μ : Measure (Fin d → ℂ)) [IsProbabilityMeasure μ] :
26    QuantumChannel (Fock d) (Fock d) :=

37theorem displacement_characteristic (z a : Fin d → ℂ) (A : TraceClass (Fock d)) :
38    tracePairing (displacementTraceMap z A) (displacement a) =
39      weylCharacter z (-a) * tracePairing A (displacement a) := by
40  have hc : weylCharacter (-z) a = weylCharacter z (-a) := by

46/-- The full quantum characteristic of the actual displaced-state integral. -/
47theorem gaussian_displacement_characteristic {b : Fin d → ℝ}
48    (hb : ∀ i, 0 < b i) (A : TraceClass (Fock d)) (a : Fin d → ℂ) :
49    tracePairing (∫ z, displacementTraceMap z A ∂gaussianProductMeasure b) (displacement a) =
50      (∏ i, Complex.exp (-((b i * ‖a i‖ ^ 2 : ℝ) : ℂ))) *
51        tracePairing A (displacement a) := by
52  letI := gaussianProductMeasure_probability hb
53  have h := (tracePairingCLM.flip (displacement a)).integral_comp_comm
54    (integrable_displacement (gaussianProductMeasure b) A)

56    ∂gaussianProductMeasure b) = tracePairing (∫ z, displacementTraceMap z A
57    ∂gaussianProductMeasure b) (displacement a) at h

64/-- Exact thermal output whenever its Weyl width is the input width plus the
65actual circular displacement variance. No covariance or moment hypothesis
66stands in for the integral. -/
67theorem gaussian_displacement_productThermal {b q x : Fin d → ℝ}
68    (hb : ∀ i, 0 < b i) (hq0 : ∀ i, 0 < q i) (hq1 : ∀ i, q i < 1)
69    (hx0 : ∀ i, 0 < x i) (hx1 : ∀ i, x i < 1)
70    (hwidth : ∀ i, geometricFourierWidth (x i) = geometricFourierWidth (q i) + b i) :
71    (∫ z, displacementTraceMap z (vectorMixture (numberBasis d) (productGeometric q))
72      ∂gaussianProductMeasure b) = vectorMixture (numberBasis d) (productGeometric x) := by
73  apply characteristic_injective
74  funext a
75  dsimp only
76  rw [gaussian_displacement_characteristic hb,
77    productThermal_characteristic hq0 hq1, productThermal_characteristic hx0 hx1,
78    ← Finset.prod_mul_distrib]
79  apply Finset.prod_congr rfl
80  intro i _
81  change Complex.exp _ * Complex.exp _ =
82    Complex.exp (-((geometricFourierWidth (x i) * ‖a i‖ ^ 2 : ℝ) : ℂ))
83  rw [hwidth, ← Complex.exp_add]
84  congr 1
85  unfold geometricFourierWidth
86  push_cast
87  ring

89/-- Circular tangent-displacement variance in the thermal coordinate `q`. -/
90def pctDisplacementVariance (g q : ℝ) : ℝ := (g - 1) * (1 + q) / (1 - q)

92lemma pctDisplacementVariance_pos {g q : ℝ} (hg : 1 < g)
93    (hq0 : 0 ≤ q) (hq1 : q < 1) : 0 < pctDisplacementVariance g q := by
94  unfold pctDisplacementVariance
95  exact div_pos (mul_pos (sub_pos.mpr hg) (by linarith)) (sub_pos.mpr hq1)

97lemma pct_width {g q : ℝ} (hg : 1 < g) (hq0 : 0 ≤ q) (hq1 : q < 1) :
98    geometricFourierWidth (Thermal.pct g q) =
99      geometricFourierWidth q + pctDisplacementVariance g q := by
100  have hd := (Thermal.pct_denominator_pos hg hq0).ne'
101  have hq := (sub_pos.mpr hq1).ne'
102  have hx := (sub_pos.mpr (Thermal.pct_lt_one hg hq0 hq1)).ne'
103  unfold geometricFourierWidth pctDisplacementVariance Thermal.pct at *
104  field_simp [hd, hq]
105  ring_nf
106  field_simp [hq]
107  <;> ring

109/-- The explicit Gaussian comparison channel sends every strict product
110thermal state to the literal product with parameters `Thermal.pct g q`. -/
111theorem gaussian_displacement_productThermal_pct (g : ℝ) (hg : 1 < g)
112    {q : Fin d → ℝ} (hq0 : ∀ i, 0 < q i) (hq1 : ∀ i, q i < 1) :
113    (∫ z, displacementTraceMap z (vectorMixture (numberBasis d) (productGeometric q))
114      ∂gaussianProductMeasure (fun i => pctDisplacementVariance g (q i))) =
115        vectorMixture (numberBasis d) (productGeometric (fun i => Thermal.pct g (q i))) := by
116  apply gaussian_displacement_productThermal
117    (fun i => pctDisplacementVariance_pos hg (hq0 i).le (hq1 i)) hq0 hq1
118  · intro i
119    exact ((hq0 i).trans (Thermal.lt_amplified hg (hq1 i))).trans
120      (Thermal.amplified_lt_pct hg (hq0 i) (hq1 i))
121  · exact fun i => Thermal.pct_lt_one hg (hq0 i).le (hq1 i)
122  · exact fun i => pct_width hg (hq0 i).le (hq1 i)
