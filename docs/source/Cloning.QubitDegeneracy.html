146theorem constantChannel_uniform {ι : Type*} {l : Filter ι}
147    (p : ι → SimpleSpectrum 2)
148    (hp : ∀ i, Tendsto (fun j => (p j).eigenvalue i) l (𝓝 (1/2:ℝ)))
149    (n m : ℕ) {ε : ℝ} (hε : 0 < ε) :
150    ∀ᶠ j in l, ∀ U : unitary (Matrix (Fin 2) (Fin 2) ℂ),
151      1-ε < spectrumPayoff n m (constantChannel n m) (p j) U := by
152  have h := (diagonal_tensor_fidelity_tendsto p hp m).eventually_const_lt
153    (show 1-ε < 1 by linarith)
154  filter_upwards [h] with j hj U
155  rw [constantChannel_payoff]

160theorem fixed_samples_value_tendsto {ι : Type*} {l : Filter ι}
161    (p : ι → SimpleSpectrum 2)

199/-- The two iterated limits are different for the genuine physical optimum:
200taking the large-sample limit first gives a strict loss, while taking the
201degeneracy limit first gives one at every sample size. -/
202theorem noncommuting_limits (p : ℕ → SimpleSpectrum 2)
203    (hp : ∀ i, Tendsto (fun j => (p j).eigenvalue i) atTop (𝓝 (1/2:ℝ)))
204    (m : ℕ → ℕ) {g : ℝ} (hg : 1 < g)
205    (hgain : Tendsto (fun n => (m n : ℝ)/n) atTop (𝓝 g)) :
206    Tendsto (fun j => limsup (fun n => knownSpectrumValue n (m n) (p j)) atTop)
207      atTop (𝓝 (Thermal.classicalBase g)) ∧
208    Tendsto (fun n => limsup (fun j => knownSpectrumValue n (m n) (p j)) atTop)
209      atTop (𝓝 1) ∧ Thermal.classicalBase g < 1 := by
210  constructor
211  · have he (j : ℕ) := (knownSpectrumValue_tendsto (p j) m g hg hgain).limsup_eq
212    simpa only [he] using orbitalValue_degeneracy p hp hg
213  constructor
214  · have he (n : ℕ) := (fixed_samples_value_tendsto p hp n (m n)).limsup_eq
215    simpa only [he] using (tendsto_const_nhds : Tendsto (fun _ : ℕ => (1:ℝ)) atTop (𝓝 1))
216  · exact Thermal.classicalBase_lt_one hg

218/-- An explicit full-rank simple qubit spectrum with a vanishing gap. -/
219def approachingSpectrum (j : ℕ) : SimpleSpectrum 2 where
220  eigenvalue := ![(1+((j:ℝ)+3)⁻¹)/2, (1-((j:ℝ)+3)⁻¹)/2]
221  positive := by
222    have hj : 0 < (j:ℝ)+3 := by positivity
223    have hi : ((j:ℝ)+3)⁻¹ < 1 := (inv_lt_one₀ hj).mpr (by linarith [Nat.cast_nonneg (α := ℝ) j])
224    intro i
225    fin_cases i
226    · change 0 < (1+((j:ℝ)+3)⁻¹)/2
227      positivity
228    · change 0 < (1-((j:ℝ)+3)⁻¹)/2
229      linarith
230  strictAnti := by
231    have hi : 0 < ((j:ℝ)+3)⁻¹ := by positivity
232    intro i k hik
233    have hi0 : i = 0 := by apply Fin.ext; have := k.isLt; change i.val < k.val at hik; omega
234    have hk1 : k = 1 := by apply Fin.ext; have := k.isLt; change i.val < k.val at hik; omega
235    subst i
236    subst k
237    change (1-((j:ℝ)+3)⁻¹)/2 < (1+((j:ℝ)+3)⁻¹)/2
238    linarith
239  normalized := by simp [Fin.sum_univ_two]; ring
