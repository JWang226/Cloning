51theorem sectorPartitionFunction_div_highest_tendsto
52    (mu : ℕ → Fin d → ℕ) (hmu : ∀ N, Antitone (mu N))
53    (δ : ℕ → ℝ) (hδ : Tendsto δ atTop atTop)
54    (hgap : ∀ᶠ N in atTop, ∀ a : PositiveRoot d,
55      δ N ≤ (mu N a.val.1 : ℝ) - mu N a.val.2)
56    (pN : ℕ → Fin d → ℝ) (p : Fin d → ℝ) (hp : ∀ a, 0 < p a) (hord : StrictAnti p)
57    (hlim : ∀ a, Tendsto (fun N => pN N a) atTop (𝓝 (p a))) :
58    Tendsto (fun N =>
59      sectorPartitionFunction (partitionHighestTensor (mu N) (hmu N)) (mu N)
60        (partitionHighestTensor_cartan (mu N) (hmu N))
61        (partitionHighestTensor_raising_zero (mu N) (hmu N)) (pN N) /
62          (∏ a, pN N a ^ mu N a)) atTop
63      (𝓝 (∏ r : PositiveRoot d, (1-rootBoltzmann p r)⁻¹)) := by
64  have hpN : ∀ᶠ N in atTop, ∀ a, 0 < pN N a :=
65    Filter.eventually_all.mpr (fun a => (hlim a).eventually (eventually_gt_nhds (hp a)))
66  obtain ⟨θ, hθ, hθ1, _, hroot⟩ := exists_eventual_root_height_decay pN p hp hord hlim
67  have hpoint (H : ℕ) : Tendsto (fun N => partitionHeightMass (mu N) (hmu N) (pN N) H)
68      atTop (𝓝 (bosonicHeightMass p H)) :=
69    (bosonicHeightMass_tendsto pN p hp hlim H).congr'
70      (by
