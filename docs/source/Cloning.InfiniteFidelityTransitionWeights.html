14def transitionTargetWeight (q : ι → κ → ℝ) (t : κ → ℝ) (i : ι) (j : κ) : ℝ :=
15  q i j * (t j / ∑ k, q k j)

17theorem transitionTargetWeight_nonneg (q : ι → κ → ℝ) (t : κ → ℝ)
18    (hq : ∀ i j, 0 ≤ q i j) (ht : ∀ j, 0 ≤ t j) (i : ι) (j : κ) :
19    0 ≤ transitionTargetWeight q t i j :=
20  mul_nonneg (hq i j) (div_nonneg (ht j) (Finset.sum_nonneg (fun k _ ↦ hq k j)))

42theorem transitionTargetWeight_affinity (q : ι → κ → ℝ) (t : κ → ℝ)
43    (hq : ∀ i j, 0 ≤ q i j) (ht : ∀ j, 0 ≤ t j) :
44    (∑ z : ι × κ, Real.sqrt (q z.1 z.2) * Real.sqrt (transitionTargetWeight q t z.1 z.2)) =
45      ∑ j, Real.sqrt (∑ i, q i j) * Real.sqrt (t j) := by
46  rw [Fintype.sum_prod_type, Finset.sum_comm]
47  apply Finset.sum_congr rfl
48  intro j hj
49  have he (i : ι) : Real.sqrt (q i j) * Real.sqrt (transitionTargetWeight q t i j) =
50      q i j * Real.sqrt (t j / ∑ k, q k j) := by
51    rw [transitionTargetWeight, Real.sqrt_mul (hq i j), ← mul_assoc, Real.mul_self_sqrt (hq i j)]
52  simp_rw [he]
53  rw [← Finset.sum_mul, Real.sqrt_div (ht j)]
54  have hs : 0 ≤ ∑ i, q i j := Finset.sum_nonneg (fun i _ ↦ hq i j)
55  by_cases hz : (∑ i, q i j) = 0
56  · simp [hz]
57  · have hr : Real.sqrt (∑ i, q i j) ≠ 0 := Real.sqrt_ne_zero'.mpr (lt_of_le_of_ne hs (Ne.symm hz))
58    field_simp
59    rw [Real.sq_sqrt hs]

79/-- A global finite transition lower bound for arbitrary actual positive
80trace-class states. The retained conditional fidelities need no orthogonal
81support hypothesis; all normalization losses are explicit. -/
82theorem transition_rootFidelity_lower (q : ι → κ → ℝ) (t : κ → ℝ)
83    (hq : ∀ i j, 0 ≤ q i j) (ht : ∀ j, 0 ≤ t j) (htsum : ∑ j, t j ≤ 1)
84    (A : ι → κ → PositiveTraceClass H) (B : κ → PositiveTraceClass H)
85    (keep : ι → κ → Prop) (c : ℝ) (hc0 : 0 ≤ c) (hc1 : c ≤ 1)
86    (hgood : ∀ i j, keep i j → c ≤ (A i j).rootFidelity (B j)) :
87    c * (∑ j, Real.sqrt (∑ i, q i j) * Real.sqrt (t j)) -
88      Real.sqrt (∑ z : ι × κ, if keep z.1 z.2 then 0 else q z.1 z.2) ≤
89    (finiteMixture (fun z : ι × κ ↦ q z.1 z.2) (fun z ↦ hq z.1 z.2)
90      (fun z ↦ A z.1 z.2)).rootFidelity (finiteMixture t ht B) := by
91  have hr := transitionTargetWeight_nonneg q t hq ht
92  have hl := finiteMixture_rootFidelity_lower_trimmed
93    (fun z : ι × κ ↦ q z.1 z.2) (fun z : ι × κ ↦ transitionTargetWeight q t z.1 z.2)
94    (fun z ↦ hq z.1 z.2) (fun z ↦ hr z.1 z.2)
95    ((transitionTargetWeight_total_le q t ht).trans htsum)
96    (fun z ↦ A z.1 z.2) (fun z ↦ B z.2) (fun z ↦ keep z.1 z.2) c hc0 hc1
97    (fun z hz ↦ hgood z.1 z.2 hz)
98  rw [transitionTargetWeight_affinity q t hq ht, finiteMixture_product_right] at hl
99  exact hl.trans (rootFidelity_mono_right _ _ _ (finiteMixture_mono_weights _ _ _ ht
100    (transitionTargetWeight_sum_le q t ht) B))
