4/-!
5# Genuine multimode vacuum-signal/idler channel
6 
7The input is the full occupation Hilbert space, not its diagonal subspace.
8For every occupation tuple `l`, the isometry has coherent amplitudes
9`sqrt (productLaw q l k)` on `|l+k,k⟩`. Slicing the first coordinate gives
10an actual CPTP channel whose number diagonal is the negative-binomial mixture
11of the input number diagonal, for arbitrary correlated and coherent idlers.
12-/

35def columnWeight (q : Fin s → ℝ) (l : Occupation s)
36    (p : Occupation s × Occupation s) : ℝ :=
37  if p.1 = l + p.2 then productLaw q l p.2 else 0

113/-- The signal vector at a fixed output idler occupation. -/
114def sliceVector (m : Occupation s) (x : JointFock s) : Fock s := by
115  refine ⟨fun k => x (m, k), memℓp_gen ?_⟩
116  exact ((lp.memℓp x).summable (by norm_num)).comp_injective
117    (fun i j h => congrArg Prod.snd h)

119theorem sliceVector_norm_le (m : Occupation s) (x : JointFock s) :
120    ‖sliceVector m x‖ ≤ ‖x‖ := by
121  apply lp.norm_le_of_tsum_le (by norm_num) (norm_nonneg x)
122  rw [lp.norm_rpow_eq_tsum (by norm_num) x]
123  exact Summable.tsum_le_tsum_of_inj (fun k : Occupation s => (m, k))
124    (fun i j h => congrArg Prod.snd h)
125    (fun p _ => Real.rpow_nonneg (norm_nonneg _) _)
126    (fun _ => le_rfl)
