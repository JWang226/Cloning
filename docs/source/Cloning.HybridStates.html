7/-!
8# Continuous classical registers with trace-class quantum fibres
9 
10A hybrid positive state is an integrable field of positive trace-class
11operators. Positivity is intrinsic in the fibre type; measurable fidelity is
12proved from its trace-norm continuity, rather than assumed for every pair.
13The base measure may be infinite (in particular Lebesgue measure).
14-/

28/-- The closed positive cone of the genuine trace-class Banach space. -/
29def PositiveTraceClass (H : Type*) [NormedAddCommGroup H] [InnerProductSpace ℂ H]
30    [CompleteSpace H] := { A : TraceClass H // 0 ≤ A.1 }

39/-- Root fidelity of two positive trace-class fibres. -/
40def rootFidelity (A B : PositiveTraceClass H) : ℝ :=
41  fidelity A.1.1 B.1.1 A.2 B.2 A.1.2 B.1.2

43lemma rootFidelity_nonneg (A B : PositiveTraceClass H) : 0 ≤ rootFidelity A B :=

66/-- The analytic root fidelity is continuous on the positive trace-class cone. -/
67theorem continuous_rootFidelity :
68    Continuous (fun p : PositiveTraceClass H × PositiveTraceClass H =>
69      rootFidelity p.1 p.2) := by
70  apply continuous_iff_continuousAt.mpr
71  intro p
72  rw [ContinuousAt, ← tendsto_sub_nhds_zero_iff, tendsto_zero_iff_norm_tendsto_zero]
73  have hA : Tendsto (fun q : PositiveTraceClass H × PositiveTraceClass H => q.1.1)
74      (𝓝 p) (𝓝 p.1.1) := continuous_subtype_val.continuousAt.comp
75        continuous_fst.continuousAt
76  have hB : Tendsto (fun q : PositiveTraceClass H × PositiveTraceClass H => q.2.1)
77      (𝓝 p) (𝓝 p.2.1) := continuous_subtype_val.continuousAt.comp
78        continuous_snd.continuousAt
79  have hlim : Tendsto
80      (fun q : PositiveTraceClass H × PositiveTraceClass H =>
81        Real.sqrt ‖q.1.1 - p.1.1‖ * Real.sqrt ‖q.2.1‖ +
82        Real.sqrt ‖q.2.1 - p.2.1‖ * Real.sqrt ‖p.1.1‖) (𝓝 p) (𝓝 0) := by
83    convert (((hA.sub tendsto_const_nhds).norm.sqrt).mul hB.norm.sqrt).add
84      (((hB.sub tendsto_const_nhds).norm.sqrt).mul tendsto_const_nhds) using 1 <;>
85      simp
86  exact squeeze_zero (fun q => norm_nonneg _) (fun q => by
87    simpa only [Real.norm_eq_abs] using rootFidelity_continuity q.1 q.2 p.1 p.2) hlim

103/-- An integrable positive operator-valued density over a continuous classical
104register. Two representatives agreeing almost everywhere describe the same state. -/
105structure PositiveField where

107  measurable : AEStronglyMeasurable value μ
108  integrable : Integrable (fun x => (value x).1) μ

114/-- Every integrable trace-class-valued function with positive fibres gives a

116def ofIntegrable (f : Ω → TraceClass H) (hf : Integrable f μ)

159lemma traceDistance_triangle (R S T : PositiveField (H := H) μ) :
160    R.traceDistance T ≤ R.traceDistance S + S.traceDistance T := by

163        ‖(S.value x).1 - (T.value x).1‖) ∂μ := by
