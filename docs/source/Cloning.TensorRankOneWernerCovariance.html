22theorem pureWernerOutput_unitary {d s L : ℕ} (hcard : Fintype.card (Fin d)=s+1)
23    (U : Matrix (Fin d) (Fin d) ℂ) (hU : Uᴴ*U=1)
24    (ψ : Register (Fin d)) (hψ : ‖ψ‖=1) (S : Finset (Fin L)) :
25    pureWernerOutput hcard (matrixRegister U ψ) ((matrixRegister_norm U hU ψ).trans hψ) S=
26      conjugationLinearMap (tensorOperator L U) (pureWernerOutput hcard ψ hψ S) := by
27  have hUU : U*Uᴴ=1 := mul_eq_one_comm.mp hU
28  have hUs : Uᴴᴴ*Uᴴ=1 := by simpa only [Matrix.conjTranspose_conjTranspose] using hUU
29  have hvec : matrixRegister Uᴴ (matrixRegister U ψ)=ψ := by
30    change ((matrixRegister Uᴴ).comp (matrixRegister U)) ψ=ψ
31    rw [← matrixRegister_mul,hU,matrixRegister_one]
32    rfl
33  have h := pureWernerOutput_compression hcard hcard Uᴴ hUs (matrixRegister U ψ)
34    ((matrixRegister_norm U hU ψ).trans hψ) S
35  have hs : supportFactor S.card L s s=1 := div_self (wernerScale_pos S.card L s).ne'
36  apply Subtype.ext
37  have he := congrArg Subtype.val h
38  change operatorConjugation (tensorMap L Uᴴ).adjoint
39      (pureWernerOperator s (matrixRegister Uᴴ (matrixRegister U ψ)) S)=
40    (supportFactor S.card L s s:ℂ) • pureWernerOperator s (matrixRegister U ψ) S at he
41  rw [hvec,tensorMap_adjoint,Matrix.conjTranspose_conjTranspose,tensorMap_eq_tensorOperator,
42    hs,Complex.ofReal_one,one_smul] at he
