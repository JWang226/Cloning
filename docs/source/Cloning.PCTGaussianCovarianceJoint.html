115/-- The full joint characteristic, with exact classical covariance `2 v Σp`
116and independent circular orbital variances `v (pᵢ+pⱼ)/(pᵢ-pⱼ)`. The additive
117exponent is derived from the integral, with no independence hypothesis. -/
118theorem jointCoordinate_characteristic
119    (u : OrthonormalBasis (Fin (s + 1)) ℂ (Register (A × A)))
120    (p : A → ℝ) (hp : ∀ a, 0 ≤ p a)
121    (hu : u 0 = coefficientVector (schmidtCoefficients p))
122    (hgap : ∀ i j, i < j → 0 < p i - p j)
123    {v : ℝ} (hv : 0 < v) (t : A → ℝ) (ξ : OrbitalPair A → ℂ) :
124    (∫ z, Complex.exp (Complex.I * ((∑ a,
125      t a * classicalCoordinate p (frameTangentMatrix u z) a : ℝ) : ℂ) +
126        ∑ ij, (ξ ij * star (orbitalCoordinate p (frameTangentMatrix u z) ij.1.1 ij.1.2) -
127          star (ξ ij) * orbitalCoordinate p (frameTangentMatrix u z) ij.1.1 ij.1.2))
128      ∂gaussianProductMeasure (fun _ : Fin s => v)) =
129      Complex.exp (-((v * ((∑ a, p a * (t a) ^ 2) - (∑ a, p a * t a) ^ 2) +
130        ∑ ij, v * (p ij.1.1 + p ij.1.2) / (p ij.1.1 - p ij.1.2) * ‖ξ ij‖ ^ 2 : ℝ) : ℂ)) := by
131  simp_rw [← jointTest_phase, coefficientVector_frameTangentMatrix]
132  rw [frameTangent_characteristic u hv, hu, jointTest_norm_sq p t hp hgap,
133    jointTest_reference_sq p t hp]
134  congr 1
135  norm_cast
136  simp_rw [div_eq_mul_inv, mul_sub, mul_add, Finset.mul_sum, mul_assoc]
