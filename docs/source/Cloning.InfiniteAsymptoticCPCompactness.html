17/-- The uniform operator-norm bound constructs a trace-class limit; the
18asymptotic positive-input trace bound gives its sharper trace estimate. -/
19theorem exists_subsequence_cp_limit_of_eventual_trace_bound
20    (L : ℕ → TraceClass H →L[ℂ] TraceClass K) (M c : ℝ)
21    (hbound : ∀ n, ‖L n‖ ≤ M)
22    (hcp : ∀ n, IsCompletelyPositive (L n).toLinearMap)
23    (htrace : ∀ A, 0 ≤ A.1 → ∀ ε : ℝ, 0 < ε → ∀ᶠ n in atTop,
24      (traceCLM (L n A)).re ≤ c * (traceCLM A).re + ε) :
25    ∃ Ψ : TraceClass H →ₗ[ℂ] TraceClass K,
26      IsCompletelyPositive Ψ ∧
27      (∀ A, 0 ≤ A.1 → (traceCLM (Ψ A)).re ≤ c * (traceCLM A).re) ∧
28      ∃ φ : ℕ → ℕ, StrictMono φ ∧
29        (∀ A x y, Tendsto (fun n => ⟪x, (L (φ n) A).1 y⟫_ℂ) atTop
30          (𝓝 ⟪x, (Ψ A).1 y⟫_ℂ)) ∧
31        (∀ A (O : K →L[ℂ] K), IsCompactOperator O →
32          Tendsto (fun n => tracePairing (L (φ n) A) O) atTop
33            (𝓝 (tracePairing (Ψ A) O))) := by
34  have hM : 0 ≤ M := (norm_nonneg (L 0)).trans (hbound 0)
35  have htraceM (n : ℕ) (A : TraceClass H) (hA : 0 ≤ A.1) :
36      (traceCLM (L n A)).re ≤ M * (traceCLM A).re := by
37    have hp : 0 ≤ (L n A).1 := (hcp n).map_nonneg A hA
38    calc
39      (traceCLM (L n A)).re = ‖L n A‖ :=
40        (TraceClass.norm_eq_trace_re_of_nonneg _ hp).symm
41      _ ≤ ‖L n‖ * ‖A‖ := (L n).le_opNorm A
42      _ ≤ M * ‖A‖ := mul_le_mul_of_nonneg_right (hbound n) (norm_nonneg A)
43      _ = M * (traceCLM A).re := by
44        rw [TraceClass.norm_eq_trace_re_of_nonneg A hA]
45        rfl
46  obtain ⟨Ψ, hp, _, φ, hφ, hlim, hcompact⟩ :=
47    exists_subsequence_completelyPositive_limit (fun n => (L n).toLinearMap)
48      M hM hcp htraceM
49  refine ⟨Ψ, hp, ?_, φ, hφ, hlim, hcompact⟩
50  intro A hA
51  apply Cloning.InfiniteTraceClassAsymptoticBound.trace_le_of_diagonal_tendsto (l := atTop)
52    (fun n => L (φ n) A) (Ψ A) (fun n => (hcp (φ n)).map_nonneg A hA)
53    (hp.map_nonneg A hA) (c * (traceCLM A).re) (fun x => hlim A x x)
54  intro ε hε
55  exact hφ.tendsto_atTop.eventually (htrace A hA ε hε)

57/-- The full common-subsequence conclusion including exact covariance, for
58any parameter set, under eventual positive-input trace estimates. -/
59theorem exists_subsequence_covariant_cp_limit_of_eventual_trace_bound {G : Type*}
60    (L : ℕ → TraceClass H →L[ℂ] TraceClass K) (M c : ℝ)
61    (hbound : ∀ n, ‖L n‖ ≤ M)
62    (hcp : ∀ n, IsCompletelyPositive (L n).toLinearMap)
63    (htrace : ∀ A, 0 ≤ A.1 → ∀ ε : ℝ, 0 < ε → ∀ᶠ n in atTop,
64      (traceCLM (L n A)).re ≤ c * (traceCLM A).re + ε)
65    (U : G → H →L[ℂ] H) (V : G → K →L[ℂ] K)
66    (hdefect : ∀ g A, Tendsto (fun n =>
67      ‖L n (sandwichCLM (U g) (star (U g)) A) -
68        sandwichCLM (V g) (star (V g)) (L n A)‖) atTop (𝓝 (0 : ℝ))) :
69    ∃ Ψ : TraceClass H →ₗ[ℂ] TraceClass K,

72      (∀ g A, Ψ (sandwichCLM (U g) (star (U g)) A) =
73        sandwichCLM (V g) (star (V g)) (Ψ A)) ∧
74      ∃ φ : ℕ → ℕ, StrictMono φ ∧
75        (∀ A x y, Tendsto (fun n => ⟪x, (L (φ n) A).1 y⟫_ℂ) atTop
76          (𝓝 ⟪x, (Ψ A).1 y⟫_ℂ)) ∧
77        (∀ A (O : K →L[ℂ] K), IsCompactOperator O →
78          Tendsto (fun n => tracePairing (L (φ n) A) O) atTop

86private lemma eventual_trace_bound_of_limsup
87    (L : ℕ → TraceClass H →L[ℂ] TraceClass K) (M c : ℝ)
88    (hbound : ∀ n, ‖L n‖ ≤ M)
89    (hcp : ∀ n, IsCompletelyPositive (L n).toLinearMap)
90    (htrace : ∀ A, 0 ≤ A.1 →
91      Filter.limsup (fun n => (traceCLM (L n A)).re) atTop ≤ c * (traceCLM A).re)
92    (A : TraceClass H) (hA : 0 ≤ A.1) (ε : ℝ) (hε : 0 < ε) :
93    ∀ᶠ n in atTop, (traceCLM (L n A)).re ≤ c * (traceCLM A).re + ε := by
94  have hb : BddAbove (Set.range (fun n => (traceCLM (L n A)).re)) := by
95    refine ⟨M * ‖A‖, ?_⟩
96    rintro _ ⟨n, rfl⟩
97    calc
98      (traceCLM (L n A)).re = ‖L n A‖ :=
99        (TraceClass.norm_eq_trace_re_of_nonneg _ ((hcp n).map_nonneg A hA)).symm
100      _ ≤ ‖L n‖ * ‖A‖ := (L n).le_opNorm A
101      _ ≤ M * ‖A‖ := mul_le_mul_of_nonneg_right (hbound n) (norm_nonneg A)
102  exact (eventually_lt_add_pos_of_limsup_le hb.isBoundedUnder_of_range
103    (htrace A hA) hε).mono (fun _ h => h.le)
