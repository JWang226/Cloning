67/-- Commutation with every physical unitary already forces commutation with

69theorem cyclicTensorOperator_unitary_commutes_generators
70    (T : cyclicSector Ω →L[ℂ] cyclicSector Ω)
71    (hT : ∀ U : unitary (Matrix (Fin d) (Fin d) ℂ),
72      T * cyclicTensorOperator Ω mu hweight hraise U =
73        cyclicTensorOperator Ω mu hweight hraise U * T) :
74    ∀ a b, T * cyclicGenerator Ω mu hweight hraise a b =
75      cyclicGenerator Ω mu hweight hraise a b * T := by
76  let C : Matrix (Fin d) (Fin d) ℂ →ₗ[ℂ] (cyclicSector Ω →L[ℂ] cyclicSector Ω) :=
77    (sectorCompression (cyclicSector Ω)).toLinearMap.comp collectiveMatrixLM
78  have hskew (Y : Matrix (Fin d) (Fin d) ℂ) (hY : star Y = -Y) : T * C Y = C Y * T := by
79    have he (t : ℝ) : matrixExpPath Y (t : ℂ) ∈ unitary (Matrix (Fin d) (Fin d) ℂ) := by
80      apply exp_mem_unitary_of_mem_skewAdjoint
81      change star ((t : ℂ) • Y) = -((t : ℂ) • Y)
82      simp only [star_smul, Complex.star_def, Complex.conj_ofReal, hY, smul_neg]
83    have hfun : (fun t : ℝ => T * cyclicTensorOperator Ω mu hweight hraise (matrixExpPath Y (t : ℂ))) =
84        (fun t : ℝ => cyclicTensorOperator Ω mu hweight hraise (matrixExpPath Y (t : ℂ)) * T) :=
85      funext (fun t => hT ⟨_, he t⟩)
86    have hd := cyclicTensorOperator_exp_hasDerivAt_zero Ω mu hweight hraise Y
87    have hleft := hd.const_mul T
88    have hright := hd.mul_const T
89    rw [hfun] at hleft
90    exact hleft.unique hright
91  have hall (X : Matrix (Fin d) (Fin d) ℂ) : T * C X = C X * T := by
92    let Y := X - star X
93    let Z := Complex.I • (X + star X)
94    have hY : star Y = -Y := by simp [Y]
95    have hZ : star Z = -Z := by
96      simp only [Z, star_smul, star_add, star_star, Complex.star_def, Complex.conj_I,
97        neg_smul, smul_add]
98      abel
99    have h1 := hskew Y hY
100    have h2 := hskew Z hZ
101    have he : (2 : ℂ) • C X = C Y - Complex.I • C Z := by
102      rw [← map_smul, ← map_smul, ← map_sub]
103      congr 1
104      simp only [Y, Z, smul_smul, Complex.I_mul_I, neg_one_smul]
105      module
106    have hc : T * ((2 : ℂ) • C X) = ((2 : ℂ) • C X) * T := by
107      rw [he]
108      apply ContinuousLinearMap.ext
109      intro x
110      simp only [ContinuousLinearMap.mul_apply, ContinuousLinearMap.sub_apply,
111        ContinuousLinearMap.smul_apply, map_sub, map_smul]
112      have h1x := DFunLike.congr_fun h1 x
113      have h2x := DFunLike.congr_fun h2 x
114      simp only [ContinuousLinearMap.mul_apply] at h1x h2x
115      rw [h1x,h2x]
116    apply ContinuousLinearMap.ext
117    intro x
118    have hx := DFunLike.congr_fun hc x
119    simp only [ContinuousLinearMap.mul_apply, ContinuousLinearMap.smul_apply, map_smul] at hx
120    exact (smul_right_injective _ (by norm_num : (2 : ℂ) ≠ 0)) hx
121  intro a b
122  have hh := hall (Matrix.single a b (1 : ℂ))
123  simpa only [C, LinearMap.comp_apply, ContinuousLinearMap.coe_coe, collectiveMatrixLM_apply,
124    collectiveMatrix_single, one_smul, sectorCompression_collectiveGenerator Ω mu hweight hraise] using hh
