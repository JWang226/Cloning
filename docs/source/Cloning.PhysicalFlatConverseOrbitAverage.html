20theorem partition_fidelity_move_unitary (mu : Fin d → ℕ) (hmu : Antitone mu)
21    (M T : Matrix (PartitionIndex mu hmu) (PartitionIndex mu hmu) ℂ)
22    (hM : M.PosSemidef) (hT : T.PosSemidef)
23    (U : unitary (Matrix (Fin d) (Fin d) ℂ)) :
24    fidelity M (partitionActionMatrix mu hmu U * T * (partitionActionMatrix mu hmu U)ᴴ) =
25      fidelity (partitionActionMatrix mu hmu ((U⁻¹ : unitary (Matrix (Fin d) (Fin d) ℂ)) : Matrix (Fin d) (Fin d) ℂ) * M * (partitionActionMatrix mu hmu ((U⁻¹ : unitary (Matrix (Fin d) (Fin d) ℂ)) : Matrix (Fin d) (Fin d) ℂ))ᴴ) T := by
26  let V := partitionActionMatrix mu hmu U
27  have hV : Vᴴ * V = 1 := partitionActionMatrix_unitary mu hmu U
28  have hV' : V * Vᴴ = 1 := mul_eq_one_comm.mp hV
29  have he : V * (Vᴴ * M * V) * Vᴴ = M := by
30    calc
31      _ = (V*Vᴴ)*M*(V*Vᴴ) := by noncomm_ring
32      _ = M := by rw [hV',one_mul,mul_one]
33  have hh := fidelity_unitary_conjugation V hV (hM.conjTranspose_mul_mul_same V) hT
34  rw [he] at hh
35  simpa only [partitionActionMatrix_inv, Matrix.conjTranspose_conjTranspose, V] using hh
