5# Positive-definite regularization of positive complex matrices

88/-- Inverse square roots undo the positive sandwich of a positive-definite matrix. -/
89theorem sqrt_inv_sandwich {A : Matrix n n ℂ} (hA : A.PosDef) :
90    (CFC.sqrt A)⁻¹ * A * (CFC.sqrt A)⁻¹ = 1 := by
91  calc
92    (CFC.sqrt A)⁻¹ * A * (CFC.sqrt A)⁻¹ =
93        (CFC.sqrt A)⁻¹ * (CFC.sqrt A * CFC.sqrt A) * (CFC.sqrt A)⁻¹ := by
94      rw [sqrt_mul_self hA.posSemidef]
95    _ = ((CFC.sqrt A)⁻¹ * CFC.sqrt A) * (CFC.sqrt A * (CFC.sqrt A)⁻¹) := by
96      simp only [Matrix.mul_assoc]
97    _ = 1 := by rw [sqrt_inv_mul hA, sqrt_mul_inv hA, Matrix.one_mul]

99/-- A general limit lemma for the trace of the positive square root along the
100positive cone. It is useful when removing a positive-definite regularization. -/
101theorem tendsto_trace_sqrt {ι : Type*} {l : Filter ι} {A : Matrix n n ℂ}
102    {f : ι → Matrix n n ℂ} (hA : A.PosSemidef)
103    (hf : Tendsto f l (nhds A)) (hpos : ∀ᶠ i in l, (f i).PosSemidef) :
104    Tendsto (fun i => (CFC.sqrt (f i)).trace.re) l (nhds (CFC.sqrt A).trace.re) := by
105  have hin : Tendsto f l (nhdsWithin A {X : Matrix n n ℂ | X.PosSemidef}) :=
106    tendsto_nhdsWithin_iff.mpr ⟨hf, hpos⟩
107  have hs : Tendsto (CFC.sqrt : Matrix n n ℂ → Matrix n n ℂ)
108      (nhdsWithin A {X : Matrix n n ℂ | X.PosSemidef}) (nhds (CFC.sqrt A)) :=
109    continuousOn_matrix_sqrt A hA
110  have ht : Continuous (fun X : Matrix n n ℂ => X.trace.re) := by
111    unfold Matrix.trace
112    fun_prop
113  exact ht.continuousAt.tendsto.comp (hs.comp hin)

115theorem tendsto_trace_sqrt_regularize {A : Matrix n n ℂ} (hA : A.PosSemidef) :
116    Tendsto (fun ε => (CFC.sqrt (regularize A ε)).trace.re)
117      (nhdsWithin 0 (Set.Ioi 0)) (nhds (CFC.sqrt A).trace.re) := by
