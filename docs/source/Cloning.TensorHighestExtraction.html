15/-- Minimum height among nonzero projected computational basis vectors yields
16an actual joint highest vector. No spectral or representation premise is used. -/
17theorem exists_highest_projection_basis
18    (W : Submodule ℂ (TensorRegister n (Fin d)))
19    (hW : ∀ a b x, x ∈ W → collectiveGenerator n a b x ∈ W)
20    (hWne : W ≠ ⊥) :
21    ∃ w : Fin n → Fin d, W.starProjection (registerBasis _ w) ≠ 0 ∧
22      ∀ a b, a < b → collectiveGenerator n a b (W.starProjection (registerBasis _ w)) = 0 := by
23  classical
24  let S := Finset.univ.filter (fun w : Fin n → Fin d => W.starProjection (registerBasis _ w) ≠ 0)
25  have hS : S.Nonempty := by
26    obtain ⟨w, hw⟩ := invariant_exists_projection_basis_ne_zero W hWne
27    exact ⟨w, by simpa only [S, Finset.mem_filter, Finset.mem_univ, true_and] using hw⟩
28  obtain ⟨w, hw, hmin⟩ := S.exists_min_image physicalWordHeight hS
29  have hwne : W.starProjection (registerBasis _ w) ≠ 0 := (Finset.mem_filter.mp hw).2
30  refine ⟨w, hwne, ?_⟩
31  intro a b hab
32  let Ω := W.starProjection (registerBasis (Fin n → Fin d) w)
33  have hweight : ∀ i, collectiveGenerator n i i Ω = (occupancy w i : ℂ) • Ω :=
34    invariant_projection_basis_cartan W hW w
35  by_contra hz
36  obtain ⟨v, hv⟩ : ∃ v : Fin n → Fin d, collectiveGenerator n a b Ω v ≠ 0 := by
37    by_contra! hn
38    apply hz
39    ext v
40    exact hn v
41  have hvW : collectiveGenerator n a b Ω ∈ W :=
42    hW a b Ω (W.starProjection_apply_mem _)
43  have hvS : v ∈ S := Finset.mem_filter.mpr ⟨Finset.mem_univ _,
44    projection_basis_ne_zero_of_coefficient W hvW v hv⟩
45  rw [collectiveGenerator_apply] at hv
46  obtain ⟨t, _, ht⟩ := Finset.exists_ne_zero_of_sum_ne_zero hv
47  have hva : v t = a := by
48    by_contra h
49    simp only [h, if_false, ne_eq, not_true_eq_false] at ht
50  have hu : Ω (Function.update v t b) ≠ 0 := by simpa only [hva, if_true] using ht
51  have hocc := cartan_weight_coefficient Ω (occupancy w) hweight (Function.update v t b) hu
52  have hheight : physicalWordHeight (Function.update v t b) = physicalWordHeight w := by
53    simp only [physicalWordHeight_eq_occupancy, hocc]
54  have he := physicalWordHeight_update v t b
55  rw [hheight, hva] at he
56  have hle := hmin v hvS
57  have habv : a.val < b.val := hab
58  omega

60/-- Every nonzero subspace invariant under the actual tensor matrix units
61contains a unit highest tensor of a genuine natural partition of `n`. -/
62theorem exists_partitionHighest_in_invariant
63    (W : Submodule ℂ (TensorRegister n (Fin d)))
64    (hW : ∀ a b x, x ∈ W → collectiveGenerator n a b x ∈ W)
65    (hWne : W ≠ ⊥) :
66    ∃ mu : Fin d → ℕ, Antitone mu ∧ (∑ a, mu a) = n ∧
67      ∃ Ω : TensorRegister n (Fin d), Ω ∈ W ∧ ‖Ω‖ = 1 ∧
68        (∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω) ∧
69        (∀ a b, a < b → collectiveGenerator n a b Ω = 0) := by
70  obtain ⟨w, hw, hraise⟩ := exists_highest_projection_basis W hW hWne
71  let x := W.starProjection (registerBasis (Fin n → Fin d) w)
72  let Ω := (‖x‖⁻¹ : ℂ) • x
73  have hΩ : ‖Ω‖ = 1 := norm_smul_inv_norm hw
74  have hweight : ∀ a, collectiveGenerator n a a Ω = (occupancy w a : ℂ) • Ω := by
75    intro a
76    simp only [Ω, map_smul, invariant_projection_basis_cartan W hW, x]
77    exact smul_comm _ _ _
78  have hΩraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0 := by
79    intro a b hab
80    simp only [Ω, x, map_smul, hraise a b hab, smul_zero]
81  refine ⟨occupancy w, highest_weight_antitone Ω (occupancy w) hweight hΩraise hΩ,
82    sum_occupancy w, Ω, W.smul_mem _ (W.starProjection_apply_mem _), hΩ, hweight, hΩraise⟩
