19theorem fullChannel_payoff_lower (d n m : ℕ) (hnm : n<m)
20    (ρ : Cloning.MatrixFidelity.State (Fin (d+1))) :
21    Real.sqrt (wernerScale n m (d*(d+2)))≤statePayoff n m (fullChannel d n m) ρ := by
22  rw [fullChannel,extendChannel_payoff_of_gt _ n m hnm ρ]
23  have hh : Real.sqrt (wernerScale n (n+(m-n)) (d*(d+2)))≤
24      statePayoff n (n+(m-n)) (PCTGlobal.channel (purification_register_card d) n (m-n)) ρ :=
25    outputState_fidelity_lower (purification_register_card d) ρ n (m-n)
26  convert hh using 1
27  rw [Nat.add_sub_of_le hnm.le]

34lemma sqrt_wernerScale_tendsto (s : ℕ) (m : ℕ→ℕ) (γ : ℝ) (hγ : 1<γ)
35    (hgain : Tendsto (fun n => (m n:ℝ)/n) atTop (𝓝 γ)) :
36    Tendsto (fun n => Real.sqrt (wernerScale n (m n) s)) atTop (𝓝 (γ^(-(s:ℝ)/2))) := by

42/-- Every fixed gain admits the uniform PCT lower envelope. -/
43theorem eventually_value_lower (d : ℕ) (m : ℕ→ℕ) (γ : ℝ) (hγ : 1<γ)
44    (hgain : Tendsto (fun n => (m n:ℝ)/n) atTop (𝓝 γ)) (ε : ℝ) (hε : 0<ε) :
45    ∀ᶠ n in atTop,γ^(-(((d+1:ℕ):ℝ)^2-1)/2)-ε<value d n (m n) := by
46  have hh := sqrt_wernerScale_tendsto (d*(d+2)) m γ hγ hgain
47  have he : -((d*(d+2):ℕ):ℝ)/2= -(((d+1:ℕ):ℝ)^2-1)/2 := by push_cast; ring
48  rw [he] at hh
49  filter_upwards [hh.eventually (eventually_gt_nhds (sub_lt_self _ hε)),
50    eventually_output_gt_input m hγ hgain] with n hn hnm
51  exact hn.trans_le (sqrt_wernerScale_le_value d n (m n) hnm)

53/-- The manuscript all-density minimax lower bound, with no spectral or
54rank premise on the adversarial density matrices. -/
55theorem lower_le_liminf_value (d : ℕ) (m : ℕ→ℕ) (γ : ℝ) (hγ : 1<γ)
56    (hgain : Tendsto (fun n => (m n:ℝ)/n) atTop (𝓝 γ)) :
57    γ^(-(((d+1:ℕ):ℝ)^2-1)/2)≤liminf (fun n => value d n (m n)) atTop := by
58  apply (le_liminf_iff
59    (isCoboundedUnder_ge_of_le atTop (fun n => value_le_one d n (m n)))
60    (Filter.isBoundedUnder_of_eventually_ge (Eventually.of_forall (fun n => value_nonneg d n (m n))))).mpr
61  intro a ha
62  have hh := eventually_value_lower d m γ hγ hgain
63    (γ^(-(((d+1:ℕ):ℝ)^2-1)/2)-a) (sub_pos.mpr ha)
64  simpa only [sub_sub_cancel] using hh
