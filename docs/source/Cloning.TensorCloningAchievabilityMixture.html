51/-- Exact joint-mixture output of the concrete physical cloner. -/
52theorem knownSpectrumChannel_rotated_apply (n m d : ℕ) (p : Fin d → ℝ)
53    (hp : ∀ a, 0 < p a) (hs : ∑ a, p a = 1) (U : Matrix (Fin d) (Fin d) ℂ) :
54    (knownSpectrumChannel n m d p (fun a => (hp a).le) hs).toLinearMap
55      (matrixTensorPower (U * Matrix.diagonal (fun a => (p a : ℂ)) * Uᴴ) n) =
56      ∑ ij : SchurCopy n d × SchurCopy m d,
57        (jointCopyWeight n m d p ij : ℂ) •
58          (copyTransition n m d ij.1 ij.2).toLinearMap
59            (canonicalRotatedGibbs ((recursivePhysicalDecomposition n d).get ij.1) U p) := by
60  rw [knownSpectrumChannel, channel_apply]
61  simp only [matrixTensorPower_canonical_block, canonicalTensorWeight_rotated_diagonal _ U p hp,
62    map_smul, Fintype.sum_prod_type, jointCopyWeight, Complex.ofReal_mul, mul_smul]
63  apply Finset.sum_congr rfl
64  intro i _
65  apply Finset.sum_congr rfl
66  intro j _
67  exact smul_comm _ _ _

69/-- Jointly repeat the actual target copy mixture over the source-copy law. -/
70theorem rotatedTensor_eq_jointCopy_mixture (n m d : ℕ) (p : Fin d → ℝ)
71    (hp : ∀ a, 0 < p a) (hs : ∑ a, p a = 1) (U : Matrix (Fin d) (Fin d) ℂ) :
72    matrixTensorPower (U * Matrix.diagonal (fun a => (p a : ℂ)) * Uᴴ) m =
73      ∑ ij : SchurCopy n d × SchurCopy m d,
74        (jointCopyWeight n m d p ij : ℂ) •
75          conjugationLinearMap ((recursivePhysicalDecomposition m d).get ij.2).canonicalEmbedding.toContinuousLinearMap
76            (canonicalRotatedGibbs ((recursivePhysicalDecomposition m d).get ij.2) U p) := by
77  rw [matrixTensorPower_rotated_eq_copy_mixture U p hp]
78  simp only [jointCopyWeight, Fintype.sum_prod_type, Complex.ofReal_mul, mul_smul]
79  rw [Finset.sum_comm]
80  apply Finset.sum_congr rfl
81  intro j _
82  let X := conjugationLinearMap ((recursivePhysicalDecomposition m d).get j).canonicalEmbedding.toContinuousLinearMap
83    (canonicalRotatedGibbs ((recursivePhysicalDecomposition m d).get j) U p)
84  have hh := Finset.sum_smul (R := ℂ)
85    (f := fun i : SchurCopy n d => (knownCopyWeight n d p i : ℂ))
86    (s := Finset.univ) (x := (knownCopyWeight m d p j : ℂ) • X)
87  rw [← Complex.ofReal_sum, knownCopyWeight_sum n d p (fun a => (hp a).le) hs,
88    Complex.ofReal_one, one_smul] at hh
89  exact hh

92/-- The physical payoff dominates the average of the genuine conditional
93sector fidelities. No compatibility, orthogonality or asymptotic premise is
94needed for this finite inequality. -/
95theorem knownSpectrumChannel_payoff_lower {d : ℕ} (n m : ℕ)
96    (p : SimpleSpectrum (d+1)) (U : unitary (Matrix (Fin (d+1)) (Fin (d+1)) ℂ)) :
97    (∑ ij : SchurCopy n (d+1) × SchurCopy m (d+1),
98      jointCopyWeight n m (d+1) p.eigenvalue ij *
99        transitionFidelity n m (d+1) p.eigenvalue p.positive U ij.1 ij.2) ≤
100      spectrumPayoff n m (knownSpectrumChannel n m (d+1) p.eigenvalue
101        (fun a => (p.positive a).le) p.normalized) p U := by
102  let A (ij : SchurCopy n (d+1) × SchurCopy m (d+1)) :=
103    (copySectorState ((recursivePhysicalDecomposition n (d+1)).get ij.1) U p.eigenvalue p.positive).map
104      (copyTransition n m (d+1) ij.1 ij.2).toPositiveTracePreservingMap
105  let B (ij : SchurCopy n (d+1) × SchurCopy m (d+1)) :=
106    (copySectorState ((recursivePhysicalDecomposition m (d+1)).get ij.2) U p.eigenvalue p.positive).map
107      (QuantumChannel.ofIsometry ((recursivePhysicalDecomposition m (d+1)).get ij.2).canonicalEmbedding).toPositiveTracePreservingMap
108  have hmono (ij : SchurCopy n (d+1) × SchurCopy m (d+1)) :
109      transitionFidelity n m (d+1) p.eigenvalue p.positive U ij.1 ij.2 ≤
110        (A ij).rootFidelity (B ij) := by
111    let C := (recursivePhysicalDecomposition n (d+1)).get ij.1
112    let D := (recursivePhysicalDecomposition m (d+1)).get ij.2
113    let S := (copySectorState C U p.eigenvalue p.positive).map
114      (partitionTransitionChannel C.weight D.weight C.weight_antitone D.weight_antitone).toPositiveTracePreservingMap
115    let T := copySectorState D U p.eigenvalue p.positive
116    exact InfiniteFidelity.fidelity_data_processing
117      (QuantumChannel.ofIsometry D.canonicalEmbedding) S.1 T.1 S.2 T.2
118  have hconc := PositiveTraceClass.finiteMixture_rootFidelity_lower_same
119    (jointCopyWeight n m (d+1) p.eigenvalue) (jointCopyWeight_nonneg n m (d+1) p.eigenvalue) A B
120  have hA : (PositiveTraceClass.finiteMixture (jointCopyWeight n m (d+1) p.eigenvalue)
121      (jointCopyWeight_nonneg n m (d+1) p.eigenvalue) A).1 =
122      (knownSpectrumChannel n m (d+1) p.eigenvalue (fun a => (p.positive a).le) p.normalized).toLinearMap
123        (tensorState (orbitState p U) n).1 := by
124    symm
125    exact knownSpectrumChannel_rotated_apply n m (d+1) p.eigenvalue p.positive p.normalized U
126  have hB : (PositiveTraceClass.finiteMixture (jointCopyWeight n m (d+1) p.eigenvalue)
127      (jointCopyWeight_nonneg n m (d+1) p.eigenvalue) B).1 = (tensorState (orbitState p U) m).1 := by
128    symm
129    exact rotatedTensor_eq_jointCopy_mixture n m (d+1) p.eigenvalue p.positive p.normalized U
130  have hA' : PositiveTraceClass.finiteMixture (jointCopyWeight n m (d+1) p.eigenvalue)
131      (jointCopyWeight_nonneg n m (d+1) p.eigenvalue) A =
132      (tensorState (orbitState p U) n).map
133        (knownSpectrumChannel n m (d+1) p.eigenvalue (fun a => (p.positive a).le) p.normalized).toPositiveTracePreservingMap :=
134    Subtype.ext hA
135  have hB' : PositiveTraceClass.finiteMixture (jointCopyWeight n m (d+1) p.eigenvalue)
136      (jointCopyWeight_nonneg n m (d+1) p.eigenvalue) B = tensorState (orbitState p U) m :=
137    Subtype.ext hB
138  rw [hA', hB'] at hconc
139  exact (Finset.sum_le_sum (fun ij _ =>
140    mul_le_mul_of_nonneg_left (hmono ij) (jointCopyWeight_nonneg n m (d+1) p.eigenvalue ij))).trans hconc
