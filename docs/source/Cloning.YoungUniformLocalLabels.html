13def sampleLabel (d N : ℕ) (p : Fin (d + 1) → ℝ) (x : rootSpace d) : Lattice d (N : ℤ) :=

50/-- Actual rounded counts divided by sample size track every moving spectrum

52theorem sampleLabel_ratio_tendsto (d : ℕ) (n : ℕ → ℕ) (hn : Tendsto n atTop atTop)
53    (p : ℕ → Fin (d + 1) → ℝ) (hp : ∀ k, ∑ i, p k i = 1)

55    (x : rootSpace d) (i : Fin (d + 1)) :
56    Tendsto (fun k ↦ ((sampleLabel d (n k) (p k) x).1 i : ℝ) / (n k : ℝ))
57      atTop (𝓝 (p₀ i)) := by
58  have hc := (sampleLabel_center_tendsto d n hn p hp x i).mul
