16theorem coeFn_Lp_add_smul (A B : Lp (TraceClass H) 1 μ) (a b : ℝ) :
17    (a • A + b • B : Lp (TraceClass H) 1 μ) =ᵐ[μ]
18      fun y => a • A y + b • B y := by
19  filter_upwards [Lp.coeFn_add (a • A) (b • B), Lp.coeFn_smul a A,
20    Lp.coeFn_smul b B] with y hadd ha hb
21  simp only [hadd, Pi.add_apply, ha, hb, Pi.smul_apply]

23def positiveL1Pairs : Set (Lp (TraceClass H) 1 μ × Lp (TraceClass H) 1 μ) :=
24  {p | (∀ᵐ y ∂μ, 0 ≤ (p.1 y).1) ∧ (∀ᵐ y ∂μ, 0 ≤ (p.2 y).1)}

43theorem convex_positiveL1Pairs : Convex ℝ (positiveL1Pairs (H := H) (μ := μ)) := by
44  intro p hp q hq a b ha hb _
45  constructor
46  · filter_upwards [hp.1, hq.1, coeFn_Lp_add_smul p.1 q.1 a b] with y hpy hqy hy
47    change 0 ≤ ((a • p.1 + b • q.1) y).1
48    rw [hy]
49    change 0 ≤ a • (p.1 y).1 + b • (q.1 y).1
50    exact add_nonneg (smul_nonneg ha hpy) (smul_nonneg hb hqy)
51  · filter_upwards [hp.2, hq.2, coeFn_Lp_add_smul p.2 q.2 a b] with y hpy hqy hy
52    change 0 ≤ ((a • p.2 + b • q.2) y).1
53    rw [hy]
54    change 0 ≤ a • (p.2 y).1 + b • (q.2 y).1
55    exact add_nonneg (smul_nonneg ha hpy) (smul_nonneg hb hqy)

75/-- Fibrewise infinite-dimensional joint concavity integrates to the actual
76L1 fidelity; no commutativity of fibres or classical--quantum independence is used. -/
77theorem concaveOn_extendedL1RootFidelity :
78    ConcaveOn ℝ (positiveL1Pairs (H := H) (μ := μ)) extendedL1RootFidelity := by
79  refine ⟨convex_positiveL1Pairs, ?_⟩
80  intro p hp q hq a b ha hb hab
81  have hc := convex_positiveL1Pairs hp hq ha hb hab
82  have hip := PositiveL1.integrable_rootFidelity (⟨p.1, hp.1⟩ : PositiveL1 H μ) ⟨p.2, hp.2⟩
83  have hiq := PositiveL1.integrable_rootFidelity (⟨q.1, hq.1⟩ : PositiveL1 H μ) ⟨q.2, hq.2⟩
84  have hic := PositiveL1.integrable_rootFidelity
85    (⟨(a • p + b • q).1, hc.1⟩ : PositiveL1 H μ) ⟨(a • p + b • q).2, hc.2⟩
86  simp only [smul_eq_mul]
87  rw [extendedL1RootFidelity_eq_integral hp, extendedL1RootFidelity_eq_integral hq,
88    extendedL1RootFidelity_eq_integral hc, ← integral_const_mul, ← integral_const_mul,
89    ← integral_add (hip.const_mul a) (hiq.const_mul b)]
90  apply integral_mono_ae ((hip.const_mul a).add (hiq.const_mul b)) hic
91  filter_upwards [hp.1, hp.2, hq.1, hq.2,
92    coeFn_Lp_add_smul p.1 q.1 a b, coeFn_Lp_add_smul p.2 q.2 a b] with y hp₁ hp₂ hq₁ hq₂ h₁ h₂
93  change a * extendedRootFidelity (p.1 y, p.2 y) + b * extendedRootFidelity (q.1 y, q.2 y) ≤
94    extendedRootFidelity ((a • p.1 + b • q.1) y, (a • p.2 + b • q.2) y)
95  rw [h₁, h₂]
96  exact concaveOn_extendedRootFidelity.2 ⟨hp₁, hp₂⟩ ⟨hq₁, hq₂⟩ ha hb hab

102theorem integrable_family_rootFidelity (A B : Ξ → PositiveL1 H μ)
103    (hA : Integrable (fun ξ => (A ξ).1) ν) (hB : Integrable (fun ξ => (B ξ).1) ν) :
104    Integrable (fun ξ => (A ξ).rootFidelity (B ξ)) ν := by
105  have hmA : AEStronglyMeasurable A ν :=
106    Topology.IsEmbedding.subtypeVal.aestronglyMeasurable_comp_iff.mp hA.aestronglyMeasurable
107  have hmB : AEStronglyMeasurable B ν :=
108    Topology.IsEmbedding.subtypeVal.aestronglyMeasurable_comp_iff.mp hB.aestronglyMeasurable
109  apply (integrable_sqrt_mul_sqrt hA.norm hB.norm (fun _ => norm_nonneg _) (fun _ => norm_nonneg _)).mono'
110    (continuous_rootFidelity.comp_aestronglyMeasurable (hmA.prodMk hmB))
111  exact Eventually.of_forall fun ξ => by
112    rw [Real.norm_of_nonneg (rootFidelity_nonneg _ _)]
113    exact rootFidelity_le_sqrt _ _

115/-- Joint Jensen for any Bochner-integrable probability family of actual
116positive L1 states, expressed with arbitrary identified average states. -/
117theorem integral_rootFidelity_le [IsProbabilityMeasure ν]
118    (A B : Ξ → PositiveL1 H μ)
119    (hA : Integrable (fun ξ => (A ξ).1) ν) (hB : Integrable (fun ξ => (B ξ).1) ν)
120    (R S : PositiveL1 H μ)
121    (hR : R.1 = ∫ ξ, (A ξ).1 ∂ν) (hS : S.1 = ∫ ξ, (B ξ).1 ∂ν) :
122    (∫ ξ, (A ξ).rootFidelity (B ξ) ∂ν) ≤ R.rootFidelity S := by
123  let f : Ξ → Lp (TraceClass H) 1 μ × Lp (TraceClass H) 1 μ :=
124    fun ξ => ((A ξ).1, (B ξ).1)
125  have hf : Integrable f ν := hA.prodMk hB
126  have hgi : Integrable (extendedL1RootFidelity ∘ f) ν := by
127    change Integrable (fun ξ => extendedL1RootFidelity ((A ξ).1, (B ξ).1)) ν
128    simpa only [extendedL1RootFidelity_eq] using
129      integrable_family_rootFidelity A B hA hB
130  have hp : ∀ᵐ ξ ∂ν, f ξ ∈ positiveL1Pairs :=
131    Eventually.of_forall fun ξ => And.intro (A ξ).2 (B ξ).2
132  have h := (concaveOn_extendedL1RootFidelity (H := H) (μ := μ)).le_map_integral
133    (continuousOn_extendedL1RootFidelity (H := H) (μ := μ))
134    (isClosed_positiveL1Pairs (H := H) (μ := μ)) hp hf hgi
135  change (∫ ξ, extendedL1RootFidelity ((A ξ).1, (B ξ).1) ∂ν) ≤
136    extendedL1RootFidelity (∫ ξ, ((A ξ).1, (B ξ).1) ∂ν) at h
137  simp only [extendedL1RootFidelity_eq] at h
138  rw [integral_pair hA hB, ← hR, ← hS, extendedL1RootFidelity_eq] at h
139  exact h
