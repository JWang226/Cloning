16/-- The Gaussian translation estimate applies directly to every operator-valued
17L¹ class, with its actual L¹ norm on the right. -/
18theorem gaussian_shift_weighted_L1norm_le
19    (R : Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ)))
20    (a : Fin k → ℝ) (ha : ∀ i, 0 < a i) {g : ℝ} (hg : 1 ≤ g) :
21    (∫ h : Fin k → ℝ, GaussianAffinity.productDensity a h *
22      ∫ v : Fin k → ℝ, GaussianAffinity.productWitness a (fun i => a i / g)
23        (fun i => v i + Real.sqrt g * h i) * ‖R v‖) ≤
24      Thermal.classicalBase g ^ ((k : ℝ) / 2) * ‖R‖ := by
25  let G := GaussianAffinity.productDensity a
26  let χ := GaussianAffinity.productWitness a (fun i => a i / g)
27  let B : (Fin k → ℝ) → ℝ := fun v => ‖R v‖
28  let F : (Fin k → ℝ) × (Fin k → ℝ) → ℝ := fun z =>
29    G z.1 * B z.2 * χ (fun i => z.2 i + Real.sqrt g * z.1 i)
30  have hg0 : 0 < g := lt_of_lt_of_le zero_lt_one hg
31  have hχp (v : Fin k → ℝ) : 0 < χ v :=
32    GaussianAffinity.productWitness_pos a _ ha (fun i => div_pos (ha i) hg0) v
33  have hχb (v : Fin k → ℝ) : ‖χ v‖ ≤
34      ∏ i, Real.sqrt (Real.sqrt (a i) / Real.sqrt (a i / g)) := by
35    rw [Real.norm_eq_abs, abs_of_pos (hχp v)]
36    apply GaussianAffinity.productWitness_le a _ ha (fun i => div_pos (ha i) hg0)
37    intro i
38    exact (div_le_iff₀ hg0).mpr (by nlinarith [ha i])
39  have hχc : Continuous χ := GaussianAffinity.continuous_productWitness a _ ha
40    (fun i => div_pos (ha i) hg0)
41  have hF : Integrable F (volume.prod volume) := by
42    apply ((GaussianAffinity.integrable_productDensity a ha).mul_prod
43      (L1.integrable_coeFn R).norm).mul_bdd
44      ((hχc.comp (by fun_prop)).aestronglyMeasurable)
45    exact Eventually.of_forall fun z => hχb _
46  have he (v : Fin k → ℝ) :
47      (∫ h : Fin k → ℝ, F (h, v)) = B v *
48        (∫ h : Fin k → ℝ, G h * χ (fun i => v i + Real.sqrt g * h i)) := by
49    rw [← integral_const_mul]
50    apply integral_congr_ae
51    exact Eventually.of_forall fun h => by dsimp [F]; ring
52  calc
53    _ = ∫ h : Fin k → ℝ, ∫ v : Fin k → ℝ, F (h, v) := by
54      apply integral_congr_ae
55      exact Eventually.of_forall fun h => by
56        dsimp only
57        rw [← integral_const_mul]
58        apply integral_congr_ae
59        exact Eventually.of_forall fun v => by dsimp [F, G, B, χ]; ring
60    _ = ∫ v : Fin k → ℝ, ∫ h : Fin k → ℝ, F (h, v) := integral_integral_swap hF
61    _ ≤ ∫ v : Fin k → ℝ, B v * Thermal.classicalBase g ^ ((k : ℝ) / 2) := by
62      apply integral_mono hF.integral_prod_right ((L1.integrable_coeFn R).norm.mul_const _)
63      intro v
64      dsimp only
65      rw [he]
66      apply mul_le_mul_of_nonneg_left _ (norm_nonneg _)
67      simpa only [Fintype.card_fin] using
68        GaussianAffinity.integral_product_translated_witness_le a ha hg v
69    _ = _ := by rw [integral_mul_const, L1.norm_eq_integral_norm]; exact mul_comm _ _

140/-- The actual Gaussian average of translated weighted traces has the sharp
141classical affinity bound, with no assumption on the input correlations. -/
142theorem gaussian_average_weightedL1Trace_le
143    (R : Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ)))
144    (a : Fin k → ℝ) (ha : ∀ i, 0 < a i) {g : ℝ} (hg : 1 ≤ g)
145    (hχ : AEStronglyMeasurable
146      (GaussianAffinity.productWitness a (fun i => a i / g)) volume)
147    {C : ℝ} (hbound : ∀ y,
148      ‖GaussianAffinity.productWitness a (fun i => a i / g) y‖ ≤ C) :
149    (∫ h : Fin k → ℝ, GaussianAffinity.productDensity a h *
150      weightedL1Trace (GaussianAffinity.productWitness a (fun i => a i / g)) hχ hbound
151        (classicalTranslation (Real.sqrt g • h) R)) ≤
152      Thermal.classicalBase g ^ ((k : ℝ) / 2) * ‖R‖ := by
153  let G := GaussianAffinity.productDensity a
154  let χ := GaussianAffinity.productWitness a (fun i => a i / g)
155  let Q := weightedL1Trace (H := H) χ hχ hbound
156  have hχc : Continuous χ := GaussianAffinity.continuous_productWitness a _ ha
157    (fun i => div_pos (ha i) (lt_of_lt_of_le zero_lt_one hg))
158  have hχ0 (y : Fin k → ℝ) : 0 ≤ χ y :=
159    (GaussianAffinity.productWitness_pos a _ ha
160      (fun i => div_pos (ha i) (lt_of_lt_of_le zero_lt_one hg)) y).le
161  have hi : Integrable (fun h : Fin k → ℝ => G h * Q
162      (classicalTranslation (Real.sqrt g • h) R)) := by
163    apply (GaussianAffinity.integrable_productDensity a ha).mul_bdd
164      (Q.continuous.comp ((continuous_classicalTranslation R).comp
165        (continuous_const.smul continuous_id))).aestronglyMeasurable
166    exact Eventually.of_forall fun h => by
167      simpa only [norm_classicalTranslation] using
168        norm_weightedL1Trace_le χ hχ hbound (classicalTranslation (Real.sqrt g • h) R)
169  have hF : Integrable (fun z : (Fin k → ℝ) × (Fin k → ℝ) =>
170      G z.1 * ‖R z.2‖ * χ (z.2 + Real.sqrt g • z.1)) (volume.prod volume) := by
171    apply ((GaussianAffinity.integrable_productDensity a ha).mul_prod
172      (L1.integrable_coeFn R).norm).mul_bdd
173      ((hχc.comp (by fun_prop)).aestronglyMeasurable)
174    exact Eventually.of_forall fun z => hbound _
175  have hj : Integrable (fun h : Fin k → ℝ =>
176      G h * ∫ y : Fin k → ℝ, χ (y + Real.sqrt g • h) * ‖R y‖) := by
177    convert hF.integral_prod_left using 1
178    funext h
179    rw [← integral_const_mul]
180    congr 1
181    funext y
182    ring
183  have hb : (∫ h : Fin k → ℝ, G h * Q (classicalTranslation (Real.sqrt g • h) R)) ≤
184      ∫ h : Fin k → ℝ, G h * ∫ y : Fin k → ℝ, χ (y + Real.sqrt g • h) * ‖R y‖ := by
185    apply integral_mono hi hj
186    intro h
187    exact mul_le_mul_of_nonneg_left
188      (weightedL1Trace_classicalTranslation_le χ hχ hχ0 hbound _ R)
189      (GaussianAffinity.productDensity_nonneg a h)
190  exact hb.trans (gaussian_shift_weighted_L1norm_le R a ha hg)
