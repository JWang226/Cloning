18theorem cyclicSector_intertwiner_eq_zero
19    (Ω : TensorRegister n (Fin d)) (Ψ : TensorRegister m (Fin d))
20    (mu nu : Fin d → ℂ) (hne : mu ≠ nu)
21    (hΩ : ‖Ω‖=1) (hΨ : ‖Ψ‖=1)
22    (hΩweight : ∀ a, collectiveGenerator n a a Ω = mu a • Ω)
23    (hΨweight : ∀ a, collectiveGenerator m a a Ψ = nu a • Ψ)
24    (hΩraise : ∀ a b, a<b → collectiveGenerator n a b Ω=0)
25    (hΨraise : ∀ a b, a<b → collectiveGenerator m a b Ψ=0)
26    (T : TensorRegister m (Fin d) →ₗ[ℂ] TensorRegister n (Fin d))
27    (hT : ∀ x ∈ cyclicSector Ψ, T x ∈ cyclicSector Ω)
28    (hcomm : ∀ a b x, x ∈ cyclicSector Ψ →
29      T (collectiveGenerator m a b x) = collectiveGenerator n a b (T x)) :
30    ∀ x ∈ cyclicSector Ψ, T x = 0 := by
31  let c := ⟪Ω,T Ψ⟫_ℂ
32  have htop : T Ψ = c • Ω := by
33    apply cyclicSector_raising_kernel Ω hΩ hΩraise (hT Ψ (highest_mem_cyclicSector Ψ))
34    intro a b hab
35    rw [← hcomm a b Ψ (highest_mem_cyclicSector Ψ),hΨraise a b hab,map_zero]
36  have hΩne : Ω ≠ 0 := by intro h; simpa [h] using hΩ
37  have hc : c=0 := by
38    by_contra hc
39    apply hne
40    funext a
41    have hh := hcomm a a Ψ (highest_mem_cyclicSector Ψ)
42    rw [hΨweight,map_smul,htop,map_smul,hΩweight,smul_smul,smul_smul] at hh
43    have hs : nu a*c = c*mu a := (smul_left_injective ℂ hΩne) hh
44    exact (mul_right_cancel₀ hc (by simpa only [mul_comm c] using hs)).symm
45  have hz : T Ψ=0 := by rw [htop,hc,zero_smul]
46  have hw : ∀ w, T (loweringWord Ψ w)=0 := by
47    intro w
48    induction w with
49    | nil => exact hz
50    | cons a w ih =>
51      rw [loweringWord,hcomm a.val.2 a.val.1 _ (loweringWord_mem_cyclicSector Ψ w),ih,map_zero]
52  intro x hx
53  induction hx using Submodule.span_induction with
54  | mem x hx => obtain ⟨w,rfl⟩ := hx; exact hw w
55  | zero => exact map_zero T
56  | add x y hx hy ihx ihy => rw [map_add,ihx,ihy,add_zero]
57  | smul a x hx ih => rw [map_smul,ih,smul_zero]

151theorem cyclicTensorOperator_unitary_intertwiner_eq_zero
152    (hne : mu ≠ nu) (hΩ : ‖Ω‖=1) (hΨ : ‖Ψ‖=1)
153    (T : cyclicSector Ψ →L[ℂ] cyclicSector Ω)
154    (hT : ∀ U : unitary (Matrix (Fin d) (Fin d) ℂ),
155      T.comp (cyclicTensorOperator Ψ nu hΨweight hΨraise U) =
156        (cyclicTensorOperator Ω mu hΩweight hΩraise U).comp T) : T=0 := by
157  let S := cyclicSector Ψ
158  let F := (cyclicSector Ω).subtype.comp (T.toLinearMap.comp S.orthogonalProjection.toLinearMap)
159  have hproj (x : S) : S.orthogonalProjection x=x := by
160    apply Subtype.ext
161    exact Submodule.starProjection_eq_self_iff.mpr x.property
162  have hF (x : S) : F x = (T x : TensorRegister n (Fin d)) := by
163    change (T (S.orthogonalProjection x) : TensorRegister n (Fin d)) = _
164    rw [hproj]
165  have hh := cyclicSector_intertwiner_eq_zero Ω Ψ mu nu hne hΩ hΨ hΩweight hΨweight hΩraise hΨraise F
166    (fun x _ => (T (S.orthogonalProjection x)).property) (by
167      intro a b x hx
168      let xx : S := ⟨x,hx⟩
169      have he : F (collectiveGenerator m a b x) =
170          (T (cyclicGenerator Ψ nu hΨweight hΨraise a b xx) : TensorRegister n (Fin d)) :=
171        hF (cyclicGenerator Ψ nu hΨweight hΨraise a b xx)
172      rw [he]
173      have ht := DFunLike.congr_fun
174        (cyclicTensorOperator_unitary_intertwines_generators Ω Ψ mu nu hΩweight hΨweight hΩraise hΨraise T hT a b) xx
