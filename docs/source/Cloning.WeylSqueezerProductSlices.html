141    signalSlice m (displacement (Fin.append z 0) x)=displacement z (signalSlice m x) := by
142  have he : (signalSlice m).comp (displacement (Fin.append z 0))=
143      (displacement z).comp (signalSlice m) := by
144    apply coherent_ext
145    intro u
