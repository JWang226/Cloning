13    (∫⁻ x, ENNReal.ofReal (limitDensity d x)) ≤ 1 := by
14  have hn : Tendsto (fun n : ℕ ↦ n+1) atTop atTop :=
15    tendsto_atTop_mono (fun n ↦ Nat.le_succ n) tendsto_id

17      Filter.liminf (fun n ↦ ENNReal.ofReal (flatDensity d (n+1) x)) atTop =
18        ENNReal.ofReal (limitDensity d x) :=
19    (ENNReal.continuous_ofReal.continuousAt.tendsto.comp
20      (flatDensity_moving_tendsto d (fun n ↦ n+1) hn (fun n ↦ Nat.succ_pos n)
21        (fun _ ↦ x) x (fun _ ↦ tendsto_const_nhds))).liminf_eq

25      ENNReal.ofReal_one]
