16theorem integral_complex_gaussian_linear (u v : ℂ) :

18      (Real.pi : ℂ) * Complex.exp (u * v) := by

20    Complex.measurableEquivPi.symm.measurableEmbedding]
21  have he (x : Fin 2 → ℝ) :
22      -((‖Complex.measurableEquivPi.symm x‖ ^ 2 : ℝ) : ℂ) +
23        u * Complex.measurableEquivPi.symm x + v * starRingEnd ℂ (Complex.measurableEquivPi.symm x) =
24      -(1 : ℂ) * (∑ i : Fin 2, (x i : ℂ) ^ 2) +
25        ∑ i : Fin 2, (![u + v, Complex.I * (u - v)] i) * x i := by
