18def extendChannel
19    (Φ : ∀ n t : ℕ, QuantumChannel (TensorRegister n (Fin d)) (TensorRegister (n+t) (Fin d)))
20    (n m : ℕ) : QuantumChannel (TensorRegister n (Fin d)) (TensorRegister m (Fin d)) :=
21  if h : m≤n then restrictionChannel n m d h else by
22    have he : n+(m-n)=m := Nat.add_sub_of_le (le_of_lt (lt_of_not_ge h))
23    exact he ▸ Φ n (m-n)

59theorem extendChannel_payoff_of_le [Nonempty (Fin d)]
60    (Φ : ∀ n t : ℕ, QuantumChannel (TensorRegister n (Fin d)) (TensorRegister (n+t) (Fin d)))
61    (n m : ℕ) (h : m≤n) (ρ : Cloning.MatrixFidelity.State (Fin d)) :
62    statePayoff n m (extendChannel Φ n m) ρ=1 := by
63  have he : (tensorState ρ n).map (extendChannel Φ n m).toPositiveTracePreservingMap=
