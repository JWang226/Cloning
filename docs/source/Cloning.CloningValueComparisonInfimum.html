11theorem modeFactor_zero {g : ℝ} (hg : 0<g) :
12    Thermal.modeFactor g 0=(Real.sqrt g)⁻¹ := by
13  simp only [Thermal.modeFactor,zero_mul,Real.sqrt_zero,add_zero]
14  have hs : Real.sqrt g ≠ 0 := (Real.sqrt_pos.mpr hg).ne'
15  apply (mul_eq_one_iff_eq_inv₀ hs).mp
16  rw [div_mul_eq_mul_div,←pow_two,Real.sq_sqrt hg.le,div_self hg.ne']
