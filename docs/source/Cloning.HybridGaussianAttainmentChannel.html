82lemma classicalDilation_completelyPositive (r : ℝ) (hr : 0 < r) :
83    L1CompletelyPositive (classicalDilation (H := H) (k := k) r hr).toLinearMap := by
84  intro n A hA
85  have hp := (Measure.quasiMeasurePreserving_smul (volume : Measure (Fin k → ℝ))
86    (inv_ne_zero hr.ne')).ae hA
87  have he : ∀ᵐ y ∂volume, ∀ i j : Fin n,
88      classicalDilation r hr (A i j) y = (((r ^ k)⁻¹ : ℝ) : ℂ) • A i j (r⁻¹ • y) :=
89    ae_all_iff.mpr fun i => ae_all_iff.mpr fun j => classicalDilation_ae r hr (A i j)
90  filter_upwards [hp, he] with y hp he
91  change BlockPositive (fun i j => classicalDilation r hr (A i j) y)
92  simp only [he]
93  intro v
94  change 0 ≤ ∑ i, ∑ j, ⟪v i, (((r ^ k)⁻¹ : ℝ) : ℂ) • ((A i j (r⁻¹ • y)).1 (v j))⟫_ℂ
95  simp only [inner_smul_right, ← Finset.mul_sum]
96  exact mul_nonneg (Complex.zero_le_real.mpr (inv_nonneg.mpr (pow_nonneg hr.le _))) (hp v)

121/-- A genuine quantum channel applied independently to every classical fibre. -/
122def fibreChannel {Ω : Type*} [MeasurableSpace Ω] (μ : Measure Ω)
123    (Φ : QuantumChannel H K) : Channel H K μ where
124  map := Φ.toPositiveTracePreservingMap.toContinuousLinearMap.compLpL 1 μ
125  completelyPositive := by
126    intro n A hA
127    have he : ∀ᵐ y ∂μ, ∀ i j : Fin n,
128        Φ.toPositiveTracePreservingMap.toContinuousLinearMap.compLpL 1 μ (A i j) y =
129          Φ.toLinearMap (A i j y) :=
130      ae_all_iff.mpr fun i => ae_all_iff.mpr fun j =>
131        Φ.toPositiveTracePreservingMap.toContinuousLinearMap.coeFn_compLpL (A i j)
132    filter_upwards [he, hA] with y hy hp
133    change BlockPositive (fun i j => Φ.toPositiveTracePreservingMap.toContinuousLinearMap.compLpL 1 μ (A i j) y)
134    simpa only [hy] using Φ.completelyPositive n (fun i j => A i j y) hp
135  tracePreserving := by
136    intro A
137    apply integral_congr_ae
138    filter_upwards [Φ.toPositiveTracePreservingMap.toContinuousLinearMap.coeFn_compLpL A]
139      with y hy
140    change traceCLM (Φ.toPositiveTracePreservingMap.toContinuousLinearMap.compLpL 1 μ A y) = _
141    rw [hy]
142    exact Φ.trace_preserving _
