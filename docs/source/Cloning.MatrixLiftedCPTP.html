3/-!
4# A genuine channel for finite sector transitions
5 
6The map is defined on every complex matrix. It measures the sector, discards
7its input multiplicity, applies an arbitrary completely positive sector map,
8samples the output sector, and prepares the normalized output multiplicity.
9Complete positivity is proved at every ancilla dimension directly from that
10of the sector maps; no Kraus representation hypothesis is imposed.
11-/

31def inputReduction (i : μ) (X : Matrix (Σ i, α i × χ i) (Σ i, α i × χ i) ℂ) :
32    Matrix (α i) (α i) ℂ :=
33  ∑ t : χ i, X.submatrix (fun a => ⟨i, (a, t)⟩) (fun a => ⟨i, (a, t)⟩)

65def outputBlock (Φ : ∀ i j, MatrixChannel (α i) (β j)) (K : μ → ν → ℝ)
66    (j : ν) (X : Matrix (Σ i, α i × χ i) (Σ i, α i × χ i) ℂ) :
67    Matrix (β j) (β j) ℂ :=
68  ∑ i, K i j • (Φ i j).toFun (inputReduction i X)

92theorem outputBlock_completely_positive (Φ : ∀ i j, MatrixChannel (α i) (β j))
93    (K : μ → ν → ℝ) (hK : ∀ i j, 0 ≤ K i j) (j : ν) (k : ℕ)
94    (X : Matrix (Fin k × (Σ i, α i × χ i)) (Fin k × (Σ i, α i × χ i)) ℂ)
95    (hX : X.PosSemidef) : (amplify (outputBlock Φ K j) X).PosSemidef := by
96  have heq : amplify (outputBlock Φ K j) X =
97      ∑ i, K i j • amplify (Φ i j).toFun (amplify (inputReduction i) X) := by

106/-- Measure, discard, transition, and prepare, on the full matrix algebra. -/
107def channelMap (Φ : ∀ i j, MatrixChannel (α i) (β j))
108    (K : μ → ν → ℝ) (τ : ∀ j, State (κ j))
109    (X : Matrix (Σ i, α i × χ i) (Σ i, α i × χ i) ℂ) :
110    Matrix (Σ j, β j × κ j) (Σ j, β j × κ j) ℂ :=
111  Matrix.blockDiagonal' (fun j => outputBlock Φ K j X ⊗ₖ (τ j).matrix)
112 
113theorem channelMap_add (Φ : ∀ i j, MatrixChannel (α i) (β j))
114    (K : μ → ν → ℝ) (τ : ∀ j, State (κ j))
115    (X Y : Matrix (Σ i, α i × χ i) (Σ i, α i × χ i) ℂ) :
116    channelMap Φ K τ (X + Y) = channelMap Φ K τ X + channelMap Φ K τ Y := by
117  simp only [channelMap, outputBlock_add, Matrix.add_kronecker]
118  exact Matrix.blockDiagonal'_add _ _

127theorem channelMap_trace (Φ : ∀ i j, MatrixChannel (α i) (β j))
128    (K : μ → ν → ℝ) (hKsum : ∀ i, ∑ j, K i j = 1) (τ : ∀ j, State (κ j))
129    (X : Matrix (Σ i, α i × χ i) (Σ i, α i × χ i) ℂ) :
130    Matrix.trace (channelMap Φ K τ X) = Matrix.trace X := by
131  simp only [channelMap, Matrix.trace_blockDiagonal', Matrix.trace_kronecker,
132    State.trace_one, mul_one, outputBlock_trace]
133  rw [Finset.sum_comm]
134  simp_rw [← Finset.sum_mul, ← Complex.ofReal_sum, hKsum, Complex.ofReal_one, one_mul]
135  exact inputReduction_trace_sum X

137theorem channelMap_completely_positive (Φ : ∀ i j, MatrixChannel (α i) (β j))
138    (K : μ → ν → ℝ) (hK : ∀ i j, 0 ≤ K i j) (τ : ∀ j, State (κ j)) (k : ℕ)
139    (X : Matrix (Fin k × (Σ i, α i × χ i)) (Fin k × (Σ i, α i × χ i)) ℂ)
140    (hX : X.PosSemidef) : (amplify (channelMap Φ K τ) X).PosSemidef := by
141  let B : ∀ j, Matrix (Fin k × (β j × κ j)) (Fin k × (β j × κ j)) ℂ :=
142    fun j => ((amplify (outputBlock Φ K j) X) ⊗ₖ (τ j).matrix).submatrix
143      (fun a => ((a.1, a.2.1), a.2.2)) (fun a => ((a.1, a.2.1), a.2.2))
144  have hB : ∀ j, (B j).PosSemidef := fun j =>
145    ((outputBlock_completely_positive Φ K hK j k X hX).kronecker
146      (τ j).positive).submatrix _
147  have hd := (blockDiagonal'_posSemidef B hB).submatrix
148    (fun a : Fin k × (Σ j, β j × κ j) => ⟨a.2.1, (a.1, a.2.2)⟩)
149  convert hd using 1
150  ext ⟨a, j, b, t⟩ ⟨a', j', b', t'⟩
151  by_cases hj : j = j'
152  · subst j'
153    simp [channelMap, amplify, B, Matrix.kroneckerMap_apply]
154  · simp [channelMap, amplify, B, Matrix.blockDiagonal'_apply_ne, hj]

156/-- The sector construction is a genuine CPTP map for arbitrary sector
157channels and stochastic kernel, with no assumptions on its input matrix. -/
158def channel (Φ : ∀ i j, MatrixChannel (α i) (β j))
159    (K : μ → ν → ℝ) (hK : ∀ i j, 0 ≤ K i j)
160    (hKsum : ∀ i, ∑ j, K i j = 1) (τ : ∀ j, State (κ j)) :
161    MatrixChannel (Σ i, α i × χ i) (Σ j, β j × κ j) where
162  toFun := channelMap Φ K τ
163  map_add := channelMap_add Φ K τ
164  map_smul := channelMap_smul Φ K τ
165  trace_preserving := channelMap_trace Φ K hKsum τ
166  completely_positive := channelMap_completely_positive Φ K hK τ
