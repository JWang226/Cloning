4/-! Exact Boltzmann weights of physical lowering words. These identities
5connect the literal sector density to occupation probabilities and the root
6height, before any asymptotic basis or tail estimate is invoked. -/

15def rootBoltzmann (p : Fin d → ℝ) (r : PositiveRoot d) : ℝ :=
16  p r.val.2 / p r.val.1

18def wordBoltzmann (p : Fin d → ℝ) (w : List (PositiveRoot d)) : ℝ :=
19  (w.map (rootBoltzmann p)).prod

21theorem prod_zpow_root_weight (p : Fin d → ℝ) (hp : ∀ a, p a ≠ 0)
22    (r : PositiveRoot d) : (∏ a, p a ^ r.weight a) = rootBoltzmann p r := by
23  simp only [PositiveRoot.weight, zpow_sub₀ (hp _), Finset.prod_div_distrib]
24  simp [rootBoltzmann]

26theorem prod_zpow_loweringWeight (p : Fin d → ℝ) (hp : ∀ a, p a ≠ 0)
27    (w : List (PositiveRoot d)) :
28    (∏ a, p a ^ loweringWeight w a) = wordBoltzmann p w := by
29  induction w with
30  | nil => simp [wordBoltzmann]
31  | cons r w ih =>
32    simp only [loweringWeight_cons, Pi.add_apply, zpow_add₀ (hp _),
33      Finset.prod_mul_distrib, prod_zpow_root_weight p hp, ih]
34    rfl

36theorem shiftedWeight_boltzmann (p : Fin d → ℝ) (hp : ∀ a, p a ≠ 0)
37    (mu : Fin d → ℕ) (w : List (PositiveRoot d)) :
38    (∏ a, p a ^ ((mu a : ℤ) + loweringWeight w a)) =
39      (∏ a, p a ^ mu a) * wordBoltzmann p w := by
40  simp only [zpow_add₀ (hp _), zpow_natCast, Finset.prod_mul_distrib,

43theorem wordBoltzmann_pos (p : Fin d → ℝ) (hp : ∀ a, 0 < p a)
44    (w : List (PositiveRoot d)) : 0 < wordBoltzmann p w := by
45  induction w with

50/-- Height decay is a derived product inequality once each finite physical
51root ratio is bounded. No state-tail or operator approximation is an input. -/
52theorem wordBoltzmann_le_pow_height (p : Fin d → ℝ) (hp : ∀ a, 0 ≤ p a)
53    (θ : ℝ) (hθ : 0 ≤ θ)
54    (hroot : ∀ r : PositiveRoot d, rootBoltzmann p r ≤ θ ^ r.height)
55    (w : List (PositiveRoot d)) : wordBoltzmann p w ≤ θ ^ loweringHeight w := by
56  induction w with
57  | nil => simp [wordBoltzmann, loweringHeight]
58  | cons r w ih =>
59    change rootBoltzmann p r * wordBoltzmann p w ≤ θ ^ (r.height + loweringHeight w)
60    rw [pow_add]
61    exact mul_le_mul (hroot r) ih
62      (by unfold wordBoltzmann; apply List.prod_nonneg; intro x hx
63          obtain ⟨a, _, rfl⟩ := List.mem_map.mp hx
64          exact div_nonneg (hp _) (hp _))
65      (pow_nonneg hθ _)

67theorem tensorOperator_diagonal_loweringWord
68    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
69    (hweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
70    (p : Fin d → ℝ) (hp : ∀ a, 0 < p a) (w : List (PositiveRoot d)) :
71    tensorOperator n (Matrix.diagonal (fun a => (p a : ℂ))) (loweringWord Ω w) =
72      (((∏ a, p a ^ mu a) * wordBoltzmann p w : ℝ) : ℂ) • loweringWord Ω w := by
73  have h := tensorOperator_diagonal_weight (fun a => (p a : ℂ)) (loweringWord Ω w)
74    (fun a => (mu a : ℤ) + loweringWeight w a) (fun a => by
75      simpa only [Int.cast_add, Int.cast_natCast] using
76        cartan_loweringWord Ω (fun a => (mu a : ℂ)) hweight w a)
77  have hc : (∏ a, (p a : ℂ) ^ ((mu a : ℤ) + loweringWeight w a)) =
78      (((∏ a, p a ^ mu a) * wordBoltzmann p w : ℝ) : ℂ) := by
79    rw [← shiftedWeight_boltzmann p (fun a => (hp a).ne') mu w]
80    simp only [Complex.ofReal_prod, Complex.ofReal_zpow]
81  rw [hc] at h
