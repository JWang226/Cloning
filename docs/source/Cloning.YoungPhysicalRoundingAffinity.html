13/-- Physical lattice affinity is exactly the continuous affinity of the actual output cells. -/
14theorem tensorYoungRawOutput_affinity_eq_integral (d n m : ℕ) (hm : 0 < m) (γ : ℝ)
15    (p : Fin (d+1) → ℝ) (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) :
16    CountableScheffe.affinity (probability (tensorYoungRawOutput d n m γ p hp hs))
17      (probability (tensorYoungIntegerPMF d m p hp hs)) =
18    ∫ x, Real.sqrt (roundedDensity (Real.sqrt (m : ℝ))⁻¹ (headAnchor d m p) γ
19      (tensorYoungHeadPMF d n p hp hs) x)*Real.sqrt (headDensity d m p hp hs x) := by
20  rw [tensorYoungRawOutput_affinity, roundedDensity, headDensity,
21    affineDensity_affinity d _ (by positivity)]
22  exact (YoungRounding.interpolate_affinity
23    (probability_nonneg (transport γ (tensorYoungHeadPMF d n p hp hs)))
24    (probability_nonneg (tensorYoungHeadPMF d m p hp hs))
25    (hasSum_probability (transport γ (tensorYoungHeadPMF d n p hp hs))).summable
26    (hasSum_probability (tensorYoungHeadPMF d m p hp hs)).summable).symm

76    (hp : ∀ p ∈ K, ∑ i, p i = 1) (hp0 : ∀ p ∈ K, ∀ i, 0 < p i)

91  choose n hn p hpK hb using hbad
