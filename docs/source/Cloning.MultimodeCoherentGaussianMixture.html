5/-!
6# Finite multimode Gaussian coherent-state mixtures
7 
8The Bochner integral of the actual multimode coherent projector under independent
9circular Gaussian coordinates equals the product-geometric occupation density.
10The proof uses finite-product Fubini and the proved one-mode Gaussian moments;
11the equality holds in the trace-class Banach space, including for zero modes.
12-/

25/-- Independent circular Gaussian amplitudes with the specified positive mean
26squared amplitude in each mode. -/
27def gaussianProductMeasure {d : ℕ} (s : Fin d → ℝ) : Measure (Fin d → ℂ) :=
28  Measure.pi (fun i ↦ CoherentGaussianMixture.gaussianMeasure (s i))

50/-- Each matrix coefficient of the actual multimode rank-one operator is the
51product of the corresponding one-mode coefficients. -/
52theorem projector_coefficient {d : ℕ} (z : Fin d → ℂ) (n m : Fin d → ℕ) :
53    ⟪numberBasis d n, (coherentProjector z).1 (numberBasis d m)⟫_ℂ =
54      ∏ i, ComplexCoherent.coherentVector (z i) (n i) *
55        star (ComplexCoherent.coherentVector (z i) (m i)) := by
56  change ⟪numberBasis d n,
57    (InnerProductSpace.rankOne ℂ (coherentVector z) (coherentVector z))
58      (numberBasis d m)⟫_ℂ = _
59  rw [InnerProductSpace.rankOne_apply, inner_smul_right, inner_numberBasis,
60    ← inner_conj_symm (coherentVector z) (numberBasis d m), inner_numberBasis]
61  rw [mul_comm]
62  simp only [coherentVector_apply, map_prod, Finset.prod_mul_distrib]
63  rfl

71/-- Fubini and the actual one-mode Gaussian moments evaluate every multimode
72matrix coefficient. -/
73theorem integral_coherent_coefficients {d : ℕ} {s : Fin d → ℝ}
74    (hs : ∀ i, 0 < s i) (n m : Fin d → ℕ) :
75    (∫ z : Fin d → ℂ, ∏ i, ComplexCoherent.coherentVector (z i) (n i) *
76      star (ComplexCoherent.coherentVector (z i) (m i)) ∂gaussianProductMeasure s) =
77      if n = m then ((∏ i, Thermal.geometric (s i / (1 + s i)) (n i) : ℝ) : ℂ)
78      else 0 := by
79  classical
80  letI (i : Fin d) : IsProbabilityMeasure (CoherentGaussianMixture.gaussianMeasure (s i)) :=
81    CoherentGaussianMixture.gaussianMeasure_probability (hs i)
82  rw [gaussianProductMeasure, integral_fin_nat_prod_eq_prod
83    (μ := fun i : Fin d ↦ CoherentGaussianMixture.gaussianMeasure (s i))
84    (fun i (z : ℂ) ↦ ComplexCoherent.coherentVector z (n i) *
85      star (ComplexCoherent.coherentVector z (m i)))]
86  simp_rw [CoherentGaussianMixture.integral_coherent_coefficients (hs _)]
87  by_cases hnm : n = m
88  · subst m
89    simp only [ite_true, Complex.ofReal_prod]
90  · rw [if_neg hnm]
91    obtain ⟨i, hi⟩ : ∃ i, n i ≠ m i := by
92      by_contra h
93      apply hnm
94      funext i
95      exact not_not.mp (not_exists.mp h i)
96    exact Finset.prod_eq_zero (Finset.mem_univ i) (if_neg hi)

98/-- The actual Gaussian mixture in the trace-class Banach space. -/
99def gaussianMixture {d : ℕ} (s : Fin d → ℝ) : TraceClass (Fock d) :=
100  ∫ z : Fin d → ℂ, coherentProjector z ∂gaussianProductMeasure s

102theorem gaussianMixture_coefficient {d : ℕ} {s : Fin d → ℝ}
103    (hs : ∀ i, 0 < s i) (n m : Fin d → ℕ) :
104    ⟪numberBasis d n, (gaussianMixture s).1 (numberBasis d m)⟫_ℂ =
105      if n = m then ((∏ i, Thermal.geometric (s i / (1 + s i)) (n i) : ℝ) : ℂ)
106      else 0 := by
107  change traceClassMatrixCoefficient (numberBasis d n) (numberBasis d m)
108    (gaussianMixture s) = _
109  rw [gaussianMixture, ← ContinuousLinearMap.integral_comp_comm _
110    (integrable_coherentProjector_gaussian hs)]
111  simp_rw [traceClassMatrixCoefficient_apply, projector_coefficient]
112  exact integral_coherent_coefficients hs n m

114/-- Independent Gaussian mixing of actual multimode coherent projectors gives
115the product-geometric density, with `qᵢ = sᵢ / (1 + sᵢ)`. This is an equality of
116trace-class operators, not merely an equality of formal matrix coefficients. -/
117theorem integral_coherentProjector_gaussian_eq_productThermal {d : ℕ}
118    {s : Fin d → ℝ} (hs : ∀ i, 0 < s i) :
119    (∫ z : Fin d → ℂ, coherentProjector z ∂gaussianProductMeasure s) =
120      vectorMixture (numberBasis d)
121        (fun k ↦ ∏ i, Thermal.geometric (s i / (1 + s i)) (k i)) := by
122  classical
123  have hq0 (i : Fin d) : 0 ≤ s i / (1 + s i) := by
124    have hi := hs i
125    positivity
126  have hq1 (i : Fin d) : s i / (1 + s i) < 1 := by
127    have hi := hs i
128    exact (div_lt_one (by positivity)).mpr (by linarith)
129  change gaussianMixture s = _
130  apply Subtype.ext
131  apply ContinuousLinearMap.ext_on
132    (Submodule.dense_iff_topologicalClosure_eq_top.mpr (numberBasis d).dense_span)
133  rintro _ ⟨m, rfl⟩
134  apply (numberBasis d).repr.injective
135  ext n
136  rw [(numberBasis d).repr_apply_apply, (numberBasis d).repr_apply_apply,
137    InfiniteOccupationStates.vectorMixture_apply_basis (numberBasis d) _
138      (Thermal.multimode_geometric_hasSum hq0 hq1).summable m, inner_smul_right,
139    orthonormal_iff_ite.mp (numberBasis d).orthonormal]
140  rw [gaussianMixture_coefficient hs]
141  by_cases hnm : n = m
142  · subst n
143    simp
144  · simp [hnm]
