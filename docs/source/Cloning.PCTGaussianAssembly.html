16theorem fullFrame_of_orthonormal {s : ℕ} (hcard : Fintype.card C = s + 1)
17    {u : Fin (s + 1) → Register C} (hu : Orthonormal ℂ u) :
18    ∃ b : OrthonormalBasis (Fin (s + 1)) ℂ (Register C), (b : Fin (s + 1) → Register C) = u := by
19  letI : FiniteDimensional ℂ (Register C) :=
20    (registerBasis C).toOrthonormalBasis.toBasis.finiteDimensional_of_finite
21  have hc : Module.finrank ℂ (Register C) = Fintype.card (Fin (s + 1)) := by
22    rw [Module.finrank_eq_card_basis (registerBasis C).toOrthonormalBasis.toBasis]
23    simpa using hcard
24  have hv : Orthonormal ℂ ((Set.univ : Set (Fin (s + 1))).restrict u) :=
25    hu.comp _ Subtype.val_injective
26  obtain ⟨b, hb⟩ := hv.exists_orthonormalBasis_extension_of_card_eq hc
27  exact ⟨b, funext fun i => hb i (Set.mem_univ i)⟩

29/-- Werner's original physical output on a pure tensor input, defined using
30only the input vector and the canonical physical symmetric projector. -/
31def pureWernerOperator {L : ℕ} (s : ℕ) (ψ : Register C) (S : Finset (Fin L)) :
32    Register (Fin L → C) →L[ℂ] Register (Fin L → C) :=
33  (((Nat.choose (S.card + s) s : ℝ) / Nat.choose (L + s) s : ℝ) : ℂ) •
34    (physicalProjector L * pureSlotsOperator ψ S * physicalProjector L)

62/-- The actual one-particle reduced density matrix of a purification. -/
63def reducedDensityMatrix (ψ : Register (A × B)) : Matrix A A ℂ :=
64  fun a c => ∑ b, ψ (a, b) * star (ψ (c, b))

142/-- The manuscript's physical PCT-to-Gaussian-mixture step. It starts with the
143canonical physical Werner sandwich on randomized purifications, traces all
144environment slots, and ends at the actual mixture of reduced tensor powers.
145No covariance, frame existence, occupation limit, or integrability premise is
146assumed; only the physical dimensions, unit vector, and cloning-ratio limit. -/
147theorem randomizedPurification_gaussian_product_mixture {s : ℕ} (hs : 1 ≤ s)
148    (hcard : Fintype.card (A × B) = s + 1)
149    (u : OrthonormalBasis (Fin (s + 1)) ℂ (Register (A × B)))
150    {Ω : Type*} [MeasurableSpace Ω] (μ : Measure Ω) [IsProbabilityMeasure μ]
151    (W : Ω → Register B →ₗᵢ[ℂ] Register B)
152    (m : ℕ → ℕ) (hm : ∀ n, n ≤ m n) {γ : ℝ} (hγ : 1 < γ)
153    (h : Tendsto (fun n ↦ (m n : ℝ) / n) atTop (𝓝 γ)) :
154    Tendsto (fun n => ‖randomizedPurificationOutput hcard (u 0) (u.orthonormal.norm_eq_one 0)
155      μ W (inputSlots n (m n) (hm n)) -
156      ∫ z, partialTraceChannel.toLinearMap (vectorProjector
157        (regroup (m n) (tensorVector (fun _ : Fin (m n) => frameParticle u z (m n)))))

163/-- The physical randomized PCT output converges in trace norm to the mixture
164`∫ ρ_{z,m}^{⊗m} dμ_{γ-1}(z)`, with each `ρ_{z,m}` the actual partial trace of
165the normalized one-particle purification. The RHS consists of literal finite
166matrix tensor powers, not an assumed identification of limit models. -/
167theorem pct_product_mixture {s : ℕ} (hs : 1 ≤ s)
168    (hcard : Fintype.card (A × B) = s + 1)
169    (u : OrthonormalBasis (Fin (s + 1)) ℂ (Register (A × B)))
170    {Ω : Type*} [MeasurableSpace Ω] (μ : Measure Ω) [IsProbabilityMeasure μ]
171    (W : Ω → Register B →ₗᵢ[ℂ] Register B)
172    (m : ℕ → ℕ) (hm : ∀ n, n ≤ m n) {γ : ℝ} (hγ : 1 < γ)
173    (h : Tendsto (fun n ↦ (m n : ℝ) / n) atTop (𝓝 γ)) :
174    Tendsto (fun n => ‖randomizedPurificationOutput hcard (u 0) (u.orthonormal.norm_eq_one 0)

176      ∫ z, matrixTensorPower (reducedDensityMatrix (frameParticle u z (m n))) (m n)

192/-- Every normalized purification admits the full physical mixture theorem;
193the required complete orthonormal frame is produced by the proof. -/
194theorem exists_pct_product_mixture {s : ℕ} (hs : 1 ≤ s)
195    (hcard : Fintype.card (A × B) = s + 1) (ψ : Register (A × B)) (hψ : ‖ψ‖ = 1)
196    {Ω : Type*} [MeasurableSpace Ω] (μ : Measure Ω) [IsProbabilityMeasure μ]
197    (W : Ω → Register B →ₗᵢ[ℂ] Register B)
198    (m : ℕ → ℕ) (hm : ∀ n, n ≤ m n) {γ : ℝ} (hγ : 1 < γ)
199    (h : Tendsto (fun n ↦ (m n : ℝ) / n) atTop (𝓝 γ)) :
200    ∃ u : OrthonormalBasis (Fin (s + 1)) ℂ (Register (A × B)), u 0 = ψ ∧
201      Tendsto (fun n => ‖randomizedPurificationOutput hcard ψ hψ μ W
202        (inputSlots n (m n) (hm n)) -
203        ∫ z, matrixTensorPower (reducedDensityMatrix (frameParticle u z (m n))) (m n)
204          ∂MultimodeCoherentGaussianMixture.gaussianProductMeasure
205            (fun _ : Fin s => γ - 1)‖) atTop (𝓝 0) := by
206  obtain ⟨u, hu⟩ := exists_purification_frame hcard ψ hψ
207  refine ⟨u, hu, ?_⟩
208  have ht := pct_product_mixture hs hcard u μ W m hm hγ h
209  simpa only [hu] using ht
