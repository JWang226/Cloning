19theorem sqrtSpectrum_norm (p : A → ℝ) (hp : ∀ a, 0 ≤ p a) (hs : ∑ a, p a = 1) :

84theorem whiteningTest_reference (p : A → ℝ) (hp : ∀ a, 0 < p a)
85    (b : OrthonormalBasis (Fin (k + 1)) ℝ (EuclideanSpace ℝ A))
86    (hb : b 0 = sqrtSpectrum p) (t : Fin k → ℝ) :
87    ∑ a, p a * whiteningTest p b t a = 0 := by
88  have h := whiteningVector_orthogonal b t
89  rw [hb, EuclideanSpace.inner_eq_star_dotProduct] at h

91    simpa [dotProduct, sqrtSpectrum, mul_comm] using h
92  rw [← h']

95  unfold whiteningTest
96  have hs := Real.sq_sqrt (hp a).le
97  have hn := (Real.sqrt_pos.mpr (hp a)).ne'
