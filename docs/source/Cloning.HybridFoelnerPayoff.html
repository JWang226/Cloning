98theorem eventually_gaussianThermal_output_rootFidelity_le
99    (Λ : ℕ → Lp (TraceClass (Fock s)) 1 (volume : Measure (Fin k → ℝ)) →L[ℂ]
100      Lp (TraceClass (Fock s)) 1 (volume : Measure (Fin k → ℝ)))
101    (hΛ : ∀ n, L1CompletelyPositive (Λ n).toLinearMap)
102    (hTP : ∀ n, L1TracePreserving (Λ n).toLinearMap)
103    (a : Fin k → ℝ) (ha : ∀ i, 0 < a i) (g : ℝ) (hg : 1 < g)
104    (q : Fin s → ℝ) (hq0 : ∀ i, 0 < q i) (hq1 : ∀ i, q i < 1)
105    (b : ℕ → (Fin k → ℝ) → ℝ)
106    (hb : ∀ n, AEStronglyMeasurable (b n) volume)
107    {B : ℝ} (hbbound : ∀ n h, ‖b n h‖ ≤ B)
108    (hlim : ∀ h, Tendsto (fun n => b n h) atTop (𝓝 0))

110      classicalTranslation (Real.sqrt g • h) (Λ n R)‖ ≤ b n h * ‖R‖)
111    (hquantum : ∀ z R, Tendsto (fun n =>
112      ‖Λ n (quantumL1Action z R) - quantumL1Action (Real.sqrt g • z) (Λ n R)‖)

118      (R n).rootFidelity (gaussianThermalField a ha q (fun i => (hq0 i).le) hq1) ≤
119        Thermal.classicalBase g ^ ((k : ℝ) / 2) *
120          (∏ i, Thermal.fidelity (q i) (Thermal.amplified g (q i))) + ε := by
121  classical

123  let C := ∏ i, Real.sqrt (Real.sqrt (a i) / Real.sqrt (a i / g))
124  have hg0 : 0 < g := by linarith
125  have hχ : AEStronglyMeasurable χ volume :=

139    (Thermal.fidelity_pos (hq0 i).le (hq1 i)
140      ((hq0 i).trans (Thermal.lt_amplified hg (hq1 i))).le
