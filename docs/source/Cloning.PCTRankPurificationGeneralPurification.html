35/-- The actual finite environment transformation is constructed by extending
36the isometry between the spans of the purification rows. -/
37theorem exists_environment_isometry_of_reduced_eq (ψ φ : Register (A×E))
38    (h : reducedDensityMatrix ψ=reducedDensityMatrix φ) :
39    ∃W : Register E→ₗᵢ[ℂ] Register E, environmentRotate W ψ=φ := by
40  let f := rowCombination ψ
41  let g := rowCombination φ
42  let e := sameNormRangeIsometry f g (rowCombination_norm_eq ψ φ h)
43  let L : LinearMap.range f→ₗᵢ[ℂ] Register E := (LinearMap.range g).subtypeₗᵢ.comp e.toLinearIsometry
44  refine ⟨L.extend,?_⟩
45  have he (x : A→ℂ) : L.extend (f x)=g x := by
46    have ht := LinearIsometry.extend_apply L ⟨f x,LinearMap.mem_range_self f x⟩
47    rw [ht]
48    exact sameNormRangeIsometry_apply f g (rowCombination_norm_eq ψ φ h) x _
49  have hrow (a : A) : L.extend (environmentRow ψ a)=environmentRow φ a := by
50    have hh := he (Pi.single a 1)
51    simpa only [f,g,rowCombination,LinearMap.coe_mk,AddHom.coe_mk,Pi.single_apply,
52      ite_smul,zero_smul,one_smul,Finset.sum_ite_eq',Finset.mem_univ,if_true] using hh
53  ext ⟨a,b⟩
54  simp only [environmentRotate_apply,hrow,environmentRow_apply]

58/-- Every finite environment isometry is the physical matrix action of a
59unitary, with the transpose convention used by Haar purification. -/
60theorem exists_purificationEnvironment (W : Register E→ₗᵢ[ℂ] Register E) :
61    ∃U : unitary (Matrix E E ℂ),purificationEnvironment U=W := by
62  let Q : Matrix E E ℂ := fun a b=>W (lp.single 2 b 1) a
63  have hQ : Qᴴ*Q=1 := by
64    ext a b
65    have h := W.inner_map_map (lp.single 2 a 1) (lp.single 2 b 1)
66    have hbase : ⟪(lp.single 2 a 1 : Register E),lp.single 2 b 1⟫_ℂ=
67        if a=b then 1 else 0 := by
68      simp [register_inner_single,lp.single_apply,Pi.single_apply,eq_comm]
69    rw [hbase] at h
70    simpa only [lp.inner_eq_tsum,tsum_fintype,RCLike.inner_apply,
71      Matrix.mul_apply,Matrix.conjTranspose_apply,Matrix.one_apply,starRingEnd_apply,
72      Q,mul_comm] using h
73  let V : unitary (Matrix E E ℂ) := ⟨Q,Matrix.mem_unitaryGroup_iff'.mpr hQ⟩
74  refine ⟨Matrix.UnitaryGroup.transpose V,?_⟩
75  ext x a
76  have hx : x=∑b,x b • (lp.single 2 b 1 : Register E) := by
77    ext b
78    simp [lp.single_apply,Pi.single_apply]
79  conv_rhs => rw [hx]
80  simp only [purificationEnvironment,unitaryRegister_apply,Matrix.UnitaryGroup.transpose,
81    Matrix.transpose_apply,V,Q,map_sum,map_smul,lp.coeFn_sum,Finset.sum_apply,
82    lp.coeFn_smul,Pi.smul_apply,smul_eq_mul]
83  apply Finset.sum_congr rfl
84  intro b _
85  exact mul_comm _ _

87/-- Any two purifications in the same environment differ by an actual
88unitary, without a full-rank hypothesis. -/
89theorem exists_environment_unitary_of_reduced_eq (ψ φ : Register (A×E))
90    (h : reducedDensityMatrix ψ=reducedDensityMatrix φ) :
91    ∃U : unitary (Matrix E E ℂ),environmentRotate (purificationEnvironment U) ψ=φ := by
92  obtain ⟨W,hW⟩ := exists_environment_isometry_of_reduced_eq ψ φ h
93  obtain ⟨U,hU⟩ := exists_purificationEnvironment W
94  exact ⟨U,hU.symm ▸ hW⟩
