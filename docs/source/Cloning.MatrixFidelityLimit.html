24theorem weighted_lifted_fidelity_converges
25    {I : ℕ → Type*} [∀ n, Fintype (I n)] [∀ n, DecidableEq (I n)]
26    {d s : (n : ℕ) → I n → Type*}
27    [∀ n i, Fintype (d n i)] [∀ n i, DecidableEq (d n i)]
28    [∀ n i, Fintype (s n i)] [∀ n i, DecidableEq (s n i)]
29    (σ ρ : (n : ℕ) → (i : I n) → State (d n i))
30    (τ : (n : ℕ) → (i : I n) → State (s n i))
31    (p q : (n : ℕ) → I n → ℝ) (η : ℕ → ℝ) (c l : ℝ)
32    (hp : ∀ n i, 0 ≤ p n i) (hq : ∀ n i, 0 ≤ q n i)
33    (hps : ∀ n, ∑ i, p n i ≤ 1) (hqs : ∀ n, ∑ i, q n i ≤ 1)
34    (hη : ∀ n, 0 ≤ η n)
35    (hsector : ∀ n i, 0 < q n i →
36      |fidelity (σ n i).matrix (ρ n i).matrix - c| ≤ η n)
37    (hηlimit : Tendsto η atTop (𝓝 0))
38    (hlabel : Tendsto (fun n => Cloning.BlockFidelity.classicalAffinity (q n) (p n))
39      atTop (𝓝 l)) :
40    Tendsto (fun n =>
41      fidelity (Matrix.blockDiagonal' (fun i => (q n i • (σ n i).matrix) ⊗ₖ (τ n i).matrix))
42        (Matrix.blockDiagonal' (fun i => (p n i • (ρ n i).matrix) ⊗ₖ (τ n i).matrix)))
43      atTop (𝓝 (c * l)) := by
44  let actual := fun n =>
45    fidelity (Matrix.blockDiagonal' (fun i => (q n i • (σ n i).matrix) ⊗ₖ (τ n i).matrix))
46      (Matrix.blockDiagonal' (fun i => (p n i • (ρ n i).matrix) ⊗ₖ (τ n i).matrix))
47  have herr : ∀ n, |actual n - c * Cloning.BlockFidelity.classicalAffinity (q n) (p n)|
48      ≤ η n := by
49    intro n
50    apply fidelity_weighted_lifted_blocks_error
51      (fun i => (σ n i).matrix) (fun i => (ρ n i).matrix) (fun i => (τ n i).matrix)
52      (p n) (q n) c (η n)
53      (fun i => (σ n i).positive) (fun i => (ρ n i).positive)
54      (fun i => (τ n i).positive) (hp n) (hq n)
55      (fun i => by rw [(τ n i).trace_one]; rfl) (hps n) (hqs n) (hη n) (hsector n)
56  have hdiff : Tendsto
57      (fun n => actual n - c * Cloning.BlockFidelity.classicalAffinity (q n) (p n))
58      atTop (𝓝 0) := by
59    apply Metric.tendsto_nhds.mpr
60    intro ε hε
61    filter_upwards [hηlimit.eventually_lt_const hε] with n hn
62    simpa only [Real.dist_eq, sub_zero] using (herr n).trans_lt hn
63  have htotal := hdiff.add (hlabel.const_mul c)
64  simpa only [sub_add_cancel, zero_add] using htotal
