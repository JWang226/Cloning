21/-- Number probabilities for vacuum signal and an idler seed of `l` quanta. -/
22def seededLaw (r : ℝ) (l k : ℕ) : ℝ :=
23  ((l + k).choose l : ℝ) * (1 - r) ^ (l + 1) * r ^ k

25/-- The geometric probability generating function. -/
26def pgfBase (r z : ℝ) : ℝ := (1 - r) / (1 - r * z)

28theorem seededLaw_nonneg {r : ℝ} (hr0 : 0 ≤ r) (hr1 : r < 1) (l k : ℕ) :
29    0 ≤ seededLaw r l k := by
30  unfold seededLaw
31  exact mul_nonneg (mul_nonneg (Nat.cast_nonneg _) (pow_nonneg (by linarith) _))
32    (pow_nonneg hr0 _)

34/-- Exact seeded number generating function, including `z=1` and zero noise. -/
35theorem seededLaw_pgf_hasSum {r z : ℝ} (hr0 : 0 ≤ r) (hr1 : r < 1)
36    (hz0 : 0 ≤ z) (hz1 : z ≤ 1) (l : ℕ) :
37    HasSum (fun k ↦ seededLaw r l k * z ^ k) (pgfBase r z ^ (l + 1)) := by
38  have hrz0 : 0 ≤ r * z := mul_nonneg hr0 hz0
39  have hrz1 : r * z < 1 := (mul_le_of_le_one_right hr0 hz1).trans_lt hr1

41  have hs := (hasSum_choose_mul_geometric_of_norm_lt_one l hnorm).mul_left ((1 - r) ^ (l + 1))
42  convert hs using 1
43  · ext k
44    simp only [seededLaw, Nat.add_comm, mul_pow]
45    ring
46  · simp only [pgfBase, div_pow]

72/-- Every seeded exponential moment is bounded by the vacuum-idler moment. -/
73theorem seededLaw_moment_le {r z : ℝ} (hr0 : 0 ≤ r) (hr1 : r < 1)
74    (hz0 : 0 ≤ z) (hz1 : z ≤ 1) (l : ℕ) :
75    (∑' k, seededLaw r l k * z ^ k) ≤ pgfBase r z := by
76  rw [(seededLaw_pgf_hasSum hr0 hr1 hz0 hz1 l).tsum_eq]
77  have h0 := pgfBase_nonneg hr0 hr1 hz1

92/-- Binomial convolution proves the actual thermal-input output probabilities. -/
93theorem thermalOutput_eq (r q : ℝ) (t : ℕ) :
94    thermalOutput r q t = Cloning.Thermal.geometric (r + (1 - r) * q) t := by
95  have he : 1 - (r + (1 - r) * q) = (1 - q) * (1 - r) := by ring
96  have hb : r + (1 - r) * q = q * (1 - r) + r := by ring
97  rw [thermalOutput, Cloning.Thermal.geometric, he, hb, add_pow, Finset.mul_sum]
98  apply Finset.sum_congr rfl
99  intro n hn
100  have hnt : n ≤ t := Nat.lt_succ_iff.mp (Finset.mem_range.mp hn)
101  rw [seededLaw, show n + (t - n) = t by omega, Cloning.Thermal.geometric,
102    pow_succ, mul_pow]
103  ring

178/-- Arbitrarily correlated idler-number mixing; no product assumption on `a`. -/
179def mixtureLaw {s : ℕ} (r : Fin s → ℝ) (a : (Fin s → ℕ) → ℝ)
180    (k : Fin s → ℕ) : ℝ := ∑' l, a l * productLaw r l k

182theorem mixtureLaw_nonneg {s : ℕ} {r : Fin s → ℝ} {a : (Fin s → ℕ) → ℝ}
183    (hr0 : ∀ i, 0 ≤ r i) (hr1 : ∀ i, r i < 1) (ha0 : ∀ l, 0 ≤ a l)

187theorem mixtureJoint_summable {s : ℕ} {r : Fin s → ℝ} {a : (Fin s → ℕ) → ℝ}
188    (hr0 : ∀ i, 0 ≤ r i) (hr1 : ∀ i, r i < 1) (ha0 : ∀ l, 0 ≤ a l)
189    (ha : Summable a) :
190    Summable (fun lk : (Fin s → ℕ) × (Fin s → ℕ) ↦ a lk.1 * productLaw r lk.1 lk.2) := by
191  apply (summable_prod_of_nonneg (fun lk ↦
192    mul_nonneg (ha0 lk.1) (productLaw_nonneg hr0 hr1 _ _))).mpr
193  refine ⟨fun l ↦ (productLaw_hasSum hr0 hr1 l).summable.mul_left (a l), ?_⟩
194  convert ha using 1
195  ext l
196  change (∑' k, a l * productLaw r l k) = a l
197  rw [tsum_mul_left, (productLaw_hasSum hr0 hr1 l).tsum_eq, mul_one]

199/-- The correlated output law has exactly the idler's total mass. -/
200theorem mixtureLaw_hasSum {s : ℕ} {r : Fin s → ℝ} {a : (Fin s → ℕ) → ℝ} {c : ℝ}
201    (hr0 : ∀ i, 0 ≤ r i) (hr1 : ∀ i, r i < 1) (ha0 : ∀ l, 0 ≤ a l)
202    (ha : HasSum a c) : HasSum (mixtureLaw r a) c := by
203  have hs := mixtureJoint_summable hr0 hr1 ha0 ha.summable
204  have hcol : Summable (mixtureLaw r a) := hs.prod_symm.prod
205  have heq : (∑' k, mixtureLaw r a k) = c := by
206    unfold mixtureLaw
207    rw [hs.tsum_comm (f := fun l k ↦ a l * productLaw r l k)]
208    simp_rw [tsum_mul_left, (productLaw_hasSum hr0 hr1 _).tsum_eq, mul_one]
209    exact ha.tsum_eq
210  rw [← heq]
211  exact hcol.hasSum

215theorem mixtureLaw_pgf_eq {s : ℕ} {r z : Fin s → ℝ} {a : (Fin s → ℕ) → ℝ}

218    (ha0 : ∀ l, 0 ≤ a l) (ha : Summable a) :
219    (∑' k, mixtureLaw r a k * productTest z k) =
220      ∑' l, a l * ∏ i, pgfBase (r i) (z i) ^ (l i + 1) := by

231  rw [hw.tsum_comm (f := fun l k ↦ a l * productLaw r l k * productTest z k)]
232  simp_rw [mul_assoc, tsum_mul_left, (productLaw_pgf_hasSum hr0 hr1 hz0 hz1 _).tsum_eq]

234/-- The least-noise exponential-moment inequality for any correlated idler
235mixing law, including deficient total mass `c`. -/
236theorem mixtureLaw_moment_le {s : ℕ} {r z : Fin s → ℝ}
237    {a : (Fin s → ℕ) → ℝ} {c : ℝ}
238    (hr0 : ∀ i, 0 ≤ r i) (hr1 : ∀ i, r i < 1)
239    (hz0 : ∀ i, 0 ≤ z i) (hz1 : ∀ i, z i ≤ 1)
240    (ha0 : ∀ l, 0 ≤ a l) (ha : HasSum a c) :
241    (∑' k, mixtureLaw r a k * productTest z k) ≤ c * ∏ i, pgfBase (r i) (z i) := by
242  rw [mixtureLaw_pgf_eq hr0 hr1 hz0 hz1 ha0 ha.summable]
243  have hM0 (l : Fin s → ℕ) : 0 ≤ ∏ i, pgfBase (r i) (z i) ^ (l i + 1) :=
244    Finset.prod_nonneg (fun i _ ↦ pow_nonneg (pgfBase_nonneg (hr0 i) (hr1 i) (hz1 i)) _)
245  have hM (l : Fin s → ℕ) : (∏ i, pgfBase (r i) (z i) ^ (l i + 1)) ≤
246      ∏ i, pgfBase (r i) (z i) := by
247    rw [← (productLaw_pgf_hasSum hr0 hr1 hz0 hz1 l).tsum_eq]
248    exact productLaw_moment_le hr0 hr1 hz0 hz1 l
249  have hmajor := ha.summable.mul_right (∏ i, pgfBase (r i) (z i))
250  have hminor : Summable (fun l ↦ a l * ∏ i, pgfBase (r i) (z i) ^ (l i + 1)) := by
251    apply hmajor.of_norm_bounded
252    intro l
253    rw [Real.norm_eq_abs, abs_of_nonneg (mul_nonneg (ha0 l) (hM0 l))]
254    exact mul_le_mul_of_nonneg_left (hM l) (ha0 l)
255  calc
256    _ ≤ ∑' l, a l * ∏ i, pgfBase (r i) (z i) :=
257      hminor.tsum_le_tsum (fun l ↦ mul_le_mul_of_nonneg_left (hM l) (ha0 l)) hmajor
258    _ = _ := by rw [tsum_mul_right, ha.tsum_eq]

298/-- The exact product thermal witness obeys the least-noise bound for every
299correlated countable idler law. Deficient mass is allowed. -/
300theorem mixtureLaw_witness_moment_le {s : ℕ} {q x : Fin s → ℝ}
301    {a : (Fin s → ℕ) → ℝ} {c : ℝ}
302    (hq0 : ∀ i, 0 < q i) (hqx : ∀ i, q i < x i) (hx1 : ∀ i, x i < 1)
303    (ha0 : ∀ l, 0 ≤ a l) (ha : HasSum a c) :
304    (∑' k, mixtureLaw x a k * ∏ i, Cloning.Thermal.witness (q i) (x i) (k i)) ≤
305      c * ∏ i, Cloning.Thermal.fidelity (q i) (x i) := by
306  let C : ℝ := ∏ i, Real.sqrt ((1 - q i) / (1 - x i))
307  let z : Fin s → ℝ := fun i ↦ Real.sqrt (q i / x i)
308  have hx0 (i) : 0 < x i := (hq0 i).trans (hqx i)
309  have hz0 (i) : 0 ≤ z i := Real.sqrt_nonneg _
310  have hz1 (i) : z i ≤ 1 :=
311    Real.sqrt_le_one.mpr ((div_le_one (hx0 i)).mpr (hqx i).le)
312  have hC : 0 ≤ C := Finset.prod_nonneg (fun i _ ↦ Real.sqrt_nonneg _)
313  have hterm (k : Fin s → ℕ) :
314      (∏ i, Cloning.Thermal.witness (q i) (x i) (k i)) = C * productTest z k := by
315    simp only [Cloning.Thermal.witness_closed (hq0 _).le ((hqx _).trans (hx1 _))
316      (hx0 _).le (hx1 _), Finset.prod_mul_distrib, C, productTest, z]
317  simp_rw [hterm, ← mul_assoc, mul_right_comm (mixtureLaw x a _) C]
318  rw [tsum_mul_right]
319  calc
320    _ ≤ (c * ∏ i, pgfBase (x i) (z i)) * C :=
321      mul_le_mul_of_nonneg_right
322        (mixtureLaw_moment_le (fun i ↦ (hx0 i).le) hx1 hz0 hz1 ha0 ha) hC
323    _ = _ := by
324      rw [mul_assoc, mul_comm (∏ i, pgfBase (x i) (z i)) C, ← Finset.prod_mul_distrib]
325      congr 1
