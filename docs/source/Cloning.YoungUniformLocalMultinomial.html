15/-- Genuine uniformity in both the spectrum and the central lattice point. -/
16theorem uniform_multinomial_local_limit {ι : Type*} [Fintype ι]
17    (a R : ℝ) (ha : 0 < a) (hR : 0 ≤ R) (ε : ℝ) (hε : 0 < ε) :
18    ∃ N₀ : ℕ, ∀ N ≥ N₀, ∀ (p x : ι → ℝ) (μ : ι → ℕ),
19      (∀ i, a ≤ p i) → (∑ i, p i = 1) → (∑ i, x i = 0) →
20      (∀ i, |x i| ≤ R) →
21      (∀ i, (μ i : ℝ) = N * (p i + x i / Real.sqrt N)) →
22      |multinomialMass N p μ / gaussianMass N p x - 1| < ε := by
23  classical
24  by_contra h
25  push_neg at h
26  choose n hn p x μ hp hs hz hx hc hb using h
27  have hnlim : Tendsto n atTop atTop := tendsto_atTop_mono hn tendsto_id
28  have ht := multinomialMass_div_gaussianMass_tendsto_one_subsequence
29    n hnlim p x μ a R ha hR (Eventually.of_forall hp) (Eventually.of_forall hx) hs hz hc
30  have he : ∀ᶠ k : ℕ in atTop,
31      |multinomialMass (n k) (p k) (μ k) / gaussianMass (n k) (p k) (x k) - 1| < ε := by
32    simpa only [Real.dist_eq] using (Metric.tendsto_nhds.mp ht ε hε)
33  obtain ⟨k, hk⟩ := he.exists
34  exact (not_lt_of_ge (hb k)) hk
35 
36end Cloning.YoungMultinomial
