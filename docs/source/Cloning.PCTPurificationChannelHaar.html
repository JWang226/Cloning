9# A canonical Haar square-root purification channel

11The compact unitary group, normalized Haar probability, entangled tensor
12moment and resulting finite-sample CPTP map are all constructed here. No
13integrability or normalization premises remain in `haarPurificationChannel`.

135/-- Left Haar invariance gives exact invariance of the entangled moment under
136all tensor powers of one-particle unitaries on the input register. -/
137theorem haarMoment_unitary_invariant (n : ℕ) (V : unitary (Matrix A A ℂ)) :
138    leftTensor n (V : Matrix A A ℂ) * haarMoment n *
139      (leftTensor n (V : Matrix A A ℂ))ᴴ = haarMoment n := by
140  have h := (matrixSandwichCLM (leftTensor n (V : Matrix A A ℂ))).integral_comp_comm
141    (integrable_haar_entangledMoment (A := A) n)
142  change (∫ U : unitary (Matrix A A ℂ), leftTensor n (V : Matrix A A ℂ) *
143      entangledMoment n (U : Matrix A A ℂ) * (leftTensor n (V : Matrix A A ℂ))ᴴ
144        ∂unitaryHaar) =
145      leftTensor n (V : Matrix A A ℂ) * haarMoment n *
146        (leftTensor n (V : Matrix A A ℂ))ᴴ at h
147  rw [← h]
148  simp_rw [← entangledMoment_mul]
149  exact integral_mul_left_eq_self
150    (μ := unitaryHaar) (fun U : unitary (Matrix A A ℂ) => entangledMoment n (U : Matrix A A ℂ)) V

152theorem haarMoment_commute_unitary (n : ℕ) (V : unitary (Matrix A A ℂ)) :
153    Commute (haarMoment (A := A) n) (leftTensor n (V : Matrix A A ℂ)) := by
154  have hn : (leftTensor n (V : Matrix A A ℂ))ᴴ * leftTensor n (V : Matrix A A ℂ) = 1 := by
155    have hV : (V : Matrix A A ℂ)ᴴ * V = 1 := Unitary.star_mul_self_of_mem V.property
156    rw [← leftTensor_star, ← leftTensor_mul, hV, leftTensor_one]
157  have h := congrArg (fun X => X * leftTensor n (V : Matrix A A ℂ))
158    (haarMoment_unitary_invariant n V)
159  simp only [Matrix.mul_assoc, hn, Matrix.mul_one] at h
160  exact h.symm
