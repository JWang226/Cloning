17/-- Weighted quantum covariance follows from the actual hybrid covariance
18defect on each fixed input, without an operator-norm covariance premise. -/
19lemma weightedQuantumMap_weyl_defect_tendsto
20    (Λ : ℕ → Lp (TraceClass (Fock s)) 1 (volume : Measure (Fin k → ℝ)) →ₗ[ℂ]
21      Lp (TraceClass (Fock s)) 1 (volume : Measure (Fin k → ℝ)))
22    (r : ℝ) (G : (Fin k → ℝ) → ℝ) (hG : Integrable G)
23    (χ : (Fin k → ℝ) → ℝ) (hχ : AEStronglyMeasurable χ volume)
24    {C : ℝ} (hbound : ∀ y, ‖χ y‖ ≤ C)
25    (hquantum : ∀ z R, Tendsto (fun n =>
26      ‖Λ n (quantumL1Action z R) - quantumL1Action (r • z) (Λ n R)‖) atTop (𝓝 0))
27    (z : Fin s → ℂ) (X : TraceClass (Fock s)) :
28    Tendsto (fun n => ‖weightedQuantumMap (Λ n) G hG χ hχ hbound
29      (displacementTraceMap z X) - displacementTraceMap (r • z)
30        (weightedQuantumMap (Λ n) G hG χ hχ hbound X)‖) atTop (𝓝 0) := by
31  have hb (n : ℕ) : ‖weightedQuantumMap (Λ n) G hG χ hχ hbound
32      (displacementTraceMap z X) - displacementTraceMap (r • z)
33        (weightedQuantumMap (Λ n) G hG χ hχ hbound X)‖ ≤
34      C * ‖Λ n (quantumL1Action z (prepareL1 G hG X)) -
35        quantumL1Action (r • z) (Λ n (prepareL1 G hG X))‖ := by
36    change ‖weightedL1Integral χ hχ hbound
37        (Λ n (prepareL1 G hG (displacementTraceMap z X))) -
38      displacementTraceMap (r • z) (weightedL1Integral χ hχ hbound
39        (Λ n (prepareL1 G hG X)))‖ ≤ _
40    rw [prepareL1_compLp, ← weightedL1Integral_compLp, ← map_sub]
41    exact norm_weightedL1Integral_le χ hχ hbound _
42  exact squeeze_zero (fun _ => norm_nonneg _) hb
43    (by simpa using (hquantum z (prepareL1 G hG X)).const_mul C)

45/-- One subsequence of the actual weighted hybrid maps converges against
46every compact quantum observable to a CP covariant residual with the sharp
47classical Gaussian trace bound. -/
48theorem exists_gaussian_weighted_covariant_residual
49    (Λ : ℕ → Lp (TraceClass (Fock s)) 1 (volume : Measure (Fin k → ℝ)) →L[ℂ]
50      Lp (TraceClass (Fock s)) 1 (volume : Measure (Fin k → ℝ)))
51    (hΛ : ∀ n, L1CompletelyPositive (Λ n).toLinearMap)
52    (hTP : ∀ n, L1TracePreserving (Λ n).toLinearMap)
53    (a : Fin k → ℝ) (ha : ∀ i, 0 < a i) {g : ℝ} (hg : 1 ≤ g)
54    (hχ : AEStronglyMeasurable
55      (GaussianAffinity.productWitness a (fun i => a i / g)) volume)
56    {C : ℝ} (hC : 0 ≤ C) (hχbound : ∀ y,
57      ‖GaussianAffinity.productWitness a (fun i => a i / g) y‖ ≤ C)
58    (b : ℕ → (Fin k → ℝ) → ℝ)
59    (hb : ∀ n, AEStronglyMeasurable (b n) volume)
60    {B : ℝ} (hbbound : ∀ n h, ‖b n h‖ ≤ B)
61    (hlim : ∀ h, Tendsto (fun n => b n h) atTop (𝓝 0))
62    (hclassical : ∀ n h R, ‖Λ n (classicalTranslation h R) -
63      classicalTranslation (Real.sqrt g • h) (Λ n R)‖ ≤ b n h * ‖R‖)
64    (hquantum : ∀ z R, Tendsto (fun n =>
65      ‖Λ n (quantumL1Action z R) - quantumL1Action (Real.sqrt g • z) (Λ n R)‖)
66        atTop (𝓝 0)) :
67    ∃ Γ : TraceClass (Fock s) →ₗ[ℂ] TraceClass (Fock s),
68      IsCompletelyPositive Γ ∧
69      (∀ X, 0 ≤ X.1 → (traceCLM (Γ X)).re ≤
70        Thermal.classicalBase g ^ ((k : ℝ) / 2) * (traceCLM X).re) ∧
71      (∀ z X, Γ (displacementTraceMap z X) =
72        displacementTraceMap (Real.sqrt g • z) (Γ X)) ∧
73      ∃ φ : ℕ → ℕ, StrictMono φ ∧
74        ∀ X (O : Fock s →L[ℂ] Fock s), IsCompactOperator O →
75          Tendsto (fun n => tracePairing
76            (weightedQuantumMap (Λ (φ n)).toLinearMap (GaussianAffinity.productDensity a)
77              (GaussianAffinity.integrable_productDensity a ha)
78              (GaussianAffinity.productWitness a (fun i => a i / g)) hχ hχbound X) O)
79            atTop (𝓝 (tracePairing (Γ X) O)) := by
80  let G := GaussianAffinity.productDensity a
81  let χ := GaussianAffinity.productWitness a (fun i => a i / g)
82  let hG := GaussianAffinity.integrable_productDensity a ha
83  let L := fun n => weightedQuantumMap (Λ n).toLinearMap G hG χ hχ hχbound
84  have hG0 := GaussianAffinity.productDensity_nonneg a
85  have hG1 : ∫ y, G y = 1 := GaussianAffinity.integral_productDensity a ha
86  have hχ0 (y : Fin k → ℝ) : 0 ≤ χ y :=
87    (GaussianAffinity.productWitness_pos a _ ha
88      (fun i => div_pos (ha i) (lt_of_lt_of_le zero_lt_one hg)) y).le
89  have hCP (n : ℕ) : IsCompletelyPositive (L n) :=
90    weightedQuantumMap_completelyPositive _ (hΛ n) G hG hG0 χ hχ hχ0 hχbound
91  have htrace (n : ℕ) (X : TraceClass (Fock s)) (hX : 0 ≤ X.1) :
92      (traceCLM (L n X)).re ≤ C * (traceCLM X).re :=
93    weightedQuantumMap_trace_le _ (hΛ n) (hTP n) G hG hG0 hG1 χ hχ hχbound X hX
94  have hdef (z : Fin s → ℂ) (X : TraceClass (Fock s)) :
95      Tendsto (fun n => ‖L n (sandwichCLM (displacement z) (star (displacement z)) X) -
96        sandwichCLM (displacement (Real.sqrt g • z)) (star (displacement (Real.sqrt g • z)))
97          (L n X)‖) atTop (𝓝 0) := by
98    simpa only [ContinuousLinearMap.star_eq_adjoint, displacement_adjoint,
99      displacementTraceMap] using weightedQuantumMap_weyl_defect_tendsto
100        (fun n => (Λ n).toLinearMap) (Real.sqrt g) G hG χ hχ hχbound hquantum z X
101  obtain ⟨Γ, hΓ, _, hcov, φ, hφ, hcoeff, hcompact⟩ :=
102    exists_subsequence_covariant_completelyPositive_limit L C hC hCP htrace
103      displacement (fun z => displacement (Real.sqrt g • z)) hdef
104  refine ⟨Γ, hΓ, ?_, ?_, φ, hφ, hcompact⟩
105  · intro X hX
106    apply InfiniteTraceClassAsymptoticBound.trace_le_of_diagonal_tendsto
107      (fun n => L (φ n) X) (Γ X) (fun n => (hCP (φ n)).map_nonneg X hX)
108      (hΓ.map_nonneg X hX) (Thermal.classicalBase g ^ ((k : ℝ) / 2) * (traceCLM X).re)
109      (fun x => hcoeff X x x)
110    intro ε hε
111    have ht : 0 ≤ (traceCLM X).re := trace_re_nonneg hX X.2
112    have hd : 0 < (traceCLM X).re + 1 := by linarith
113    have he := eventually_weightedQuantumMap_gaussian_trace_le Λ hΛ hTP a ha hg hχ hC
114      hχbound b hb hbbound hlim hclassical (ε / ((traceCLM X).re + 1)) (div_pos hε hd)
115    filter_upwards [hφ.tendsto_atTop.eventually he] with n hn
116    apply (hn X hX).trans
117    have hmul : ε / ((traceCLM X).re + 1) * (traceCLM X).re ≤ ε := by
118      calc
119        _ ≤ ε / ((traceCLM X).re + 1) * ((traceCLM X).re + 1) :=
120          mul_le_mul_of_nonneg_left (by linarith) (div_pos hε hd).le
121        _ = ε := div_mul_cancel₀ _ hd.ne'
122    nlinarith
123  · simpa only [ContinuousLinearMap.star_eq_adjoint, displacement_adjoint,
124      displacementTraceMap] using hcov
