5/-!
6# The actual operator-valued L¹ space behind hybrid states
7 
8Positive fields map to Mathlib's Banach space of integrable functions modulo
9almost-everywhere equality. Their mass and trace distance agree exactly with
10the `L¹` norm. Forgetting the continuous classical register produces a positive
11trace-class operator, and its trace is the integrated trace.
12-/
13noncomputable section

22def toL1 (R : PositiveField (H := H) μ) : Lp (TraceClass H) 1 μ :=

32/-- Hybrid trace distance is exactly the norm difference in operator-valued L¹. -/
33theorem traceDistance_eq_norm (R S : PositiveField (H := H) μ) :
34    R.traceDistance S = ‖R.toL1 - S.toL1‖ := by
35  rw [L1.norm_eq_integral_norm]
36  apply integral_congr_ae
37  filter_upwards [Lp.coeFn_sub R.toL1 S.toL1, R.coe_toL1, S.coe_toL1] with x hsub hR hS
38  dsimp only [Pi.sub_apply] at hsub
39  rw [hsub, hR, hS]

62/-- The operator-valued Bochner integral is a positive trace-class operator. -/
63theorem quantumMarginal_nonneg (R : PositiveField (H := H) μ) :
64    0 ≤ R.quantumMarginal.1 := by
65  apply R.quantumMarginal.1.nonneg_iff_isPositive.mpr
66  apply (ContinuousLinearMap.isPositive_iff_complex R.quantumMarginal.1).mpr
67  intro v
68  have hnonneg : 0 ≤ ⟪v, R.quantumMarginal.1 v⟫_ℂ := by
69    have hi := (traceClassMatrixCoefficient v v).integrable_comp R.integrable
70    have h := integral_complex_nonneg hi (fun x =>
71      (((R.value x).1.1).nonneg_iff_isPositive.mp (R.value x).2).inner_nonneg_right v)
72    have he := (traceClassMatrixCoefficient v v).integral_comp_comm R.integrable
73    change (∫ x, ⟪v, (R.value x).1.1 v⟫_ℂ ∂μ) = ⟪v, R.quantumMarginal.1 v⟫_ℂ at he
74    simp only [traceClassMatrixCoefficient_apply] at h
75    rwa [he] at h
76  have hleft : 0 ≤ ⟪R.quantumMarginal.1 v, v⟫_ℂ := by
77    rw [← inner_conj_symm (R.quantumMarginal.1 v) v]
78    apply Complex.nonneg_iff.mpr
79    obtain ⟨hre, him⟩ := Complex.nonneg_iff.mp hnonneg
80    simpa only [Complex.conj_re, Complex.conj_im, ← him, neg_zero] using And.intro hre (rfl : (0 : ℝ) = 0)
81  obtain ⟨hre, him⟩ := Complex.nonneg_iff.mp hleft
82  refine ⟨?_, hre⟩
83  apply Complex.ext
84  · simp
85  · simpa using him
