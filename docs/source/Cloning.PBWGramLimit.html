4/-!
5# Normalized PBW Gram limits
6 
7Factorial normalization turns the general-word Wick leading term into the
8identity on distinct occupation lists. Local cutoff commutator errors tending
9to zero therefore imply convergence of the actual normalized vector Gram
10entries. The representation spaces may vary with the asymptotic parameter.
11-/

22/-- The occupation factorial is always positive, including the vacuum word. -/
23theorem occupationFactorial_pos (w : List ι) : 0 < occupationFactorial w := by
24  induction w with
25  | nil => simp [occupationFactorial]
26  | cons i w ih => exact Nat.mul_pos (by omega) ih

28/-- The normalization is determined only by root occupation numbers. -/
29theorem occupationFactorial_eq_of_perm [Fintype ι] {u v : List ι} (h : u.Perm v) :
30    occupationFactorial u = occupationFactorial v := by
31  simp only [occupationFactorial_eq_prod]
32  exact Finset.prod_congr rfl (fun i _ => congrArg Nat.factorial (h.count_eq i))

34/-- Ordered PBW vector divided by the square root of its occupation factorial. -/
35def normalizedWord (F : ι → H →L[ℂ] H) (w : List ι) (Ω : H) : H :=
36  ((Real.sqrt (occupationFactorial w : ℝ))⁻¹ : ℂ) • word F w Ω

38private theorem sqrt_factorial_ge_one (w : List ι) :
39    1 ≤ Real.sqrt (occupationFactorial w : ℝ) := by
40  apply Real.one_le_sqrt.mpr
41  exact_mod_cast occupationFactorial_pos w

43private theorem norm_normalization_le_one (w : List ι) :
44    ‖((Real.sqrt (occupationFactorial w : ℝ))⁻¹ : ℂ)‖ ≤ 1 := by
45  rw [norm_inv, Complex.norm_real, Real.norm_eq_abs, abs_of_nonneg (by positivity)]
46  exact (inv_le_one₀ (by have := sqrt_factorial_ge_one w; linarith)).mpr
47    (sqrt_factorial_ge_one w)

49/-- Factorial normalization can only improve the raw Wick Gram error. -/
50theorem normalized_gram_error_le [Fintype ι]
51    (F : ι → H →L[ℂ] H) (Ω : H) (u v : List ι) :
52    ‖⟪normalizedWord F u Ω, normalizedWord F v Ω⟫_ℂ -
53      (if u.Perm v then 1 else 0 : ℂ)‖ ≤
54    ‖⟪word F u Ω, word F v Ω⟫_ℂ -
55      (if u.Perm v then (occupationFactorial u : ℂ) else 0)‖ := by
56  let a : ℂ := ((Real.sqrt (occupationFactorial u : ℝ))⁻¹ : ℂ)
57  let b : ℂ := ((Real.sqrt (occupationFactorial v : ℝ))⁻¹ : ℂ)
58  have ha : ‖a‖ ≤ 1 := norm_normalization_le_one u
59  have hb : ‖b‖ ≤ 1 := norm_normalization_le_one v
60  have hlead : (a * b) * (if u.Perm v then (occupationFactorial u : ℂ) else 0) =
61      (if u.Perm v then 1 else 0 : ℂ) := by
62    split_ifs with huv
63    · have heq := occupationFactorial_eq_of_perm huv
64      dsimp [a, b]
65      rw [← heq]
66      have hr : (Real.sqrt (occupationFactorial u : ℝ))⁻¹ *
67          (Real.sqrt (occupationFactorial u : ℝ))⁻¹ * (occupationFactorial u : ℝ) = 1 := by
68        have hs : Real.sqrt (occupationFactorial u : ℝ) ≠ 0 :=
69          ne_of_gt (by have := sqrt_factorial_ge_one u; linarith)
70        field_simp
71        exact (Real.sq_sqrt (by positivity)).symm
72      exact_mod_cast hr
73    · simp
74  have heq : ⟪normalizedWord F u Ω, normalizedWord F v Ω⟫_ℂ =
75      (a * b) * ⟪word F u Ω, word F v Ω⟫_ℂ := by
76    simp [normalizedWord, inner_smul_left, inner_smul_right, a, b, mul_assoc, mul_left_comm]
77  rw [heq, ← hlead, ← mul_sub, norm_mul]
78  calc
79    _ ≤ 1 * ‖⟪word F u Ω, word F v Ω⟫_ℂ -
80        (if u.Perm v then (occupationFactorial u : ℂ) else 0)‖ := by
81      apply mul_le_mul_of_nonneg_right _ (norm_nonneg _)
82      simpa only [norm_mul, one_mul] using mul_le_mul ha hb (norm_nonneg b) (by norm_num : (0:ℝ) ≤ 1)
83    _ = _ := one_mul _

85/-- Normalized fixed-height PBW Gram bound, with all inputs stated at the
86operator and local commutator level. -/
87theorem normalized_gram_error_le_of_cutoff [Fintype ι]
88    (F A : ι → H →L[ℂ] H)
89    (K : ℕ → Submodule ℂ H) (hK : Monotone K) (Ω : H) (hΩmem : Ω ∈ K 0)
90    (hnorm : ⟪Ω, Ω⟫_ℂ = 1) (hΩ : ∀ i, A i Ω = 0)
91    (hadj : ∀ i x y, ⟪F i x, y⟫_ℂ = ⟪x, A i y⟫_ℂ)
92    (hF : ∀ i r x, x ∈ K r → F i x ∈ K (r + 1))
93    (hA : ∀ i r x, x ∈ K r → A i x ∈ K (r + 1))
94    (L : ℕ) (M ε : ℝ) (hM : 1 ≤ M) (hε : 0 ≤ ε)
95    (hbound : ∀ i r, r ≤ L → ∀ x ∈ K r, ‖F i x‖ ≤ M * ‖x‖)
96    (hcomm : ∀ i j r, r ≤ L → ∀ x ∈ K r,
97      ‖commutatorDefect F A i j x‖ ≤ ε * ‖x‖)
98    (u v : List ι) (hu : u.length ≤ L) (hv : v.length ≤ L) :
99    ‖⟪normalizedWord F u Ω, normalizedWord F v Ω⟫_ℂ -
100      (if u.Perm v then 1 else 0 : ℂ)‖ ≤
101      (gramConstant L u.length : ℝ) * M ^ (2 * L) * ε :=
102  (normalized_gram_error_le F Ω u v).trans
103    (gram_error_le_of_cutoff F A K hK Ω hΩmem hnorm hΩ hadj hF hA
104      L M ε hM hε hbound hcomm u v hu hv)

106/-- Local normalized commutator convergence yields normalized Gram convergence,
107even when every highest weight lives in a different Hilbert space. -/
108theorem normalized_gram_tendsto_of_cutoff [Fintype ι]
109    (E : ℕ → Type*) [∀ n, NormedAddCommGroup (E n)] [∀ n, InnerProductSpace ℂ (E n)]
110    (F A : ∀ n, ι → E n →L[ℂ] E n)
111    (K : ∀ n, ℕ → Submodule ℂ (E n)) (Ω : ∀ n, E n)
112    (L : ℕ) (M : ℝ) (ε : ℕ → ℝ) (hM : 1 ≤ M) (hε : ∀ n, 0 ≤ ε n)
113    (hεlim : Tendsto ε atTop (𝓝 0))
114    (hK : ∀ n, Monotone (K n)) (hΩmem : ∀ n, Ω n ∈ K n 0)
115    (hnorm : ∀ n, ⟪Ω n, Ω n⟫_ℂ = 1) (hΩ : ∀ n i, A n i (Ω n) = 0)
116    (hadj : ∀ n i x y, ⟪F n i x, y⟫_ℂ = ⟪x, A n i y⟫_ℂ)
