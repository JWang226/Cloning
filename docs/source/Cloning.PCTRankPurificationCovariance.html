15/-- Exact conjugation covariance of the square-root sandwich from actual
16commutation with the input unitary and its adjoint. -/
17theorem purificationMap_conjugate_of_commute (R : Matrix (A×B) (A×B) ℂ)
18    (V X : Matrix A A ℂ)
19    (hV : Commute R (V ⊗ₖ (1 : Matrix B B ℂ)))
20    (hVs : Commute R (Vᴴ ⊗ₖ (1 : Matrix B B ℂ))) :
21    purificationMap R (V*X*Vᴴ)=
22      (V ⊗ₖ (1 : Matrix B B ℂ))*purificationMap R X*(V ⊗ₖ (1 : Matrix B B ℂ))ᴴ := by
23  have hs : Commute (CFC.sqrt R) (V ⊗ₖ (1 : Matrix B B ℂ)) := hV.cfcₙ_nnreal NNReal.sqrt
24  have hss : Commute (CFC.sqrt R) ((V ⊗ₖ (1 : Matrix B B ℂ))ᴴ) := by
25    simpa only [Matrix.conjTranspose_kronecker,Matrix.conjTranspose_one] using
26      hVs.cfcₙ_nnreal NNReal.sqrt
27  have ht : (V*X*Vᴴ) ⊗ₖ (1 : Matrix B B ℂ)=
28      (V ⊗ₖ (1 : Matrix B B ℂ))*(X ⊗ₖ (1 : Matrix B B ℂ))*(V ⊗ₖ (1 : Matrix B B ℂ))ᴴ := by
29    simp only [Matrix.conjTranspose_kronecker,Matrix.conjTranspose_one,
30      ← Matrix.mul_kronecker_mul,Matrix.one_mul]
31  unfold purificationMap
32  rw [ht]
33  calc
34    _ = (CFC.sqrt R*(V ⊗ₖ (1 : Matrix B B ℂ)))*(X ⊗ₖ (1 : Matrix B B ℂ))*
35        ((V ⊗ₖ (1 : Matrix B B ℂ))ᴴ*CFC.sqrt R) := by noncomm_ring
36    _ = ((V ⊗ₖ (1 : Matrix B B ℂ))*CFC.sqrt R)*(X ⊗ₖ (1 : Matrix B B ℂ))*
37        (CFC.sqrt R*(V ⊗ₖ (1 : Matrix B B ℂ))ᴴ) := by rw [hs.eq,← hss.eq]
38    _ = _ := by noncomm_ring
