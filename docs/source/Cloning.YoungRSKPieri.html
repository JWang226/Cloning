16theorem tableauRemovable_iff_shape {d : ℕ} (T : List (RowContent d)) (r : ℕ) :
17    tableauRemovable T r ↔ shapeRemovable (tableauShape T) r := by
18  induction T generalizing r with
19  | nil => simp [tableauRemovable, tableauShape, shapeRemovable]
20  | cons U T ih =>
21    cases r with
22    | zero =>
23      cases T <;> simp [tableauRemovable, tableauShape, shapeRemovable, rowMass]
24    | succ r => exact ih r

26theorem semistandardContent_antitone {d N : ℕ} {μ : Fin d → ℕ}
27    (A : SemistandardContent N μ) : Antitone μ := by
28  intro i j hij
29  rcases eq_or_lt_of_le hij with rfl | hij
30  · rfl
31  · have h := rowInterlaces_mass_le (fun k => (A.val i k).val)
32      (fun k => (A.val j k).val) (semistandardContent_rowInterlaces A.property i j hij)

46def contentInsertion {d N : ℕ} (μ : Fin d → ℕ) (hN : ∀ i, μ i ≤ N) :
47    SemistandardContent N μ × Fin d → ContentSuccessor N μ :=

50theorem contentInsertion_injective {d N : ℕ} (μ : Fin d → ℕ) (hN : ∀ i, μ i ≤ N) :
51    Function.Injective (contentInsertion μ hN) := by
52  rintro ⟨A, x⟩ ⟨B, y⟩ he
53  have hc := congrArg (fun z : ContentSuccessor N μ => z.1.val) he
54  have hr := congrArg (fun z : ContentSuccessor N μ => contentRows z.2.val) he
55  change (contentInsertCorner A x).val = (contentInsertCorner B y).val at hc
56  change contentRows (contentInsert A hN x).val = contentRows (contentInsert B hN y).val at hr
57  rw [contentInsert_rows, contentInsert_rows] at hr
58  have ha := tableauReverse_after_insert_fixed (contentRows A.val) x (contentRows_length _)
59  have hb := tableauReverse_after_insert_fixed (contentRows B.val) y (contentRows_length _)
60  change (tableauInsert (contentRows A.val) x).2 =
61    (tableauInsert (contentRows B.val) y).2 at hc
62  rw [hr, hc] at ha
63  have hp := Option.some.inj (ha.symm.trans hb)
64  have hAB : A = B := Subtype.ext (contentRows_injective (congrArg Prod.fst hp))
65  exact Prod.ext hAB (congrArg (fun z : List (RowContent d) × Fin d => z.2) hp)

67theorem contentInsertion_surjective {d N : ℕ} (μ : Fin d → ℕ)
68    (hμ : Antitone μ) (hN : ∀ i, μ i ≤ N) :
69    Function.Surjective (contentInsertion μ hN) := by
70  rintro ⟨r, B⟩
71  have hrem : tableauRemovable (contentRows B.val) r.val := by
72    rw [tableauRemovable_iff_shape, contentRows_shape]
73    exact shapeRemovable_addBox μ hμ r
74  obtain ⟨S, x, hrev, hp⟩ := tableauReverse_exists_pairwise (contentRows B.val)
75    (contentRows_pairwise B) r.val hrem
76  have hlen : S.length = d :=
77    (tableauReverse_length _ _ _ _ hrev).trans (contentRows_length _)
78  have hshape : tableauShape S = List.ofFn μ := by
79    apply shapeAdd_injective_of_length (r := r.val)
80    · simpa only [tableauShape, List.length_map, List.length_ofFn] using hlen
81    · rw [tableauReverse_shape _ _ _ _ hrev, contentRows_shape, shapeAdd_ofFn]
82  let A := rowsSemistandardContent S hlen hshape hp hN
83  have hA : contentRows A.val = S := rowsSemistandardContent_rows _ _ _ _ _
84  have hins : tableauInsert (contentRows A.val) x = (contentRows B.val, r.val) := by
85    rw [hA]
86    exact tableauInsert_after_reverse _ _ _ _ hrev
87  have hcorner : contentInsertCorner A x = r := Fin.ext (congrArg Prod.snd hins)
88  refine ⟨(A, x), ?_⟩
89  apply Sigma.ext hcorner
90  have hrows : contentRows (contentInsert A hN x).val = contentRows B.val := by
91    rw [contentInsert_rows]
92    exact congrArg Prod.fst hins
93  have hval := contentRows_injective hrows
94  exact (Subtype.heq_iff_coe_eq (fun _ => by
95    change IsSemistandardContent (addBox μ (contentInsertCorner A x)) _ ↔
96      IsSemistandardContent (addBox μ r) _
97    rw [hcorner])).mpr hval

99def contentInsertionEquiv {d N : ℕ} (μ : Fin d → ℕ)
100    (hμ : Antitone μ) (hN : ∀ i, μ i ≤ N) :
101    (SemistandardContent N μ × Fin d) ≃ ContentSuccessor N μ :=

105/-- Exact weighted Pieri identity for the literal finite-array Schur sum. -/
106theorem schurPolynomial_pieri {d : ℕ} (N : ℕ) (p : Fin d → ℝ) (μ : Fin d → ℕ)
107    (hμ : Antitone μ) (hN : ∀ i, μ i ≤ N) :
108    (∑ i : Fin d, schurPolynomial (N + 1) p (addBox μ i)) =
109      schurPolynomial N p μ * ∑ i, p i := by
110  simp only [schurPolynomial_eq_content_sum]
111  rw [← Fintype.sum_sigma (fun z : ContentSuccessor N μ => contentWeight p z.2.val)]
112  rw [← (contentInsertionEquiv μ hμ hN).sum_comp]
113  change (∑ z : SemistandardContent N μ × Fin d,
114    contentWeight p (contentInsert z.1 hN z.2).val) = _
115  simp only [contentInsert_weight, Fintype.sum_prod_type]
116  simp_rw [← Finset.mul_sum]
117  exact (Finset.sum_mul _ _ _).symm
