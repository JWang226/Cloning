4/-!
5# Positive trace pairing of Hilbert–Schmidt operators
6 
7Two positive Hilbert–Schmidt operators have nonnegative trace pairing even
8when neither factor is trace class. The proof uses positive finite matrix
9compressions and the absolutely summable matrix expansion of their product.
10The directed family of finite subsets is unrestricted, so no separability
11assumption is required.
12-/

25/-- A finite matrix of a positive operator against any family of vectors is
26positive semidefinite. No orthogonality is required. -/
27theorem matrixCoefficients_posSemidef {n : Type*} [Fintype n] [DecidableEq n]
28    {R : H →L[ℂ] H} (hR : 0 ≤ R) (v : n → H) :
29    Matrix.PosSemidef (fun i j => ⟪v i, R (v j)⟫_ℂ) := by
30  have hpos := R.nonneg_iff_isPositive.mp hR
31  apply Matrix.PosSemidef.of_dotProduct_mulVec_nonneg
32  · ext i j
33    change star ⟪v j, R (v i)⟫_ℂ = ⟪v i, R (v j)⟫_ℂ
34    exact (inner_conj_symm (R (v i)) (v j)).trans
35      (hpos.isSelfAdjoint.isSymmetric (v i) (v j))
36  · intro z
37    have h := hpos.inner_nonneg_right (∑ i, z i • v i)
38    simp only [map_sum, map_smul, sum_inner, inner_sum, inner_smul_left,
39      inner_smul_right, Finset.mul_sum] at h
40    rw [Finset.sum_comm] at h
41    simpa only [map_sum, map_smul, sum_inner, inner_sum, inner_smul_left,
42      inner_smul_right, dotProduct, Matrix.mulVec, Pi.star_apply, Finset.mul_sum,
43      starRingEnd_apply, mul_assoc, mul_comm, mul_left_comm] using h

45/-- Positivity of each finite square matrix pairing. -/
46theorem finite_trace_pairing_nonneg {ι : Type*} {R S : H →L[ℂ] H}
47    (hR : 0 ≤ R) (hS : 0 ≤ S) (v : ι → H) (s : Finset ι) :
48    0 ≤ (∑ i ∈ s, ∑ j ∈ s, ⟪v i, R (v j)⟫_ℂ * ⟪v j, S (v i)⟫_ℂ).re := by
49  classical
50  have h := MatrixFidelity.trace_mul_re_nonneg
51    (matrixCoefficients_posSemidef hR (fun i : s => v i))
52    (matrixCoefficients_posSemidef hS (fun i : s => v i))
53  simp only [Matrix.trace, Matrix.diag, Matrix.mul_apply] at h
54  have heq : (∑ i : s, ∑ j : s, ⟪v i, R (v j)⟫_ℂ * ⟪v j, S (v i)⟫_ℂ) =
55      ∑ i ∈ s, ∑ j ∈ s, ⟪v i, R (v j)⟫_ℂ * ⟪v j, S (v i)⟫_ℂ := by
56    calc
57      _ = ∑ i : s, ∑ j ∈ s, ⟪v i, R (v j)⟫_ℂ * ⟪v j, S (v i)⟫_ℂ := by
58        apply Finset.sum_congr rfl
59        intro i _
60        exact Finset.sum_coe_sort s
61          (fun j : ι => ⟪v i, R (v j)⟫_ℂ * ⟪v j, S (v i)⟫_ℂ)
62      _ = _ := Finset.sum_coe_sort s
63        (fun i : ι => ∑ j ∈ s, ⟪v i, R (v j)⟫_ℂ * ⟪v j, S (v i)⟫_ℂ)
64  rw [heq] at h
65  exact h

67/-- Finite square subsets form a cofinal family among subsets of a Cartesian
68square. This avoids selecting a countable Hilbert basis. -/
69theorem tendsto_finset_square_atTop (ι : Type*) :
70    Tendsto (fun s : Finset ι => s ×ˢ s) atTop atTop := by
71  classical
72  apply Monotone.tendsto_atTop_atTop
73  · intro s t hst
74    exact Finset.product_subset_product hst hst
75  · intro t
76    refine ⟨t.image Prod.fst ∪ t.image Prod.snd, ?_⟩
77    intro p hp
78    simp only [Finset.mem_product, Finset.mem_union, Finset.mem_image]
79    exact ⟨Or.inl ⟨p, hp, rfl⟩, Or.inr ⟨p, hp, rfl⟩⟩

81/-- The trace pairing of two nonnegative Hilbert–Schmidt operators is
82nonnegative. Neither factor is assumed to have finite trace. -/
83theorem trace_re_mul_nonneg_of_isHilbertSchmidt {R S : H →L[ℂ] H}
84    (hR : HilbertSchmidt.IsHilbertSchmidt R) (hS : HilbertSchmidt.IsHilbertSchmidt S)
85    (hR0 : 0 ≤ R) (hS0 : 0 ≤ S) :
86    0 ≤ (trace (R * S) (isTraceClass_mul_of_isHilbertSchmidt hR hS)).re := by
87  classical
88  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
89  let F : w × w → ℂ := fun p =>
90    ⟪b p.1, R (b p.2)⟫_ℂ * ⟪b p.2, S (b p.1)⟫_ℂ
91  have hF : Summable F := by
92    simpa only [F, Function.uncurry, ContinuousLinearMap.adjoint_inner_left] using
93      HilbertSchmidt.summable_matrix_of_hilbertSchmidt b b hR hS
94  have htotal : (∑' p, F p) = trace (R * S) (isTraceClass_mul_of_isHilbertSchmidt hR hS) := by
95    rw [hF.tsum_prod, trace_eq_of_hilbertBasis _ b]
96    apply tsum_congr
97    intro i
98    simpa only [F, ContinuousLinearMap.adjoint_inner_left] using
99      (HilbertSchmidt.hasSum_matrix_row_of_hilbertSchmidt (R := R) (S := S) b b i).tsum_eq
100  have hlim : Tendsto (fun s : Finset w => ∑ p ∈ s ×ˢ s, F p) atTop
101      (𝓝 (trace (R * S) (isTraceClass_mul_of_isHilbertSchmidt hR hS))) := by
102    rw [← htotal]
103    exact hF.hasSum.comp (tendsto_finset_square_atTop w)
104  have hre := Complex.continuous_re.continuousAt.tendsto.comp hlim
105  apply le_of_tendsto_of_tendsto tendsto_const_nhds hre
106  apply Eventually.of_forall
107  intro s
108  simpa only [Finset.sum_product, F] using finite_trace_pairing_nonneg hR0 hS0 b s
