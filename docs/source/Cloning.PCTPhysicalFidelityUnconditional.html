16/-- Unconditional physical PCT root-fidelity limit on every unitary orbit of
17a positive simple spectrum. -/
18theorem physical_pct_fidelity_unitary
19    (p : SimpleSpectrum (k+1)) (U : unitary (Matrix (Fin (k+1)) (Fin (k+1)) ℂ))
20    (hs : 1 ≤ s) (hcard : Fintype.card (Fin (k+1) × Fin (k+1)) = s+1)
21    (r : ℕ → ℕ) (γ : ℝ) (hγ : 1 < γ)
22    (hr : Tendsto (fun n => ((n+r n : ℕ) : ℝ) / n) atTop (𝓝 γ)) :
23    Tendsto (fun n =>
24      (outputState hcard (conjugatedState (baseState p) U) n (r n)).rootFidelity
25      (tensorState (conjugatedState (baseState p) U) (n+r n))) atTop (𝓝 (pctValue γ p)) := by
26  obtain ⟨b,hb⟩ := exists_whitening_frame (by simp : Fintype.card (Fin (k+1)) = k+1)
27    p.eigenvalue (fun a => (p.positive a).le) p.normalized
28  let e := (Fintype.equivFin (PairIndex (k+1))).symm
29  exact physical_pct_fidelity_unitary_of_compactWindowLAN p b hb e
