29/-- The actual product Gaussian measure has the exact symplectic Fourier
30transform, with the sign convention of the constructed Weyl operators. -/
31theorem integral_weylCharacter_gaussianProductMeasure {s : Fin d → ℝ}
32    (hs : ∀ i, 0 < s i) (a : Fin d → ℂ) :
33    (∫ z, weylCharacter z a ∂gaussianProductMeasure s) =
34      ∏ i, Complex.exp (-((s i * ‖a i‖ ^ 2 : ℝ) : ℂ)) := by
35  letI (i : Fin d) : IsProbabilityMeasure (Cloning.CoherentGaussianMixture.gaussianMeasure (s i)) :=
36    Cloning.CoherentGaussianMixture.gaussianMeasure_probability (hs i)
37  simp_rw [weylCharacter_eq_prod_exp]
38  rw [gaussianProductMeasure, integral_fin_nat_prod_eq_prod
39    (μ := fun i : Fin d => Cloning.CoherentGaussianMixture.gaussianMeasure (s i))
40    (fun i (z : ℂ) => Complex.exp
41      (z * (starRingEnd ℂ) (a i) - (starRingEnd ℂ) z * a i))]
42  apply Finset.prod_congr rfl
43  intro i _
44  have heq (z : ℂ) : z * (starRingEnd ℂ) (a i) - (starRingEnd ℂ) z * a i =
45      (-a i) * (starRingEnd ℂ) z - (starRingEnd ℂ) (-a i) * z := by
46    rw [map_neg]

51/-- Lebesgue Fourier weight of the product thermal state with mean occupation
52numbers `sᵢ`. The per-mode normalization is exactly `1/π`. -/
53def thermalFourierWeight (s : Fin d → ℝ) (a : Fin d → ℂ) : ℝ :=
54  ∏ i, Real.exp (-(s i + 1 / 2) * ‖a i‖ ^ 2) / Real.pi

56lemma thermalFourierWeight_factor (s : Fin d → ℝ) (a : Fin d → ℂ) :
57    (weylGaussianWeight a : ℂ) *
58      (∏ i, Complex.exp (-((s i * ‖a i‖ ^ 2 : ℝ) : ℂ))) =
59      (thermalFourierWeight s a : ℂ) := by
60  simp only [weylGaussianWeight, thermalFourierWeight, Complex.ofReal_prod,
61    ← Finset.prod_mul_distrib, ComplexCoherent.weylGaussianWeight,
62    Complex.ofReal_div, ← Complex.ofReal_neg, ← Complex.ofReal_exp]
63  apply Finset.prod_congr rfl
64  intro i _
65  have hreal : Real.exp (-‖a i‖ ^ 2 / 2) / Real.pi *
66      Real.exp (-(s i * ‖a i‖ ^ 2)) =
67      Real.exp (-(s i + 1 / 2) * ‖a i‖ ^ 2) / Real.pi := by
68    rw [div_mul_eq_mul_div, ← Real.exp_add]
69    congr 2
70    ring

88/-- The same witness bridge in geometric parameters `0<qᵢ<1`, as used by
89the manuscript's product thermal state and compact witness. -/
90theorem tracePairing_geometricThermal_fourier {q : Fin d → ℝ}
91    (hq0 : ∀ i, 0 < q i) (hq1 : ∀ i, q i < 1) (T : TraceClass (Fock d)) :
92    tracePairing T (vectorMixture (numberBasis d)
93      (fun k => ∏ i, Cloning.Thermal.geometric (q i) (k i))).1 =
94      ∫ a : Fin d → ℂ,
95        (∏ i, (Real.exp (-((1 + q i) / (2 * (1 - q i))) * ‖a i‖ ^ 2) / Real.pi : ℂ)) *
96          tracePairing T (displacement a) := by
97  let s : Fin d → ℝ := fun i => q i / (1 - q i)
98  have hs (i : Fin d) : 0 < s i := div_pos (hq0 i) (sub_pos.mpr (hq1 i))
99  have hq (i : Fin d) : s i / (1 + s i) = q i := by
100    dsimp only [s]
101    field_simp [ne_of_gt (sub_pos.mpr (hq1 i))]
102    <;> ring
103  have hc (i : Fin d) : s i + 1 / 2 = (1 + q i) / (2 * (1 - q i)) := by
104    dsimp only [s]
105    field_simp [ne_of_gt (sub_pos.mpr (hq1 i))]
106    <;> ring
107  have h := tracePairing_productThermal_fourier hs T
