21/-- Rectangular trace duality for a conjugation between different Hilbert spaces. -/
22theorem tracePairing_conjugation (V : H →L[ℂ] K) (T : TraceClass H)
23    (A : K →L[ℂ] K) :
24    tracePairing (conjugationLinearMap V T) A =
25      tracePairing T (V.adjoint.comp (A.comp V)) := by
26  let f : TraceClass H →ₗ[ℂ] ℂ :=
27    (tracePairingCLM.flip A).toLinearMap.comp (conjugationLinearMap V)
28  let g : TraceClass H →ₗ[ℂ] ℂ :=
29    (tracePairingCLM.flip (V.adjoint.comp (A.comp V))).toLinearMap
30  have hpos (S : TraceClass H) (hS : 0 ≤ S.1) : f S = g S := by
31    obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
32    have hf := (conjugationLinearMap_positive_hasSum V S hS b).mapL (tracePairingCLM.flip A)
33    have hg := (positive_rankOne_series hS S.2 b).mapL
34      (tracePairingCLM.flip (V.adjoint.comp (A.comp V)))
35    have heq : (fun i : w => tracePairing (vectorProjector (V (CFC.sqrt S.1 (b i)))) A) =
36        (fun i : w => tracePairing (vectorProjector (CFC.sqrt S.1 (b i)))
37          (V.adjoint.comp (A.comp V))) := by
38      funext i
39      change tracePairing (rankOneOperator _ _) _ = tracePairing (rankOneOperator _ _) _
40      simp only [tracePairing_rankOneOperator, ContinuousLinearMap.comp_apply,
41        ContinuousLinearMap.adjoint_inner_right]
42    exact (show HasSum _ (f S) from hf).unique (heq ▸ (show HasSum _ (g S) from hg))
43  have hsa (S : TraceClass H) (hS : IsSelfAdjoint S.1) : f S = g S := by
44    rw [← TraceClass.positivePart_sub_negativePart S hS, map_sub, map_sub,
45      hpos _ (TraceClass.positivePart_nonneg S hS),
46      hpos _ (TraceClass.negativePart_nonneg S hS)]
47  change f T = g T
48  rw [← TraceClass.realComponent_add_I_smul_imaginaryComponent T,
49    map_add, map_add, map_smul, map_smul,
50    hsa _ (TraceClass.realComponent_isSelfAdjoint T),
51    hsa _ (TraceClass.imaginaryComponent_isSelfAdjoint T)]

53/-- A rectangular sandwich with distinct left and right factors, constructed
54inside the genuine trace-class space by complex polarization. -/
55def rectangularSandwich (V W : H →L[ℂ] K) (T : TraceClass H) : TraceClass K :=
56  (1 / 4 : ℂ) • (conjugationLinearMap (V + W) T - conjugationLinearMap (V - W) T +
57    Complex.I • (conjugationLinearMap (V + Complex.I • W) T -
58      conjugationLinearMap (V - Complex.I • W) T))
59 
60theorem rectangularSandwich_coe (V W : H →L[ℂ] K) (T : TraceClass H) :
61    (rectangularSandwich V W T).1 = V.comp (T.1.comp W.adjoint) := by
62  change inclusionCLM (rectangularSandwich V W T) = _
63  simp only [rectangularSandwich, map_smul, map_add, map_sub,
64    inclusionCLM_apply, conjugationLinearMap_coe]
65  ext x
66  simp only [ContinuousLinearMap.smul_apply,
67    ContinuousLinearMap.add_apply, ContinuousLinearMap.sub_apply,
68    operatorConjugation_apply, ContinuousLinearMap.comp_apply]
69  have hI : (Complex.I • W).adjoint = -Complex.I • W.adjoint := by
70    simpa using (map_smulₛₗ (ContinuousLinearMap.adjoint (𝕜 := ℂ)) Complex.I W)
71  simp only [map_add, map_sub, hI, ContinuousLinearMap.add_apply,
72    ContinuousLinearMap.sub_apply, ContinuousLinearMap.smul_apply, map_smul]
73  match_scalars <;> ring_nf <;> norm_num [Complex.I_sq]

97/-- The actual finite Hilbert direct sum used for an ancilla. -/
98abbrev BlockHilbert (ι : Type*) (H : Type*) [Fintype ι]
99    [NormedAddCommGroup H] := PiLp 2 (fun _ : ι => H)

101/-- Coordinate injection into the finite Hilbert direct sum. -/
102def blockInjection (i : ι) : H →L[ℂ] BlockHilbert ι H :=
103  (PiLp.continuousLinearEquiv 2 ℂ (fun _ : ι => H)).symm.toContinuousLinearMap.comp
104    (ContinuousLinearMap.single ℂ (fun _ : ι => H) i)
105 
106@[simp] theorem blockInjection_apply (i j : ι) (x : H) :
107    blockInjection i x j = if j = i then x else 0 := by
108  simp [blockInjection, Pi.single_apply]

110@[simp] theorem blockInjection_adjoint_apply (i : ι) (x : BlockHilbert ι H) :

122/-- Bounded matrix action on the entire finite Hilbert direct sum. -/
123def blockOperator (A : ι → ι → H →L[ℂ] H) : BlockHilbert ι H →L[ℂ] BlockHilbert ι H :=
124  ∑ i, ∑ j, (blockInjection i).comp ((A i j).comp (blockInjection j).adjoint)

126theorem blockOperator_apply (A : ι → ι → H →L[ℂ] H) (x : BlockHilbert ι H) (i : ι) :

147/-- A finite trace-class block is trace class on the actual Hilbert direct sum. -/
148def traceClassBlock (B : ι → ι → TraceClass H) : TraceClass (BlockHilbert ι H) :=
149  ∑ i, ∑ j, rectangularSandwich (blockInjection i) (blockInjection j) (B i j)

151theorem traceClassBlock_coe (B : ι → ι → TraceClass H) :
152    (traceClassBlock B).1 = blockOperator (fun i j => (B i j).1) := by
153  change (inclusionCLM (H := BlockHilbert ι H)) (traceClassBlock B) = _
154  simp only [traceClassBlock, map_sum, inclusionCLM_apply, rectangularSandwich_coe,
155    blockOperator]

157theorem traceClassBlock_nonneg (B : ι → ι → TraceClass H) (hB : BlockPositive B) :

162/-- The exact block trace pairing, with the mandatory transposition of indices. -/
