7/-!
8# Subadditivity of the trace of the positive square root
9 
10The positive definite estimate follows by comparison of inverse square roots.
11The positive semidefinite extension is obtained by regularization and continuity.
12No trace-norm triangle inequality or fidelity law is assumed.
13-/

25/-- Cancellation with the inverse positive square root of a positive definite
26matrix, using the concrete nonsingular matrix inverse. -/
27lemma mul_inv_sqrt_eq_sqrt {X : Matrix n n ℂ} (hX : X.PosDef) :
28    X * (CFC.sqrt X)⁻¹ = CFC.sqrt X := by
29  have hS : (CFC.sqrt X).PosDef := hX.isStrictlyPositive.sqrt.posDef
30  have hdet : IsUnit (CFC.sqrt X).det :=
31    (Matrix.isUnit_iff_isUnit_det _).mp hS.isUnit
32  calc
33    X * (CFC.sqrt X)⁻¹ = (CFC.sqrt X * CFC.sqrt X) * (CFC.sqrt X)⁻¹ := by
34      rw [sqrt_mul_self hX.posSemidef]
35    _ = CFC.sqrt X := by
36      rw [Matrix.mul_assoc, Matrix.mul_nonsing_inv _ hdet, Matrix.mul_one]

38/-- Subadditivity of the square-root trace for positive definite matrices. -/
39lemma trace_sqrt_add_le_posDef {X Y : Matrix n n ℂ}
40    (hX : X.PosDef) (hY : Y.PosDef) :
41    (Matrix.trace (CFC.sqrt (X + Y))).re ≤
42      (Matrix.trace (CFC.sqrt X)).re + (Matrix.trace (CFC.sqrt Y)).re := by
43  have hS : (X + Y).PosDef := hX.add hY
44  have hXS : X ≤ X + Y := le_add_of_nonneg_right hY.posSemidef.nonneg
45  have hYS : Y ≤ X + Y := le_add_of_nonneg_left hX.posSemidef.nonneg
46  calc
47    (Matrix.trace (CFC.sqrt (X + Y))).re =
48        (Matrix.trace ((X + Y) * (CFC.sqrt (X + Y))⁻¹)).re := by
49      rw [mul_inv_sqrt_eq_sqrt hS]
50    _ = (Matrix.trace (X * (CFC.sqrt (X + Y))⁻¹)).re +
51        (Matrix.trace (Y * (CFC.sqrt (X + Y))⁻¹)).re := by
52      rw [Matrix.add_mul, Matrix.trace_add, Complex.add_re]
53    _ ≤ (Matrix.trace (X * (CFC.sqrt X)⁻¹)).re +
54        (Matrix.trace (Y * (CFC.sqrt Y)⁻¹)).re :=
55      add_le_add
56        (trace_mul_re_mono_right hX.posSemidef (sqrt_inv_antitone hX hS hXS))
57        (trace_mul_re_mono_right hY.posSemidef (sqrt_inv_antitone hY hS hYS))
58    _ = _ := by rw [mul_inv_sqrt_eq_sqrt hX, mul_inv_sqrt_eq_sqrt hY]

60/-- Subadditivity of the trace of the positive square root, including singular
61positive semidefinite matrices. The proof removes strictly positive white-noise
62regularization by continuity along the positive cone. -/
63theorem trace_sqrt_add_le {X Y : Matrix n n ℂ}
64    (hX : X.PosSemidef) (hY : Y.PosSemidef) :
65    (Matrix.trace (CFC.sqrt (X + Y))).re ≤
66      (Matrix.trace (CFC.sqrt X)).re + (Matrix.trace (CFC.sqrt Y)).re := by
67  have hSum : Tendsto
68      (fun ε => (CFC.sqrt (regularize X ε + regularize Y ε)).trace.re)
69      (nhdsWithin 0 (Set.Ioi 0)) (nhds (CFC.sqrt (X + Y)).trace.re) := by
70    apply tendsto_trace_sqrt (hX.add hY)
71      ((tendsto_regularize_pos X).add (tendsto_regularize_pos Y))
72    filter_upwards [self_mem_nhdsWithin] with ε hε
73    exact (regularize_posSemidef hX hε.le).add (regularize_posSemidef hY hε.le)
74  exact le_of_tendsto_of_tendsto hSum
75    ((tendsto_trace_sqrt_regularize hX).add (tendsto_trace_sqrt_regularize hY))
76    (by
77      filter_upwards [self_mem_nhdsWithin] with ε hε
78      exact trace_sqrt_add_le_posDef (regularize_posDef hX hε) (regularize_posDef hY hε))
