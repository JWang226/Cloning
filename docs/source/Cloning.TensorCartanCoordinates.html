69/-- The coordinate matrix is actually isometric, because the target frame is
70complete on the literal product-sector subspace. -/
71theorem cartanMatrix_isometry
72    (mu nu : Fin d → ℕ) (hmu : Antitone mu) (hnu : Antitone nu) :
73    (cartanMatrix mu nu hmu hnu)ᴴ * cartanMatrix mu nu hmu hnu = 1 := by
74  ext i j
75  simp only [Matrix.mul_apply, Matrix.conjTranspose_apply, cartanMatrix_apply]

94theorem productPartitionGenerator_matrix
95    (mu nu : Fin d → ℕ) (hmu : Antitone mu) (hnu : Antitone nu) (a b : Fin d) :
96    LinearMap.toMatrix (cartanProductBasis mu nu hmu hnu).toBasis
97      (cartanProductBasis mu nu hmu hnu).toBasis
98      (productPartitionGenerator mu nu hmu hnu a b).toLinearMap =
99      partitionGeneratorMatrix mu hmu a b ⊗ₖ (1 : Matrix (PartitionIndex nu hnu) _ ℂ) +
100      (1 : Matrix (PartitionIndex mu hmu) _ ℂ) ⊗ₖ partitionGeneratorMatrix nu hnu a b := by
