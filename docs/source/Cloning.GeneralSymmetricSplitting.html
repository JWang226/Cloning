21def wordPairEquiv (n r d : ℕ) : Word n d × Word r d ≃ Word (n + r) d where
22  toFun p := Fin.append p.1 p.2
23  invFun w := (fun i => w (Fin.castAdd r i), fun j => w (Fin.natAdd n j))
24  left_inv p := by ext i <;> simp
25  right_inv w := Fin.append_castAdd_natAdd

39/-- The summed occupation profile of two tensor factors. -/
40def merge {n r d : ℕ} (p : Occupation n d) (q : Occupation r d) : Occupation (n + r) d :=
41  (compositionEquiv (n + r) d).symm
42    ⟨p.val + q.val, by

54theorem merge_left_injective {n r d : ℕ} (q : Occupation r d) :
55    Function.Injective (fun p : Occupation n d => merge p q) := by
56  intro p t h
57  apply Subtype.ext

76/-- Exact symmetric splitting amplitudes, with their computational multiplicities. -/
77def splittingMatrix (n r d : ℕ) :
78    Matrix (Occupation n d × Occupation r d) (Occupation (n + r) d) ℂ :=
79  fun p t => if t = merge p.1 p.2 then
80    (Real.sqrt (multiplicity p.1 : ℝ) : ℂ) * (Real.sqrt (multiplicity p.2 : ℝ) : ℂ) /
81      (Real.sqrt (multiplicity t : ℝ) : ℂ) else 0

88/-- Exact physical identification of the Cartan inclusion: its coefficients
89are derived from the actual normalized computational occupation fibers. -/
90theorem splittingMatrix_physical (n r d : ℕ) :
91    (columnMatrix n d ⊗ₖ columnMatrix r d) * splittingMatrix n r d =
92      splitColumnMatrix n r d := by
93  classical
94  ext ⟨w, z⟩ t
95  simp only [Matrix.mul_apply, Matrix.kronecker_apply, Fintype.sum_prod_type,
96    columnMatrix, column_apply]
97  rw [Finset.sum_eq_single (label w)]
98  · rw [Finset.sum_eq_single (label z)]
99    · simp only [splittingMatrix, splitColumnMatrix, wordPairEquiv, Equiv.coe_fn_mk, column_apply, ite_true]
100      rw [← merge_label]
101      by_cases ht : t = merge (label w) (label z)
102      · rw [if_pos ht, if_pos ht.symm]
103        have hp : (Real.sqrt (multiplicity (label w) : ℝ) : ℂ) ≠ 0 := by
104          exact_mod_cast (Real.sqrt_pos.mpr (by exact_mod_cast multiplicity_pos (label w))).ne'
105        have hq : (Real.sqrt (multiplicity (label z) : ℝ) : ℂ) ≠ 0 := by
106          exact_mod_cast (Real.sqrt_pos.mpr (by exact_mod_cast multiplicity_pos (label z))).ne'
107        field_simp
108      · rw [if_neg ht, if_neg (Ne.symm ht)]
109        simp
110    · intro q hq hn
111      simp [Ne.symm hn]
112    · simp
113  · intro p hp hn
114    simp [Ne.symm hn]
115  · simp

117theorem splitColumnMatrix_isometry (n r d : ℕ) :
118    (splitColumnMatrix n r d).conjTranspose * splitColumnMatrix n r d = 1 := by
119  classical
120  ext p q
121  simp only [Matrix.mul_apply, Matrix.conjTranspose_apply, splitColumnMatrix]
122  rw [(wordPairEquiv n r d).sum_comp (fun w => star (column p w) * column q w)]
123  exact congrArg (fun M : Matrix (Occupation (n + r) d) (Occupation (n + r) d) ℂ => M p q)
124    (columnMatrix_isometry (n + r) d)

126/-- The constructed symmetric splitting matrix is an isometry. -/
127theorem splittingMatrix_isometry (n r d : ℕ) :
128    (splittingMatrix n r d).conjTranspose * splittingMatrix n r d = 1 := by
129  have h := splitColumnMatrix_isometry n r d
130  rw [← splittingMatrix_physical, Matrix.conjTranspose_mul] at h
131  have hT : (columnMatrix n d ⊗ₖ columnMatrix r d).conjTranspose *
132      (columnMatrix n d ⊗ₖ columnMatrix r d) = 1 := by
133    rw [Matrix.conjTranspose_kronecker, ← Matrix.mul_kronecker_mul,
134      columnMatrix_isometry, columnMatrix_isometry, Matrix.one_kronecker_one]
135  simpa only [Matrix.mul_assoc, ← Matrix.mul_assoc (columnMatrix n d ⊗ₖ columnMatrix r d).conjTranspose,
136    hT, Matrix.one_mul] using h
