17/-- The scalar kernel dictated by the input and output Weyl cocycles. -/
18def diagonalQuantumPositiveKernel (f : (Fin d → ℂ) → ℂ) (r : Fin d → ℝ)
19    (a b : Fin d → ℂ) : ℂ :=
20  displacementPhase (-a) b / displacementPhase (-(diagonalScale r a)) (diagonalScale r b) * f (b - a)

22/-- Quantum positive-definiteness at the covariance scaling `r`. This tests
23all finite phase-space configurations and all complex scalar coefficients. -/
24def IsDiagonalQuantumPositive (f : (Fin d → ℂ) → ℂ) (r : Fin d → ℝ) : Prop :=
25  ∀ n : ℕ, ∀ b : Fin n → (Fin d → ℂ), ∀ c : Fin n → ℂ,
26    0 ≤ ∑ i, ∑ j, star (c i) * c j * diagonalQuantumPositiveKernel f r (b i) (b j)

28/-- A matrix coefficient that removes the output Weyl Gram factor exactly. -/
29lemma diagonal_displaced_gram_coefficient (r : Fin d → ℝ) (a b : Fin d → ℂ) (u : Fock d) :
30    ⟪displacement (-(diagonalScale r a)) u,
31      displacement (diagonalScale r (b - a)) (displacement (-(diagonalScale r b)) u)⟫_ℂ =
32    (displacementPhase (-(diagonalScale r a)) (diagonalScale r b))⁻¹ * ⟪u, u⟫_ℂ := by
33  have hsum : -(diagonalScale r a) + diagonalScale r b = diagonalScale r (b - a) := by
34    rw [diagonalScale_sub]

41    (displacementPhase (-(diagonalScale r a)) (diagonalScale r b))⁻¹ • v) h

45/-- Every completely positive Weyl multiplier has the quantum-positive
46kernel determined by the difference between its two Weyl cocycles. -/
47theorem multiplier_isDiagonalQuantumPositive
48    (Ψ : (Fock d →L[ℂ] Fock d) →ₗ[ℂ] (Fock d →L[ℂ] Fock d))
49    (hCP : Cloning.BoundedOperator.IsCompletelyPositive Ψ)
50    (r : Fin d → ℝ) (f : (Fin d → ℂ) → ℂ)
51    (hmult : ∀ b, Ψ (displacement b) = f b • displacement (diagonalScale r b)) :
52    IsDiagonalQuantumPositive f r := by
53  intro n b c
54  have hgram := Cloning.BoundedOperator.gram_blockPositive (fun i : Fin n => displacement (b i))
55  have h := hCP n _ hgram (fun i => c i •
56    displacement (-(diagonalScale r (b i))) (coherentVector (0 : Fin d → ℂ)))
57  convert h using 1

63  rw [displacement_adjoint, displacement_comp]

75lemma IsDiagonalQuantumPositive.two_point (hf : IsDiagonalQuantumPositive f r)
76    (hf0 : f 0 = 1) (b : Fin d → ℂ) (u v : ℂ) :
77    0 ≤ star u * u + star u * v * f b + star v * u * f (-b) + star v * v := by

95/-- The diagonal gain case has the same unit modulus bound. -/
96theorem IsDiagonalQuantumPositive.norm_le_one (hf : IsDiagonalQuantumPositive f r)
97    (hf0 : f 0 = 1) (b : Fin d → ℂ) : ‖f b‖ ≤ 1 := by
98  have h := (Complex.nonneg_iff.mp (hf.two_point hf0 b 1 (-star (f b)))).1
99  rw [hf.conj_symm hf0 b] at h
100  have heq : star (1 : ℂ) * 1 + star (1 : ℂ) * (-star (f b)) * f b +
101      star (-star (f b)) * 1 * star (f b) + star (-star (f b)) * (-star (f b)) =
102      1 - star (f b) * f b := by simp; ring
103  rw [heq] at h
104  change 0 ≤ (1 - (starRingEnd ℂ) (f b) * f b).re at h
105  rw [← Complex.normSq_eq_conj_mul_self] at h
106  simp only [Complex.sub_re, Complex.one_re, Complex.ofReal_re,
107    Complex.normSq_eq_norm_sq] at h
108  nlinarith [norm_nonneg (f b)]

110/-- All regular quantum-positive multiplier data are derived from the actual

112theorem quantumChannel_regular_diagonalQuantumPositive_multiplier
113    (Φ : QuantumChannel (Fock d) (Fock d)) (r : Fin d → ℝ)
114    (hΦ : ∀ a T, Φ.toLinearMap (displacementTraceMap a T) =
115      displacementTraceMap (diagonalScale r a) (Φ.toLinearMap T)) :
116    (∀ b, Φ.heisenberg (displacement b) =
117      diagonalWeylMultiplier Φ.heisenberg r b • displacement (diagonalScale r b)) ∧
118    diagonalWeylMultiplier Φ.heisenberg r 0 = 1 ∧

120    IsDiagonalQuantumPositive (diagonalWeylMultiplier Φ.heisenberg r) r ∧
121    (∀ b, ‖diagonalWeylMultiplier Φ.heisenberg r b‖ ≤ 1) := by
