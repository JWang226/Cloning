20theorem channel_embeddedState {s : ℕ} (k n t : ℕ)
21    (hcard : Fintype.card (Fin (r+k)×Fin r)=s+1)
22    (b0 : (Fin n → Fin (r+k))×(Fin n → Fin r))
23    (J : Matrix (Fin (r+k)) (Fin r) ℂ) (hJ : Jᴴ*J=1)
24    (ρ : Cloning.MatrixFidelity.State (Fin r)) :
25    (PCTRankGlobal.channel hcard n t (Cloning.TensorLie.coordinateInclusionMatrix r k) b0).toLinearMap
26      (tensorState (embeddedState J hJ ρ) n).1 =
27      (embeddedPurificationOutput hcard J hJ ρ n t).1 := by

30  rw [PCTRankGlobal.channel_matrix,rankPurificationChannel_embedded_positive n k b0 J hJ ρ.matrix ρ.positive,
