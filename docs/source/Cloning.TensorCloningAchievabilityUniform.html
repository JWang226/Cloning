14/-- Uniform control over a varying finite family follows by selecting a
15violating subsequence; no globally admissible reference family is needed. -/
16theorem eventually_uniform_of_subsequences {A : ℕ → Type*}
17    (S P : ∀ N, A N → Prop)
18    (hseq : ∀ (φ : ℕ → ℕ), Tendsto φ atTop atTop →
19      ∀ (x : ∀ k, A (φ k)), (∀ᶠ k in atTop, S (φ k) (x k)) →
20        ∀ᶠ k in atTop, P (φ k) (x k)) :
21    ∀ᶠ N in atTop, ∀ x, S N x → P N x := by
22  by_contra hbad
23  rw [eventually_atTop] at hbad
24  push_neg at hbad
25  choose φ hφ x hx hfail using hbad
26  have hφlim : Tendsto φ atTop atTop := tendsto_atTop_mono hφ tendsto_id
27  obtain ⟨k,hk⟩ := (hseq φ hφlim x (Eventually.of_forall hx)).exists
28  exact hfail k hk
