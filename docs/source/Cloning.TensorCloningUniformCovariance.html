37    (U : unitary (Matrix A A ℂ)) (ρ σ : PositiveTraceClass (Register A)) :
38    (ρ.map (unitaryChannel U).toPositiveTracePreservingMap).rootFidelity
39      (σ.map (unitaryChannel U).toPositiveTracePreservingMap) = ρ.rootFidelity σ := by
40  have hρ : (ρ.map (unitaryChannel U).toPositiveTracePreservingMap).map

54theorem spectrumPayoff_unitary_of_covariant {d : ℕ} (n m : ℕ)
55    (Φ : QuantumChannel (TensorRegister n (Fin (d+1))) (TensorRegister m (Fin (d+1))))
56    (p : SimpleSpectrum (d+1)) (U : unitary (Matrix (Fin (d+1)) (Fin (d+1)) ℂ))
57    (hcov : ∀ A, Φ.toLinearMap (conjugationLinearMap (tensorOperator n U) A) =
58      conjugationLinearMap (tensorOperator m U) (Φ.toLinearMap A)) :
59    spectrumPayoff n m Φ p U = spectrumPayoff n m Φ p 1 := by
60  let ρ := diagonalState p.eigenvalue (fun a => (p.positive a).le) p.normalized
61  let A := (tensorState ρ n).map Φ.toPositiveTracePreservingMap
62  let B := tensorState ρ m
63  have hA : (tensorState (orbitState p U) n).map Φ.toPositiveTracePreservingMap =
64      A.map (unitaryChannel (tensorUnitary U m)).toPositiveTracePreservingMap := by
65    apply Subtype.ext
66    change Φ.toLinearMap (tensorState (conjugatedState ρ U) n).1 = _
67    rw [← tensorState_conjugated, tensor_unitaryChannel_eq_conjugation,
68      hcov, ← tensor_unitaryChannel_eq_conjugation]
69    rfl
70  have hB : tensorState (orbitState p U) m = B.map (unitaryChannel (tensorUnitary U m)).toPositiveTracePreservingMap :=
71    Subtype.ext (tensorState_conjugated ρ U m).symm
72  have hρ : orbitState p (1 : unitary (Matrix (Fin (d+1)) (Fin (d+1)) ℂ)) = ρ := by
73    have he : (orbitState p (1 : unitary (Matrix (Fin (d+1)) (Fin (d+1)) ℂ))).matrix = ρ.matrix := by
74      simp [orbitState, conjugatedState, ρ, diagonalState]
75    exact state_eq_of_matrix_eq _ _ he
76  change ((tensorState (orbitState p U) n).map Φ.toPositiveTracePreservingMap).rootFidelity
77    (tensorState (orbitState p U) m) = _
78  rw [hA,hB,rootFidelity_unitaryChannel]
79  change _ = ((tensorState (orbitState p 1) n).map Φ.toPositiveTracePreservingMap).rootFidelity
80    (tensorState (orbitState p 1) m)
81  rw [hρ]
