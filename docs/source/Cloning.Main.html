23/-- A strictly ordered positive spectrum, normalized exactly as in the paper. -/
24structure SimpleSpectrum (d : ℕ) where
25  eigenvalue : Fin d → ℝ
26  positive : ∀ i, 0 < eigenvalue i
27  strictAnti : StrictAnti eigenvalue
28  normalized : ∑ i, eigenvalue i = 1
29 
30abbrev PairIndex (d : ℕ) := {ij : Fin d × Fin d // ij.1 < ij.2}
31 
32def SimpleSpectrum.ratio {d : ℕ} (p : SimpleSpectrum d) (ij : PairIndex d) : ℝ :=
33  p.eigenvalue ij.val.2 / p.eigenvalue ij.val.1
34 
35theorem SimpleSpectrum.ratio_pos {d : ℕ} (p : SimpleSpectrum d) (ij : PairIndex d) :
36    0 < p.ratio ij := div_pos (p.positive _) (p.positive _)
37 
38theorem SimpleSpectrum.ratio_lt_one {d : ℕ} (p : SimpleSpectrum d) (ij : PairIndex d) :
39    p.ratio ij < 1 := (div_lt_one (p.positive _)).mpr (p.strictAnti ij.property)
40 
41/-- Exact formula in `eq:main-values`, with each i<j pair counted once. -/
42def orbitalValue {d : ℕ} (g : ℝ) (p : SimpleSpectrum d) : ℝ :=
43  ∏ ij : PairIndex d, Thermal.modeFactor g (p.ratio ij)
44 
45def classicalValue (g : ℝ) (d : ℕ) : ℝ :=
46  Thermal.classicalBase g ^ (((d : ℝ) - 1) / 2)
47 
48def universalValue {d : ℕ} (g : ℝ) (p : SimpleSpectrum d) : ℝ :=
49  classicalValue g d * orbitalValue g p
50 
51/-- PCT's formula expressed through its geometric thermal parameters. -/
52def pctValue {d : ℕ} (g : ℝ) (p : SimpleSpectrum d) : ℝ :=
53  classicalValue (2 * g - 1) d *
54    ∏ ij : PairIndex d, Thermal.fidelity (p.ratio ij) (Thermal.pct g (p.ratio ij))
55 
56theorem orbitalValue_pos {d : ℕ} {g : ℝ} (hg : 1 < g) (p : SimpleSpectrum d) :
57    0 < orbitalValue g p := by
58  apply Finset.prod_pos
59  intro ij _
60  exact Thermal.modeFactor_pos hg (p.ratio_pos ij).le (p.ratio_lt_one ij)
61 
62theorem orbitalValue_le_one {d : ℕ} {g : ℝ} (hg : 1 < g) (p : SimpleSpectrum d) :
63    orbitalValue g p ≤ 1 := by
64  apply Finset.prod_le_one
65  · intro ij _
66    exact (Thermal.modeFactor_pos hg (p.ratio_pos ij).le (p.ratio_lt_one ij)).le
67  · intro ij _
68    exact Thermal.modeFactor_le_one hg (p.ratio_pos ij).le (p.ratio_lt_one ij)
69 
70theorem classicalValue_pos {g : ℝ} (hg : 0 < g) (d : ℕ) :
71    0 < classicalValue g d := Real.rpow_pos_of_pos (Thermal.classicalBase_pos hg) _
72 
73theorem classicalValue_le_one {g : ℝ} (hg : 0 < g) {d : ℕ} (hd : 1 ≤ d) :
74    classicalValue g d ≤ 1 := by
75  apply Real.rpow_le_one (Thermal.classicalBase_pos hg).le
76    (Thermal.classicalBase_le_one hg.le)
77  have hd' : (1 : ℝ) ≤ d := by exact_mod_cast hd
78  linarith
79 
80theorem universalValue_pos {d : ℕ} {g : ℝ} (hg : 1 < g) (p : SimpleSpectrum d) :
81    0 < universalValue g p :=
82  mul_pos (classicalValue_pos (by linarith) d) (orbitalValue_pos hg p)
83 
84theorem universalValue_le_orbital {d : ℕ} {g : ℝ} (hg : 1 < g)
85    (hd : 1 ≤ d) (p : SimpleSpectrum d) : universalValue g p ≤ orbitalValue g p := by
86  exact (mul_le_mul_of_nonneg_right (classicalValue_le_one (by linarith) hd)
87    (orbitalValue_pos hg p).le).trans_eq (one_mul _)

89/-- Strict PCT loss for the explicit limiting formulas, in every dimension
90`d ≥ 2`. Identifying these formulas with channel limits is a separate task. -/
91theorem pctValue_lt_universalValue {d : ℕ} (hd : 2 ≤ d) {g : ℝ}
92    (hg : 1 < g) (p : SimpleSpectrum d) : pctValue g p < universalValue g p := by
93  have hg' : 0 < 2 * g - 1 := by linarith
94  have hd' : (2 : ℝ) ≤ d := by exact_mod_cast hd
95  have he : (0 : ℝ) < ((d : ℝ) - 1) / 2 := by linarith
96  have hc : classicalValue (2 * g - 1) d < classicalValue g d :=
97    Real.rpow_lt_rpow (Thermal.classicalBase_pos hg').le
98      (Thermal.pct_classicalBase_lt hg) he
99  have hprod : (∏ ij : PairIndex d,
100        Thermal.fidelity (p.ratio ij) (Thermal.pct g (p.ratio ij))) ≤ orbitalValue g p := by
101    apply Finset.prod_le_prod
102    · intro ij _
103      have hqa := Thermal.lt_amplified hg (p.ratio_lt_one ij)
104      have hap := Thermal.amplified_lt_pct hg (p.ratio_pos ij) (p.ratio_lt_one ij)
105      exact (Thermal.fidelity_pos (p.ratio_pos ij).le (p.ratio_lt_one ij)
106        ((p.ratio_pos ij).trans (hqa.trans hap)).le
107        (Thermal.pct_lt_one hg (p.ratio_pos ij).le (p.ratio_lt_one ij))).le
108    · intro ij _
109      exact (Thermal.pct_fidelity_lt_modeFactor hg
110        (p.ratio_pos ij) (p.ratio_lt_one ij)).le
111  exact (mul_le_mul_of_nonneg_left hprod (classicalValue_pos hg' d).le).trans_lt
112    (mul_lt_mul_of_pos_right hc (orbitalValue_pos hg p))

114/-- Fully proved convergence of the lifted fidelity from its finite quantitative
115ingredients. The label type depends on sample size, as Young diagrams do.
116This abstract version retains `hquantum` and `hsector` as interfaces.
117`MatrixFidelityAsymptotics` supplies `hquantum` from proved concrete matrix
118deletion. The sector, discarded-mass, and label limits remain asymptotic inputs.
119The classical trimming, block summation, and convergence are proved here. -/
120theorem lifted_fidelity_converges
121    {I : ℕ → Type*} [∀ n, Fintype (I n)]
122    (p q qG f : (n : ℕ) → I n → ℝ)
123    (actual η : ℕ → ℝ) (c l : ℝ)
124    (hp : ∀ n i, 0 ≤ p n i) (hqG : ∀ n i, 0 ≤ qG n i)
125    (hle : ∀ n i, qG n i ≤ q n i)
126    (hps : ∀ n, ∑ i, p n i = 1) (hqs : ∀ n, ∑ i, q n i ≤ 1)
127    (hc0 : 0 ≤ c) (hc1 : c ≤ 1) (hη : ∀ n, 0 ≤ η n)
128    (hsector : ∀ n i, 0 < qG n i → |f n i - c| ≤ η n)
129    (hquantum : ∀ n,
130      |actual n - ∑ i, Real.sqrt (qG n i) * Real.sqrt (p n i) * f n i| ≤
131        Real.sqrt (∑ i, (q n i - qG n i)))
132    (hηlimit : Tendsto η atTop (𝓝 0))
133    (hmass : Tendsto (fun n => ∑ i, (q n i - qG n i)) atTop (𝓝 0))
134    (hlabel : Tendsto (fun n => BlockFidelity.classicalAffinity (q n) (p n))
135      atTop (𝓝 l)) : Tendsto actual atTop (𝓝 (c * l)) := by
136  have herr := fun n => BlockFidelity.lifted_factorization_bound
137    (p n) (q n) (qG n) (f n) (actual n) c (η n)
138    (hp n) (hqG n) (hle n) (hps n) (hqs n) hc0 hc1 (hη n) (hsector n) (hquantum n)
139  have hsqrt : Tendsto (fun n => Real.sqrt (∑ i, (q n i - qG n i))) atTop (𝓝 0) := by
140    simpa using Real.continuous_sqrt.continuousAt.tendsto.comp hmass
141  have herror : Tendsto
142      (fun n => η n + 2 * Real.sqrt (∑ i, (q n i - qG n i))) atTop (𝓝 0) := by
143    simpa using hηlimit.add (hsqrt.const_mul 2)
144  have hdiff : Tendsto
145      (fun n => actual n - c * BlockFidelity.classicalAffinity (q n) (p n))
146      atTop (𝓝 0) := by
147    apply Metric.tendsto_nhds.mpr
148    intro ε hε
149    filter_upwards [herror.eventually_lt_const hε] with n hn
150    simpa only [Real.dist_eq, sub_zero] using (herr n).trans_lt hn
151  have htotal := hdiff.add (hlabel.const_mul c)
152  simpa only [sub_add_cancel, zero_add] using htotal

154/-- A uniform sector-factorization error and a uniform label-affinity error
155combine without increasing the latter, since the orbital factor is at most 1. -/
156theorem uniform_lifted_error {P : Type*}
157    (actual affinity c l : P → ℝ) (η ε ζ : ℝ)
158    (hc0 : ∀ p, 0 ≤ c p) (hc1 : ∀ p, c p ≤ 1)
159    (hζ : 0 ≤ ζ)
160    (hfactor : ∀ p, |actual p - c p * affinity p| ≤ η + 2 * Real.sqrt ε)
161    (hlabel : ∀ p, |affinity p - l p| ≤ ζ) :
162    ∀ p, |actual p - c p * l p| ≤ η + 2 * Real.sqrt ε + ζ := by
163  intro p
164  have hprod : |c p * affinity p - c p * l p| ≤ ζ := by
165    rw [← mul_sub, abs_mul, abs_of_nonneg (hc0 p)]
166    exact (mul_le_mul_of_nonneg_left (hlabel p) (hc0 p)).trans
167      (by simpa using mul_le_mul_of_nonneg_right (hc1 p) hζ)
168  exact (abs_sub_le (actual p) (c p * affinity p) (c p * l p)).trans
169    (add_le_add (hfactor p) hprod)

171/-- Conditional known-spectrum optimum. The payoff is not secretly defined to
172be the answer: it is arbitrary, with candidate lower bounds and two-scale
173converse estimates supplied. No quantum/LAN estimate is assumed as an axiom. -/
174theorem known_spectrum_optimum_of_bounds
175    {d : ℕ} {g : ℝ} (p : SimpleSpectrum d)
176    {I : ℕ → Type*} {Orbit : Type*} [Nonempty Orbit]
177    (payoff : (n : ℕ) → I n → Orbit → ℝ) (candidate : (n : ℕ) → I n)
178    (a b : ℕ → ℝ) (error : ℕ → ℕ → ℝ)
179    (hzero : ∀ n i u, 0 ≤ payoff n i u) (hone : ∀ n i u, payoff n i u ≤ 1)
180    (hcandidate : ∀ n u, a n ≤ payoff n (candidate n) u)
181    (ha : Tendsto a atTop (𝓝 (orbitalValue g p)))
182    (hupper : ∀ L, ∀ᶠ n in atTop, LAN.minimaxValue (payoff n) ≤ b L + error L n)
183    (herror : ∀ L, Tendsto (error L) atTop (𝓝 0))
184    (hb : Tendsto b atTop (𝓝 (orbitalValue g p))) :
185    Tendsto (fun n => LAN.minimaxValue (payoff n)) atTop (𝓝 (orbitalValue g p)) :=
186  LAN.minimax_converges_of_candidates payoff candidate a b error _
187    hzero hone hcandidate ha hupper herror hb

189/-- Conditional unknown-spectrum minimax theorem. Here `P` can be the
190compact spectral set times the orbit. The uniform candidate approximation
191is converted to a bound at the infimum over all parameters. The dense-interior
192extension is proved separately in `LAN.upper_bound_infimum_of_dense`.
193 
194The uniform quantum achievability and the local Gaussian upper estimates
195are still hypotheses, so this is not the unconditional manuscript theorem. -/
196theorem unknown_spectrum_optimum_of_bounds
197    {d : ℕ} {g : ℝ} (hg : 1 < g)
198    {I : ℕ → Type*} {P : Type*} [Nonempty P]
199    (spectrum : P → SimpleSpectrum d)
200    (payoff : (n : ℕ) → I n → P → ℝ) (candidate : (n : ℕ) → I n)
201    (δ b : ℕ → ℝ) (error : ℕ → ℕ → ℝ)
202    (hzero : ∀ n i p, 0 ≤ payoff n i p) (hone : ∀ n i p, payoff n i p ≤ 1)
203    (hcandidate : ∀ n p,
204      |payoff n (candidate n) p - universalValue g (spectrum p)| ≤ δ n)
205    (hδ : Tendsto δ atTop (𝓝 0))
206    (hupper : ∀ L, ∀ᶠ n in atTop, LAN.minimaxValue (payoff n) ≤ b L + error L n)
207    (herror : ∀ L, Tendsto (error L) atTop (𝓝 0))
208    (hb : Tendsto b atTop (𝓝 (⨅ p, universalValue g (spectrum p)))) :
209    Tendsto (fun n => LAN.minimaxValue (payoff n)) atTop
210      (𝓝 (⨅ p, universalValue g (spectrum p))) := by
211  let V := ⨅ p, universalValue g (spectrum p)
212  have hbelow : BddBelow (Set.range (fun p => universalValue g (spectrum p))) := by
213    refine ⟨0, ?_⟩
214    rintro x ⟨p, rfl⟩
215    exact (universalValue_pos hg (spectrum p)).le
216  have hlow : ∀ n p, V - δ n ≤ payoff n (candidate n) p := by
217    intro n p
218    have hv : V ≤ universalValue g (spectrum p) := ciInf_le hbelow p
219    have ha := (abs_le.mp (hcandidate n p)).1
220    linarith
221  apply LAN.minimax_converges_of_candidates payoff candidate (fun n => V - δ n)
222    b error V hzero hone hlow
223  · simpa using tendsto_const_nhds.sub hδ
224  · exact hupper
225  · exact herror
226  · exact hb
