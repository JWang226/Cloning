17theorem isClosed_Lp_ae_mem {E : Type*} [NormedAddCommGroup E]
18    {S : Set E} (hS : IsClosed S) : IsClosed {f : Lp E 1 μ | ∀ᵐ y ∂μ, f y ∈ S} := by
19  apply IsSeqClosed.isClosed
20  intro f g hf hfg
21  obtain ⟨ns, _, he⟩ := (tendstoInMeasure_of_tendsto_Lp hfg).exists_seq_tendsto_ae
22  have hm : ∀ᵐ y ∂μ, ∀ n : ℕ, f (ns n) y ∈ S := ae_all_iff.mpr fun n => hf (ns n)
23  filter_upwards [he, hm] with y hy hm
24  exact hS.mem_of_tendsto hy (Eventually.of_forall hm)

26theorem coeFn_Lp_finset_sum {I E : Type*} [NormedAddCommGroup E]
27    (t : Finset I) (A : I → Lp E 1 μ) :
28    ((∑ i ∈ t, A i : Lp E 1 μ) : Ω → E) =ᵐ[μ] fun y => ∑ i ∈ t, A i y := by
29  classical
30  induction t using Finset.induction_on with
31  | empty => simpa using (Lp.coeFn_zero E 1 μ)
32  | @insert i t hi ih =>
33    simp only [Finset.sum_insert hi]
34    filter_upwards [Lp.coeFn_add (A i) (∑ j ∈ t, A j), ih] with y hy hs
35    simpa only [Pi.add_apply, hs] using hy

40/-- Assemble finitely many L1 coordinates into the actual L1 of the product. -/
41def assembleL1 : (I → Lp E 1 μ) →L[ℂ] Lp (I → E) 1 μ :=
42  ∑ i, ((ContinuousLinearMap.single ℂ (fun _ : I => E) i).compLpL 1 μ).comp
43    (ContinuousLinearMap.proj i)

45theorem assembleL1_ae (A : I → Lp E 1 μ) :
46    assembleL1 A =ᵐ[μ] fun y i => A i y := by
47  have he : ∀ᵐ y ∂μ, ∀ i : I,
48      ((ContinuousLinearMap.single ℂ (fun _ : I => E) i).compLpL 1 μ (A i)) y =
49        Pi.single i (A i y) :=
50    ae_all_iff.mpr fun i => (ContinuousLinearMap.single ℂ (fun _ : I => E) i).coeFn_compLpL (A i)
51  simp only [assembleL1, ContinuousLinearMap.sum_apply, ContinuousLinearMap.comp_apply,
52    ContinuousLinearMap.proj_apply]

54    (fun i => (ContinuousLinearMap.single ℂ (fun _ : I => E) i).compLpL 1 μ (A i)), he] with y hy he

63/-- Positivity of a finite matrix of L1 quantum densities. -/
64def L1BlockPositive (A : Fin n → Fin n → Lp (TraceClass H) 1 μ) : Prop :=
65  ∀ᵐ y ∂μ, BlockPositive (fun i j => A i j y)

67theorem isClosed_flatBlockPositive :
68    IsClosed {A : (Fin n × Fin n) → TraceClass H | BlockPositive (fun i j => A (i, j))} := by
69  simp only [BlockPositive, Set.setOf_forall]
70  apply isClosed_iInter
71  intro v
72  have hc : Continuous (fun A : (Fin n × Fin n) → TraceClass H =>
73      ∑ i, ∑ j, ⟪v i, (A (i, j)).1 (v j)⟫_ℂ) := by
74    apply continuous_finset_sum _
75    intro i _
76    apply continuous_finset_sum _
77    intro j _
78    exact (traceClassMatrixCoefficient (v i) (v j)).continuous.comp (continuous_apply (i, j))
79  exact isClosed_le continuous_const hc

81theorem isClosed_L1BlockPositive :
82    IsClosed {A : Fin n → Fin n → Lp (TraceClass H) 1 μ | L1BlockPositive A} := by
83  have hc : Continuous (fun A : Fin n → Fin n → Lp (TraceClass H) 1 μ =>
84      assembleL1 (fun ij : Fin n × Fin n => A ij.1 ij.2)) :=
85    assembleL1.continuous.comp (continuous_pi fun ij =>
86      (continuous_apply ij.2).comp (continuous_apply ij.1))
87  have hs := (isClosed_Lp_ae_mem (μ := μ) (isClosed_flatBlockPositive (H := H) (n := n))).preimage hc
88  convert hs using 1
89  ext A
90  change (∀ᵐ y ∂μ, BlockPositive (fun i j => A i j y)) ↔
91    (∀ᵐ y ∂μ, BlockPositive (fun i j => assembleL1 (fun ij : Fin n × Fin n => A ij.1 ij.2) y (i, j)))
92  apply eventually_congr
93  filter_upwards [assembleL1_ae (fun ij : Fin n × Fin n => A ij.1 ij.2)] with y hy
94  rw [hy]

96theorem convex_L1BlockPositive :
97    Convex ℝ {A : Fin n → Fin n → Lp (TraceClass H) 1 μ | L1BlockPositive A} := by
98  intro A hA B hB a b ha hb _
99  have he : ∀ᵐ y ∂μ, ∀ i j : Fin n,
100      (a • A i j + b • B i j) y = a • A i j y + b • B i j y := by
101    apply ae_all_iff.mpr
102    intro i
103    apply ae_all_iff.mpr
104    intro j
105    filter_upwards [Lp.coeFn_add (a • A i j) (b • B i j),
106      Lp.coeFn_smul a (A i j), Lp.coeFn_smul b (B i j)] with y hadd hsmulA hsmulB
107    simp only [hadd, Pi.add_apply, hsmulA, hsmulB, Pi.smul_apply]
108  filter_upwards [hA, hB, he] with y hAy hBy hy
109  intro v
110  change 0 ≤ ∑ i, ∑ j, ⟪v i, ((a • A i j + b • B i j) y).1 (v j)⟫_ℂ
111  simp only [hy]
112  change 0 ≤ ∑ i, ∑ j, ⟪v i, (a • (A i j y).1 + b • (B i j y).1) (v j)⟫_ℂ
113  simp only [ContinuousLinearMap.add_apply, ContinuousLinearMap.smul_apply,
114    inner_add_right, inner_smul_right_eq_smul, Finset.sum_add_distrib, ← Finset.smul_sum]
115  exact add_nonneg (smul_nonneg ha (hAy v)) (smul_nonneg hb (hBy v))

119theorem L1BlockPositive_integral {Ξ : Type*} [MeasurableSpace Ξ] {ν : Measure Ξ}
