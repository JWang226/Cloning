19/-- Standard product circular Gaussian, as a literal Lebesgue density. -/
20def priorDensity (z : Fin d → ℂ) : ℝ :=
21  ∏ i, CoherentGaussianMixture.gaussianDensity 1 (z i)

23lemma priorDensity_pos (z : Fin d → ℂ) : 0 < priorDensity z := by
24  apply Finset.prod_pos
25  intro i _
26  exact CoherentGaussianMixture.gaussianDensity_pos (by norm_num) _

28lemma continuous_priorDensity : Continuous (priorDensity (d := d)) := by

31    (CoherentGaussianMixture.continuous_gaussianDensity 1).comp (continuous_apply i))

42/-- The base probability prior; each complex coordinate has variance one. -/
43def gaussianPrior (d : ℕ) : Measure (Fin d → ℂ) :=
44  volume.withDensity (fun z => ENNReal.ofReal (priorDensity z))

46instance gaussianPrior_probability (d : ℕ) : IsProbabilityMeasure (gaussianPrior d) := by
47  constructor
48  rw [gaussianPrior, withDensity_apply _ MeasurableSet.univ, Measure.restrict_univ]
49  rw [← ofReal_integral_eq_lintegral_ofReal integrable_priorDensity
50    (Eventually.of_forall (fun z => (priorDensity_pos z).le)), priorDensity_integral]

53/-- Expansion by a factor `n+1`, giving variance `(n+1)^2` in every mode. -/
54def expandingGaussianPrior (d n : ℕ) : Measure (Fin d → ℂ) :=
55  (gaussianPrior d).map (fun a => ((n : ℝ) + 1) • a)

62/-- L¹ translation continuity is proved from Mathlib's continuous
63measure-preserving action on L¹, for the actual finite-dimensional Haar measure. -/
64theorem integral_abs_translate_sub_tendsto (p : (Fin d → ℂ) → ℝ) (hp : Integrable p) :
65    Tendsto (fun a : Fin d → ℂ => ∫ x, |p (a + x) - p x|) (𝓝 0) (𝓝 0) := by
66  let f : Lp ℝ 1 (volume : Measure (Fin d → ℂ)) := hp.toL1 p
67  let shifts : C((Fin d → ℂ), C((Fin d → ℂ), (Fin d → ℂ))) :=
68    (ContinuousMap.mk (fun z : (Fin d → ℂ) × (Fin d → ℂ) => z.1 + z.2) continuous_add).curry
69  have hm (a : Fin d → ℂ) : MeasurePreserving (shifts a) volume volume :=
70    measurePreserving_add_left volume a
71  have hc : Continuous (fun a => Lp.compMeasurePreserving (shifts a) (hm a) f) :=
72    continuous_const.compMeasurePreservingLp shifts.continuous hm (by norm_num)
73  have he (a : Fin d → ℂ) :
74      ‖Lp.compMeasurePreserving (shifts a) (hm a) f - f‖ = ∫ x, |p (a + x) - p x| := by
75    have hi := (hm a).integrable_comp_of_integrable hp
76    change ‖hi.toL1 (fun x => p (a + x)) - hp.toL1 p‖ = _
77    rw [← Integrable.toL1_sub, L1.norm_of_fun_eq_integral_norm]
78    simp only [Pi.sub_apply, Real.norm_eq_abs]
79  have hz : Lp.compMeasurePreserving (shifts 0) (hm 0) f = f := by
80    apply Lp.ext
81    filter_upwards [Lp.coeFn_compMeasurePreserving f (hm 0)] with x hx
82    simpa [shifts] using hx
83  have h := ((hc.tendsto 0).sub_const f).norm
84  simpa only [hz, sub_self, norm_zero, he] using h

86/-- The translation error for the rescaled prior tends to zero for every
87fixed physical displacement. -/
88theorem prior_translation_error_tendsto (b : Fin d → ℂ) :
89    Tendsto (fun n : ℕ => ∫ x, |priorDensity (-(((n : ℝ) + 1)⁻¹ • b) + x) -
90      priorDensity x|) atTop (𝓝 0) := by
91  have hi : Tendsto (fun n : ℕ => ((n : ℝ) + 1)⁻¹) atTop (𝓝 (0 : ℝ)) := by
92    simpa only [one_div] using (tendsto_one_div_add_atTop_nhds_zero_nat (𝕜 := ℝ))
93  have h := (hi.smul_const b).neg
94  have h' : Tendsto (fun n : ℕ => -(((n : ℝ) + 1)⁻¹ • b)) atTop (𝓝 0) := by
95    simpa only [zero_smul, neg_zero] using h
