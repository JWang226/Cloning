18theorem oscillatorBox_convex (s : ℕ) :
19    Convex ℝ (Cloning.MultimodeCoherent.phaseSpaceUnitBox s) := by
20  intro x hx y hy a c ha hc hac
21  have hbound {u v : ℝ} (hu : |u|≤1) (hv : |v|≤1) : |a*u+c*v|≤1 := by
22    calc
23      |a*u+c*v|≤|a*u|+|c*v| := abs_add_le _ _
24      _ = a*|u|+c*|v| := by rw [abs_mul,abs_mul,abs_of_nonneg ha,abs_of_nonneg hc]
25      _ ≤ 1 := by nlinarith
26  intro j
27  exact ⟨by simpa using hbound (hx j).1 (hy j).1,
28    by simpa using hbound (hx j).2 (hy j).2⟩

30theorem oscillatorBox_zero_interior (s : ℕ) :
31    (0 : Fin s→ℂ) ∈ interior (Cloning.MultimodeCoherent.phaseSpaceUnitBox s) := by
32  apply mem_interior_iff_mem_nhds.mpr
33  apply mem_of_superset (Metric.ball_mem_nhds (0 : Fin s→ℂ) (by norm_num : (0:ℝ)<1))
34  intro z hz j

45theorem mem_scoreUnitBody (p : A→ℝ) (hp : ∀a,0<p a)
46    (b : OrthonormalBasis (Fin (k+1)) ℝ (EuclideanSpace ℝ A))
47    (hb : b 0 = sqrtSpectrum p) (s : ℕ) (ξ : PhaseSpace k s) :
48    ξ∈scoreUnitBody p hp b hb s ↔
49      (∀a,|unwhiten p b ξ.1 a|≤1) ∧ (∀j,|(ξ.2 j).re|≤1 ∧ |(ξ.2 j).im|≤1) := by
50  have he : ξ.1∈(whiteningContinuousEquiv p hp b hb) ''
51      Metric.closedBall (0 : scoreHyperplane A) 1 ↔ ∀a,|unwhiten p b ξ.1 a|≤1 := by
52    let e := whiteningContinuousEquiv p hp b hb
53    change ξ.1 ∈ e.toEquiv '' Metric.closedBall (0 : scoreHyperplane A) 1 ↔ _
54    rw [e.toEquiv.image_eq_preimage_symm]
55    change dist (e.symm ξ.1) 0 ≤ 1 ↔ _
56    rw [dist_zero_right]
57    change ‖unwhiten p b ξ.1‖ ≤ 1 ↔ _
58    rw [pi_norm_le_iff_of_nonneg (by norm_num : (0:ℝ)≤1)]
59    simp only [Real.norm_eq_abs]
60  exact and_congr_left (fun _ => he)

91/-- Exact scaling identifies the nonrectangular whitened body with the
92literal score-hyperplane and oscillator-coordinate box at radius L. -/
93theorem smul_scoreWindow_iff (p : A→ℝ) (hp : ∀a,0<p a)
94    (b : OrthonormalBasis (Fin (k+1)) ℝ (EuclideanSpace ℝ A))
95    (hb : b 0 = sqrtSpectrum p) (s : ℕ) {L : ℝ} (hL : 0<L) (ξ : PhaseSpace k s) :
96    ((∀a,|unwhiten p b (L • ξ).1 a|≤L) ∧
97      (∀j,|((L • ξ).2 j).re|≤L ∧ |((L • ξ).2 j).im|≤L)) ↔
98      ξ∈scoreUnitBody p hp b hb s := by
99  have he (t : ℝ) : |L*t|≤L ↔ |t|≤1 := by
100    rw [abs_mul,abs_of_pos hL]
101    simpa only [mul_one] using
102      (mul_le_mul_iff_right₀ hL : L*|t|≤L*1 ↔ |t|≤1)
103  rw [mem_scoreUnitBody]
104  simp only [Prod.smul_fst,Prod.smul_snd,unwhiten_smul p hp b hb,
105    Pi.smul_apply,Complex.smul_re,Complex.smul_im,smul_eq_mul,he]

107theorem scoreWindow_eq_dilate (p : A→ℝ) (hp : ∀a,0<p a)
108    (b : OrthonormalBasis (Fin (k+1)) ℝ (EuclideanSpace ℝ A))
109    (hb : b 0 = sqrtSpectrum p) (s : ℕ) {L : ℝ} (hL : 0<L) :
110    {ξ : PhaseSpace k s | (∀a,|unwhiten p b ξ.1 a|≤L) ∧
111      (∀j,|(ξ.2 j).re|≤L ∧ |(ξ.2 j).im|≤L)} =
112      (fun ξ => L • ξ) '' scoreUnitBody p hp b hb s := by
113  ext ξ
114  constructor
115  · intro hξ
116    have he : L • (L⁻¹ • ξ)=ξ := smul_inv_smul₀ hL.ne' ξ
117    refine ⟨L⁻¹ • ξ,?_,he⟩
118    apply (smul_scoreWindow_iff p hp b hb s hL (L⁻¹ • ξ)).mp
119    simpa only [he] using hξ
120  · rintro ⟨z,hz,rfl⟩
121    exact (smul_scoreWindow_iff p hp b hb s hL z).mpr hz
