13def restrictCLM (L : ℕ) : Fock →L[ℂ] OccupationSpace L :=
14  LinearMap.mkContinuous
15    { toFun := restrict L
16      map_add' := by intro x y; ext j; rfl
17      map_smul' := by intro c x; ext j; rfl }

33def tailKraus (L j : ℕ) : Fock →L[ℂ] TensorSpace L :=
34  if L < j then (ContinuousLinearMap.toSpanSingleton ℂ (column L 0)).comp (coordinate j) else 0

36@[simp] theorem tailKraus_apply (L j : ℕ) (x : Fock) :

75/-- The retained coherent block, followed by one vacuum-replacement operator
76for each discarded occupation. -/
77def occupationKraus (L : ℕ) : ℕ → Fock →L[ℂ] TensorSpace L
78  | 0 => compression L
79  | j + 1 => tailKraus L j
