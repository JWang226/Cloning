15/-- Any finite list of column heights gives an actual normalized highest tensor. -/
16theorem exists_highestTensor_of_columns {k d : ℕ} (h : Fin k → ℕ)
17    (hd : ∀ j, h j ≤ d) :
18    ∃ Ω : TensorRegister (∑ j, h j) (Fin d), ‖Ω‖ = 1 ∧
19      (∀ a b : Fin d, a < b → collectiveGenerator (∑ j, h j) a b Ω = 0) ∧
20      (∀ a : Fin d, collectiveGenerator (∑ j, h j) a a Ω =
21        ((∑ j : Fin k, if a.val < h j then 1 else 0 : ℕ) : ℂ) • Ω) := by
22  induction k with
23  | zero =>
24    simp only [Fin.sum_univ_zero]
25    refine ⟨registerBasis (Fin 0 → Fin d) Fin.elim0,
26      (registerBasis _).orthonormal.norm_eq_one _, ?_, ?_⟩
27    · intro a b _
28      simp [collectiveGenerator]
29    · intro a
30      simp [collectiveGenerator]
31  | succ k ih =>
32    obtain ⟨Ω, hΩ, hraise, hcartan⟩ := ih (fun j => h j.succ) (fun j => hd j.succ)
33    rw [Fin.sum_univ_succ]
34    refine ⟨tensorJoin (columnHighestTensor (hd 0)) Ω,
35      tensorJoin_norm_one _ _ (columnHighestTensor_norm _) hΩ, ?_, ?_⟩
36    · intro a b hab
37      exact tensorJoin_raising_zero a b _ _ (columnHighestTensor_raising_zero _ a b hab)
38        (hraise a b hab)
39    · intro a
40      have he := tensorJoin_cartan a (columnHighestTensor (hd 0)) Ω
41        (if a.val < h 0 then (1 : ℂ) else 0)
42        ((∑ j : Fin k, if a.val < h j.succ then 1 else 0 : ℕ) : ℂ)
43        (columnHighestTensor_cartan _ a) (hcartan a)
44      simpa only [Fin.sum_univ_succ, Nat.cast_add, Nat.cast_ite, Nat.cast_one,
45        Nat.cast_zero] using he

47/-- Height of the column with zero-based index `j`. -/
48def partitionColumnHeight {d : ℕ} (μ : Fin d → ℕ) (j : ℕ) : ℕ :=
49  (Finset.univ.filter (fun i => j < μ i)).card

51theorem partitionColumnHeight_le {d : ℕ} (μ : Fin d → ℕ) (j : ℕ) :
52    partitionColumnHeight μ j ≤ d := by
53  exact (Finset.card_filter_le ..).trans_eq (Fintype.card_fin d)

55/-- Decreasing rows make each column an initial segment of physical letters. -/
56theorem lt_partitionColumnHeight_iff {d : ℕ} (μ : Fin d → ℕ)
57    (hμ : Antitone μ) (i : Fin d) (j : ℕ) :
58    i.val < partitionColumnHeight μ j ↔ j < μ i := by
59  constructor
60  · intro hi
61    by_contra hj
62    have hs : Finset.univ.filter (fun t => j < μ t) ⊆ Finset.Iio i := by
63      intro t ht
64      have htm : j < μ t := (Finset.mem_filter.mp ht).2
65      apply Finset.mem_Iio.mpr
66      by_contra hti
67      exact (not_lt_of_ge ((hμ (le_of_not_gt hti)).trans (le_of_not_gt hj))) htm
68    have hc := Finset.card_le_card hs
69    rw [Fin.card_Iio] at hc
70    exact Nat.not_lt_of_ge hc hi
71  · intro hj
72    have hs : Finset.Iic i ⊆ Finset.univ.filter (fun t => j < μ t) := by
73      intro t ht
74      exact Finset.mem_filter.mpr ⟨Finset.mem_univ _, hj.trans_le (hμ (Finset.mem_Iic.mp ht))⟩
75    have hc := Finset.card_le_card hs
76    rw [Fin.card_Iic] at hc
77    exact hc

79/-- Count a bounded initial segment of the column indices. -/
80theorem sum_column_indicators {N m : ℕ} (hm : m ≤ N) :
81    (∑ j : Fin N, if j.val < m then 1 else 0 : ℕ) = m := by
82  rw [← Finset.card_filter]
83  by_cases hmN : m < N
84  · let i : Fin N := ⟨m, hmN⟩
85    have he : Finset.univ.filter (fun j : Fin N => j.val < m) = Finset.Iio i := by
86      ext j
87      simp only [Finset.mem_filter, Finset.mem_univ, true_and, Finset.mem_Iio]
88      rfl
89    rw [he, Fin.card_Iio]
90  · have he : m = N := le_antisymm hm (le_of_not_gt hmN)
91    subst m
92    simp

94/-- The number of columns containing row `i` is exactly that row's length. -/
95theorem sum_partitionColumnHeight_indicators {d N : ℕ} (μ : Fin d → ℕ)
96    (hμ : Antitone μ) (hN : ∀ i, μ i ≤ N) (i : Fin d) :
97    (∑ j : Fin N, if i.val < partitionColumnHeight μ j.val then 1 else 0 : ℕ) = μ i := by
98  simp_rw [lt_partitionColumnHeight_iff μ hμ]
99  exact sum_column_indicators (hN i)

101/-- Counting the boxes by columns equals counting them by rows. -/
102theorem sum_partitionColumnHeight {d N : ℕ} (μ : Fin d → ℕ)
103    (hN : ∀ i, μ i ≤ N) :
104    (∑ j : Fin N, partitionColumnHeight μ j.val) = ∑ i, μ i := by
105  simp only [partitionColumnHeight, Finset.card_filter]
106  rw [Finset.sum_comm]
107  apply Finset.sum_congr rfl
108  intro i _
109  exact sum_column_indicators (hN i)

111/-- Every finite partition has an actual unit highest tensor of its exact weight.
112The ambient space is the literal `sum μ`-fold physical tensor register. -/
113theorem exists_partitionHighestTensor {d : ℕ} (μ : Fin d → ℕ) (hμ : Antitone μ) :
114    ∃ Ω : TensorRegister (∑ i, μ i) (Fin d), ‖Ω‖ = 1 ∧
115      (∀ a b : Fin d, a < b → collectiveGenerator (∑ i, μ i) a b Ω = 0) ∧
116      (∀ a : Fin d, collectiveGenerator (∑ i, μ i) a a Ω = (μ a : ℂ) • Ω) := by
117  have hN : ∀ i, μ i ≤ ∑ t, μ t := fun i => Finset.single_le_sum
118    (fun _ _ => Nat.zero_le _) (Finset.mem_univ i)
119  have he := exists_highestTensor_of_columns
120    (fun j : Fin (∑ i, μ i) => partitionColumnHeight μ j.val)
121    (fun j => partitionColumnHeight_le μ j.val)
122  have hsum : (∑ j : Fin (∑ i, μ i), partitionColumnHeight μ j.val) = ∑ i, μ i :=
123    sum_partitionColumnHeight μ hN
124  simp_rw [sum_partitionColumnHeight_indicators μ hμ hN] at he
125  exact hsum ▸ he

127/-- A fixed actual highest tensor for each finite partition. -/
128def partitionHighestTensor {d : ℕ} (μ : Fin d → ℕ) (hμ : Antitone μ) :
129    TensorRegister (∑ i, μ i) (Fin d) :=
130  Classical.choose (exists_partitionHighestTensor μ hμ)

132theorem partitionHighestTensor_norm {d : ℕ} (μ : Fin d → ℕ) (hμ : Antitone μ) :
