3/-!
4# Fixed-cutoff PBW Gram estimates from local commutator bounds
5 
6For a filtered Hilbert space, creators and annihilators shift the cutoff by at
7most one. A local creator bound `M` and local canonical-commutator defect `ε`
8imply an explicit fixed-word Gram error bounded by `C(L,k) M^(2L) ε`.
9 
10In the highest-weight application one step of the filtration can be taken to
11mean `d-1` units of simple-root height. Thus the theorem does not require every
12positive root to have simple-root height one, or any global bounded-operator
13estimate uniform in the highest weight. The construction of the representations
14and proof of their local root estimates remain separate obligations.
15-/

25/-- A creation word of length `k` maps a vacuum in cutoff zero into cutoff `k`. -/
26theorem word_mem_cutoff (F : ι → H →L[ℂ] H) (K : ℕ → Submodule ℂ H)
27    (Ω : H) (hΩ : Ω ∈ K 0)
28    (hF : ∀ i r x, x ∈ K r → F i x ∈ K (r + 1)) (w : List ι) :
29    word F w Ω ∈ K w.length := by
30  induction w with
31  | nil => exact hΩ
32  | cons i w ih => exact hF i w.length _ ih

34/-- Local operator bounds control all bounded-length creation words. -/
35theorem word_norm_le_cutoff (F : ι → H →L[ℂ] H) (K : ℕ → Submodule ℂ H)
36    (Ω : H) (hΩ : Ω ∈ K 0) (hΩnorm : ‖Ω‖ ≤ 1)
37    (hF : ∀ i r x, x ∈ K r → F i x ∈ K (r + 1))
38    (L : ℕ) (M : ℝ) (hM : 0 ≤ M)
39    (hbound : ∀ i r, r ≤ L → ∀ x ∈ K r, ‖F i x‖ ≤ M * ‖x‖)
40    (w : List ι) (hw : w.length ≤ L) : ‖word F w Ω‖ ≤ M ^ w.length := by
41  induction w with
42  | nil => simpa using hΩnorm
43  | cons i w ih =>
44      have hw' : w.length ≤ L := by simp only [List.length_cons] at hw; omega
45      calc
46        ‖word F (i :: w) Ω‖ ≤ M * ‖word F w Ω‖ :=
47          hbound i w.length hw' _ (word_mem_cutoff F K Ω hΩ hF w)
48        _ ≤ M * M ^ w.length := mul_le_mul_of_nonneg_left (ih hw') hM
49        _ = M ^ (i :: w).length := by simp [pow_succ, mul_comm]

51/-- A commutator defect has cutoff at most two greater than its input. -/
52theorem commutatorDefect_mem_cutoff (F A : ι → H →L[ℂ] H)
53    (K : ℕ → Submodule ℂ H) (hK : Monotone K)
54    (hF : ∀ i r x, x ∈ K r → F i x ∈ K (r + 1))
55    (hA : ∀ i r x, x ∈ K r → A i x ∈ K (r + 1))
56    (i j : ι) (r : ℕ) (x : H) (hx : x ∈ K r) :
57    commutatorDefect F A i j x ∈ K (r + 2) := by
58  unfold commutatorDefect
59  refine (K (r + 2)).sub_mem ((K (r + 2)).sub_mem
60    (hA i (r + 1) _ (hF j r x hx)) (hF j (r + 1) _ (hA i r x hx))) ?_
61  split_ifs
62  · exact hK (by omega) hx
63  · exact (K _).zero_mem

65/-- Every error term from a word of length `k` lies in cutoff `k+1`. -/
66theorem errorTerms_mem_cutoff (F A : ι → H →L[ℂ] H)
67    (K : ℕ → Submodule ℂ H) (hK : Monotone K) (Ω : H) (hΩ : Ω ∈ K 0)
68    (hF : ∀ i r x, x ∈ K r → F i x ∈ K (r + 1))
69    (hA : ∀ i r x, x ∈ K r → A i x ∈ K (r + 1))
70    (i : ι) (w : List ι) (x : H) (hx : x ∈ errorTerms F A Ω i w) :
71    x ∈ K (w.length + 1) := by
72  induction w generalizing x with
73  | nil => simp [errorTerms] at hx
74  | cons j w ih =>
75      simp only [errorTerms, List.mem_cons, List.mem_map] at hx
76      rcases hx with rfl | ⟨y, hy, rfl⟩
77      · exact commutatorDefect_mem_cutoff F A K hK hF hA i j w.length _
78          (word_mem_cutoff F K Ω hΩ hF w)
79      · exact hF j (w.length + 1) y (ih y hy)

81/-- The transported commutator errors are controlled by local cutoff estimates.
82This discharges the error-term premise of `gram_error_le`. -/
83theorem errorTerms_norm_le_cutoff (F A : ι → H →L[ℂ] H)
84    (K : ℕ → Submodule ℂ H) (hK : Monotone K) (Ω : H)
85    (hΩ : Ω ∈ K 0) (hΩnorm : ‖Ω‖ ≤ 1)
86    (hF : ∀ i r x, x ∈ K r → F i x ∈ K (r + 1))
87    (hA : ∀ i r x, x ∈ K r → A i x ∈ K (r + 1))
88    (L : ℕ) (M ε : ℝ) (hM : 1 ≤ M) (hε : 0 ≤ ε)
89    (hbound : ∀ i r, r ≤ L → ∀ x ∈ K r, ‖F i x‖ ≤ M * ‖x‖)
90    (hcomm : ∀ i j r, r ≤ L → ∀ x ∈ K r,
91      ‖commutatorDefect F A i j x‖ ≤ ε * ‖x‖)
92    (i : ι) (w : List ι) (hw : w.length ≤ L)
93    (x : H) (hx : x ∈ errorTerms F A Ω i w) :
94    ‖x‖ ≤ ε * M ^ w.length := by
95  induction w generalizing x with
96  | nil => simp [errorTerms] at hx
97  | cons j w ih =>
98      have hw' : w.length ≤ L := by simp only [List.length_cons] at hw; omega
99      simp only [errorTerms, List.mem_cons, List.mem_map] at hx
100      rcases hx with rfl | ⟨y, hy, rfl⟩
101      · calc
102          _ ≤ ε * ‖word F w Ω‖ := hcomm i j w.length hw' _
103            (word_mem_cutoff F K Ω hΩ hF w)
104          _ ≤ ε * M ^ w.length := mul_le_mul_of_nonneg_left
105            (word_norm_le_cutoff F K Ω hΩ hΩnorm hF L M (by linarith) hbound w hw') hε
106          _ ≤ ε * M ^ (j :: w).length := by
107            apply mul_le_mul_of_nonneg_left _ hε
108            exact pow_le_pow_right₀ hM (by simp)
109      · calc
110          ‖F j y‖ ≤ M * ‖y‖ := hbound j (w.length + 1) hw _
111            (errorTerms_mem_cutoff F A K hK Ω hΩ hF hA i w y hy)
112          _ ≤ M * (ε * M ^ w.length) :=
113            mul_le_mul_of_nonneg_left (ih hw' y hy) (by linarith)
114          _ = ε * M ^ (j :: w).length := by simp [pow_succ]; ring

116/-- General-rank fixed-cutoff PBW Gram estimate. The hypotheses concern only
117local root actions and their commutators; the factorial Gram leading term is
118proved by normal ordering. -/
119theorem gram_error_le_of_cutoff (F A : ι → H →L[ℂ] H)
120    (K : ℕ → Submodule ℂ H) (hK : Monotone K) (Ω : H) (hΩmem : Ω ∈ K 0)
121    (hnorm : ⟪Ω, Ω⟫_ℂ = 1) (hΩ : ∀ i, A i Ω = 0)
122    (hadj : ∀ i x y, ⟪F i x, y⟫_ℂ = ⟪x, A i y⟫_ℂ)
123    (hF : ∀ i r x, x ∈ K r → F i x ∈ K (r + 1))
124    (hA : ∀ i r x, x ∈ K r → A i x ∈ K (r + 1))
125    (L : ℕ) (M ε : ℝ) (hM : 1 ≤ M) (hε : 0 ≤ ε)
126    (hbound : ∀ i r, r ≤ L → ∀ x ∈ K r, ‖F i x‖ ≤ M * ‖x‖)
127    (hcomm : ∀ i j r, r ≤ L → ∀ x ∈ K r,
128      ‖commutatorDefect F A i j x‖ ≤ ε * ‖x‖)
129    (u v : List ι) (hu : u.length ≤ L) (hv : v.length ≤ L) :
130    ‖⟪word F u Ω, word F v Ω⟫_ℂ -
131      (if u.Perm v then (occupationFactorial u : ℂ) else 0)‖ ≤
132      (gramConstant L u.length : ℝ) * M ^ (2 * L) * ε := by
133  have hΩnorm : ‖Ω‖ ≤ 1 := by
134    have hh : ‖Ω‖ ^ 2 = 1 := by
135      rw [@norm_sq_eq_re_inner ℂ, hnorm]
136      rfl
137    nlinarith [norm_nonneg Ω]
138  have hw : ∀ w : List ι, w.length ≤ L → ‖word F w Ω‖ ≤ M ^ L := by
139    intro w hw
140    exact (word_norm_le_cutoff F K Ω hΩmem hΩnorm hF L M
141      (by linarith) hbound w hw).trans (by gcongr; exact hM)
142  have he : ∀ i w, w.length ≤ L →
143      ∀ x ∈ errorTerms F A Ω i w, ‖x‖ ≤ ε * M ^ L := by
144    intro i w hw x hx
145    exact (errorTerms_norm_le_cutoff F A K hK Ω hΩmem hΩnorm hF hA
146      L M ε hM hε hbound hcomm i w hw x hx).trans (by gcongr; exact hM)
147  have hh := gram_error_le F A Ω L (M ^ L) (ε * M ^ L)
148    (by positivity) (by positivity) hnorm hΩ hadj hw he u v hu hv
149  rw [wick_eq_occupationFactorial] at hh
150  simp only [Nat.cast_ite, Nat.cast_zero] at hh
151  convert hh using 1; ring
