12theorem rootKey_lt_of_first_lt (a b : PositiveRoot d) (h : a.val.1 < b.val.1) :
13    rootKey a < rootKey b := by
14  have hb := Nat.mul_le_mul_left d (Nat.succ_le_of_lt h)

17  simp only [rootKey]

26/-- At an adjacent inversion in this order, the negative commutator branch is
27impossible. The remaining merged root has exactly the sum grading. -/
28theorem loweringWord_swap_merge (Ω : TensorRegister n (Fin d))
29    (p w : List (PositiveRoot d)) (a b : PositiveRoot d)
30    (hinv : rootKey b < rootKey a) (h : a.val.1 = b.val.2) :
31    loweringWord Ω (p ++ a :: b :: w) =
32      loweringWord Ω (p ++ b :: a :: w) +
33      loweringWord Ω (p ++ PositiveRoot.merge a b h :: w) := by
34  have he := congrArg (fun T : TensorRegister n (Fin d) →L[ℂ] TensorRegister n (Fin d) =>
35    T (loweringWord Ω w))
36    (collectiveGenerator_commutator (n := n) a.val.2 a.val.1 b.val.2 b.val.1)
37  simp only [ContinuousLinearMap.sub_apply, ContinuousLinearMap.mul_apply,
38    if_pos h, if_neg (inverted_roots_not_reverse_composable a b hinv),
39    ContinuousLinearMap.zero_apply, sub_zero] at he
40  have he' : loweringWord Ω (a :: b :: w) =
41      loweringWord Ω (b :: a :: w) + loweringWord Ω (PositiveRoot.merge a b h :: w) := by
42    simpa only [loweringWord, PositiveRoot.merge] using
43      (sub_eq_iff_eq_add.mp he).trans (add_comm _ _)
44  simpa only [map_add, loweringOperator_apply, ← loweringWord_append] using
45    congrArg (loweringOperator n p) he'

47theorem loweringWord_swap_commute (Ω : TensorRegister n (Fin d))
48    (p w : List (PositiveRoot d)) (a b : PositiveRoot d)
49    (hinv : rootKey b < rootKey a) (h : a.val.1 ≠ b.val.2) :
50    loweringWord Ω (p ++ a :: b :: w) = loweringWord Ω (p ++ b :: a :: w) := by
51  have he := congrArg (fun T : TensorRegister n (Fin d) →L[ℂ] TensorRegister n (Fin d) =>
52    T (loweringWord Ω w))
53    (collectiveGenerator_commutator (n := n) a.val.2 a.val.1 b.val.2 b.val.1)
54  simp only [ContinuousLinearMap.sub_apply, ContinuousLinearMap.mul_apply,
55    if_neg h, if_neg (inverted_roots_not_reverse_composable a b hinv),
56    ContinuousLinearMap.zero_apply, sub_self] at he
57  have he' : loweringWord Ω (a :: b :: w) = loweringWord Ω (b :: a :: w) :=
58    sub_eq_zero.mp he
59  simpa only [loweringOperator_apply, ← loweringWord_append] using
60    congrArg (loweringOperator n p) he'

63theorem lowering_merge_grading (p w : List (PositiveRoot d)) (a b : PositiveRoot d)
64    (h : a.val.1 = b.val.2) :
65    loweringHeight (p ++ PositiveRoot.merge a b h :: w) =
66      loweringHeight (p ++ a :: b :: w) ∧
67    loweringWeight (p ++ PositiveRoot.merge a b h :: w) =
68      loweringWeight (p ++ a :: b :: w) ∧
69    (p ++ PositiveRoot.merge a b h :: w).length < (p ++ a :: b :: w).length := by
70  constructor

79/-- Straightening in a fixed submodule. The well-founded measure is the
80lexicographic pair (word length, number of inversions). Swaps reduce the second
81coordinate; commutator terms reduce the first coordinate. -/
82theorem loweringWord_mem_of_ordered
83    (Ω : TensorRegister n (Fin d)) (S : Submodule ℂ (TensorRegister n (Fin d)))
84    (N H : ℕ) (ν : Fin d → ℤ)
85    (hordered : ∀ v, RootOrdered v → v.length ≤ N → loweringHeight v = H →
86      loweringWeight v = ν → loweringWord Ω v ∈ S)
87    (w : List (PositiveRoot d)) (hlen : w.length ≤ N)
88    (hheight : loweringHeight w = H) (hweight : loweringWeight w = ν) :
89    loweringWord Ω w ∈ S := by
90  have main : ∀ L, ∀ w : List (PositiveRoot d), w.length = L → w.length ≤ N →
91      loweringHeight w = H → loweringWeight w = ν → loweringWord Ω w ∈ S := by
92    intro L
93    induction L using Nat.strong_induction_on with
94    | h L ihL =>
95      have inner : ∀ I, ∀ w : List (PositiveRoot d), w.length = L →
96          rootInversions w = I → w.length ≤ N → loweringHeight w = H →
97          loweringWeight w = ν → loweringWord Ω w ∈ S := by
98        intro I
99        induction I using Nat.strong_induction_on with
100        | h I ihI =>
101          intro w hwL hwI hwN hwH hwν
102          by_cases hs : RootOrdered w
103          · exact hordered w hs hwN hwH hwν
104          obtain ⟨p, a, b, t, rfl, hab⟩ := exists_adjacent_rootInversion w hs
105          have hswaplen : (p ++ b :: a :: t).length = L := by simpa using hwL
106          have hswapI : rootInversions (p ++ b :: a :: t) < I := by
107            have hh := rootInversions_swap p t a b hab
108            omega
109          have hswapH : loweringHeight (p ++ b :: a :: t) = H := by
110            simp only [loweringHeight_append, loweringHeight] at hwH ⊢
111            omega
112          have hswapν : loweringWeight (p ++ b :: a :: t) = ν := by
113            simp only [loweringWeight_append, loweringWeight_cons] at hwν ⊢
114            convert hwν using 1 <;> abel
115          have hswap := ihI _ hswapI _ hswaplen rfl (by simpa using hwN) hswapH hswapν
116          by_cases hc : a.val.1 = b.val.2
117          · rw [loweringWord_swap_merge Ω p t a b hab hc]
118            apply S.add_mem hswap
119            obtain ⟨hmH, hmν, hmlen⟩ := lowering_merge_grading p t a b hc
120            exact ihL _ (by omega) _ rfl (by omega) (hmH.trans hwH) (hmν.trans hwν)
121          · rw [loweringWord_swap_commute Ω p t a b hab hc]
122            exact hswap
123      intro w hw hwN hwH hwν
124      exact inner (rootInversions w) w hw rfl hwN hwH hwν
125  exact main w.length w rfl hlen hheight hweight

127/-- The span of ordered words at one exact weight and height, with bounded
128length. All grading data are retained by physical straightening. -/
129def orderedLoweringSpan (Ω : TensorRegister n (Fin d))
130    (N H : ℕ) (ν : Fin d → ℤ) : Submodule ℂ (TensorRegister n (Fin d)) :=
131  Submodule.span ℂ {x | ∃ v : List (PositiveRoot d), RootOrdered v ∧ v.length ≤ N ∧
132    loweringHeight v = H ∧ loweringWeight v = ν ∧ x = loweringWord Ω v}

141/-- The actual cutoff is spanned by ordered words within the same exact root

143theorem cyclicCutoff_eq_ordered_span (Ω : TensorRegister n (Fin d)) (R : ℕ) :
144    cyclicCutoff Ω (R : ℤ) = Submodule.span ℂ
145      {x | ∃ w : List (PositiveRoot d), RootOrdered w ∧ loweringHeight w ≤ R ∧
146        x = loweringWord Ω w} := by
147  apply le_antisymm
