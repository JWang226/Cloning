14/-- The composite classification applied to the literal thermal input.
15The joint idler may be correlated, non-Gaussian and non-diagonal. -/
16theorem thermalOutput_exists_idler (Λ : QuantumChannel (Fock d) (Fock d))
17    (γ : ℝ) (hγ : 1<γ)
18    (hΛ : ∀ a A, Λ.toLinearMap (displacementTraceMap a A)=
19      displacementTraceMap (Real.sqrt γ • a) (Λ.toLinearMap A))
20    (q : Fin d → ℝ) (hq0 : ∀ i,0≤q i) (hq1 : ∀ i,q i<1) :
21    ∃ σ : TraceClass (Fock d), 0≤σ.1 ∧ traceCLM σ=1 ∧
22      ∀ a : Fin d → ℂ,
23        tracePairing (Λ.toLinearMap (vectorMixture (numberBasis d) (productGeometric q)))
24          (displacement a)=
25        tracePairing σ (displacement (fun i =>
26          (Real.sqrt (modeGain γ (q i)-1) : ℂ)*star (a i))) *
27          ∏ i, (Real.exp (-(modeGain γ (q i)*‖a i‖^2)/2) : ℂ) := by
28  let G : Fin d → ℝ := fun i => modeGain γ (q i)
29  have hG (i : Fin d) : 1<G i := modeGain_gt_one hγ (hq0 i) (hq1 i)
30  have hcov : ∀ a A, (compositeChannel Λ q hq0 hq1).toLinearMap (displacementTraceMap a A)=
31      displacementTraceMap (diagonalScale (diagonalGainAmplitude G) a)
32        ((compositeChannel Λ q hq0 hq1).toLinearMap A) :=
33    compositeChannel_covariant Λ γ (by linarith) hΛ q hq0 hq1
34  obtain ⟨σ,hσ,ht,_,hc⟩ := quantumChannel_diagonal_idler_representation
35    (compositeChannel Λ q hq0 hq1) G hG hcov
36  refine ⟨σ,hσ,ht,fun a => ?_⟩
37  have hh := hc (coherentProjector (0 : Fin d → ℂ)) a
38  rw [compositeChannel_vacuum] at hh
39  have hv : tracePairing (coherentProjector (0 : Fin d → ℂ))
40      (displacement (diagonalScale (diagonalGainAmplitude G) a))=
41      ∏ i, (Real.exp (-(G i*‖a i‖^2)/2) : ℂ) := by
42    change tracePairing (rankOneOperator (coherentVector 0) (coherentVector 0)) _=_
43    rw [tracePairing_rankOneOperator, vacuum_characteristic]
44    apply Finset.prod_congr rfl
45    intro i _
46    congr 2
47    simp only [diagonalScale, diagonalGainAmplitude, norm_mul, Complex.norm_real,
48      Real.norm_eq_abs, abs_of_nonneg (Real.sqrt_nonneg _), mul_pow,
49      Real.sq_sqrt (le_trans zero_le_one (hG i).le)]
50  rw [hv] at hh
51  exact hh
