60theorem environmentRotate_cross (W : Register B →ₗᵢ[ℂ] Register B)
61    (v w : Register (A × B)) (a c : A) :
62    (∑ b, environmentRotate W v (a, b) * star (environmentRotate W w (c, b))) =
