20/-- Positivity on the finite Hilbert direct sum for bounded operator blocks. -/
21def BlockPositive (A : ι → ι → H →L[ℂ] H) : Prop :=
22  ∀ x : ι → H, 0 ≤ ∑ i, ∑ j, ⟪x i, A i j (x j)⟫_ℂ

24/-- Complete positivity of a bounded-operator map, at every finite ancilla. -/
25def IsCompletelyPositive (Ψ : (H →L[ℂ] H) →ₗ[ℂ] (K →L[ℂ] K)) : Prop :=
26  ∀ n : ℕ, ∀ A : Fin n → Fin n → H →L[ℂ] H,
27    BlockPositive A → BlockPositive (fun i j => Ψ (A i j))

29/-- Operator Gram matrices are positive at arbitrary finite size. -/
30theorem gram_blockPositive (V : ι → H →L[ℂ] K) :
31    BlockPositive (fun i j => (V i).adjoint.comp (V j)) := by
32  intro x
33  have h : (0 : ℂ) ≤ ⟪∑ i, V i (x i), ∑ j, V j (x j)⟫_ℂ := by
34    apply RCLike.nonneg_iff.mpr
35    exact ⟨inner_self_nonneg, inner_self_im _⟩
36  rw [sum_inner] at h
37  simpa only [inner_sum, ContinuousLinearMap.comp_apply,
38    ContinuousLinearMap.adjoint_inner_right] using h

49/-- The scalar kernel dictated by the input and output Weyl cocycles. -/
50def quantumPositiveKernel (f : (Fin d → ℂ) → ℂ) (r : ℝ)
51    (a b : Fin d → ℂ) : ℂ :=
52  displacementPhase (-a) b / displacementPhase (-(r • a)) (r • b) * f (b - a)
53 
54/-- Quantum positive-definiteness at the covariance scaling `r`. This tests
55all finite phase-space configurations and all complex scalar coefficients. -/
56def IsQuantumPositive (f : (Fin d → ℂ) → ℂ) (r : ℝ) : Prop :=
57  ∀ n : ℕ, ∀ b : Fin n → (Fin d → ℂ), ∀ c : Fin n → ℂ,
58    0 ≤ ∑ i, ∑ j, star (c i) * c j * quantumPositiveKernel f r (b i) (b j)

64/-- A matrix coefficient that removes the output Weyl Gram factor exactly. -/
65lemma displaced_gram_coefficient (r : ℝ) (a b : Fin d → ℂ) (u : Fock d) :
66    ⟪displacement (-(r • a)) u,
67      displacement (r • (b - a)) (displacement (-(r • b)) u)⟫_ℂ =
68    (displacementPhase (-(r • a)) (r • b))⁻¹ * ⟪u, u⟫_ℂ := by
69  have hsum : -(r • a) + r • b = r • (b - a) := by
70    rw [smul_sub]
71    abel
72  have h := congrArg (fun T : Fock d →L[ℂ] Fock d => T (displacement (-(r • b)) u))
73    (displacement_comp (-(r • a)) (r • b))
74  simp only [ContinuousLinearMap.comp_apply, displacement_cancel_neg,
75    ContinuousLinearMap.smul_apply, hsum] at h
76  have h' := congrArg (fun v : Fock d =>
77    (displacementPhase (-(r • a)) (r • b))⁻¹ • v) h
78  simp only [smul_smul, inv_mul_cancel₀ (displacementPhase_ne_zero _ _), one_smul] at h'

81/-- Every completely positive Weyl multiplier has the quantum-positive
82kernel determined by the difference between its two Weyl cocycles. -/
83theorem multiplier_isQuantumPositive
84    (Ψ : (Fock d →L[ℂ] Fock d) →ₗ[ℂ] (Fock d →L[ℂ] Fock d))
85    (hCP : Cloning.BoundedOperator.IsCompletelyPositive Ψ)
86    (r : ℝ) (f : (Fin d → ℂ) → ℂ)
87    (hmult : ∀ b, Ψ (displacement b) = f b • displacement (r • b)) :
88    IsQuantumPositive f r := by
89  intro n b c
90  have hgram := Cloning.BoundedOperator.gram_blockPositive (fun i : Fin n => displacement (b i))
91  have h := hCP n _ hgram (fun i => c i •
92    displacement (-(r • b i)) (coherentVector (0 : Fin d → ℂ)))
93  convert h using 1
94  apply Finset.sum_congr rfl
95  intro i _
96  apply Finset.sum_congr rfl
97  intro j _
98  dsimp only
99  rw [displacement_adjoint, displacement_comp]
100  simp only [map_smul, hmult, ContinuousLinearMap.smul_apply, inner_smul_left,
101    inner_smul_right]
102  rw [show -b i + b j = b j - b i by abel, displaced_gram_coefficient]
103  simp only [inner_self_eq_norm_sq_to_K, coherentVector_norm,
104    RCLike.ofReal_eq_complex_ofReal, one_pow,
105    Complex.ofReal_one, mul_one, quantumPositiveKernel, starRingEnd_apply, div_eq_mul_inv]
