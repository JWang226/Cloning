5/-! Finite-mode quantum Bochner reconstruction: normalized continuous Weyl

17/-- Every normalized continuous Weyl-positive function has an actual density
18operator on the standard Fock space, in any finite number of modes. -/
19theorem exists_density_characteristic (g : (Fin d → ℂ) → ℂ)
20    (hg0 : g 0=1) (hgcont : Continuous g) (hgpos : IsWeylCharacteristicPositive g) :
21    ∃ σ : TraceClass (Fock d), 0≤σ.1 ∧ traceCLM σ=1 ∧
22      ∀ a : Fin d → ℂ, tracePairing σ (displacement a)=g a := by
23  letI : Fact (PositiveKernel.IsNormalizedPositive (kernel g)) :=
24    ⟨normalizedPositive_kernel hgpos hg0⟩
25  obtain ⟨σ,hσ,ht,hc⟩ := (regularWeyl g hgcont).exists_density_characteristic
26    (feature g 0) (norm_feature g 0)
27  refine ⟨σ,hσ,ht,fun a => ?_⟩
28  exact (hc a).trans (origin_characteristic g a)

30/-- The reconstructed density operator is unique among all trace-class
31operators with the prescribed characteristic function. -/
32theorem existsUnique_density_characteristic (g : (Fin d → ℂ) → ℂ)
33    (hg0 : g 0=1) (hgcont : Continuous g) (hgpos : IsWeylCharacteristicPositive g) :
34    ∃! σ : TraceClass (Fock d), 0≤σ.1 ∧ traceCLM σ=1 ∧
35      ∀ a : Fin d → ℂ, tracePairing σ (displacement a)=g a := by
36  obtain ⟨σ,hσ,ht,hc⟩ := exists_density_characteristic g hg0 hgcont hgpos
37  refine ⟨σ,⟨hσ,ht,hc⟩,?_⟩
38  intro τ hτ
39  apply characteristic_injective
40  funext a
41  exact (hτ.2.2 a).trans (hc a).symm

43/-- The scalar-gain covariant channel's joint idler now exists as an actual

45diagonal or a product state. -/
46theorem covariantChannel_existsUnique_idler
47    (Φ : QuantumChannel (Fock d) (Fock d)) (r : ℝ) (hr : 1<r)
48    (hΦ : ∀ a T, Φ.toLinearMap (displacementTraceMap a T)=
49      displacementTraceMap (r • a) (Φ.toLinearMap T)) :

51      ∀ a, tracePairing σ (displacement a)=amplifierIdlerFunction (weylMultiplier Φ.heisenberg r) r a := by
52  obtain ⟨hg0,hgc,hgp,_,_⟩ := quantumChannel_idler_characteristic_data Φ r hr hΦ
53  exact existsUnique_density_characteristic _ hg0 hgc hgp
