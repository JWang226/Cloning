12/-- Initial-segment inclusion of a column's letters into the physical alphabet. -/
13def columnLetters {h d : ℕ} (hd : h ≤ d) : Fin h ↪ Fin d :=
14  ⟨fun i => ⟨i.val, lt_of_lt_of_le i.isLt hd⟩,

21def columnTensorRaw {h d : ℕ} (hd : h ≤ d) : TensorRegister h (Fin d) :=

27/-- Every physical raising matrix unit annihilates the antisymmetric column. -/
28theorem columnTensorRaw_raising_zero {h d : ℕ} (hd : h ≤ d)
29    (a b : Fin d) (hab : a < b) :
30    collectiveGenerator h a b (columnTensorRaw hd) = 0 := by
31  rw [columnTensorRaw, collectiveGenerator_wedgeTensor]
32  apply Finset.sum_eq_zero
33  intro i _
34  rw [matrixUnit_basis]
35  by_cases hbi : b = columnLetters hd i
36  · rw [if_pos hbi]
37    have ha : a.val < h := by
38      have hb : b.val = i.val := congrArg Fin.val hbi
39      exact lt_trans (show a.val < b.val from hab) (hb ▸ i.isLt)
40    let j : Fin h := ⟨a.val, ha⟩
41    have haj : columnLetters hd j = a := Fin.ext rfl
42    have hij : i ≠ j := by
43      intro he
44      have hba : b = a := hbi.trans (congrArg (columnLetters hd) he |>.trans haj)
45      exact (ne_of_lt hab) hba.symm
46    apply (wedgeTensor h).map_eq_zero_of_eq (i := i) (j := j)
47    · simp only [Function.update_self, Function.update_of_ne hij.symm, haj]
48    · exact hij
49  · rw [if_neg hbi, AlternatingMap.map_update_zero]

51/-- The column contains each letter below its height exactly once. -/
52theorem columnTensorRaw_cartan {h d : ℕ} (hd : h ≤ d) (a : Fin d) :
53    collectiveGenerator h a a (columnTensorRaw hd) =
54      (if a.val < h then (1 : ℂ) else 0) • columnTensorRaw hd := by
55  rw [columnTensorRaw, collectiveGenerator_wedgeTensor]
56  by_cases ha : a.val < h
57  · rw [if_pos ha, one_smul]
58    let i : Fin h := ⟨a.val, ha⟩
59    have hai : a = columnLetters hd i := Fin.ext rfl
60    rw [Finset.sum_eq_single i]
61    · rw [matrixUnit_basis, if_pos hai, hai, Function.update_eq_self]
62    · intro j _ hji
63      have haj : a ≠ columnLetters hd j := by
64        intro he
65        exact hji ((columnLetters hd).injective (he.symm.trans hai))
66      rw [matrixUnit_basis, if_neg haj, AlternatingMap.map_update_zero]
67    · simp
68  · rw [if_neg ha, zero_smul]
69    apply Finset.sum_eq_zero
70    intro i _
71    have hai : a ≠ columnLetters hd i := by
72      intro he
73      exact ha ((congrArg Fin.val he) ▸ i.isLt)
74    rw [matrixUnit_basis, if_neg hai, AlternatingMap.map_update_zero]
