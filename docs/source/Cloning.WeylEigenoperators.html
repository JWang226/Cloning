16/-- Character appearing when two actual displacement operators are commuted. -/
17def weylCharacter (a b : Fin d → ℂ) : ℂ :=
18  displacementPhase a b / displacementPhase b a

20lemma displacementPhase_ne_zero (a b : Fin d → ℂ) : displacementPhase a b ≠ 0 := by
21  intro h
22  have hn := displacementPhase_norm a b
23  rw [h, norm_zero] at hn
24  norm_num at hn

26/-- The exact Weyl commutation character for bounded operators. -/
27theorem displacement_commutation (a b : Fin d → ℂ) :
28    (displacement a).comp (displacement b) =
29      weylCharacter a b • (displacement b).comp (displacement a) := by
30  rw [displacement_comp, displacement_comp, smul_smul, weylCharacter,
31    div_mul_cancel₀ _ (displacementPhase_ne_zero b a), add_comm b a]

33/-- An operator transforming with the same commutation character as `D(b)`
34becomes an operator in the ordinary commutant after multiplying by `D(-b)`. -/
35lemma eigenoperator_mul_inverse_commutes (T : Fock d →L[ℂ] Fock d) (b : Fin d → ℂ)
36    (hT : ∀ a : Fin d → ℂ, (displacement a).comp T =
37      weylCharacter a b • T.comp (displacement a)) :
38    ∀ a : Fin d → ℂ, (T.comp (displacement (-b))).comp (displacement a) =
39      (displacement a).comp (T.comp (displacement (-b))) := by
40  intro a
41  apply ContinuousLinearMap.ext
42  intro v
43  have hW := congrArg (fun L : Fock d →L[ℂ] Fock d ↦ L (displacement (-b) v))
44    (displacement_commutation a b)
45  simp only [ContinuousLinearMap.comp_apply, ContinuousLinearMap.smul_apply,
46    displacement_cancel_neg] at hW
47  have hW' := congrArg (displacement (-b)) hW
48  simp only [map_smul, displacement_neg_cancel] at hW'
49  have h := congrArg (fun L : Fock d →L[ℂ] Fock d ↦ L (displacement (-b) v)) (hT a)
50  simp only [ContinuousLinearMap.comp_apply, ContinuousLinearMap.smul_apply] at h
51  change T (displacement (-b) (displacement a v)) = displacement a (T (displacement (-b) v))
52  rw [hW', map_smul, ← h]

54/-- Every bounded eigenoperator of the Weyl conjugation action is a scalar
55multiple of the displacement with the matching character. -/
56theorem eigenoperator_eq_scalar_displacement (T : Fock d →L[ℂ] Fock d) (b : Fin d → ℂ)
57    (hT : ∀ a : Fin d → ℂ, (displacement a).comp T =
58      weylCharacter a b • T.comp (displacement a)) :
59    T = ⟪coherentVector 0, T (displacement (-b) (coherentVector 0))⟫_ℂ • displacement b := by
60  have hscalar := displacement_commutant_scalar (T.comp (displacement (-b)))
61    (eigenoperator_mul_inverse_commutes T b hT)
62  have h := congrArg (fun L : Fock d →L[ℂ] Fock d ↦ L.comp (displacement b)) hscalar
63  apply ContinuousLinearMap.ext
64  intro v
65  have hv := congrArg (fun L : Fock d →L[ℂ] Fock d ↦ L v) h
66  simpa only [ContinuousLinearMap.comp_apply, ContinuousLinearMap.smul_apply,
67    ContinuousLinearMap.id_apply, displacement_neg_cancel] using hv

69/-- Existence form of the bounded Weyl eigenoperator classification. -/
70theorem exists_scalar_eigenoperator (T : Fock d →L[ℂ] Fock d) (b : Fin d → ℂ)
71    (hT : ∀ a : Fin d → ℂ, (displacement a).comp T =
72      weylCharacter a b • T.comp (displacement a)) :
73    ∃ c : ℂ, T = c • displacement b :=
74  ⟨_, eigenoperator_eq_scalar_displacement T b hT⟩
