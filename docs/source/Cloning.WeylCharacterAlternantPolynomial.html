69theorem alternant_expansion {d : ℕ} (a : Fin d → ℕ) :
70    alternant a = ∑ σ : Equiv.Perm (Fin d),
71      ((Equiv.Perm.sign σ : ℤ) : ℂ) • monomial (exponentOf (a ∘ σ.symm)) 1 := by

82theorem eulerSquares_alternant {d : ℕ} (a : Fin d → ℕ) :
83    eulerSquares (alternant a) = (∑ i, (a i : ℂ)^2) • alternant a := by
84  apply MvPolynomial.ext
85  intro m
86  rw [coeff_eulerSquares, coeff_smul, smul_eq_mul]
87  by_cases hm : coeff m (alternant a) = 0
88  · simp [hm]
89  have hex : ∃ σ : Equiv.Perm (Fin d), m = exponentOf (a ∘ σ.symm) := by
90    by_contra! hh
91    apply hm
92    rw [alternant_expansion, coeff_sum]
93    apply Finset.sum_eq_zero
94    intro σ hσ
95    simp [coeff_smul, coeff_monomial, Ne.symm (hh σ)]
96  obtain ⟨σ, rfl⟩ := hex
97  simp only [exponentOf_apply, Function.comp_apply]
98  rw [Equiv.sum_comp σ.symm (fun i ↦ (a i : ℂ)^2)]
