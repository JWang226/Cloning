4/-!
5# Exponential concentration and reciprocal Weyl moments for two rows
6 
7The concrete tableau-path law is bounded by `(N+1)` times the fair binary
8path law. An exact generating-function calculation proves a fixed-fraction
9tail bound `(N+1)(9/10)^N`, without any concentration hypothesis.

75/-- A genuine exponential tail bound for the concrete Schur path law. -/
76theorem largeGapProbability_le (N : ℕ) :
77    largeGapProbability N ≤ ((N : ℝ) + 1) * (9 / 10 : ℝ) ^ N := by
78  have hs0 : 0 < Real.sqrt (8 : ℝ) := Real.sqrt_pos.mpr (by norm_num)
79  have hpoint (p : Path N) :
80      (if N < 2 * spin N p then weight N p else 0) ≤
81        (((N : ℝ) + 1) / ((2 : ℝ) ^ N * Real.sqrt 8 ^ N)) * 4 ^ upperRow N p := by
82    by_cases ht : N < 2 * spin N p
83    · rw [if_pos ht]
84      by_cases hp : legal N p
85      · have hratio : (1 : ℝ) ≤ 4 ^ upperRow N p / Real.sqrt 8 ^ N :=
86          (one_le_div (pow_pos hs0 N)).mpr (sqrt_eight_pow_le N p hp ht)
87        calc
88          _ ≤ ((N : ℝ) + 1) / (2 : ℝ) ^ N := weight_le_uniform_factor N p
89          _ ≤ (((N : ℝ) + 1) / (2 : ℝ) ^ N) *
90              (4 ^ upperRow N p / Real.sqrt 8 ^ N) :=
91            le_mul_of_one_le_right (by positivity) hratio
92          _ = _ := by ring
93      · rw [weight_eq_zero_of_not_legal N p hp]
94        positivity
95    · rw [if_neg ht]
96      positivity
97  have hq0 : (0 : ℝ) ≤ 5 / (2 * Real.sqrt 8) := by positivity
98  have hq : (5 : ℝ) / (2 * Real.sqrt 8) ≤ 9 / 10 := by
99    apply (div_le_iff₀ (mul_pos (by norm_num) hs0)).mpr
100    nlinarith [Real.sq_sqrt (show (0 : ℝ) ≤ 8 by norm_num)]
101  calc
102    _ ≤ ∑ p : Path N,
103        (((N : ℝ) + 1) / ((2 : ℝ) ^ N * Real.sqrt 8 ^ N)) * 4 ^ upperRow N p :=
104      Finset.sum_le_sum (fun p _ => hpoint p)
105    _ = ((N : ℝ) + 1) * (5 / (2 * Real.sqrt 8)) ^ N := by
106      rw [← Finset.mul_sum, sum_four_pow_upperRow, div_pow, mul_pow]
107      ring
108    _ ≤ _ := mul_le_mul_of_nonneg_left (pow_le_pow_left₀ hq0 hq N) (by positivity)
