4/-!
5# Symmetric occupation vectors in the computational tensor basis
6 
7A subset of `Fin L` labels the tensor-basis vector whose excited slots are
8exactly that subset. The `j`-excitation symmetric vector is the normalized
9uniform sum of those basis vectors. Its normalization and orthogonality are
10proved from the cardinality of `powersetCard`, rather than supplied as data.
11-/

133/-- The occupation construction has exactly the coefficients of the normalized
134product `( |0⟩ + sqrt(t/L) |1⟩ )^⊗L`. -/
135theorem productTensor_apply (t : ℝ) (ht : 0 ≤ t) (L : ℕ) (S : Finset (Fin L)) :
136    productTensor t ht L S =
137      ((Real.sqrt (t / L) ^ S.card / Real.sqrt (1 + t / L) ^ L : ℝ) : ℂ) := by

160theorem restrict_norm_le (L : ℕ) (v : lp (fun _ : ℕ => ℂ) 2) :
161    ‖restrict L v‖ ≤ ‖v‖ := by
162  have hs : HasSum (fun j : ℕ => ‖v j‖ ^ 2) (‖v‖ ^ 2) := by
163    simpa only [ENNReal.toReal_ofNat, Real.rpow_two] using
164      lp.hasSum_norm (show 0 < (2 : ENNReal).toReal by norm_num) v
165  have heq : ‖restrict L v‖ ^ 2 = ∑ j : Fin (L + 1), ‖v j‖ ^ 2 := by
166    simpa only [ENNReal.toReal_ofNat, Real.rpow_two, tsum_fintype, restrict_apply] using
167      lp.norm_rpow_eq_tsum (p := 2) (by norm_num) (restrict L v)
168  have hle := hs.summable.sum_le_tsum (Finset.range (L + 1)) (fun j hj => sq_nonneg ‖v j‖)
169  rw [hs.tsum_eq, ← Fin.sum_univ_eq_sum_range, ← heq] at hle
170  nlinarith [norm_nonneg (restrict L v), norm_nonneg v]

290/-- Literal tensor-product factorization in the computational basis: every
291slot has the same normalized one-particle amplitudes. -/
292theorem productTensor_apply_eq_tensorPower (t : ℝ) (ht : 0 ≤ t) (L : ℕ)
293    (S : Finset (Fin L)) :
294    productTensor t ht L S = ∏ i : Fin L,
295      (((if i ∈ S then Real.sqrt (t / L) else 1) / Real.sqrt (1 + t / L) : ℝ) : ℂ) := by
296  rw [productTensor_apply, ← Complex.ofReal_prod]
297  congr 1
298  rw [Finset.prod_div_distrib]
299  have hprod : (∏ i : Fin L, if i ∈ S then Real.sqrt (t / L) else 1) =
300      Real.sqrt (t / L) ^ S.card := by
301    rw [Finset.prod_ite]
302    simp [div_pow]
303  rw [hprod]
304  simp
