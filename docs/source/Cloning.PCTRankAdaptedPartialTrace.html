16theorem partialTrace_rectangular (J : Matrix B A ℂ)
17    (X : Matrix (A × E) (A × E) ℂ) :
18    partialTrace ((J ⊗ₖ (1 : Matrix E E ℂ)) * X * (J ⊗ₖ (1 : Matrix E E ℂ))ᴴ) =
19      J * partialTrace X * Jᴴ := by
20  apply Matrix.ext_iff_trace_mul_left.mpr
21  intro T
22  calc
23    _ = Matrix.trace ((T ⊗ₖ (1 : Matrix E E ℂ)) *
24        ((J ⊗ₖ (1 : Matrix E E ℂ)) * X * (J ⊗ₖ (1 : Matrix E E ℂ))ᴴ)) := (trace_tensor_pairing T _).symm
25    _ = Matrix.trace (((J ⊗ₖ (1 : Matrix E E ℂ))ᴴ *
26        (T ⊗ₖ (1 : Matrix E E ℂ)) * (J ⊗ₖ (1 : Matrix E E ℂ))) * X) := by
27      rw [← Matrix.mul_assoc, ← Matrix.mul_assoc, Matrix.trace_mul_cycle]
28      simp only [Matrix.mul_assoc]
29    _ = Matrix.trace (((Jᴴ * T * J) ⊗ₖ (1 : Matrix E E ℂ)) * X) := by
30      rw [Matrix.conjTranspose_kronecker, Matrix.conjTranspose_one,
31        ← Matrix.mul_kronecker_mul, ← Matrix.mul_kronecker_mul]
32      simp only [Matrix.one_mul, Matrix.mul_one]
33    _ = Matrix.trace ((Jᴴ * T * J) * partialTrace X) := trace_tensor_pairing _ X
34    _ = Matrix.trace (T * (J * partialTrace X * Jᴴ)) := by
35      calc
36        _ = Matrix.trace (Jᴴ * (T * (J * partialTrace X))) := by simp only [Matrix.mul_assoc]
37        _ = Matrix.trace ((T * (J * partialTrace X)) * Jᴴ) := Matrix.trace_mul_comm _ _
38        _ = _ := by simp only [Matrix.mul_assoc]

72theorem regroup_tensorMap (L : ℕ) (J : Matrix B A ℂ) :
73    (regroup (A:=B) (B:=E) L).toContinuousLinearMap.comp
74      (tensorMap L (J ⊗ₖ (1 : Matrix E E ℂ))) =
75      (matrixRegister (tensorPower L J ⊗ₖ (1 : Matrix (Fin L → E) (Fin L → E) ℂ))).comp
