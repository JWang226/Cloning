18/-- The Hermitian tangent furnished by the actual partial trace differential. -/
19def tangentDifferential (p : A → ℝ) (Z : Matrix A A ℂ) : Hermitian A :=
20  ⟨partialTraceDifferential p Z, partialTraceDifferential_hermitian p Z⟩

22/-- The exact rational curve determined by normalized purification, centered
23at the base state. The projection only packages its Hermitian codomain. -/
24def tangentCurve (p : A → ℝ) (Z : Matrix A A ℂ) (t : ℝ) : Hermitian A :=
25  hermitianPart ((1 + t ^ 2 * ‖coefficientVector Z‖ ^ 2)⁻¹ •
26    (base p + t • partialTraceDifferential p Z + t ^ 2 • (Z * Zᴴ)) - base p)

31lemma reducedDensityMatrix_hermitian (ψ : Register (A × A)) :
32    (reducedDensityMatrix ψ).IsHermitian := by
33  let M : Matrix A A ℂ := fun a b => ψ (a,b)
34  change (M * Mᴴ)ᴴ = M * Mᴴ
35  simp

37/-- The curve is exactly the actual reduced normalized purification, with no
38change caused by the Hermitian packaging projection. -/
39theorem tangentCurve_coe (p : A → ℝ) (hp : ∀ a, 0 ≤ p a)
40    (Z : Matrix A A ℂ) (t : ℝ) :
41    (tangentCurve p Z t : Matrix A A ℂ) =
42      reducedDensityMatrix (normalizedTangentVector p Z (‖coefficientVector Z‖ ^ 2) t) - base p := by
43  have hh := (reducedDensityMatrix_hermitian
44    (normalizedTangentVector p Z (‖coefficientVector Z‖ ^ 2) t)).sub (base_hermitian p)

47  rw [normalizedTangent_reduced_exact p hp Z _ (sq_nonneg _) t]

50/-- The exact normalized physical curve has the stated first derivative. -/
51theorem tangentCurve_hasDerivAt_zero (p : A → ℝ) (Z : Matrix A A ℂ) :
52    HasDerivAt (tangentCurve p Z) (tangentDifferential p Z) 0 := by
53  have ht2 : HasDerivAt (fun t : ℝ => t ^ 2) 0 0 := by
54    simpa using (hasDerivAt_id (0 : ℝ)).pow 2
55  have hn : HasDerivAt (fun t : ℝ => base p + t • partialTraceDifferential p Z +
56      t ^ 2 • (Z * Zᴴ)) (partialTraceDifferential p Z) 0 := by
57    simpa using (((hasDerivAt_id (0 : ℝ)).smul_const (partialTraceDifferential p Z)).const_add
58      (base p)).add (ht2.smul_const (Z * Zᴴ))
59  have hi : HasDerivAt (fun t : ℝ => (1 + t ^ 2 * ‖coefficientVector Z‖ ^ 2)⁻¹) 0 0 := by
60    simpa using ((ht2.mul_const (‖coefficientVector Z‖ ^ 2)).const_add 1).inv (by norm_num)
61  have hr := (hi.smul hn).sub_const (base p)
62  have hr' : HasDerivAt (fun t : ℝ => (1 + t ^ 2 * ‖coefficientVector Z‖ ^ 2)⁻¹ •
63      (base p + t • partialTraceDifferential p Z + t ^ 2 • (Z * Zᴴ)) - base p)
64      (partialTraceDifferential p Z) 0 := by simpa using hr
65  have h := (hermitianPart (A := A)).hasFDerivAt.comp_hasDerivAt 0 hr'
66  have he : hermitianPart (partialTraceDifferential p Z) = tangentDifferential p Z := by
67    apply Subtype.ext
68    exact (partialTraceDifferential_hermitian p Z).isSelfAdjoint.coe_selfAdjointPart_apply ℝ
69  simpa only [he, tangentCurve] using h

86/-- The scaled exact coordinates converge. No quadratic bound on the inverse,
87or uniform control over unbounded tangents, is needed. -/
88theorem scaledTangentCoordinates_tendsto (p : A → ℝ) (hp : Function.Injective p)
89    (Z : Matrix A A ℂ) :
90    Tendsto (fun t : ℝ => t⁻¹ • tangentCoordinates p hp Z t) (𝓝[≠] 0)

95/-- For every fixed tangent, inverse-chart equality holds at all sufficiently
96small scales. The matrices here are the physical normalized reduced states. -/
97theorem eventually_tangent_exact_chart (p : A → ℝ) (hp : Function.Injective p)
98    (hp0 : ∀ a, 0 ≤ p a) (Z : Matrix A A ℂ) :
99    ∀ᶠ t : ℝ in 𝓝 0,
100      NormedSpace.exp (generator p (tangentCoordinates p hp Z t)) *
101        (base p + diagonalPart (tangentCoordinates p hp Z t : Matrix A A ℂ)) *
102          NormedSpace.exp (-generator p (tangentCoordinates p hp Z t)) =
103        reducedDensityMatrix (normalizedTangentVector p Z (‖coefficientVector Z‖ ^ 2) t) := by
104  have hc : Tendsto (tangentCurve p Z) (𝓝 0) (𝓝 0) := by
105    simpa only [tangentCurve_zero] using (tangentCurve_hasDerivAt_zero p Z).continuousAt.tendsto
106  filter_upwards [hc.eventually (eventually_chart_localInverse p hp)] with t ht
107  have h := congrArg (fun H : Hermitian A => (H : Matrix A A ℂ)) ht
108  dsimp only at h
109  rw [chart_coe, tangentCurve_coe p hp0] at h
110  change _ - base p = _ - base p at h
111  exact sub_left_injective h
