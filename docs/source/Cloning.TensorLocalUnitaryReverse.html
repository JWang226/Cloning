13theorem adjoint_contraction (V : H →L[ℂ] K) (hV : ∀ x, ‖V x‖ ≤ ‖x‖) (y : K) :
14    ‖V.adjoint y‖ ≤ ‖y‖ := by
15  have hop : ‖V‖ ≤ 1 := ContinuousLinearMap.opNorm_le_bound _ (by norm_num)
16    (fun x => by simpa only [one_mul] using hV x)

18    ‖V.adjoint y‖ ≤ ‖V.adjoint‖ * ‖y‖ := V.adjoint.le_opNorm y
19    _ ≤ 1 * ‖y‖ := by
20      rw [ContinuousLinearMap.adjoint.norm_map]
21      exact mul_le_mul_of_nonneg_right hop (norm_nonneg _)
22    _ = ‖y‖ := one_mul _

26theorem reverse_vector_norm_sq_le (V : H →L[ℂ] K) (hV : ∀ x, ‖V x‖ ≤ ‖x‖)
27    (x : H) (y : K) (hy : ‖y‖ = 1) :
28    ‖V.adjoint y - x‖^2 ≤ ‖x‖^2 - 1 + 2*‖V x-y‖ := by
29  have hnorm := adjoint_contraction V hV y
30  rw [hy] at hnorm
31  have hinner : 1 - (⟪y, V x⟫_ℂ).re ≤ ‖V x-y‖ := by
32    have h := (Complex.re_le_norm ⟪y, y-V x⟫_ℂ).trans (norm_inner_le_norm _ _)
33    have hyy : (⟪y,y⟫_ℂ).re = 1 := by
34      change RCLike.re (⟪y,y⟫_ℂ) = 1
35      rw [← norm_sq_eq_re_inner (𝕜 := ℂ), hy, one_pow]
