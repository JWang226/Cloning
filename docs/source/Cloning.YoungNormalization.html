5/-!
6# Normalization of the literal tableau formula and its actual PMF
7 
8The reversible finite-array Pieri bijection and standard-word branching
9prove that the total tableau weight is `(∑ pᵢ)^N`. Thus a probability spectrum
10defines a genuine Young-label PMF directly from `standardCount * schurPolynomial`.
11Neither normalization nor a pointwise distribution identification is assumed.
12-/

19theorem totalTableauMass_succ_mul (d N : ℕ) (p : Fin d → ℝ) :
20    totalTableauMass d (N + 1) p = totalTableauMass d N p * ∑ i, p i := by
21  rw [totalTableauMass_succ, totalTableauMass_eq_standardSum]
22  calc
23    _ = weightedStandardSum N (fun μ => schurPolynomial N p μ * ∑ i, p i) := by
24      apply weightedStandardSum_congr_on_shapes
25      intro μ hμ hsum
26      have hN : ∀ i, μ i ≤ N := fun i => by
27        simpa only [hsum] using row_le_sum μ i
28      calc
29        _ = ∑ i : Fin d, schurPolynomial (N + 1) p (addBox μ i) := by
30          apply Finset.sum_congr rfl
31          intro i _
32          by_cases hi : Antitone (addBox μ i)
33          · simp only [if_pos hi]
34          · simp only [if_neg hi, schurPolynomial_eq_zero_of_not_antitone (N + 1) p _ hi]
35        _ = _ := schurPolynomial_pieri N p μ hμ hN
36    _ = _ := by
37      unfold weightedStandardSum
38      simp only [ite_mul, zero_mul, Finset.sum_mul]

40/-- Exact normalization polynomial in every alphabet size, including zero

42theorem totalTableauMass_eq_pow (d N : ℕ) (p : Fin d → ℝ) :
43    totalTableauMass d N p = (∑ i, p i) ^ N := by
44  induction N with
45  | zero => simp only [totalTableauMass_zero, pow_zero]
46  | succ N ih => rw [totalTableauMass_succ_mul, ih, pow_succ]

57/-- The actual normalized tableau law, constructed from its proved finite
58weights. -/
59def tableauPMF {d : ℕ} (N : ℕ) (p : Fin d → ℝ)
60    (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) : PMF (Shape d N) :=
61  PMF.ofFintype (fun μ => ENNReal.ofReal (youngWeight N p (fun i => (μ i).val))) (by
62    rw [← ENNReal.ofReal_sum_of_nonneg (fun μ _ => youngWeight_nonneg N p hp _),
63      sum_youngWeight_eq_one N p hs, ENNReal.ofReal_one])

83/-- The proved tail bound now applies to a constructed PMF, with no formula

85theorem tableauPMF_tailProbability_le {d : ℕ} (N : ℕ) (p : Fin d → ℝ)
86    (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) (horder : Antitone p)
87    (ε : ℝ) (hε : 0 ≤ ε) (hε2 : ε ≤ 2) :
88    tailProbability (tableauPMF N p hp hs) p ε ≤
89      ((N : ℝ) + 1) ^ (d * d) * (2 * d * Real.exp (-(N : ℝ) * ε ^ 2 / 4)) :=

92/-- Shrinking-window concentration for the actual PMFs, uniformly for moving

94theorem tableauPMF_tailProbability_tendsto_zero {d : ℕ} (p : ℕ → Fin d → ℝ)
