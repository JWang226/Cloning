14lemma amplifier_phase_split {s t r : ℝ} (h : s + t ^ 2 * r = r) (a v : ℂ) :
15    displacementPhase (-a) ((s : ℂ) * v) *
16      displacementPhase (-((t * r : ℝ) : ℂ) * a) ((t : ℂ) * v) =
17      displacementPhase (-((r : ℂ) * a)) v := by
18  simp only [displacementPhase, ← Complex.exp_add]

20  push_cast
21  simp only [map_neg, map_mul, Complex.conj_ofReal]

35/-- Exact covariance of the constructed countable-Kraus amplifier on all
36complex trace-class inputs. The gain is the physical reciprocal attenuation. -/
37theorem channel_weyl_covariant (q : ℝ) (hq0 : 0 ≤ q) (hq1 : q < 1)
38    (a : ℂ) (A : TraceClass Fock) :
39    (channel q hq0 hq1).toLinearMap (displacementTraceMap a A) =
40      displacementTraceMap ((((Real.sqrt (1 - q))⁻¹ : ℝ) : ℂ) * a)
41        ((channel q hq0 hq1).toLinearMap A) := by
42  let s : ℝ := Real.sqrt (1 - q)
43  let t : ℝ := Real.sqrt q
44  let r : ℝ := s⁻¹
45  have hs : 0 < s := Real.sqrt_pos.mpr (sub_pos.mpr hq1)
46  have hs2 : s ^ 2 = 1 - q := Real.sq_sqrt (sub_nonneg.mpr hq1.le)
47  have ht2 : t ^ 2 = q := Real.sq_sqrt hq0
48  have hsr : (s : ℂ) * (r : ℂ) = 1 := by
49    change (s : ℂ) * ((s⁻¹ : ℝ) : ℂ) = 1
50    rw [← Complex.ofReal_mul, mul_inv_cancel₀ hs.ne', Complex.ofReal_one]
51  have hsplit : s + t ^ 2 * r = r := by
52    dsimp only [r]
53    apply (mul_right_cancel₀ hs.ne')
54    field_simp
55    nlinarith [hs2, ht2]
56  apply Subtype.ext
57  apply continuousLinearMap_ext_coherent
58  intro w
59  apply sub_eq_zero.mp
60  apply coherentVector_total
61  intro v
62  change ⟪coherentVector v,
63    ((channel q hq0 hq1).toLinearMap (displacementTraceMap a A)).1 (coherentVector w) -
64      (displacementTraceMap ((r : ℂ) * a) ((channel q hq0 hq1).toLinearMap A)).1
65        (coherentVector w)⟫_ℂ = 0
66  rw [inner_sub_right, sub_eq_zero]
67  rw [channel_coherent_matrixCoefficient, displacement_coherent_matrixCoefficient,
68    displacement_coherent_matrixCoefficient, channel_coherent_matrixCoefficient]
69  change ((1 - q : ℂ) * ⟪coherentVector ((t : ℂ) * v), coherentVector ((t : ℂ) * w)⟫_ℂ) *
70      (star (displacementPhase (-a) ((s : ℂ) * v)) * displacementPhase (-a) ((s : ℂ) * w) *
71        ⟪coherentVector (-a + (s : ℂ) * v), A.1 (coherentVector (-a + (s : ℂ) * w))⟫_ℂ) =
72    star (displacementPhase (-((r : ℂ) * a)) v) * displacementPhase (-((r : ℂ) * a)) w *
73      (((1 - q : ℂ) *
74        ⟪coherentVector ((t : ℂ) * (-((r : ℂ) * a) + v)),
75          coherentVector ((t : ℂ) * (-((r : ℂ) * a) + w))⟫_ℂ) *
76        ⟪coherentVector ((s : ℂ) * (-((r : ℂ) * a) + v)),
77          A.1 (coherentVector ((s : ℂ) * (-((r : ℂ) * a) + w)))⟫_ℂ)
78  have harg (z : ℂ) : (s : ℂ) * (-((r : ℂ) * a) + z) = -a + (s : ℂ) * z := by
79    calc
80      _ = -((s : ℂ) * (r : ℂ)) * a + (s : ℂ) * z := by ring
81      _ = _ := by rw [hsr]; ring
82  rw [harg, harg]
83  have hnoise := inner_displacedCoherent (-((t * r : ℝ) : ℂ) * a) ((t : ℂ) * v) ((t : ℂ) * w)
84  simp only [displacedCoherent, inner_smul_left, inner_smul_right] at hnoise
85  have hargt (z : ℂ) : -((t * r : ℝ) : ℂ) * a + (t : ℂ) * z =
86      (t : ℂ) * (-((r : ℂ) * a) + z) := by push_cast; ring
87  rw [hargt, hargt] at hnoise
88  rw [← hnoise, ← amplifier_phase_split hsplit a v, ← amplifier_phase_split hsplit a w]
89  simp only [Complex.star_def, map_mul]
90  ring

93theorem channel_weyl_covariant_gain (g : ℝ) (hg : 1 < g) (a : ℂ) (A : TraceClass Fock) :
94    (channel (1 - 1 / g) (gainNoise_nonneg hg) (gainNoise_lt_one hg)).toLinearMap
95      (displacementTraceMap a A) =
96    displacementTraceMap ((Real.sqrt g : ℂ) * a)
97      ((channel (1 - 1 / g) (gainNoise_nonneg hg) (gainNoise_lt_one hg)).toLinearMap A) := by
98  have he : (Real.sqrt (1 - (1 - 1 / g)))⁻¹ = Real.sqrt g := by
99    rw [show 1 - (1 - 1 / g) = g⁻¹ by ring, Real.sqrt_inv, inv_inv]

101    (gainNoise_nonneg hg) (gainNoise_lt_one hg) a A
