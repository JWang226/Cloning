48/-- Exact quantum-limited Weyl multiplier of the physical amplifier. -/
49theorem gainChannel_weylMultiplier (g : ℝ) (hg : 1 < g) (a : Fin d → ℂ) :
50    weylMultiplier (gainChannel g hg).heisenberg (Real.sqrt g) a =
51      ∏ i, Complex.exp (-((((g - 1) / 2) * ‖a i‖ ^ 2 : ℝ) : ℂ)) := by
52  let Φ := gainChannel (d := d) g hg
53  have hcov := gainChannel_covariant (d := d) g hg
54  have hm := (quantumChannel_weyl_multiplier Φ (Real.sqrt g) hcov).1 a
55  have hv := heisenbergDual_pairing Φ.toPositiveTracePreservingMap.toContinuousLinearMap
56    (coherentProjector (0 : Fin d → ℂ)) (displacement a)
57  change tracePairing (coherentProjector (0 : Fin d → ℂ)) (Φ.heisenberg (displacement a)) =
58    tracePairing (Φ.toLinearMap (coherentProjector 0)) (displacement a) at hv
59  rw [hm, map_smul, smul_eq_mul, vacuumProjector_characteristic,
60    show Φ.toLinearMap (coherentProjector 0) =
61      vectorMixture (numberBasis d) (productGeometric (fun _ => 1 - 1 / g))
62      from gainChannel_vacuum g hg,

65  have hsq : Real.sqrt g ^ 2 = g := Real.sq_sqrt (by linarith)

67      ∏ i, Complex.exp (-(((g / 2) * ‖a i‖ ^ 2 : ℝ) : ℂ)) := by
68    apply Finset.prod_congr rfl
69    intro i _

88/-- Actual thermal amplification on the full multimode Fock space. -/
89theorem gainChannel_productThermal (g : ℝ) (hg : 1 < g) {q : Fin d → ℝ}
90    (hq0 : ∀ i, 0 < q i) (hq1 : ∀ i, q i < 1) :
91    (gainChannel g hg).toLinearMap (vectorMixture (numberBasis d) (productGeometric q)) =
92      vectorMixture (numberBasis d) (productGeometric (fun i => Cloning.Thermal.amplified g (q i))) := by
93  apply characteristic_injective
94  funext a
95  dsimp only
96  have hx0 (i : Fin d) : 0 < Cloning.Thermal.amplified g (q i) :=
97    (hq0 i).trans (Cloning.Thermal.lt_amplified hg (hq1 i))
98  have hx1 (i : Fin d) : Cloning.Thermal.amplified g (q i) < 1 :=
99    Cloning.Thermal.amplified_lt_one (by linarith) (hq1 i)
100  rw [productThermal_characteristic hx0 hx1]
101  have h := covariant_thermal_output_characteristic
102    (gainChannel (d := d) g hg).toPositiveTracePreservingMap.toContinuousLinearMap
103    (Real.sqrt g) (gainChannel_covariant g hg) hq0 hq1 a
104  change tracePairing ((gainChannel g hg).toLinearMap
105      (vectorMixture (numberBasis d) (productGeometric q))) (displacement a) =
106    weylMultiplier (gainChannel g hg).heisenberg (Real.sqrt g) a * _ at h
107  rw [h, gainChannel_weylMultiplier, ← Finset.prod_mul_distrib]
108  apply Finset.prod_congr rfl
109  intro i _
110  rw [Real.sq_sqrt (by linarith : 0 ≤ g), ← Complex.exp_add]
111  change Complex.exp _ = Complex.exp (-((geometricFourierWidth (Cloning.Thermal.amplified g (q i)) * ‖a i‖ ^ 2 : ℝ) : ℂ))
112  rw [geometricFourierWidth_amplified (by linarith : g ≠ 0) (hq1 i).ne]
