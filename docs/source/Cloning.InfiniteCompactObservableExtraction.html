16/-- Actual extraction yields simultaneous convergence against every compact

18theorem exists_subsequence_compact_observable_limit [TopologicalSpace.SeparableSpace H]
19    (A : ℕ → TraceClass H) (hA : ∀ n, 0 ≤ (A n).1) (C : ℝ)
20    (htrace : ∀ n, (traceCLM (A n)).re ≤ C) :
21    ∃ T : TraceClass H, 0 ≤ T.1 ∧ (traceCLM T).re ≤ C ∧
22      ∃ φ : ℕ → ℕ, StrictMono φ ∧
23        (∀ x y : H, Tendsto (fun n => ⟪y, (A (φ n)).1 x⟫_ℂ) atTop
24          (𝓝 ⟪y, T.1 x⟫_ℂ)) ∧
25        ∀ K : H →L[ℂ] H, IsCompactOperator K →
26          Tendsto (fun n => tracePairing (A (φ n)) K) atTop (𝓝 (tracePairing T K)) := by
27  obtain ⟨T, hT, hTC, φ, hφ, hlim⟩ :=
28    exists_subsequence_weak_operator_limit A hA C htrace
29  refine ⟨T, hT, hTC, φ, hφ, hlim, ?_⟩
30  intro K hK
31  exact tracePairing_compact_tendsto_of_nonneg atTop (fun n => A (φ n)) T C
32    (fun n => hA (φ n)) (fun n => htrace (φ n)) hlim K hK

34end
