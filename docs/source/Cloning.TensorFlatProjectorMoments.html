24    Finset.sum_nonneg (fun mu _ => mul_nonneg ENNReal.toReal_nonneg (abs_nonneg _)))) ?_ hbound

32theorem tensorFlatYoungPMF_mean_inverse_dimension_error (r k : ℕ) (hr : 0 < r) :
