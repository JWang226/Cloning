4/-!
5# Positive trace-class weak limits with possible trace loss
6 
7A weak-operator limit of positive trace-class operators with uniformly bounded
8trace remains positive and trace class. Only an upper bound on its trace
9survives: mass may escape to infinitely many occupation modes. The limit is
10not assumed trace class. The proof passes each finite diagonal sum through
11the limit and then uses bounded nonnegative partial sums.
12-/

25/-- Positivity is closed under convergence of all quadratic matrix coefficients. -/
26theorem nonneg_of_diagonal_tendsto (A : α → H →L[ℂ] H) (T : H →L[ℂ] H)
27    (hA : ∀ n, 0 ≤ A n)
28    (hlim : ∀ x, Tendsto (fun n ↦ ⟪A n x, x⟫_ℂ) l (𝓝 ⟪T x, x⟫_ℂ)) :
29    0 ≤ T := by
30  apply T.nonneg_iff_isPositive.mpr
31  apply (ContinuousLinearMap.isPositive_iff_complex T).mpr
32  intro x
33  have hnonneg : 0 ≤ ⟪T x, x⟫_ℂ := ge_of_tendsto (hlim x)
34    (Eventually.of_forall fun n ↦ ((A n).nonneg_iff_isPositive.mp (hA n)).inner_nonneg_left x)
35  obtain ⟨hre, him⟩ := Complex.nonneg_iff.mp hnonneg
36  refine ⟨?_, hre⟩
37  apply Complex.ext
38  · simp
39  · simpa using him

41/-- Every finite diagonal sum of a positive trace-class operator is bounded
42by its analytic trace. -/
43theorem finite_diagonal_le_trace {T : H →L[ℂ] H} (hT : 0 ≤ T)
44    (hTC : IsTraceClass T) {w : Set H} (b : HilbertBasis w ℂ H) (s : Finset w) :
45    (∑ i ∈ s, (⟪b i, T (b i)⟫_ℂ).re) ≤ (trace T hTC).re := by
46  rw [trace_re_eq_traceNorm hT hTC, traceNorm_eq_of_hilbertBasis hTC b,
47    CFC.abs_of_nonneg T hT]
48  have hs := summable_inner_abs_of_hilbertBasis hTC b
49  rw [CFC.abs_of_nonneg T hT] at hs
50  exact hs.sum_le_tsum s (fun i _ ↦
51    (T.nonneg_iff_isPositive.mp hT).re_inner_nonneg_right (b i))

53/-- The finite-sum Fatou step needs only diagonal weak convergence. -/
54theorem finite_diagonal_le_of_tendsto (A : α → H →L[ℂ] H) (T : H →L[ℂ] H)
55    (hA : ∀ n, 0 ≤ A n) (hTC : ∀ n, IsTraceClass (A n)) (C : ℝ)
56    (hbound : ∀ n, (trace (A n) (hTC n)).re ≤ C)
57    (hlim : ∀ x, Tendsto (fun n ↦ ⟪A n x, x⟫_ℂ) l (𝓝 ⟪T x, x⟫_ℂ))
58    {w : Set H} (b : HilbertBasis w ℂ H) (s : Finset w) :
59    (∑ i ∈ s, (⟪b i, T (b i)⟫_ℂ).re) ≤ C := by
60  have hd (i : w) : Tendsto (fun n ↦ (⟪b i, A n (b i)⟫_ℂ).re) l
61      (𝓝 (⟪b i, T (b i)⟫_ℂ).re) := by
62    convert (Complex.continuous_re.tendsto _).comp (hlim (b i)) using 1
63    · ext n; exact inner_re_symm (𝕜 := ℂ) (b i) (A n (b i))
64    · congr 1; exact inner_re_symm (𝕜 := ℂ) (b i) (T (b i))
65  have hsum := tendsto_finset_sum s (fun i _ ↦ hd i)
66  exact le_of_tendsto hsum (Eventually.of_forall fun n ↦
67    (finite_diagonal_le_trace (hA n) (hTC n) b s).trans (hbound n))

69/-- A positive bounded operator whose finite diagonal sums are bounded is
70trace class; this is the nonnegative-series form of Fatou's argument. -/
71theorem isTraceClass_of_finite_diagonal_bound {T : H →L[ℂ] H} (hT : 0 ≤ T)
72    {w : Set H} (b : HilbertBasis w ℂ H) (C : ℝ)
73    (hbound : ∀ s : Finset w, (∑ i ∈ s, (⟪b i, T (b i)⟫_ℂ).re) ≤ C) :
74    IsTraceClass T := by
75  refine ⟨w, b, ?_⟩
76  rw [CFC.abs_of_nonneg T hT]
77  exact summable_of_sum_le
78    (fun i ↦ (T.nonneg_iff_isPositive.mp hT).re_inner_nonneg_right (b i)) hbound

80/-- Diagonal weak limits preserve positivity and an upper trace bound,
81without requiring the limit to be trace class in advance. -/
82theorem traceClass_of_diagonal_tendsto (A : α → H →L[ℂ] H) (T : H →L[ℂ] H)
83    (hA : ∀ n, 0 ≤ A n) (hTC : ∀ n, IsTraceClass (A n)) (C : ℝ)
84    (hbound : ∀ n, (trace (A n) (hTC n)).re ≤ C)
85    (hlim : ∀ x, Tendsto (fun n ↦ ⟪A n x, x⟫_ℂ) l (𝓝 ⟪T x, x⟫_ℂ)) :
86    0 ≤ T ∧ ∃ hT : IsTraceClass T, (trace T hT).re ≤ C := by
87  have hT := nonneg_of_diagonal_tendsto A T hA hlim
88  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
89  have hb := finite_diagonal_le_of_tendsto A T hA hTC C hbound hlim b
90  have hclass := isTraceClass_of_finite_diagonal_bound hT b C hb
91  refine ⟨hT, hclass, ?_⟩
92  rw [trace_re_eq_traceNorm hT hclass, traceNorm_eq_of_hilbertBasis hclass b,
93    CFC.abs_of_nonneg T hT]
94  exact Real.tsum_le_of_sum_le
95    (fun i ↦ (T.nonneg_iff_isPositive.mp hT).re_inner_nonneg_right (b i)) hb
