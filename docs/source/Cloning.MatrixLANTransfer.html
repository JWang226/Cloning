29/-- Actual matrix channels are closed under composition. -/
30def compose (T : MatrixChannel b h) (M : MatrixChannel a b) : MatrixChannel a h where
31  toFun := fun X => T.toFun (M.toFun X)
32  map_add := by intro X Y; rw [M.map_add, T.map_add]
33  map_smul := by intro c X; rw [M.map_smul, T.map_smul]
34  trace_preserving := by intro X; rw [T.trace_preserving, M.trace_preserving]
35  completely_positive := by
36    intro k X hX
37    change (amplify T.toFun (amplify M.toFun X)).PosSemidef
38    exact T.completely_positive k _ (M.completely_positive k X hX)

48/-- Swapping the arguments of the trace-norm error does not change it. -/
49theorem traceNorm_sub_symm (A B : Matrix a a ℂ) :
50    matrixTraceNorm (A - B) = matrixTraceNorm (B - A) := by
51  have hneg : ∀ X : Matrix a a ℂ, matrixTraceNorm (-X) = matrixTraceNorm X := by
52    intro X
53    simp [matrixTraceNorm]
54  rw [← neg_sub B A, hneg]

57/-- The matrix version of the manuscript's channel-composition transfer
58estimate. No contractivity, data-processing, or continuity laws are assumed. -/
59theorem fidelity_transfer
60    (S : MatrixChannel g a) (M : MatrixChannel a b) (T : MatrixChannel b h)
61    (ρ : State a) (σ : State b) (Φ : State g) (Ψ : State h) {δ η : ℝ}
62    (hδ : matrixTraceNorm (S.toFun Φ.matrix - ρ.matrix) ≤ δ)
63    (hη : matrixTraceNorm (T.toFun σ.matrix - Ψ.matrix) ≤ η) :
64    fidelity (M.toFun ρ.matrix) σ.matrix ≤
65      fidelity (T.toFun (M.toFun (S.toFun Φ.matrix))) Ψ.matrix +
66        Real.sqrt δ + Real.sqrt η := by
67  have hSΦ := map_positive S Φ.positive
68  have hMρ := map_positive M ρ.positive
69  have hMSΦ := map_positive M hSΦ
70  have hTMρ := map_positive T hMρ
71  have hTMSΦ := map_positive T hMSΦ
72  have hTσ := map_positive T σ.positive
73  have hδ' : matrixTraceNorm (ρ.matrix - S.toFun Φ.matrix) ≤ δ := by
74    rwa [traceNorm_sub_symm]
75  have herror : matrixTraceNorm
76      (T.toFun (M.toFun ρ.matrix) - T.toFun (M.toFun (S.toFun Φ.matrix))) ≤ δ :=
77    (T.traceNorm_contract_sub hMρ hMSΦ).trans
78      ((M.traceNorm_contract_sub ρ.positive hSΦ).trans hδ')
79  have hcont := fidelity_continuity_of_traceNorm_le hTMρ hTMSΦ hTσ Ψ.positive
80    (by rw [T.trace_preserving, σ.trace_one]; norm_num)
81    (by rw [T.trace_preserving, M.trace_preserving, S.trace_preserving, Φ.trace_one]; norm_num)
82    herror hη
83  have hdata := fidelity_data_processing T hMρ σ.positive
84  have hle := (le_abs_self
85    (fidelity (T.toFun (M.toFun ρ.matrix)) (T.toFun σ.matrix) -
86      fidelity (T.toFun (M.toFun (S.toFun Φ.matrix))) Ψ.matrix)).trans hcont

90/-- Uniform comparison of finite matrix experiments gives the actual minimax
91versus Bayes bound. The only analytic premise beyond approximation is
92integrability of the comparison payoff under the chosen probability prior. -/
93theorem minimax_le_bayesValue
94    {I P Θ : Type*} [Nonempty I] [MeasurableSpace Θ]
95    (prior : Measure Θ) [IsProbabilityMeasure prior]
96    (S : MatrixChannel g a) (M : I → MatrixChannel a b) (T : MatrixChannel b h)
97    (ρ : P → State a) (σ : P → State b) (Φ : Θ → State g) (Ψ : Θ → State h)
98    (localChart : Θ → P) (δ η : ℝ)
99    (hδ : ∀ θ, matrixTraceNorm (S.toFun (Φ θ).matrix - (ρ (localChart θ)).matrix) ≤ δ)
100    (hη : ∀ θ, matrixTraceNorm (T.toFun (σ (localChart θ)).matrix - (Ψ θ).matrix) ≤ η)
101    (hintegrable : ∀ N : MatrixChannel g h,
102      Integrable (fun θ => fidelity (N.toFun (Φ θ).matrix) (Ψ θ).matrix) prior) :
103    Cloning.LAN.minimaxValue
104      (fun i p => fidelity ((M i).toFun (ρ p).matrix) (σ p).matrix) ≤
105      Cloning.LAN.bayesValue prior
106        (fun (N : MatrixChannel g h) θ => fidelity (N.toFun (Φ θ).matrix) (Ψ θ).matrix) +
107          (Real.sqrt δ + Real.sqrt η) := by
108  apply Cloning.LAN.minimax_le_bayesValue prior _ _ localChart
109    (fun i => compose T (compose (M i) S)) _
110    (fun _ _ => fidelity_nonneg _ _)
111  · intro i θ
112    simpa only [compose, add_assoc] using fidelity_transfer S (M i) T
113      (ρ (localChart θ)) (σ (localChart θ)) (Φ θ) (Ψ θ) (hδ θ) (hη θ)
114  · exact hintegrable
115  · intro N θ
116    exact fidelity_le_one (map_positive N (Φ θ).positive) (Ψ θ).positive
117      (by rw [N.trace_preserving, (Φ θ).trace_one]; norm_num)
118      (by rw [(Ψ θ).trace_one]; norm_num)
