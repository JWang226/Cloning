41/-- **The trace norm as an `AddGroupNorm`** on the trace-class submodule: nonnegativity, the
42triangle inequality (`traceNorm_add_le`), invariance under negation, and the zero-detection
43property (via `opNorm_le_traceNorm`). -/
44noncomputable def traceClassAddGroupNorm : AddGroupNorm (TraceClass H) where
45  toFun T := traceNorm T.1 (isTraceClass_coe T)
46  map_zero' := traceNorm_zero
47  neg' T := traceNorm_neg (isTraceClass_coe T) (isTraceClass_coe (-T))
48  add_le' T T' := traceNorm_add_le (isTraceClass_coe T) (isTraceClass_coe T')
49    (isTraceClass_coe (T + T'))
50  eq_zero_of_map_eq_zero' T hT0 := by
51    have hop : ‖T.1‖ ≤ traceNorm T.1 (isTraceClass_coe T) :=
52      opNorm_le_traceNorm (isTraceClass_coe T)
53    rw [hT0] at hop
54    have : T.1 = 0 := norm_le_zero_iff.mp hop
55    exact Subtype.ext this

144/-- **The trace-class operators form a Banach space.** The partial sums of an absolutely convergent
145series converge in operator norm, and lower semicontinuity bounds the trace norm of the tails. -/
146noncomputable instance instCompleteSpace : CompleteSpace (TraceClass H) := by
147  apply NormedAddCommGroup.completeSpace_of_summable_imp_tendsto
148  intro u hu
149  set psum : ℕ → ℝ := fun n => ∑ i ∈ Finset.range n, ‖u i‖ with hpartialdef
150  set M : ℝ := ∑' n, ‖u n‖ with hMdef
151  set Sn : ℕ → TraceClass H := fun n => ∑ i ∈ Finset.range n, u i with hSndef
152  set Tn : ℕ → H →L[ℂ] H := fun n => (Sn n).1 with hTndef
153  have hpartial_le_M : ∀ n, psum n ≤ M := fun n => hu.sum_le_tsum _ (fun i _ => norm_nonneg _)
154  have hpartial_tendsto : Tendsto psum atTop (𝓝 M) := hu.hasSum.tendsto_sum_nat
155  set tail : ℕ → ℝ := fun n => M - psum n with htaildef
156  have htail_tendsto : Tendsto tail atTop (𝓝 0) := by
157    have := hpartial_tendsto.const_sub M
158    simpa [htaildef] using this

160    intro n m hnm
161    simp only [hSndef]
162    rw [Finset.sum_Ico_eq_sub _ hnm]
163  have hpsum_split : ∀ n m : ℕ, n ≤ m →
164      (∑ i ∈ Finset.Ico n m, ‖u i‖) = psum m - psum n := by
165    intro n m hnm
166    simp only [hpartialdef]
167    rw [Finset.sum_Ico_eq_sub _ hnm]
168  have htrace_tail_bound : ∀ n m : ℕ, n ≤ m → ‖Sn m - Sn n‖ ≤ tail n := by
169    intro n m hnm
170    rw [hSn_diff n m hnm]
171    calc ‖∑ i ∈ Finset.Ico n m, u i‖ ≤ ∑ i ∈ Finset.Ico n m, ‖u i‖ := norm_sum_le _ _
172      _ = psum m - psum n := hpsum_split n m hnm
173      _ ≤ tail n := by simp only [htaildef]; linarith [hpartial_le_M m]
174  have hSn_diff_coe : ∀ n m : ℕ, ((Sn m - Sn n : TraceClass H)).1 = Tn m - Tn n :=
175    fun n m => Submodule.coe_sub _ _ _

198    intro n
199    rw [dist_eq_norm]
200    have hVtrace : ∀ m, IsTraceClass (Tn (n + m) - Tn n) := by
201      intro m
202      rw [← hSn_diff_coe n (n + m)]
203      exact isTraceClass_coe (Sn (n + m) - Sn n)

216      have h1 : Tendsto (fun m => Tn (n + m)) atTop (𝓝 S) := by
