5/-!
6# Quantitative finite-word PBW normal ordering
7 
8This file isolates the normal-ordering argument in the fixed-height PBW proof.
9It works for any collection of lowering operators and their adjoints, with no
10restriction on rank. The input is a bound on transported commutator defects,
11not a bound on Gram matrices. The output is an explicit finite-word Gram bound
12against the bosonic Wick contraction coefficient. No highest-weight module is
13postulated or constructed here: its local root-operator estimates are the
14separate input needed to instantiate this general theorem.
15-/

26noncomputable def word (F : ι → H →L[ℂ] H) : List ι → H → H
27  | [], x => x
28  | i :: w, x => F i (word F w x)

34/-- One deletion for every occurrence of `i`; multiplicities are retained. -/
35noncomputable def removeOne (i : ι) : List ι → List (List ι)
36  | [] => []
37  | j :: w => (if i = j then [w] else []) ++ (removeOne i w).map (List.cons j)

65/-- Bosonic vacuum contraction number, computable recursively for any two words.
66It counts bijections matching equal root labels. -/
67noncomputable def wick : List ι → List ι → ℕ
68  | [], v => if v = [] then 1 else 0
69  | i :: u, v => ((removeOne i v).map (wick u)).sum

77/-- The normalized canonical commutator defect, applied to a vector. -/
78def commutatorDefect (F A : ι → H →L[ℂ] H) (i j : ι) (x : H) : H :=
79  A i (F j x) - F j (A i x) - if i = j then x else 0

81/-- Terms produced by commuting one annihilator through a creation word.
82Each term is one commutator defect with the preceding creators applied. -/
83noncomputable def errorTerms (F A : ι → H →L[ℂ] H) (Ω : H) (i : ι) : List ι → List H
84  | [] => []
85  | j :: w => commutatorDefect F A i j (word F w Ω) ::
86      (errorTerms F A Ω i w).map (F j)

106/-- Exact normal ordering, with every commutator error retained explicitly. -/
107theorem annihilator_word (F A : ι → H →L[ℂ] H) (Ω : H)
108    (hΩ : ∀ i, A i Ω = 0) (i : ι) (w : List ι) :
109    A i (word F w Ω) =
110      ((removeOne i w).map (fun t => word F t Ω)).sum +
111        (errorTerms F A Ω i w).sum := by
112  induction w with
113  | nil => simp [word, removeOne, errorTerms, hΩ]
114  | cons j w ih =>
115      simp only [word_cons, removeOne, List.map_append, List.sum_append,
116        List.map_map, Function.comp_def, errorTerms, List.sum_cons]
117      have hm :
118          ((removeOne i w).map (fun t => F j (word F t Ω))).sum =
119            F j (((removeOne i w).map (fun t => word F t Ω)).sum) := by
120        rw [map_sum_clm, List.map_map]
121        rfl
122      rw [hm, ← map_sum_clm]
123      unfold commutatorDefect
124      rw [ih, map_add]
125      split_ifs <;> simp only [List.map_cons, List.map_nil, List.sum_cons,
126        List.sum_nil, add_zero, zero_add] <;> abel

128private theorem norm_list_sum_le (l : List H) (ε : ℝ) (_hε : 0 ≤ ε)
129    (hl : ∀ x ∈ l, ‖x‖ ≤ ε) : ‖l.sum‖ ≤ l.length * ε := by
130  induction l with
131  | nil => simp

133      have hx := hl x (by simp)
134      have ht := ih (fun y hy => hl y (by simp [hy]))
135      simpa [Nat.cast_add, Nat.cast_one, add_mul, add_comm] using (norm_add_le x l.sum).trans
136        (add_le_add hx ht)

138/-- Quantitative normal ordering from bounds on transported commutators. -/
139theorem annihilator_word_error_le (F A : ι → H →L[ℂ] H) (Ω : H)
140    (hΩ : ∀ i, A i Ω = 0) (i : ι) (w : List ι) (ε : ℝ) (hε : 0 ≤ ε)
141    (herr : ∀ x ∈ errorTerms F A Ω i w, ‖x‖ ≤ ε) :
142    ‖A i (word F w Ω) -
143      ((removeOne i w).map (fun t => word F t Ω)).sum‖ ≤ w.length * ε := by
144  rw [annihilator_word F A Ω hΩ, add_sub_cancel_left]
145  simpa using norm_list_sum_le (errorTerms F A Ω i w) ε hε herr

147/-- An explicit rank-independent constant for a right word of length at most `L`.
148The recurrence gives `L + L² + ⋯ + L^k`. -/
149noncomputable def gramConstant (L : ℕ) : ℕ → ℕ
150  | 0 => 0
151  | k + 1 => L * (gramConstant L k + 1)

153private theorem complex_list_sum_bound (l : List ι) (f g : ι → ℂ) (C : ℝ)
154    (_hC : 0 ≤ C) (h : ∀ t ∈ l, ‖f t - g t‖ ≤ C) :
155    ‖(l.map f).sum - (l.map g).sum‖ ≤ l.length * C := by
156  induction l with
157  | nil => simp
158  | cons t l ih =>
159      have ht := h t (by simp)
160      have hl := ih (fun s hs => h s (by simp [hs]))
161      simp only [List.map_cons, List.sum_cons, List.length_cons]
162      rw [show f t + (l.map f).sum - (g t + (l.map g).sum) =
163        (f t - g t) + ((l.map f).sum - (l.map g).sum) by ring]
164      simpa [Nat.cast_add, Nat.cast_one, add_mul, add_comm] using
165        (norm_add_le _ _).trans (add_le_add ht hl)

167/-- Fixed-word Gram estimate derived from annihilation of the vacuum, adjointness,
168bounded generated vectors, and small transported commutator defects.
169 
170Only finitely many words, of length at most `L`, occur in the assumptions.
171Consequently local cutoff operator bounds suffice; a uniform global bound on
172creation operators is neither required nor asserted. -/
173theorem gram_error_le (F A : ι → H →L[ℂ] H) (Ω : H) (L : ℕ)
174    (B ε : ℝ) (hB : 0 ≤ B) (hε : 0 ≤ ε)
175    (hnorm : ⟪Ω, Ω⟫_ℂ = 1) (hΩ : ∀ i, A i Ω = 0)
176    (hadj : ∀ i x y, ⟪F i x, y⟫_ℂ = ⟪x, A i y⟫_ℂ)
177    (hword : ∀ u : List ι, u.length ≤ L → ‖word F u Ω‖ ≤ B)
178    (herr : ∀ i w, w.length ≤ L →
179      ∀ x ∈ errorTerms F A Ω i w, ‖x‖ ≤ ε)
180    (u v : List ι) (hu : u.length ≤ L) (hv : v.length ≤ L) :
181    ‖⟪word F u Ω, word F v Ω⟫_ℂ - (wick u v : ℂ)‖ ≤
182      (gramConstant L u.length : ℝ) * B * ε := by
183  induction u generalizing v with
184  | nil =>
185      cases v with
186      | nil => simp only [word_nil, wick_nil_nil, hnorm, Nat.cast_one, sub_self, norm_zero,
187          List.length_nil, gramConstant, Nat.cast_zero, zero_mul, le_refl]
188      | cons j v =>
189          have hz : ⟪Ω, F j (word F v Ω)⟫_ℂ = 0 := by
190            rw [← inner_conj_symm, hadj, hΩ, inner_zero_right, map_zero]
191          simp [word, wick, hz, gramConstant]
192  | cons i u ih =>
193      have hu' : u.length ≤ L := by simp only [List.length_cons] at hu; omega
194      have step := annihilator_word_error_le F A Ω hΩ i v ε hε (herr i v hv)
195      let S : H := ((removeOne i v).map (fun t => word F t Ω)).sum
196      have hfirst :
197          ‖⟪word F u Ω, A i (word F v Ω)⟫_ℂ - ⟪word F u Ω, S⟫_ℂ‖ ≤
198            B * ((v.length : ℝ) * ε) := by
199        rw [← inner_sub_right]
200        exact (norm_inner_le_norm _ _).trans (mul_le_mul
201          (hword u hu') step (norm_nonneg _) hB)
202      have hsum :
203          ‖⟪word F u Ω, S⟫_ℂ - (wick (i :: u) v : ℂ)‖ ≤
204            (v.length : ℝ) * ((gramConstant L u.length : ℝ) * B * ε) := by
205        have hi : ∀ t ∈ removeOne i v,
206            ‖⟪word F u Ω, word F t Ω⟫_ℂ - (wick u t : ℂ)‖ ≤
207              (gramConstant L u.length : ℝ) * B * ε := by
208          intro t ht
209          exact ih t hu' (by have := length_of_mem_removeOne i v t ht; omega)
210        have hc : 0 ≤ (gramConstant L u.length : ℝ) * B * ε := by positivity
211        have hh := complex_list_sum_bound (removeOne i v)
212          (fun t => ⟪word F u Ω, word F t Ω⟫_ℂ)
213          (fun t => (wick u t : ℂ)) _ hc hi
214        have castSum : (wick (i :: u) v : ℂ) =
215            ((removeOne i v).map (fun t => (wick u t : ℂ))).sum := by
216          simp [wick, Function.comp_def]
217        rw [castSum]
218        dsimp [S]
219        rw [inner_list_sum, List.map_map]
220        exact hh.trans (mul_le_mul_of_nonneg_right
221          (by exact_mod_cast removeOne_length_le i v) hc)
222      calc
223        _ = ‖⟪word F u Ω, A i (word F v Ω)⟫_ℂ - (wick (i :: u) v : ℂ)‖ := by
224          rw [word_cons, hadj]
225        _ ≤ ‖⟪word F u Ω, A i (word F v Ω)⟫_ℂ - ⟪word F u Ω, S⟫_ℂ‖ +
226              ‖⟪word F u Ω, S⟫_ℂ - (wick (i :: u) v : ℂ)‖ := by
227          simpa only [dist_eq_norm] using dist_triangle
228            ⟪word F u Ω, A i (word F v Ω)⟫_ℂ ⟪word F u Ω, S⟫_ℂ (wick (i :: u) v : ℂ)
229        _ ≤ B * ((v.length : ℝ) * ε) +
230              (v.length : ℝ) * ((gramConstant L u.length : ℝ) * B * ε) :=
231          add_le_add hfirst hsum
232        _ = (v.length : ℝ) * ((gramConstant L u.length : ℝ) + 1) * B * ε := by ring
233        _ ≤ (L : ℝ) * ((gramConstant L u.length : ℝ) + 1) * B * ε := by
234          gcongr
235        _ = _ := by simp [gramConstant, Nat.cast_mul, Nat.cast_add, Nat.cast_one]

276/-- Wick contractions vanish unless occupations agree, and otherwise equal the
277product of the occupation factorials. This proves the bosonic leading term
278rather than assuming it as part of the normal-ordering estimate. -/
279theorem wick_eq_occupationFactorial (u v : List ι) :
280    wick u v = if u.Perm v then occupationFactorial u else 0 := by
281  induction u generalizing v with
282  | nil => cases v <;> simp [wick, occupationFactorial]
283  | cons i u ih =>
284      have hterm : ∀ t ∈ removeOne i v,
285          wick u t = if (i :: u).Perm v then occupationFactorial u else 0 := by
286        intro t ht
287        rw [ih]
288        have hp := perm_cons_of_mem_removeOne i v t ht
289        have heq : u.Perm t ↔ (i :: u).Perm v := by
290          constructor
291          · intro h; exact (h.cons i).trans hp
292          · intro h; exact (List.perm_cons i).mp (h.trans hp.symm)
293        simp only [heq]
294      rw [wick, sum_map_constant _ _ _ hterm, removeOne_length]
295      by_cases hp : (i :: u).Perm v
296      · rw [if_pos hp, if_pos hp, occupationFactorial]
297        have hc := hp.count_eq i
298        simpa using congrArg (fun n => n * occupationFactorial u) hc.symm
299      · simp [hp]

301/-- For a finite root set this is the usual explicit factorial product. -/
302theorem occupationFactorial_eq_prod [Fintype ι] (u : List ι) :
303    occupationFactorial u = ∏ i, (u.count i).factorial := by
304  induction u with
305  | nil => simp [occupationFactorial]
306  | cons j u ih =>
307      rw [occupationFactorial, ih]
308      have hcount : ∀ i, ((j :: u).count i).factorial =
309          (if i = j then u.count j + 1 else 1) * (u.count i).factorial := by
310        intro i
311        by_cases hij : i = j
312        · subst i; simp [Nat.factorial_succ]
313        · simp [hij, Ne.symm hij]
314      simp_rw [hcount]
315      rw [Finset.prod_mul_distrib]
316      simp

318/-- Explicit leading Gram matrix for any finite collection of roots. -/
319theorem wick_eq_factorial_product [Fintype ι] (u v : List ι) :
