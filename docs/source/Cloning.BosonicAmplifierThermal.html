32/-- The output of a number input has the padded negative-binomial spectrum. -/
33theorem channel_numberProjector_apply_basis (r : ℝ) (hr0 : 0 ≤ r) (hr1 : r < 1)
34    (n t : ℕ) :
35    ((channel r hr0 hr1).toLinearMap (vectorProjector (numberBasis n))).1 (numberBasis t) =
36      (transition r n t : ℂ) • numberBasis t := by
37  classical
38  rw [channel_numberProjector]
39  let L : TraceClass Fock →L[ℂ] Fock :=
40    (ContinuousLinearMap.apply ℂ Fock (numberBasis t)).comp inclusionCLM
41  change L (vectorMixture (fun k => numberBasis (n + k)) (weight r n)) = _
42  rw [vectorMixture, L.map_tsum (summable_weighted_projectors _
43    (fun k => numberBasis.orthonormal.norm_eq_one (n + k)) _ (weight_hasSum hr0 hr1 n).summable)]
44  have hterm (k : ℕ) : L (vectorProjector (numberBasis (n + k))) =
45      if n + k = t then numberBasis t else 0 := by
46    change InnerProductSpace.rankOne ℂ (numberBasis (n + k)) (numberBasis (n + k))
47      (numberBasis t) = _
48    rw [InnerProductSpace.rankOne_apply,
49      orthonormal_iff_ite.mp numberBasis.orthonormal (n + k) t]
50    split_ifs with h
51    · simp [h]
52    · simp
53  simp only [map_smul, hterm]
54  by_cases hnt : n ≤ t
55  · rw [tsum_eq_single (t - n)]
56    · simp only [Nat.add_sub_of_le hnt, transition, if_pos hnt]
57      rfl
58    · intro k hk
59      have hne : n + k ≠ t := by omega
60      simp only [if_neg hne, smul_zero]
61  · have hne (k : ℕ) : n + k ≠ t := by omega
62    simp only [hne, if_false, smul_zero, tsum_zero, transition, if_neg hnt, Complex.ofReal_zero,
63      zero_smul]

65/-- Exact action on every summable diagonal input, proved using continuity of
66the actual channel. Each output coefficient is a finite convolution. -/
67theorem channel_vectorMixture_apply_basis (r : ℝ) (hr0 : 0 ≤ r) (hr1 : r < 1)
68    (p : ℕ → ℝ) (hp : Summable p) (t : ℕ) :
69    ((channel r hr0 hr1).toLinearMap (vectorMixture numberBasis p)).1 (numberBasis t) =
70      ((∑ n ∈ Finset.range (t + 1), p n * seededLaw r n (t - n) : ℝ) : ℂ) • numberBasis t := by
71  classical
72  let Φ := (channel r hr0 hr1).toPositiveTracePreservingMap.toContinuousLinearMap
73  let L : TraceClass Fock →L[ℂ] Fock :=
74    ((ContinuousLinearMap.apply ℂ Fock (numberBasis t)).comp inclusionCLM).comp Φ
75  change L (vectorMixture numberBasis p) = _
76  rw [vectorMixture, L.map_tsum
77    (summable_weighted_projectors _ numberBasis.orthonormal.norm_eq_one p hp)]
78  have hterm (n : ℕ) : L (vectorProjector (numberBasis n)) =
79      (transition r n t : ℂ) • numberBasis t :=
80    channel_numberProjector_apply_basis r hr0 hr1 n t
81  simp only [map_smul, hterm, smul_smul, ← Complex.ofReal_mul]
82  rw [tsum_eq_sum (s := Finset.range (t + 1))]
83  · rw [← Finset.sum_smul, ← Complex.ofReal_sum]
84    congr 2
85    apply Finset.sum_congr rfl
86    intro n hn
87    simp only [transition, if_pos (Nat.lt_succ_iff.mp (Finset.mem_range.mp hn))]
88  · intro n hn
89    have hnt : ¬ n ≤ t := by simpa only [Finset.mem_range, Nat.lt_succ_iff] using hn
90    simp only [transition, if_neg hnt, mul_zero, Complex.ofReal_zero, zero_smul]

101/-- The genuine infinite-dimensional quantum-limited amplifier sends a
102thermal diagonal operator to the thermal operator with the derived parameter. -/
103theorem channel_thermalOperator (r q : ℝ) (hr0 : 0 ≤ r) (hr1 : r < 1)
104    (hq0 : 0 ≤ q) (hq1 : q < 1) :
105    (channel r hr0 hr1).toLinearMap (vectorMixture numberBasis (Thermal.geometric q)) =
106      vectorMixture numberBasis (Thermal.geometric (r + (1 - r) * q)) := by
107  apply Subtype.ext
108  have hbasis (t : ℕ) :
109      ((channel r hr0 hr1).toLinearMap (vectorMixture numberBasis (Thermal.geometric q))).1
110          (numberBasis t) =
111        (vectorMixture numberBasis (Thermal.geometric (r + (1 - r) * q))).1 (numberBasis t) := by
112    rw [channel_vectorMixture_apply_basis r hr0 hr1 _ (Thermal.geometric_hasSum hq0 hq1).summable,
113      InfiniteOccupationStates.vectorMixture_apply_basis _ _
114        (Thermal.geometric_hasSum (thermalOutput_parameter_nonneg hr0 hr1 hq0)
115          (thermalOutput_parameter_lt_one hr1 hq1)).summable]
116    change (thermalOutput r q t : ℂ) • numberBasis t = _
117    rw [thermalOutput_eq]
118  apply ContinuousLinearMap.ext
119  intro v
120  have h₁ := ((channel r hr0 hr1).toLinearMap
121    (vectorMixture numberBasis (Thermal.geometric q))).1.hasSum (numberBasis.hasSum_repr v)
122  have h₂ := (vectorMixture numberBasis (Thermal.geometric (r + (1 - r) * q))).1.hasSum
123    (numberBasis.hasSum_repr v)
124  simp only [map_smul, hbasis] at h₁ h₂
125  exact h₁.unique h₂

134/-- Equality of actual normalized density states after amplification. -/
135theorem channel_geometricState (r q : ℝ) (hr0 : 0 ≤ r) (hr1 : r < 1)
136    (hq0 : 0 ≤ q) (hq1 : q < 1) :
