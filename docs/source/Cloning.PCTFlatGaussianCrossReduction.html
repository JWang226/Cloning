62/-- Every continuous cross integral is transported through an actual
63Hermitian frame with exactly the same reference vector. -/
64theorem integral_flat_cross_eq_real_coordinates
65    (u : OrthonormalBasis (Fin (s+1)) ℂ (Register (A×A)))
66    (hH0 : ∀ a b,star (u 0 (b,a))=u 0 (a,b))
67    (r : ℕ) (hr : 0<r) {δ : ℝ} (hδ : 0<δ) :
68    (∫ z, Real.exp ((r : ℝ)*(Matrix.trace
69      (partialTraceDifferential (fun _ : A => (r : ℝ)⁻¹) (frameTangentMatrix u z.1) *
70       partialTraceDifferential (fun _ : A => (r : ℝ)⁻¹) (frameTangentMatrix u z.2))).re)
71      ∂(gaussianProductMeasure (fun _ : Fin s => δ)).prod
72        (gaussianProductMeasure (fun _ : Fin s => δ)))=
73    ∫ z, Real.exp (4*∑ i,(z.1 i).re*(z.2 i).re)
74      ∂(gaussianProductMeasure (fun _ : Fin s => δ)).prod
75        (gaussianProductMeasure (fun _ : Fin s => δ)) := by
76  obtain ⟨v,hv0,hv⟩ := exists_hermitian_frame (frame_card_sq u) (u 0)
77    (u.orthonormal.norm_eq_one 0) hH0
78  have he := integral_frameTangent_prod_eq u v hv0.symm hδ
79    (flatCrossExponential r) (continuous_flatCrossExponential r)
80  simp only [flatCrossExponential_frameTangent] at he
81  rw [he]
82  apply integral_congr_ae
83  filter_upwards [] with z
84  rw [flatDifferential_rank_trace_cross v hv r hr]
