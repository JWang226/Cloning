11/-!
12# Actual lattice cells and randomized rounding
13 
14The coordinate space here is `ι → ℝ`, where `ι` indexes the first `d-1`
15coordinates of the manuscript's affine lattice.  The last coordinate is
16recovered by `Cloning.Rounding.completeLast`.  Lebesgue measure in these
17coordinates gives the unit cell volume one; the common Jacobian `√d` for
18the induced hyperplane volume cancels from all probabilities.
19 
20The rounding distribution is constructed by pushing forward restricted
21Lebesgue measure, rather than by supplying a row-sum assumption.
22-/

172/-- Discretization by any measurable countable partition contracts the actual
173`L¹` distance.  The proof uses the fiber partition, not an assumed kernel law. -/
174theorem binMass_l1_contraction (μ : Measure X) {r : X → Z} (hr : Measurable r)
175    {f g : X → ℝ} (hf : Integrable f μ) (hg : Integrable g μ) :
176    (∑' z, |binMass μ r f z - binMass μ r g z|) ≤ ∫ x, |f x - g x| ∂μ := by
177  have hsum := hasSum_binMass μ hr (hf.sub hg).abs
178  have hbound (z : Z) :
179      |binMass μ r f z - binMass μ r g z| ≤ binMass μ r (fun x ↦ |f x - g x|) z := by
180    unfold binMass
181    rw [← integral_sub hf.integrableOn hg.integrableOn]
182    simpa only [Real.norm_eq_abs] using
183      norm_integral_le_integral_norm (f := fun x ↦ f x - g x) (μ := μ.restrict (r ⁻¹' {z}))
184  have hs : Summable (fun z ↦ |binMass μ r f z - binMass μ r g z|) :=
185    hsum.summable.of_nonneg_of_le (fun _ ↦ abs_nonneg _) hbound
186  exact (hs.tsum_le_tsum hbound hsum.summable).trans_eq hsum.tsum_eq

188/-- The normalized-density estimate used in the manuscript's local-limit
189argument.  Only the reference density's tail is needed. -/
190theorem density_l1_local_tail (μ : Measure X) {f g : X → ℝ}
191    (hf : Integrable f μ) (hg : Integrable g μ)
192    (hf0 : ∀ x, 0 ≤ f x) (hg0 : ∀ x, 0 ≤ g x)
193    (hmass : (∫ x, f x ∂μ) = ∫ x, g x ∂μ)
194    {s : Set X} (hs : MeasurableSet s) :
195    (∫ x, |f x - g x| ∂μ) ≤
196      2 * (∫ x in s, |f x - g x| ∂μ) + 2 * (∫ x in sᶜ, g x ∂μ) := by
197  have hsplit : (∫ x in s, |f x - g x| ∂μ) + (∫ x in sᶜ, |f x - g x| ∂μ) =
198      ∫ x, |f x - g x| ∂μ := integral_add_compl hs (hf.sub hg).abs
199  have hsplitf := integral_add_compl hs hf
200  have hsplitg := integral_add_compl hs hg
201  have hlocal : (∫ x in s, g x ∂μ) - (∫ x in s, f x ∂μ) ≤
202      ∫ x in s, |f x - g x| ∂μ := by
203    rw [← integral_sub hg.integrableOn hf.integrableOn]
204    apply integral_mono (hg.sub hf).integrableOn (hf.sub hg).abs.integrableOn
205    intro x
206    simpa only [neg_sub] using neg_le_abs (f x - g x)
207  have htail : (∫ x in sᶜ, |f x - g x| ∂μ) ≤
208      (∫ x in sᶜ, f x ∂μ) + ∫ x in sᶜ, g x ∂μ := by
209    rw [← integral_add hf.integrableOn hg.integrableOn]
210    apply integral_mono (hf.sub hg).abs.integrableOn (hf.add hg).integrableOn
211    intro x
212    simpa only [abs_of_nonneg (hf0 x), abs_of_nonneg (hg0 x), Pi.add_apply] using
213      abs_sub (f x) (g x)

216/-- A uniform density error on a finite-volume set gives an explicit global
217`L¹` bound, without a separate concentration assumption for `f`. -/
218theorem density_l1_local_uniform_tail (μ : Measure X) {f g : X → ℝ}
219    (hf : Integrable f μ) (hg : Integrable g μ)
220    (hf0 : ∀ x, 0 ≤ f x) (hg0 : ∀ x, 0 ≤ g x)
221    (hmass : (∫ x, f x ∂μ) = ∫ x, g x ∂μ)
222    {s : Set X} (hs : MeasurableSet s) (hsfin : μ s ≠ ∞)
223    {ε : ℝ} (hlocal : ∀ x ∈ s, |f x - g x| ≤ ε) :
224    (∫ x, |f x - g x| ∂μ) ≤
225      2 * (μ s).toReal * ε + 2 * (∫ x in sᶜ, g x ∂μ) := by

248/-- Uniform local density convergence plus uniform tails of the normalized
249reference family implies uniform global `L¹` convergence.  The parameter type
250can be a compact set of spectra; no compactness or local limit is assumed
251implicitly in this statement. -/
252theorem density_l1_uniform_of_local_and_tails {P : Type*}
253    (μ : Measure X) (f : ℕ → P → X → ℝ) (g : P → X → ℝ)
254    (hf : ∀ n p, Integrable (f n p) μ) (hg : ∀ p, Integrable (g p) μ)
255    (hf0 : ∀ n p x, 0 ≤ f n p x) (hg0 : ∀ p x, 0 ≤ g p x)
256    (hmass : ∀ n p, (∫ x, f n p x ∂μ) = ∫ x, g p x ∂μ)
257    (s : ℕ → Set X) (hs : ∀ k, MeasurableSet (s k)) (hsfin : ∀ k, μ (s k) ≠ ∞)
258    (hlocal : ∀ k δ, 0 < δ → ∀ᶠ n in atTop, ∀ p x, x ∈ s k →
259      |f n p x - g p x| ≤ δ)
260    (htail : ∀ δ, 0 < δ → ∃ k, ∀ p, (∫ x in (s k)ᶜ, g p x ∂μ) ≤ δ) :
261    ∀ ε, 0 < ε → ∀ᶠ n in atTop, ∀ p, (∫ x, |f n p x - g p x| ∂μ) < ε := by
262  intro ε hε
263  obtain ⟨k, hk⟩ := htail (ε / 4) (by positivity)
264  let V := (μ (s k)).toReal
265  have hV : 0 ≤ V := ENNReal.toReal_nonneg
266  let δ := ε / (4 * (V + 1))
267  have hδ : 0 < δ := div_pos hε (by positivity)
268  have hδeq : 4 * (V + 1) * δ = ε := by
269    dsimp [δ]
270    field_simp
271  filter_upwards [hlocal k δ hδ] with n hn
272  intro p
273  have hb := density_l1_local_uniform_tail μ (hf n p) (hg p) (hf0 n p) (hg0 p)
274    (hmass n p) (hs k) (hsfin k) (hn p)
275  change (∫ x, |f n p x - g p x| ∂μ) ≤
276    2 * V * δ + 2 * (∫ x in (s k)ᶜ, g p x ∂μ) at hb
277  have hkp := hk p
278  nlinarith
