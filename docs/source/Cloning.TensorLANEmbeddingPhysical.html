67/-- Compress the complete physical sector onto its actual root-height cutoff,

69def sectorToFock (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ) (R : ℕ)
70    (hgap : ∀ a : PositiveRoot d, 0 < rootGap mu a)
71    (hli : LinearIndependent ℂ (cutoffRawFrame Ω mu R)) : QuantumChannel (cyclicSector Ω) (RootFock d) :=
72  frameForwardChannel (sectorCutoff Ω R) (sectorCutoffBasis Ω mu R hgap hli)
