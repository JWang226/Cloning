18lemma regularizedWeight_strictlyPositive {W : H →L[ℂ] H} (hW : 0 ≤ W)
19    {ε : ℝ} (hε : 0 < ε) : IsStrictlyPositive (regularizedWeight W ε) :=

22lemma trace_regularizedWeight {A W : H →L[ℂ] H} (hTA : IsTraceClass A) (ε : ℝ) :
23    trace (A * regularizedWeight W ε)
24      (isTraceClass_mul_mul (A := 1) (B := regularizedWeight W ε) hTA) =
25      trace (A * W) (isTraceClass_mul_mul (A := 1) (B := W) hTA) +

33lemma fidelity_sq_le_positive_weight {A B W : H →L[ℂ] H}
34    (hA : 0 ≤ A) (hB : 0 ≤ B) (hTA : IsTraceClass A) (hTB : IsTraceClass B)
35    (hW : IsStrictlyPositive W) :

48/-- A uniform bound on regularized inverse moments suffices for the unregularized witness
49bound, even when `W` itself has no bounded inverse. -/
50lemma fidelity_sq_le_of_regularized_inverse_moment {A B W : H →L[ℂ] H}
51    (hA : 0 ≤ A) (hB : 0 ≤ B) (hW : 0 ≤ W)
52    (hTA : IsTraceClass A) (hTB : IsTraceClass B) {M : ℝ} (hM : 0 ≤ M)
53    (hInv : ∀ ε : ℝ, 0 < ε →
54      (trace (B * CFC.rpow (regularizedWeight W ε) (-1))
55        (isTraceClass_mul_mul (A := 1)
56          (B := CFC.rpow (regularizedWeight W ε) (-1)) hTB)).re ≤ M) :
57    fidelity A B hA hB hTA hTB ^ 2 ≤
58      (trace (A * W) (isTraceClass_mul_mul (A := 1) (B := W) hTA)).re * M := by
59  have htrA : 0 ≤ (trace A hTA).re := trace_re_nonneg hA hTA
60  have htrW := trace_re_mul_nonneg_of_nonneg_left_traceClass hA hW hTA
61  apply le_of_forall_pos_le_add
62  intro δ hδ
63  let ε : ℝ := δ / ((trace A hTA).re * M + 1)
64  have hden : 0 < (trace A hTA).re * M + 1 := by positivity
65  have hε : 0 < ε := div_pos hδ hden
66  have hbound := fidelity_sq_le_positive_weight hA hB hTA hTB
67    (regularizedWeight_strictlyPositive hW hε)
68  have ht := congrArg Complex.re (trace_regularizedWeight (W := W) hTA ε)
69  simp only [Complex.add_re, Complex.mul_re, Complex.ofReal_re, Complex.ofReal_im,
70    zero_mul, sub_zero] at ht
71  rw [ht] at hbound
72  have hbound' := hbound.trans (mul_le_mul_of_nonneg_left (hInv ε hε)
73    (add_nonneg htrW (mul_nonneg hε.le htrA)))
74  have hsmall : ε * ((trace A hTA).re * M) ≤ δ := by
75    calc
76      ε * ((trace A hTA).re * M) ≤ ε * ((trace A hTA).re * M + 1) := by
77        gcongr
78        exact le_add_of_nonneg_right zero_le_one
79      _ = δ := div_mul_cancel₀ δ (ne_of_gt hden)
80  nlinarith
