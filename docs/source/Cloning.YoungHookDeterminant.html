4/-!
5# The factorial determinant and its predecessor recurrence
6 
7The reciprocal factorial is extended by zero to negative integers.  Its exact
8one-step recurrence proves the determinant predecessor identity, including
9boundary terms.  This is the algebraic input to the standard-tableau dimension
10formula; it does not assume a hook formula or a representation dimension.
11-/

18def inverseFactorial (z : ℤ) : ℝ :=
19  if 0 ≤ z then ((z.toNat.factorial : ℕ) : ℝ)⁻¹ else 0

21@[simp] theorem inverseFactorial_nat (n : ℕ) :
22    inverseFactorial n = (n.factorial : ℝ)⁻¹ := by
23  simp [inverseFactorial]

25theorem inverseFactorial_of_neg {z : ℤ} (hz : z < 0) :
26    inverseFactorial z = 0 := by
27  simp [inverseFactorial, not_le.mpr hz]

29theorem mul_inverseFactorial (z : ℤ) :
30    (z : ℝ) * inverseFactorial z = inverseFactorial (z - 1) := by
31  by_cases hz : 0 < z
32  · obtain ⟨n, rfl⟩ := Int.eq_ofNat_of_zero_le (le_of_lt hz)
33    cases n with
34    | zero => simp at hz
35    | succ n =>
36      rw [show ((n + 1 : ℕ) : ℤ) - 1 = (n : ℤ) by omega]
37      change ((n + 1 : ℕ) : ℝ) * inverseFactorial ((n + 1 : ℕ) : ℤ) =
38        inverseFactorial (n : ℤ)
39      rw [inverseFactorial_nat, inverseFactorial_nat]
40      simp only [Nat.factorial_succ, Nat.cast_add,
41        Nat.cast_one, Nat.cast_mul]
42      have h : (n : ℝ) + 1 ≠ 0 := by positivity
43      field_simp
44  · by_cases hz0 : z = 0
45    · subst z
46      norm_num [inverseFactorial]
47    · rw [inverseFactorial_of_neg (by omega),
48        inverseFactorial_of_neg (by omega)]
49      simp

51/-- Columns are indexed by rows of the partition.  Transposing this matrix
52gives the usual `1 / (a_i - i + j)!` convention. -/
53def factorialMatrix {d : ℕ} (a : Fin d → ℤ) : Matrix (Fin d) (Fin d) ℝ :=
54  fun i j ↦ inverseFactorial (a j - (j.val : ℤ) + (i.val : ℤ))

56def factorialDeterminant {d : ℕ} (a : Fin d → ℤ) : ℝ :=
57  (factorialMatrix a).det

59def lowerIntegerRow {d : ℕ} (a : Fin d → ℤ) (i : Fin d) : Fin d → ℤ :=
60  fun j ↦ a j - if j = i then 1 else 0

62theorem factorialMatrix_lower_product {d : ℕ} (a : Fin d → ℤ)
63    (σ : Equiv.Perm (Fin d)) (i : Fin d) :
64    (∏ j, factorialMatrix (lowerIntegerRow a i) (σ j) j) =
65      ((a i - (i.val : ℤ) + ((σ i).val : ℤ) : ℤ) : ℝ) *
66        ∏ j, factorialMatrix a (σ j) j := by
67  have heq : (fun j ↦ factorialMatrix (lowerIntegerRow a i) (σ j) j) =
68      Function.update (fun j ↦ factorialMatrix a (σ j) j) i
69        (inverseFactorial (a i - (i.val : ℤ) + ((σ i).val : ℤ) - 1)) := by
70    funext j
71    by_cases hji : j = i
72    · subst j
73      simp only [factorialMatrix, lowerIntegerRow, ite_true, Function.update_self]
74      congr 1
75      ring
76    · simp [factorialMatrix, lowerIntegerRow, hji]
77  rw [heq, Finset.prod_update_of_mem (Finset.mem_univ i),
78    ← mul_inverseFactorial]
79  rw [← Finset.mul_prod_erase _ _ (Finset.mem_univ i)]
80  simp only [Finset.sdiff_singleton_eq_erase, factorialMatrix, mul_assoc]

82/-- The exact predecessor sum, valid for every integer vector, with no
83partition or nonnegativity assumptions. -/
84theorem factorialDeterminant_predecessors {d : ℕ} (a : Fin d → ℤ) :
85    (∑ i, factorialDeterminant (lowerIntegerRow a i)) =
86      (∑ i, (a i : ℝ)) * factorialDeterminant a := by
87  simp only [factorialDeterminant, Matrix.det_apply']
88  rw [Finset.sum_comm, Finset.mul_sum]
89  apply Finset.sum_congr rfl
90  intro σ _
91  simp_rw [factorialMatrix_lower_product]
92  have hs : (∑ i, ((a i - (i.val : ℤ) + ((σ i).val : ℤ) : ℤ) : ℝ)) =
93      ∑ i, (a i : ℝ) := by
94    simp only [Int.cast_add, Int.cast_sub, Int.cast_natCast,
95      Finset.sum_add_distrib, Finset.sum_sub_distrib]
96    rw [Equiv.sum_comp σ (fun i : Fin d ↦ (i.val : ℝ))]
97    ring
98  calc
99    _ = (Equiv.Perm.sign σ : ℝ) *
100        ((∑ i, ((a i - (i.val : ℤ) + ((σ i).val : ℤ) : ℤ) : ℝ)) *
101          ∏ j, factorialMatrix a (σ j) j) := by
102      rw [Finset.sum_mul, Finset.mul_sum]
103    _ = _ := by rw [hs]; ring

105theorem factorialDeterminant_zero_shape (d : ℕ) :
106    factorialDeterminant (fun _ : Fin d ↦ (0 : ℤ)) = 1 := by
107  unfold factorialDeterminant
108  rw [Matrix.det_of_lowerTriangular]
109  · simp [factorialMatrix, inverseFactorial]
110  · intro i j hij
111    apply inverseFactorial_of_neg
112    change j.val > i.val at hij
113    dsimp only
114    omega

116theorem factorialDeterminant_lower_equal_next {d : ℕ} (μ : Fin d → ℕ)
117    (i j : Fin d) (hij : j.val = i.val + 1) (hμ : μ i = μ j) :
118    factorialDeterminant (lowerIntegerRow (fun k ↦ (μ k : ℤ)) i) = 0 := by
119  apply Matrix.det_zero_of_column_eq (show i ≠ j by intro h; subst j; omega)
120  intro k
121  have hji : j ≠ i := by intro h; subst j; omega
122  simp only [factorialMatrix, lowerIntegerRow, ite_true, if_neg hji, sub_zero]
123  congr 1
124  rw [hμ]

142theorem removeBox_antitone_of_strict {d : ℕ} (μ : Fin d → ℕ)
143    (hμ : Antitone μ) (i : Fin d) (hi : ∀ j, i < j → μ j < μ i) :
144    Antitone (removeBox μ i) := by
145  intro a b hab
146  have h := hμ hab
147  by_cases hai : a = i
148  · subst a
149    by_cases hbi : b = i
150    · subst b; exact le_rfl
151    · have hib : i < b := lt_of_le_of_ne hab (Ne.symm hbi)
152      have hstrict := hi b hib
153      simp only [removeBox, if_neg hbi, ite_true]
154      omega
155  · simp only [removeBox, if_neg hai]
156    split_ifs <;> omega

158theorem factorialDeterminant_lower_not_antitone {d : ℕ} (μ : Fin d → ℕ)
159    (hμ : Antitone μ) (i : Fin d) (hbad : ¬ Antitone (removeBox μ i)) :
160    factorialDeterminant (lowerIntegerRow (fun k ↦ (μ k : ℤ)) i) = 0 := by
161  have hex : ∃ j, i < j ∧ μ j = μ i := by
162    by_contra! hn
163    apply hbad
164    apply removeBox_antitone_of_strict μ hμ i
165    intro j hij
166    exact lt_of_le_of_ne (hμ (le_of_lt hij)) (hn j hij)
167  obtain ⟨j, hij, heq⟩ := hex
168  have hnext : i.val + 1 < d := by have := j.isLt; simp only [Fin.lt_def] at hij; omega
169  let k : Fin d := ⟨i.val + 1, hnext⟩
170  apply factorialDeterminant_lower_equal_next μ i k rfl
171  have hik := hμ (show i ≤ k by simp [k, Fin.le_def])
172  have hkj := hμ (show k ≤ j by simp only [Fin.le_def, k]; simp only [Fin.lt_def] at hij; omega)
173  omega
