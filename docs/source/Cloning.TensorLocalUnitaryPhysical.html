64        (1 : Matrix (Fin d) (Fin d) ℂ) (w j) (Function.update w t b j)) = 1 := by

66      intro j hj
67      rw [Function.update_of_ne (Finset.mem_erase.mp hj).1]

137/-- Exact integration of the collective Lie action, on the full tensor space. -/
138theorem tensorOperator_matrixExpPath (Y : Matrix (Fin d) (Fin d) ℂ) (z : ℂ) :
139    tensorOperator n (matrixExpPath Y z) = exp (z • collectiveMatrix (n := n) Y) := by
140  let A := collectiveMatrix (n := n) Y
141  let F (s : ℂ) := exp ((-s) • A) * tensorOperator n (matrixExpPath Y s)
142  have hd (s : ℂ) : HasDerivAt F 0 s := by
143    have hbase := hasDerivAt_exp_smul_const A (-s)
144    have hminus := (show HasDerivAt _ _ (0-s) by simpa using hbase).comp_const_sub 0 s
145    simp only [zero_sub] at hminus
146    have hplus := tensorOperator_matrixExpPath_hasDerivAt (n := n) Y s
147    have hh := hminus.mul hplus
148    convert hh using 1
149    dsimp [A]
150    simp only [neg_smul]
151    noncomm_ring
152  have hc : F z = F 0 := is_const_of_deriv_eq_zero
153    (fun s => (hd s).differentiableAt) (fun s => (hd s).deriv) z 0
154  have hcancel : exp (z • A) * exp ((-z) • A) = 1 := by
155    rw [neg_smul, ← exp_add_of_commute (Commute.refl (z • A)).neg_right,
156      add_neg_cancel, exp_zero]
157  have hh := congrArg (fun S => exp (z • A) * S) hc
158  simpa only [F, neg_zero, zero_smul, exp_zero, matrixExpPath_zero,
159    tensorOperator_one, one_mul, mul_one, ← mul_assoc, hcancel] using hh
