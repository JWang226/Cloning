3/-! The actual cyclic lowering-word filtration of an arbitrary highest tensor.
4All positive roots are included. Raising invariance follows from the literal
5matrix-unit commutator, without PBW independence or irreducibility. -/

13abbrev PositiveRoot (d : ℕ) := {p : Fin d × Fin d // p.1 < p.2}

15def PositiveRoot.height (a : PositiveRoot d) : ℕ := a.val.2.val - a.val.1.val

17theorem PositiveRoot.height_pos (a : PositiveRoot d) : 0 < a.height := by

22noncomputable def loweringHeight : List (PositiveRoot d) → ℕ
23  | [] => 0
24  | a :: w => a.height + loweringHeight w

26noncomputable def loweringWord (Ω : TensorRegister n (Fin d)) :
27    List (PositiveRoot d) → TensorRegister n (Fin d)
28  | [] => Ω
29  | a :: w => collectiveGenerator n a.val.2 a.val.1 (loweringWord Ω w)

31def cyclicCutoff (Ω : TensorRegister n (Fin d)) (r : ℤ) :
32    Submodule ℂ (TensorRegister n (Fin d)) :=
33  Submodule.span ℂ {x | ∃ w : List (PositiveRoot d), (loweringHeight w : ℤ) ≤ r ∧
34    x = loweringWord Ω w}

36theorem loweringWord_mem_cyclicCutoff (Ω : TensorRegister n (Fin d))
37    (w : List (PositiveRoot d)) {r : ℤ} (hr : (loweringHeight w : ℤ) ≤ r) :
38    loweringWord Ω w ∈ cyclicCutoff Ω r :=
39  Submodule.subset_span ⟨w, hr, rfl⟩

41theorem cyclicCutoff_monotone (Ω : TensorRegister n (Fin d)) : Monotone (cyclicCutoff Ω) := by
42  intro r s hrs
43  apply Submodule.span_mono
44  rintro x ⟨w, hw, rfl⟩
45  exact ⟨w, hw.trans hrs, rfl⟩

47theorem highest_mem_cyclicCutoff_zero (Ω : TensorRegister n (Fin d)) : Ω ∈ cyclicCutoff Ω 0 :=

50theorem cyclicCutoff_eq_bot_of_neg (Ω : TensorRegister n (Fin d)) {r : ℤ} (hr : r < 0) :
51    cyclicCutoff Ω r = ⊥ := by
52  apply le_antisymm _ bot_le
53  apply Submodule.span_le.mpr

58theorem cyclicCutoff_map_of_words (Ω : TensorRegister n (Fin d))

69theorem lowering_mem_cyclicCutoff (Ω : TensorRegister n (Fin d))
70    (a : PositiveRoot d) (r : ℤ) {x : TensorRegister n (Fin d)} (hx : x ∈ cyclicCutoff Ω r) :
71    collectiveGenerator n a.val.2 a.val.1 x ∈ cyclicCutoff Ω (r + a.height) := by
72  apply cyclicCutoff_map_of_words Ω _ r (r + a.height) _ hx
73  intro w hw
74  apply loweringWord_mem_cyclicCutoff Ω (a :: w)
75  simp only [loweringHeight, Nat.cast_add]

78/-- The Lie commutator gives every root its exact height shift on each actual
79lowering word. The Cartan and raising premises concern only the highest tensor. -/
80theorem generator_loweringWord_mem
81    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℂ)
82    (hweight : ∀ a, collectiveGenerator n a a Ω = mu a • Ω)
83    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
84    (w : List (PositiveRoot d)) (a b : Fin d) :
85    collectiveGenerator n a b (loweringWord Ω w) ∈
86      cyclicCutoff Ω ((loweringHeight w : ℤ) + a.val - b.val) := by
87  induction w generalizing a b with
88  | nil =>
89      simp only [loweringWord, loweringHeight, Nat.cast_zero, zero_add]
90      rcases lt_trichotomy a b with hab | hab | hab
91      · rw [hraise a b hab]
92        exact (cyclicCutoff Ω _).zero_mem
93      · subst b
94        simp only [sub_self, hweight]
95        exact (cyclicCutoff Ω 0).smul_mem _ (highest_mem_cyclicCutoff_zero Ω)
96      · let c : PositiveRoot d := ⟨(b, a), hab⟩
97        have hh := loweringWord_mem_cyclicCutoff Ω [c] (le_refl ((loweringHeight [c] : ℕ) : ℤ))
98        have hc : (loweringHeight [c] : ℤ) = (a.val : ℤ) - b.val := by
99          simp only [loweringHeight, PositiveRoot.height, Nat.add_zero, c]
100          rw [Int.natCast_sub (Nat.le_of_lt hab)]
101        simpa only [loweringWord, hc] using hh
102  | cons c w ih =>
103      have he := congrArg (fun T : TensorRegister n (Fin d) →L[ℂ] TensorRegister n (Fin d) =>
104        T (loweringWord Ω w)) (collectiveGenerator_commutator (n := n) a b c.val.2 c.val.1)
105      simp only [ContinuousLinearMap.sub_apply, ContinuousLinearMap.mul_apply] at he
106      have he' : collectiveGenerator n a b (loweringWord Ω (c :: w)) =
107          collectiveGenerator n c.val.2 c.val.1
108            (collectiveGenerator n a b (loweringWord Ω w)) +
109          ((if b = c.val.2 then collectiveGenerator n a c.val.1 else 0)
110            (loweringWord Ω w) -
111          (if c.val.1 = a then collectiveGenerator n c.val.2 b else 0)
112            (loweringWord Ω w)) := by
113        change _ - _ = _ at he
114        simpa only [loweringWord] using (sub_eq_iff_eq_add.mp he).trans (add_comm _ _)
115      rw [he']
116      apply (cyclicCutoff Ω _).add_mem
117      · have hh := lowering_mem_cyclicCutoff Ω c _ (ih a b)
118        convert hh using 1 <;> simp only [loweringHeight, Nat.cast_add] <;> ring
119      · apply (cyclicCutoff Ω _).sub_mem
120        · split_ifs with hb
121          · subst b
122            have hh := ih a c.val.1
123            convert hh using 1
124            simp only [loweringHeight, Nat.cast_add, PositiveRoot.height]
125            rw [Int.natCast_sub (Nat.le_of_lt c.property)]
126            ring
127          · exact (cyclicCutoff Ω _).zero_mem
128        · split_ifs with ha
129          · subst a
130            have hh := ih c.val.2 b
131            convert hh using 1
132            simp only [loweringHeight, Nat.cast_add, PositiveRoot.height]
133            rw [Int.natCast_sub (Nat.le_of_lt c.property)]
134            ring
135          · exact (cyclicCutoff Ω _).zero_mem

137theorem generator_mem_cyclicCutoff
138    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℂ)
139    (hweight : ∀ a, collectiveGenerator n a a Ω = mu a • Ω)
140    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
141    (r : ℤ) (a b : Fin d) {x : TensorRegister n (Fin d)} (hx : x ∈ cyclicCutoff Ω r) :
142    collectiveGenerator n a b x ∈ cyclicCutoff Ω (r + a.val - b.val) := by
143  apply cyclicCutoff_map_of_words Ω _ r _ _ hx
144  intro w hw
145  exact cyclicCutoff_monotone Ω (by omega) (generator_loweringWord_mem Ω mu hweight hraise w a b)

147/-- Raising annihilates the full cyclic vacuum cutoff, not merely the chosen vector. -/
148theorem raising_zero_cyclicCutoff
149    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℂ)
150    (hweight : ∀ a, collectiveGenerator n a a Ω = mu a • Ω)
151    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
152    (a b : Fin d) (hab : a < b) {x : TensorRegister n (Fin d)}
153    (hx : x ∈ cyclicCutoff Ω 0) : collectiveGenerator n a b x = 0 := by
154  have hh := generator_mem_cyclicCutoff Ω mu hweight hraise 0 a b hx
155  rw [cyclicCutoff_eq_bot_of_neg Ω (by have : a.val < b.val := hab; omega)] at hh
156  exact hh

158/-- Every raising root strictly lowers this filtration. -/
159theorem raising_lowers_cyclicCutoff
160    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℂ)
161    (hweight : ∀ a, collectiveGenerator n a a Ω = mu a • Ω)
162    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
163    (a b : Fin d) (hab : a < b) (r : ℕ) {x : TensorRegister n (Fin d)}
164    (hx : x ∈ cyclicCutoff Ω (r + 1)) : collectiveGenerator n a b x ∈ cyclicCutoff Ω r := by
165  have hh := generator_mem_cyclicCutoff Ω mu hweight hraise (r + 1) a b hx

169theorem loweringHeight_eq_zero_iff (w : List (PositiveRoot d)) :
170    loweringHeight w = 0 ↔ w = [] := by
171  cases w with
172  | nil => simp [loweringHeight]
173  | cons a w =>
174      have ha := a.height_pos
175      simp only [loweringHeight, reduceCtorEq, iff_false]
176      omega

178/-- The cyclic cutoff zero consists precisely of scalar multiples of the chosen
179highest vector; there are no extraneous ambient vectors at this cutoff. -/
180theorem cyclicCutoff_zero (Ω : TensorRegister n (Fin d)) :
181    cyclicCutoff Ω 0 = Submodule.span ℂ {Ω} := by
182  apply le_antisymm
183  · apply Submodule.span_le.mpr
184    rintro x ⟨w, hw, rfl⟩
185    have hw0 : loweringHeight w = 0 := by omega
186    rw [(loweringHeight_eq_zero_iff w).mp hw0, loweringWord]
187    exact Submodule.subset_span (Set.mem_singleton Ω)
188  · apply Submodule.span_le.mpr
189    intro x hx
190    have he : x = Ω := by simpa only [Set.mem_singleton_iff] using hx
191    rw [he]
192    exact highest_mem_cyclicCutoff_zero Ω

198theorem lowering_raises_cyclicCutoff (Ω : TensorRegister n (Fin d))
199    (a : PositiveRoot d) (r : ℕ) {x : TensorRegister n (Fin d)}
200    (hx : x ∈ cyclicCutoff Ω r) :
201    collectiveGenerator n a.val.2 a.val.1 x ∈ cyclicCutoff Ω ((r + d : ℕ) : ℤ) := by
202  have hh := lowering_mem_cyclicCutoff Ω a r hx
203  apply cyclicCutoff_monotone Ω _ hh
