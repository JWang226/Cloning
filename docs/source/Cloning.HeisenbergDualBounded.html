10/-- Boundedness of the reconstructed dual follows from the exact trace norm
11of rank-one operators and the norm-one trace pairing. -/
12theorem heisenbergDual_norm_le (Φ : TraceClass H →L[ℂ] TraceClass K)
13    (A : K →L[ℂ] K) : ‖heisenbergDual Φ A‖ ≤ ‖Φ‖ * ‖A‖ := by
14  change ‖traceClassDualOp (((tracePairingCLM (H := K)).flip A).comp Φ)‖ ≤ _
15  refine (norm_traceClassDualOp_le (H := H) (((tracePairingCLM (H := K)).flip A).comp Φ)).trans ?_
16  refine (ContinuousLinearMap.opNorm_comp_le ((tracePairingCLM (H := K)).flip A) Φ).trans ?_
17  have hpair : ‖(tracePairingCLM (H := K)).flip A‖ ≤ ‖A‖ := by
18    apply ContinuousLinearMap.opNorm_le_bound _ (norm_nonneg _)
19    intro T
20    change ‖tracePairing T A‖ ≤ ‖A‖ * ‖T‖
21    calc
22      _ ≤ ‖tracePairing T‖ * ‖A‖ := (tracePairing T).le_opNorm A
23      _ ≤ ‖T‖ * ‖A‖ := mul_le_mul_of_nonneg_right
24        ((tracePairingCLM.le_opNorm T).trans (by
25          simpa using mul_le_mul_of_nonneg_right (norm_tracePairingCLM_le (H := K)) (norm_nonneg T)))
26        (norm_nonneg A)
27      _ = _ := mul_comm _ _
28  exact (mul_le_mul_of_nonneg_right hpair (norm_nonneg Φ)).trans_eq (mul_comm _ _)

31def heisenbergDualCLM (Φ : TraceClass H →L[ℂ] TraceClass K) :
