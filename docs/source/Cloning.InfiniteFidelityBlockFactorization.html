18theorem norm_positive_sub (A B : PositiveTraceClass H) (h : B.1.1 ≤ A.1.1) :
19    ‖A.1-B.1‖ = ‖A.1‖-‖B.1‖ := by
20  have hp : 0 ≤ (A.1-B.1).1 := by
21    change 0 ≤ A.1.1-B.1.1
22    exact sub_nonneg.mpr h
23  rw [TraceClass.norm_eq_trace_re_of_nonneg _ hp, positive_norm_eq_trace A, positive_norm_eq_trace B]
24  change (traceCLM (A.1-B.1)).re = _
25  rw [map_sub, Complex.sub_re]

57/-- The actual orthogonal output fidelity is within η+2√δ of the classical
58affinity times the common retained-sector fidelity. Components with zero
59retained mass require no normalized-fidelity estimate. -/
60theorem orthogonalSum_factorization_bound
61    (V : ∀ i, E i →ₗᵢ[ℂ] H) (hV : OrthogonalFamily ℂ E V)
62    (b : ∀ i, OrthonormalBasis (J i) ℂ (E i))
63    (X C B : ∀ i, PositiveTraceClass (E i)) (p r : I → ℝ)
64    (hp : ∀ i, 0 ≤ p i) (hr : ∀ i, 0 ≤ r i)
65    (hps : ∑ i, p i = 1) (hXs : ∑ i, ‖(X i).1‖ ≤ 1)
66    (hC : ∀ i, ‖(C i).1‖ = 1) (hB : ∀ i, ‖(B i).1‖ = 1)
67    (hle : ∀ i, (PositiveTraceClass.scale ⟨r i, hr i⟩ (C i)).1.1 ≤ (X i).1.1)
68    (c η : ℝ) (hc0 : 0 ≤ c) (hc1 : c ≤ 1) (hη : 0 ≤ η)
69    (hsector : ∀ i, 0 < r i → |(C i).rootFidelity (B i)-c| ≤ η) :
70    |(orthogonalSum V X).rootFidelity
71        (orthogonalSum V (fun i ↦ PositiveTraceClass.scale ⟨p i,hp i⟩ (B i))) -
72      c * Cloning.BlockFidelity.classicalAffinity (fun i ↦ ‖(X i).1‖) p| ≤
73      η + 2 * Real.sqrt (∑ i, (‖(X i).1‖-r i)) := by
74  let G i := PositiveTraceClass.scale ⟨r i,hr i⟩ (C i)
75  let T i := PositiveTraceClass.scale ⟨p i,hp i⟩ (B i)
76  have hG i : ‖(G i).1‖ = r i := by
77    simp only [G, PositiveTraceClass.norm_scale, hC, mul_one]
78    rfl
79  have hT i : ‖(T i).1‖ = p i := by
80    simp only [T, PositiveTraceClass.norm_scale, hB, mul_one]
81    rfl
82  have hmass : ‖(orthogonalSum V T).1‖ = 1 := by
83    rw [norm_orthogonalSum]
84    simpa only [hT] using hps
85  have hri i : r i ≤ ‖(X i).1‖ := by
86    have hn := norm_nonneg ((X i).1-(G i).1)
87    rw [norm_positive_sub (X i) (G i) (hle i), hG] at hn
88    linarith
89  apply Cloning.BlockFidelity.lifted_factorization_bound p (fun i ↦ ‖(X i).1‖) r
90    (fun i ↦ (C i).rootFidelity (B i)) _ c η hp hr hri hps hXs hc0 hc1 hη hsector
91  have hf := PositiveTraceClass.rootFidelity_continuity
92    (orthogonalSum V X) (orthogonalSum V T) (orthogonalSum V G) (orthogonalSum V T)
93  rw [sub_self, norm_zero, Real.sqrt_zero, zero_mul, add_zero, hmass, Real.sqrt_one, mul_one,
94    norm_positive_sub _ _ (orthogonalSum_mono V G X hle), norm_orthogonalSum,
95    norm_orthogonalSum, ← Finset.sum_sub_distrib] at hf
96  simp only [hG] at hf
97  have he := rootFidelity_orthogonalSum V hV b G T
98  simp only [G, T, PositiveTraceClass.rootFidelity_scale] at he
99  rw [he] at hf
100  exact hf
