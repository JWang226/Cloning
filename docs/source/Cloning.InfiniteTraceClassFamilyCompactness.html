13/-- Actual simultaneous extraction from a bounded family of dual spaces.
14The index `I` may be uncountable; only the observations must be countable. -/
15theorem exists_subsequence_family_functionals {I J E : Type*} [Countable J]
16    [NormedAddCommGroup E] [NormedSpace ℂ E]
17    (u : ℕ → I → E →L[ℂ] ℂ) (C : I → ℝ)
18    (hu : ∀ n i, ‖u n i‖ ≤ C i) (O : J → I × E) :
19    ∃ f : I → E →L[ℂ] ℂ,
20      (∀ i, ‖f i‖ ≤ C i) ∧
21      (fun i => StrongDual.toWeakDual (f i)) ∈
22        closure (Set.range (fun n i => StrongDual.toWeakDual (u n i))) ∧
23      ∃ φ : ℕ → ℕ, StrictMono φ ∧ ∀ j,
24        Tendsto (fun n => u (φ n) (O j).1 (O j).2) atTop
25          (𝓝 (f (O j).1 (O j).2)) := by
26  let v : ℕ → I → WeakDual ℂ E := fun n i => StrongDual.toWeakDual (u n i)
27  let B : Set (I → WeakDual ℂ E) := Set.pi Set.univ
28    (fun i => WeakDual.toStrongDual ⁻¹' Metric.closedBall 0 (C i))
29  have hB : IsCompact B := isCompact_univ_pi
30    (fun i => WeakDual.isCompact_closedBall ℂ (0 : E →L[ℂ] ℂ) (C i))
31  have hvB : Set.range v ⊆ B := by
32    rintro _ ⟨n, rfl⟩ i _
33    simpa [v, Metric.mem_closedBall, dist_zero_right] using hu n i
34  have hcsub : closure (Set.range v) ⊆ B := closure_minimal hvB hB.isClosed
35  have hc : IsCompact (closure (Set.range v)) := hB.of_isClosed_subset isClosed_closure hcsub
36  let ev : (I → WeakDual ℂ E) → J → ℂ := fun f j => f (O j).1 (O j).2
37  have hev : Continuous ev := continuous_pi (fun j =>
38    (WeakDual.eval_continuous (O j).2).comp (continuous_apply (O j).1))
39  have hmem (n : ℕ) : ev (v n) ∈ ev '' closure (Set.range v) :=
40    ⟨v n, subset_closure (Set.mem_range_self n), rfl⟩
41  obtain ⟨g, hg, φ, hφ, hlim⟩ := (hc.image hev).tendsto_subseq hmem
42  obtain ⟨f, hf, rfl⟩ := hg
43  refine ⟨fun i => WeakDual.toStrongDual (f i), ?_, hf, φ, hφ, ?_⟩
44  · intro i
45    have hi := hcsub hf i (Set.mem_univ i)
46    simpa [Metric.mem_closedBall, dist_zero_right] using hi
47  · intro j
48    exact (continuous_apply j).tendsto _ |>.comp hlim

64lemma tendsto_rankOne_of_dense_evaluations [SeparableSpace H]
65    (u : ℕ → (H →L[ℂ] H) →L[ℂ] ℂ) (f : (H →L[ℂ] H) →L[ℂ] ℂ)
66    (C : ℝ) (hC : 0 ≤ C) (hu : ∀ n, ‖u n‖ ≤ C) (hf : ‖f‖ ≤ C)
67    (hlim : ∀ i : ℕ × ℕ, Tendsto (fun n => u n
68      (InnerProductSpace.rankOne ℂ (denseSeq H i.1) (denseSeq H i.2))) atTop
69      (𝓝 (f (InnerProductSpace.rankOne ℂ (denseSeq H i.1) (denseSeq H i.2)))))

79/-- One subsequence works simultaneously for every member of a countable
80family of positive trace-bounded operator sequences. -/
81theorem exists_subsequence_weak_operator_limits {J : Type*} [Countable J]
82    [SeparableSpace H] (A : ℕ → J → TraceClass H)
83    (hA : ∀ n j, 0 ≤ (A n j).1) (C : J → ℝ)
84    (htrace : ∀ n j, (trace (A n j).1 (A n j).2).re ≤ C j) :
85    ∃ T : J → TraceClass H,
86      (∀ j, 0 ≤ (T j).1 ∧ (trace (T j).1 (T j).2).re ≤ C j) ∧
87      ∃ φ : ℕ → ℕ, StrictMono φ ∧ ∀ j x y,
88        Tendsto (fun n => ⟪y, (A (φ n) j).1 x⟫_ℂ) atTop (𝓝 ⟪y, (T j).1 x⟫_ℂ) := by
89  let u := fun n j => tracePairing (A n j)
90  have hnorm (n : ℕ) (j : J) : ‖A n j‖ ≤ C j := by
91    rw [TraceClass.norm_eq_trace_re_of_nonneg _ (hA n j)]
92    exact htrace n j
93  have hu (n : ℕ) (j : J) : ‖u n j‖ ≤ C j :=
94    (norm_tracePairing_le _).trans (hnorm n j)
95  let O : J × (ℕ × ℕ) → J × (H →L[ℂ] H) := fun j =>
96    (j.1, InnerProductSpace.rankOne ℂ (denseSeq H j.2.1) (denseSeq H j.2.2))
97  obtain ⟨f, hf, _, φ, hφ, hlim⟩ := exists_subsequence_family_functionals u C hu O
98  have hcoeff (j : J) (x y : H) :
99      Tendsto (fun n => ⟪y, (A (φ n) j).1 x⟫_ℂ) atTop
100        (𝓝 ⟪y, normalPartOp (f j) x⟫_ℂ) := by
101    have h := tendsto_rankOne_of_dense_evaluations (fun n => u (φ n) j) (f j) (C j)
102      ((norm_nonneg (A 0 j)).trans (hnorm 0 j))
103      (fun n => hu (φ n) j) (hf j) (fun i => hlim (j, i)) x y
104    simpa only [u, tracePairing_rankOne, normalPart_inner] using h
105  have hex (j : J) : 0 ≤ normalPartOp (f j) ∧
106      ∃ ht : IsTraceClass (normalPartOp (f j)),
107        (trace (normalPartOp (f j)) ht).re ≤ C j := by
108    apply InfiniteTraceClassWeakLimit.traceClass_of_diagonal_tendsto (l := atTop)
109      (fun n => (A (φ n) j).1) (normalPartOp (f j)) (fun n => hA (φ n) j)
110      (fun n => (A (φ n) j).2) (C j) (fun n => htrace (φ n) j)
111    intro x
112    convert (hcoeff j x x).star using 1
113    · ext n; exact (inner_conj_symm _ _).symm
114    · congr 1; exact (inner_conj_symm _ _).symm
115  let T : J → TraceClass H := fun j =>
116    TraceClass.ofOperator (normalPartOp (f j)) (hex j).2.choose
117  refine ⟨T, ?_, φ, hφ, hcoeff⟩
118  intro j
119  exact ⟨(hex j).1, (hex j).2.choose_spec⟩
