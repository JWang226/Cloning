21/-- The actual quantum-limited amplifier, in noise parameter `q=1-1/gain`. -/
22def channel (q : ℝ) (hq0 : 0 ≤ q) (hq1 : q < 1) : QuantumChannel Fock Fock :=
23  QuantumChannel.ofKraus (kraus q hq0 hq1) (kraus_norm_sq_hasSum q hq0 hq1)

25/-- The defining Kraus series converges in the actual trace norm. -/
26theorem channel_hasSum (q : ℝ) (hq0 : 0 ≤ q) (hq1 : q < 1) (A : TraceClass Fock) :
27    HasSum (fun k => krausTerm (kraus q hq0 hq1 k) A) ((channel q hq0 hq1).toLinearMap A) :=
28  QuantumChannel.ofKraus_hasSum _ _ A

30/-- Scalar square-root amplitudes produce exactly their squared probability. -/
31theorem vectorProjector_sqrt_smul {H : Type*}
32    [NormedAddCommGroup H] [InnerProductSpace ℂ H] [CompleteSpace H]
33    (a : ℝ) (ha : 0 ≤ a) (x : H) :
34    vectorProjector ((Real.sqrt a : ℂ) • x) = (a : ℂ) • vectorProjector x := by
35  apply Subtype.ext
36  ext y
37  change ⟪(Real.sqrt a : ℂ) • x, y⟫_ℂ • ((Real.sqrt a : ℂ) • x) =
38    (a : ℂ) • (⟪x, y⟫_ℂ • x)
39  rw [inner_smul_left, smul_smul, smul_smul]
40  have hs : (Real.sqrt a : ℂ) * (Real.sqrt a : ℂ) = (a : ℂ) := by
41    exact_mod_cast Real.mul_self_sqrt ha
42  congr 1
43  simp only [Complex.conj_ofReal]
44  calc
45    _ = ((Real.sqrt a : ℂ) * (Real.sqrt a : ℂ)) * ⟪x, y⟫_ℂ := by ring
46    _ = _ := by rw [hs]

48theorem krausTerm_numberProjector (q : ℝ) (hq0 : 0 ≤ q) (hq1 : q < 1) (n k : ℕ) :
49    krausTerm (kraus q hq0 hq1 k) (vectorProjector (numberBasis n)) =
50      (weight q n k : ℂ) • vectorProjector (numberBasis (n + k)) := by
51  rw [krausTerm_vectorProjector, kraus_numberBasis, vectorProjector_sqrt_smul]
52  exact weight_nonneg hq0 hq1 n k

54/-- Exact negative-binomial output of a number input under the actual channel. -/
55theorem channel_numberProjector (q : ℝ) (hq0 : 0 ≤ q) (hq1 : q < 1) (n : ℕ) :
56    (channel q hq0 hq1).toLinearMap (vectorProjector (numberBasis n)) =
57      vectorMixture (fun k => numberBasis (n + k)) (weight q n) := by
58  have h := channel_hasSum q hq0 hq1 (vectorProjector (numberBasis n))
59  simp only [krausTerm_numberProjector] at h
60  exact h.tsum_eq.symm
