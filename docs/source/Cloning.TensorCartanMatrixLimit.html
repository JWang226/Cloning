33/-- Exact physical matrix elements are finite sums of explicit splitting
34coefficients times the actual Gram entries in the two factor sectors. -/
35theorem cartanWordMatrixElement_eq_sum
36    (mu nu : Fin d → ℕ) (hmu : Antitone mu) (hnu : Antitone nu)
37    (hmuGap : ∀ a : PositiveRoot d, 0 < rootGap mu a)
38    (hnuGap : ∀ a : PositiveRoot d, 0 < rootGap nu a)
39    (w u v : List (PositiveRoot d)) :
40    cartanWordMatrixElement mu nu hmu hnu w u v =
41      ((loweringSplits w).map (fun ab => normalizedSplitCoefficient mu nu w ab.1 ab.2 *
42        (⟪partitionNormalizedLoweringWord mu hmu u,
43          partitionNormalizedLoweringWord mu hmu ab.1⟫_ℂ *
44         ⟪partitionNormalizedLoweringWord nu hnu v,
45          partitionNormalizedLoweringWord nu hnu ab.2⟫_ℂ))).sum := by
46  unfold cartanWordMatrixElement
47  rw [cartanAmbientInclusion_word_split mu nu hmu hnu hmuGap hnuGap]
48  change innerSL ℂ (tensorJoin (partitionNormalizedLoweringWord mu hmu u)
49    (partitionNormalizedLoweringWord nu hnu v)) _ = _
50  rw [map_list_sum]
51  simp only [List.map_map, Function.comp_def, innerSL_apply_apply, inner_smul_right,
52    tensorJoin_inner, partitionNormalizedLoweringWord]

54/-- The actual Cartan matrix-element limit, with every physical Gram and
55intertwining premise discharged by the tensor constructions. Both sector root
56gaps diverge and their fractions converge; the three words remain fixed. -/
57theorem cartanWordMatrixElement_tendsto
58    (mu nu : ℕ → Fin d → ℕ) (hmu : ∀ N, Antitone (mu N)) (hnu : ∀ N, Antitone (nu N))
59    (δmu δnu : ℕ → ℝ) (hδmu : Tendsto δmu atTop atTop) (hδnu : Tendsto δnu atTop atTop)
60    (hmuGap : ∀ᶠ N in atTop, ∀ a : PositiveRoot d, δmu N ≤ rootGap (mu N) a)
61    (hnuGap : ∀ᶠ N in atTop, ∀ a : PositiveRoot d, δnu N ≤ rootGap (nu N) a)
62    (t : PositiveRoot d → ℝ)
63    (ht : ∀ a, Tendsto (fun N => rootFraction (mu N) (nu N) a) atTop (𝓝 (t a)))
64    (w u v : List (PositiveRoot d)) :
65    Tendsto (fun N => cartanWordMatrixElement (mu N) (nu N) (hmu N) (hnu N) w u v)
66      atTop (𝓝 (occupationSplitMatrixElement t w u v)) := by
67  have hμpos : ∀ᶠ N in atTop, ∀ a : PositiveRoot d, 0 < rootGap (mu N) a := by
68    filter_upwards [hmuGap, hδmu.eventually (eventually_gt_atTop 0)] with N hN hp
69    exact fun a => hp.trans_le (hN a)
70  have hνpos : ∀ᶠ N in atTop, ∀ a : PositiveRoot d, 0 < rootGap (nu N) a := by
71    filter_upwards [hnuGap, hδnu.eventually (eventually_gt_atTop 0)] with N hN hp
72    exact fun a => hp.trans_le (hN a)
73  have hsum : Tendsto (fun N =>
74      ((loweringSplits w).map (fun ab => normalizedSplitCoefficient (mu N) (nu N) w ab.1 ab.2 *
75        (⟪partitionNormalizedLoweringWord (mu N) (hmu N) u,
76          partitionNormalizedLoweringWord (mu N) (hmu N) ab.1⟫_ℂ *
77         ⟪partitionNormalizedLoweringWord (nu N) (hnu N) v,
78          partitionNormalizedLoweringWord (nu N) (hnu N) ab.2⟫_ℂ))).sum)
79      atTop (𝓝 (occupationSplitMatrixElement t w u v)) := by
80    apply tendsto_list_sum
81    intro ab hab
82    have hA := partition_normalized_gram_tendsto (fun a : PositiveRoot d => a)
83      Function.injective_id mu hmu δmu hδmu hmuGap u ab.1
84    have hB := partition_normalized_gram_tendsto (fun a : PositiveRoot d => a)
85      Function.injective_id nu hnu δnu hδnu hnuGap v ab.2
86    exact (normalizedSplitCoefficient_tendsto_complement mu nu hμpos hνpos t ht w ab.1 ab.2 hab).mul
87      (hA.mul hB)
88  apply hsum.congr'
89  filter_upwards [hμpos, hνpos] with N hμ hν
90  exact (cartanWordMatrixElement_eq_sum (mu N) (nu N) (hmu N) (hnu N) hμ hν w u v).symm
