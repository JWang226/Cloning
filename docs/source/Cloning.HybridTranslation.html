6# Classical translations of operator-valued L¹

8Translation acts on actual L¹ equivalence classes by `A(y-h)`.  It is a

22/-- Translation of the classical register, with the density convention
23`(T h A)(y) = A(y-h)`. -/
24def classicalTranslation (h : Fin k → ℝ) :
25    Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ)) →ₗᵢ[ℂ]
26      Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ)) :=
27  Lp.compMeasurePreservingₗᵢ ℂ (fun y => y - h)
28    (measurePreserving_sub_right volume h)

30theorem classicalTranslation_ae (h : Fin k → ℝ)

32    classicalTranslation h A =ᵐ[volume] fun y => A (y - h) :=
33  Lp.coeFn_compMeasurePreserving A (measurePreserving_sub_right volume h)

46theorem classicalTranslation_add (h g : Fin k → ℝ)
47    (A : Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ))) :
48    classicalTranslation (h + g) A = classicalTranslation h (classicalTranslation g A) := by
49  apply Lp.ext
50  have he := (measurePreserving_sub_right volume h).quasiMeasurePreserving.ae
51    (classicalTranslation_ae g A)
52  filter_upwards [classicalTranslation_ae (h + g) A,
53    classicalTranslation_ae h (classicalTranslation g A), he] with y hsum hh hg
54  rw [hsum, hh, hg, sub_sub]

56theorem classicalTranslation_nonneg (h : Fin k → ℝ)
57    (A : Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ)))
58    (hA : ∀ᵐ y ∂volume, 0 ≤ (A y).1) :
59    ∀ᵐ y ∂volume, 0 ≤ (classicalTranslation h A y).1 := by
60  have hp := (measurePreserving_sub_right volume h).quasiMeasurePreserving.ae hA
61  filter_upwards [classicalTranslation_ae h A, hp] with y hy hp
62  rwa [hy]

64theorem classicalTranslation_completelyPositive (h : Fin k → ℝ) :
65    L1CompletelyPositive (classicalTranslation (H := H) h).toLinearMap := by
66  intro n A hA
67  have hp := (measurePreserving_sub_right volume h).quasiMeasurePreserving.ae hA
68  have he : ∀ᵐ y ∂volume, ∀ i j : Fin n,
69      classicalTranslation h (A i j) y = A i j (y - h) :=
70    ae_all_iff.mpr fun i => ae_all_iff.mpr fun j => classicalTranslation_ae h (A i j)
71  filter_upwards [he, hp] with y hy hp
72  change BlockPositive (fun i j => classicalTranslation h (A i j) y)
73  simpa only [hy] using hp

75theorem integral_classicalTranslation (h : Fin k → ℝ)
76    (A : Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ))) :
77    (∫ y, classicalTranslation h A y) = ∫ y, A y := by
78  calc
79    _ = ∫ y, A (y - h) := integral_congr_ae (classicalTranslation_ae h A)
80    _ = _ := integral_sub_right_eq_self A h

82theorem classicalTranslation_tracePreserving (h : Fin k → ℝ) :
83    L1TracePreserving (classicalTranslation (H := H) h).toLinearMap := by
84  intro A
85  calc
86    _ = ∫ y, traceCLM (A (y - h)) :=
87      integral_congr_ae ((classicalTranslation_ae h A).fun_comp traceCLM)

90/-- The classical action is jointly continuous in the displacement and L¹
91input, in particular strongly continuous in the displacement. -/
92theorem continuous_classicalTranslation_joint :
93    Continuous (fun p : (Fin k → ℝ) ×
94      Lp (TraceClass H) 1 (volume : Measure (Fin k → ℝ)) =>
95        classicalTranslation p.1 p.2) := by
96  let g : C((Fin k → ℝ), C((Fin k → ℝ), (Fin k → ℝ))) :=
97    (⟨fun p : (Fin k → ℝ) × (Fin k → ℝ) => p.2 - p.1,
98      continuous_snd.sub continuous_fst⟩ : C(_, _)).curry
99  exact continuous_snd.compMeasurePreservingLp (g.continuous.comp continuous_fst)
100    (fun p => measurePreserving_sub_right volume p.1) (by simp)

102theorem continuous_classicalTranslation
