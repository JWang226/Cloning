3/-!
4# Vacuum-signal output for an arbitrary idler state
5 
6Taking slices in the other output coordinate of the amplifier isometry gives
7a genuine complementary channel. For input number `l`, the physical signal
8and idler pair is `|k,k+l⟩`. The channel retains general input coherences;
9only its output number diagonal depends solely on the idler number diagonal.
10-/

20/-- A slice at a fixed output idler number. -/
21def idlerSliceVector (m : ℕ) (x : TwoModeFock) : Fock := by
22  refine ⟨fun k => x (m, k), memℓp_gen ?_⟩
23  exact ((lp.memℓp x).summable (by norm_num)).comp_injective
24    (fun i j h => congrArg Prod.snd h)

26theorem idlerSliceVector_norm_le (m : ℕ) (x : TwoModeFock) :
27    ‖idlerSliceVector m x‖ ≤ ‖x‖ := by
28  apply lp.norm_le_of_tsum_le (by norm_num) (norm_nonneg x)
29  rw [lp.norm_rpow_eq_tsum (by norm_num) x]
30  exact Summable.tsum_le_tsum_of_inj (fun k : ℕ => (m, k))
31    (fun i j h => congrArg Prod.snd h)
32    (fun p _ => Real.rpow_nonneg (norm_nonneg _) _)
33    (fun _ => le_rfl)
34    ((lp.memℓp (idlerSliceVector m x)).summable (by norm_num))
35    ((lp.memℓp x).summable (by norm_num))

37def idlerSlice (m : ℕ) : TwoModeFock →L[ℂ] Fock :=
38  LinearMap.mkContinuous
39    { toFun := idlerSliceVector m
40      map_add' := by intro x y; ext k; rfl
41      map_smul' := by intro c x; ext k; rfl }
42    1 (fun x => by simpa only [one_mul] using idlerSliceVector_norm_le m x)

47theorem idlerSlice_norm_sq_hasSum (x : TwoModeFock) :
48    HasSum (fun m => ‖idlerSlice m x‖ ^ 2) (‖x‖ ^ 2) := by
49  have hfull : HasSum (fun p : ℕ × ℕ => ‖x p‖ ^ (2 : ℕ)) (‖x‖ ^ 2) := by
50    simpa only [ENNReal.toReal_ofNat, Real.rpow_two] using
51      lp.hasSum_norm (show 0 < (2 : ENNReal).toReal by norm_num) x
52  apply hfull.prod_fiberwise
53  intro m
54  simpa only [ENNReal.toReal_ofNat, Real.rpow_two, idlerSlice_apply] using
55    lp.hasSum_norm (show 0 < (2 : ENNReal).toReal by norm_num) (idlerSlice m x)

57/-- Kraus operators for the vacuum signal with a general idler input. -/
58def idlerKraus (q : ℝ) (hq0 : 0 ≤ q) (hq1 : q < 1) (m : ℕ) : Fock →L[ℂ] Fock :=
59  (idlerSlice m).comp (isometry q hq0 hq1).toContinuousLinearMap

94/-- On a fixed output number the adjoint Kraus operator selects one idler number. -/
95theorem idlerKraus_adjoint_numberBasis_shift
96    (q : ℝ) (hq0 : 0 ≤ q) (hq1 : q < 1) (l k : ℕ) :
97    star (idlerKraus q hq0 hq1 (l + k)) (numberBasis k) =
98      (Real.sqrt (weight q l k) : ℂ) • numberBasis l := by
99  ext n
100  rw [idlerKraus_adjoint_numberBasis_apply]
101  simp only [numberBasis_eq_single, lp.coeFn_smul, Pi.smul_apply,
102    lp.single_apply, Pi.single_apply, Nat.add_right_cancel_iff, smul_eq_mul]
103  by_cases h : n = l
104  · subst n; simp
105  · have h' : l ≠ n := Ne.symm h
106    simp [h, h']

108theorem idlerKraus_adjoint_numberBasis_zero
109    (q : ℝ) (hq0 : 0 ≤ q) (hq1 : q < 1) {m k : ℕ} (hmk : m < k) :

116/-- A fixed output number selects exactly one input diagonal coefficient in each
117Kraus summand. No positivity or diagonality of the input is assumed. -/
118theorem idlerKraus_diagonal_shift
119    (q : ℝ) (hq0 : 0 ≤ q) (hq1 : q < 1) (A : TraceClass Fock) (l k : ℕ) :
120    ⟪numberBasis k, (krausTerm (idlerKraus q hq0 hq1 (l + k)) A).1
121      (numberBasis k)⟫_ℂ =
122    (weight q l k : ℂ) * ⟪numberBasis l, A.1 (numberBasis l)⟫_ℂ := by
123  rw [krausTerm, inner_sandwichCLM, idlerKraus_adjoint_numberBasis_shift,
124    map_smul, inner_smul_left, inner_smul_right, Complex.conj_ofReal]
125  have hs : (Real.sqrt (weight q l k) : ℂ) * (Real.sqrt (weight q l k) : ℂ) =
126      (weight q l k : ℂ) := by
127    exact_mod_cast Real.mul_self_sqrt (weight_nonneg hq0 hq1 l k)
128  rw [← mul_assoc, hs]

138/-- Exact output number law for an arbitrary trace-class idler, including all
139coherences. The law is derived from the norm-convergent Kraus series. -/
140theorem idlerChannel_diagonal_hasSum
141    (q : ℝ) (hq0 : 0 ≤ q) (hq1 : q < 1) (A : TraceClass Fock) (k : ℕ) :
142    HasSum (fun l => (weight q l k : ℂ) *
143      ⟪numberBasis l, A.1 (numberBasis l)⟫_ℂ)
144      ⟪numberBasis k, ((idlerChannel q hq0 hq1).toLinearMap A).1
145        (numberBasis k)⟫_ℂ := by
146  let f : ℕ → ℂ := fun m => ⟪numberBasis k,
147    (krausTerm (idlerKraus q hq0 hq1 m) A).1 (numberBasis k)⟫_ℂ
148  have hsum : HasSum f ⟪numberBasis k,
149      ((idlerChannel q hq0 hq1).toLinearMap A).1 (numberBasis k)⟫_ℂ :=
150    (idlerChannel_hasSum q hq0 hq1 A).mapL
151      (traceClassMatrixCoefficient (numberBasis k) (numberBasis k))
152  have he : Function.Injective (fun l : ℕ => l + k) := by
153    intro i j hij
154    exact Nat.add_right_cancel hij
155  have hzero : ∀ m ∉ Set.range (fun l : ℕ => l + k), f m = 0 := by
156    intro m hm
157    have hmk : m < k := by
158      by_contra hn
159      exact hm ⟨m - k, Nat.sub_add_cancel (by omega)⟩
160    exact idlerKraus_diagonal_zero q hq0 hq1 A hmk
161  have h := (he.hasSum_iff hzero).mpr hsum
162  simpa only [Function.comp_def, f, idlerKraus_diagonal_shift] using h
