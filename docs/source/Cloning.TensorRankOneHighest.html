14theorem occupancy_eq_length_iff_constant (w : Fin n → Fin (d+1)) (a : Fin (d+1)) :
15    occupancy w a=n ↔ w=(fun _=>a) := by
16  simpa only [Finset.card_univ,Fintype.card_fin,Finset.mem_univ,true_implies,
17    funext_iff,occupancy] using
18    (Finset.card_filter_eq_iff (s := (Finset.univ : Finset (Fin n))) (p := fun i=>w i=a))

20theorem top_occupation_eq_smul_highest (Ω : TensorRegister n (Fin (d+1)))
21    (hΩ : collectiveGenerator n 0 0 Ω=(n:ℂ) • Ω) :
22    Ω=Ω (fun _=>0) • highestTensor n d := by
23  ext w
24  have he := congrArg (fun x : TensorRegister n (Fin (d+1)) => x w) hΩ
25  simp only [collectiveGenerator_diagonal,lp.coeFn_smul,Pi.smul_apply,smul_eq_mul] at he
26  by_cases hw : w=(fun _=>0)
27  · subst w
28    simp [highestTensor,registerBasis_apply,lp.single_apply,Pi.single_apply]
29  · have hne : occupancy w 0 ≠ n := mt (occupancy_eq_length_iff_constant w 0).mp hw
30    have hn : (occupancy w 0 : ℂ) ≠ n := by exact_mod_cast hne
31    have hz : Ω w=0 := (mul_eq_mul_right_iff.mp he).resolve_left hn
32    simp [highestTensor,registerBasis_apply,lp.single_apply,Pi.single_apply,hw,hz]

40theorem cyclicSector_smul_highest (Ω : TensorRegister n (Fin d)) (c : ℂ) (hc : c≠0) :
41    cyclicSector (c • Ω)=cyclicSector Ω := by
42  apply le_antisymm
43  · apply Submodule.span_le.mpr
44    rintro x ⟨w,rfl⟩
45    rw [loweringWord_smul_highest]
46    exact (cyclicSector Ω).smul_mem c (loweringWord_mem_cyclicSector Ω w)
47  · apply Submodule.span_le.mpr
48    rintro x ⟨w,rfl⟩
49    have hh := (cyclicSector (c • Ω)).smul_mem c⁻¹ (loweringWord_mem_cyclicSector (c • Ω) w)
50    simpa only [loweringWord_smul_highest,smul_smul,inv_mul_cancel₀ hc,one_smul] using hh

52theorem cyclicSector_eq_physicalSymmetric_of_top_occupation
53    (Ω : TensorRegister n (Fin (d+1))) (hΩnorm : ‖Ω‖=1)
54    (hΩ : collectiveGenerator n 0 0 Ω=(n:ℂ) • Ω) :
55    cyclicSector Ω=physicalSymmetric n := by
56  have he := top_occupation_eq_smul_highest Ω hΩ
57  have hc : Ω (fun _=>0) ≠ 0 := by
58    intro hc
59    rw [hc,zero_smul] at he
60    rw [he,norm_zero] at hΩnorm
61    exact zero_ne_one hΩnorm
62  calc
63    cyclicSector Ω=cyclicSector (Ω (fun _=>0) • highestTensor n d) := congrArg cyclicSector he
64    _=cyclicSector (highestTensor n d) := cyclicSector_smul_highest _ _ hc
65    _=physicalSymmetric n := cyclicSector_highestTensor_eq_physicalSymmetric n d
