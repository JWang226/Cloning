13theorem wordBoltzmann_canonicalWord (p : Fin d → ℝ) (k : PositiveRoot d → ℕ) :
14    wordBoltzmann p (canonicalWord k) = ∏ r, rootBoltzmann p r ^ k r := by
15  unfold wordBoltzmann
16  rw [Finset.prod_list_map_count]
17  have hs : (∏ r ∈ (canonicalWord k).toFinset,
18      rootBoltzmann p r ^ (canonicalWord k).count r) =
19      ∏ r : PositiveRoot d, rootBoltzmann p r ^ (canonicalWord k).count r := by
20    apply Finset.prod_subset (Finset.subset_univ _)
21    intro r _ hr
22    simp only [List.mem_toFinset] at hr
23    rw [List.count_eq_zero.mpr hr, pow_zero]
24  calc
25    _ = ∏ r : PositiveRoot d, rootBoltzmann p r ^ (canonicalWord k).count r := by
26      simpa only [List.count, beq_eq_decide] using hs
27    _ = _ := by simp only [canonicalWord_count]

35theorem rootPowerProduct_hasSum (p : Fin d → ℝ) (hp : ∀ a, 0 < p a)
36    (hord : StrictAnti p) :
37    HasSum (fun k : PositiveRoot d → ℕ => ∏ r, rootBoltzmann p r ^ k r)
38      (∏ r : PositiveRoot d, (1-rootBoltzmann p r)⁻¹) := by
39  classical
40  let e : Fin (Fintype.card (PositiveRoot d)) ≃ PositiveRoot d := (Fintype.equivFin _).symm
41  let E := Equiv.arrowCongr e (Equiv.refl ℕ)
42  rw [← E.hasSum_iff]
43  have hs := Thermal.hasSum_fin_product (Fintype.card (PositiveRoot d))
44    (fun i k => rootBoltzmann p (e i) ^ k)
45    (fun i => (1-rootBoltzmann p (e i))⁻¹)
46    (fun i k => pow_nonneg (div_pos (hp _) (hp _)).le k)
47    (fun i => hasSum_geometric_of_lt_one (div_pos (hp _) (hp _)).le
48      (rootBoltzmann_lt_one p hp hord (e i)))
49  convert hs using 1
50  · funext k
51    change (∏ r, rootBoltzmann p r ^ k (e.symm r)) = _
52    simpa only [e.symm_apply_apply] using (e.prod_comp (fun r => rootBoltzmann p r ^ k (e.symm r))).symm
53  · exact (e.prod_comp (fun r => (1-rootBoltzmann p r)⁻¹)).symm

55theorem bosonicHeightMass_hasSum (p : Fin d → ℝ) (hp : ∀ a, 0 < p a)
56    (hord : StrictAnti p) :
57    HasSum (bosonicHeightMass p) (∏ r : PositiveRoot d, (1-rootBoltzmann p r)⁻¹) := by
58  have hs := (occupationHeightEquiv (d := d)).hasSum_iff.mpr (rootPowerProduct_hasSum p hp hord)
59  have he := hs.summable.tsum_sigma' (fun _ => (hasSum_fintype _).summable)
60  have hs' := hs.summable.sigma
61  have hf : (fun H : ℕ => ∑' k : ExactHeightOccupation d H,
62      ∏ r, rootBoltzmann p r ^ k.val.val r) = bosonicHeightMass p := by
63    funext H
64    simp only [tsum_fintype, bosonicHeightMass, wordBoltzmann_canonicalWord]
65  change Summable (fun H : ℕ => ∑' k : ExactHeightOccupation d H,
66      ∏ r, rootBoltzmann p r ^ k.val.val r) at hs'
67  rw [hf] at hs'
68  apply hs'.hasSum_iff.mpr
69  rw [← hs.tsum_eq, he]
70  change (∑' H, bosonicHeightMass p H) = ∑' H, ∑' k : ExactHeightOccupation d H,
71    ∏ r, rootBoltzmann p r ^ k.val.val r
72  rw [hf]
