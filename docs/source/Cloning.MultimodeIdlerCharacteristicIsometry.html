35lemma coherentKernel_shift (q : Fin d → ℝ)
36    (hq0 : ∀ i, 0 ≤ q i) (hq1 : ∀ i, q i < 1) (a v w : Fin d → ℂ) :
37    displacementPhase (-(amplifierScale q a)) v *
38      displacementPhase (-(complementaryScale q (star a))) w *
39      coherentKernel q (v-amplifierScale q a) (w-complementaryScale q (star a)) =
40      coherentKernel q v w * displacementPhase (-a) (vacuumScale q v) := by
41  simp only [displacementPhase, coherentKernel_eq_prod, ← Finset.prod_mul_distrib]
42  apply Finset.prod_congr rfl
43  intro i _
44  have hrel : (1-(Real.sqrt (q i))^2)*(Real.sqrt (1-q i))⁻¹=Real.sqrt (1-q i) := by
45    rw [Real.sq_sqrt (hq0 i), ← Real.sq_sqrt (sub_nonneg.mpr (hq1 i).le)]
46    have hs : Real.sqrt (1-q i) ≠ 0 := ne_of_gt (Real.sqrt_pos.mpr (sub_pos.mpr (hq1 i)))
47    simp [pow_two, hs]
48  have hh := scalarCoherentKernel_shift (Real.sqrt (1-q i)) (Real.sqrt (q i))
49    (Real.sqrt (1-q i))⁻¹ hrel (a i) (v i) (w i)
50  dsimp only [amplifierScale, complementaryScale, vacuumScale, Pi.neg_apply,
51    Pi.sub_apply, Pi.star_apply]
52  calc
53    _ = (Real.sqrt (1-q i):ℂ) *
54      (Cloning.ComplexCoherent.displacementPhase (-(((Real.sqrt (1-q i))⁻¹:ℝ):ℂ)*a i) (v i) *
55      Cloning.ComplexCoherent.displacementPhase
56        (-((Real.sqrt (q i)*(Real.sqrt (1-q i))⁻¹:ℝ):ℂ)*star (a i)) (w i) *
57      scalarCoherentKernel (Real.sqrt (q i))
58        (v i-(((Real.sqrt (1-q i))⁻¹:ℝ):ℂ)*a i)
59        (w i-((Real.sqrt (q i)*(Real.sqrt (1-q i))⁻¹:ℝ):ℂ)*star (a i))) := by
60          simp only [neg_mul]
61          ring
62    _ = _ := by
63      simp only [neg_mul] at hh ⊢
64      rw [hh]
65      ring

89lemma appendedIsometry_adjoint_intertwines (q : Fin d → ℝ)
90    (hq0 : ∀ i, 0 ≤ q i) (hq1 : ∀ i, q i < 1) (a : Fin d → ℂ) :
91    (appendedIsometry q hq0 hq1).toContinuousLinearMap.adjoint.comp
92      (displacement (-Fin.append (amplifierScale q a) (complementaryScale q (star a)))) =
93    (displacement (-a)).comp (appendedIsometry q hq0 hq1).toContinuousLinearMap.adjoint := by
94  apply coherent_ext
95  intro z
96  let v : Fin d → ℂ := fun i => z (Fin.castAdd d i)
97  let w : Fin d → ℂ := fun i => z (Fin.natAdd d i)
98  have hz : Fin.append v w = z := Fin.append_castAdd_natAdd
99  rw [← hz]
100  have hneg : -Fin.append (amplifierScale q a) (complementaryScale q (star a)) =
101      Fin.append (-(amplifierScale q a)) (-(complementaryScale q (star a))) := by
102    ext i
103    refine Fin.addCases ?_ ?_ i <;> intro j <;>
104      simp only [Pi.neg_apply, Fin.append_left, Fin.append_right]
105  have hadd : Fin.append (-(amplifierScale q a)) (-(complementaryScale q (star a)))+

108    congr 1 <;> abel

113/-- The literal isometry intertwines input Weyl displacements with the two
114output Weyl displacements, including the positive conjugated complementary shift. -/
115theorem appendedIsometry_intertwines (q : Fin d → ℝ)
116    (hq0 : ∀ i, 0 ≤ q i) (hq1 : ∀ i, q i < 1) (a : Fin d → ℂ) :
117    (appendedIsometry q hq0 hq1).toContinuousLinearMap.comp (displacement a) =
118      (displacement (Fin.append (amplifierScale q a) (complementaryScale q (star a)))).comp
119        (appendedIsometry q hq0 hq1).toContinuousLinearMap := by
120  have hh := congrArg (fun T => ContinuousLinearMap.adjoint T)
121    (appendedIsometry_adjoint_intertwines q hq0 hq1 a)
122  simpa only [ContinuousLinearMap.adjoint_comp, ContinuousLinearMap.adjoint_adjoint,
123    displacement_adjoint, neg_neg] using hh.symm
