87def phaseEquiv (c s : Fin d → ℝ) (hcs : ∀i,c i^2-s i^2=1) :
88    (Fin (d+d) → ℂ) ≃L[ℝ] (Fin (d+d) → ℂ) where
89  toFun := transform c s
90  invFun := transform c (-s)
91  left_inv := transform_inverse c s hcs
92  right_inv := by
93    intro z
94    have hh : ∀i,c i^2-(-s i)^2=1 := by simpa using hcs
95    simpa only [neg_neg] using transform_inverse c (-s) hh z
96  map_add' := transform_add c s
97  map_smul' := transform_smul c s
98  continuous_toFun := continuous_transform c s
99  continuous_invFun := continuous_transform c (-s)

114/-- The Weyl cocycle is exactly preserved, including the complex phase. -/
115theorem displacementPhase_transform (c s : Fin d → ℝ) (hcs : ∀i,c i^2-s i^2=1)
116    (a b : Fin (d+d) → ℂ) :
117    displacementPhase (transform c s a) (transform c s b)=displacementPhase a b := by
118  unfold displacementPhase
119  rw [Fin.prod_univ_add, Fin.prod_univ_add, ← Finset.prod_mul_distrib, ← Finset.prod_mul_distrib]

121  intro i _
122  simp only [transform_left, transform_right]
123  exact phase_pair (c i) (s i) (hcs i) _ _ _ _

125/-- The pulled-back action is an actual regular Weyl representation on doubled

127def regularWeyl (c s : Fin d → ℝ) (hcs : ∀i,c i^2-s i^2=1) :
128    RegularWeyl (d+d) (Fock (d+d)) where
129  toIsometry z := weylUnitary (transform c s z)
130  zero_apply := by intro v; simp
131  mul_apply := by
132    intro a b v
133    have hh := (fockRegularWeyl (d+d)).mul_apply (transform c s a) (transform c s b) v
134    rw [displacementPhase_transform c s hcs, ← transform_add] at hh
135    exact hh
