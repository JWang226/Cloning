24/-- Simultaneous slot permutations in the grouped system/environment register. -/
25def groupedSymmetric (n : ℕ) : Submodule ℂ (Register ((Fin n → A) × (Fin n → A))) where
26  carrier := {x | ∀ a b (σ : Equiv.Perm (Fin n)), x (a ∘ σ, b ∘ σ) = x (a, b)}
27  zero_mem' := by intro a b σ; rfl
28  add_mem' := by intro x y hx hy a b σ; exact congrArg₂ (· + ·) (hx a b σ) (hy a b σ)
29  smul_mem' := by intro c x hx a b σ; exact congrArg (c • ·) (hx a b σ)

122/-- A positive matrix supported by an orthogonal projector has supported square root. -/
123theorem sqrt_supported {I : Type*} [Fintype I] [DecidableEq I]
124    (P R : Matrix I I ℂ) (hP : P * P = P) (hPs : P.conjTranspose = P)
125    (hR : R.PosSemidef) (hPRP : P * R * P = R) :
126    P * CFC.sqrt R = CFC.sqrt R ∧ CFC.sqrt R * P = CFC.sqrt R := by
127  have hPR : P * R = R := by
128    calc
129      P * R = P * (P * R * P) := by rw [hPRP]
130      _ = P * R * P := by simp only [← Matrix.mul_assoc, hP]
131      _ = R := hPRP
132  have hRP : R * P = R := by
133    simpa only [Matrix.conjTranspose_mul, hPs, hR.isHermitian.eq] using
134      congrArg Matrix.conjTranspose hPR
135  have hzero : (CFC.sqrt R * (1 - P)).conjTranspose * (CFC.sqrt R * (1 - P)) = 0 := by
136    rw [Matrix.conjTranspose_mul, Matrix.conjTranspose_sub, Matrix.conjTranspose_one,
137      hPs, Cloning.MatrixFidelity.sqrt_conjTranspose]
138    calc
139      (1 - P) * CFC.sqrt R * (CFC.sqrt R * (1 - P)) =
140          (1 - P) * (CFC.sqrt R * CFC.sqrt R) * (1 - P) := by simp only [Matrix.mul_assoc]
141      _ = (1 - P) * R * (1 - P) := by rw [Cloning.MatrixFidelity.sqrt_mul_self hR]
142      _ = 0 := by rw [Matrix.sub_mul, Matrix.one_mul, hPR, sub_self, Matrix.zero_mul]
143  have hz : CFC.sqrt R * (1 - P) = 0 :=
144    (CStarRing.star_mul_self_eq_zero_iff _).mp hzero
145  have hright : CFC.sqrt R * P = CFC.sqrt R := by
146    rw [Matrix.mul_sub, Matrix.mul_one] at hz
147    exact (sub_eq_zero.mp hz).symm
148  refine ⟨?_, hright⟩
149  simpa only [Matrix.conjTranspose_mul, hPs, Cloning.MatrixFidelity.sqrt_conjTranspose] using
150    congrArg Matrix.conjTranspose hright
