17/-- Scalar dominated convergence works even when the Banach target varies with
18sample size. The error bound is the trace distance bound for normalized states. -/
19theorem norm_integral_sub_tendsto_of_pointwise
20    {E : ℕ → Type*} [∀ n, NormedAddCommGroup (E n)] [∀ n, NormedSpace ℝ (E n)]
21    [∀ n, CompleteSpace (E n)] (f g : ∀ n, Z → E n)
22    (hf : ∀ n, Integrable (f n) ν) (hg : ∀ n, Integrable (g n) ν)
23    (hb : ∀ n, ∀ᵐ z ∂ν, ‖f n z - g n z‖ ≤ 2)
24    (hpoint : ∀ᵐ z ∂ν, Tendsto (fun n => ‖f n z - g n z‖) atTop (𝓝 0)) :
25    Tendsto (fun n => ‖(∫ z, f n z ∂ν) - ∫ z, g n z ∂ν‖) atTop (𝓝 0) := by
26  have hi : Tendsto (fun n => ∫ z, ‖f n z - g n z‖ ∂ν) atTop (𝓝 0) := by
27    have h := tendsto_integral_of_dominated_convergence (fun _ : Z => (2 : ℝ))
28      (fun n => ((hf n).sub (hg n)).norm.aestronglyMeasurable) (integrable_const _)
29      (fun n => (hb n).mono fun z hz => by simpa only [norm_norm] using hz) hpoint
30    simpa using h
31  apply squeeze_zero (fun n => norm_nonneg _) _ hi
32  intro n
33  rw [← integral_sub (hf n) (hg n)]
34  exact norm_integral_le_integral_norm _

41/-- Integrate forward pointwise approximation using only a finite mixing measure.
42All physical and comparison integrands are genuine normalized positive states. -/
43theorem forward_mixture_tendsto
44    (T : ∀ n, QuantumToHybrid (Q n) K μ)
45    (ρ : ∀ n, Z → PositiveTraceClass (Q n)) (B : Z → PositiveL1 K μ)
46    (hρ : ∀ n z, ‖(ρ n z).1‖ = 1) (hB : ∀ z, ‖(B z).1‖ = 1)
47    (hiρ : ∀ n, Integrable (fun z => (ρ n z).1) ν)
48    (hiB : Integrable (fun z => (B z).1) ν)
49    (hpoint : ∀ᵐ z ∂ν, Tendsto (fun n => ‖(T n).map (ρ n z).1 - (B z).1‖)
50      atTop (𝓝 0)) :
51    Tendsto (fun n => ‖(T n).map (∫ z, (ρ n z).1 ∂ν) - ∫ z, (B z).1 ∂ν‖)
52      atTop (𝓝 0) := by
53  have h := norm_integral_sub_tendsto_of_pointwise
54    (fun n z => (T n).map (ρ n z).1) (fun _ z => (B z).1)
55    (fun n => (T n).map.integrable_comp (hiρ n)) (fun _ => hiB)
56    (fun n => Eventually.of_forall fun z => by
57      calc
58        _ ≤ ‖(T n).map (ρ n z).1‖ + ‖(B z).1‖ := norm_sub_le _ _
59        _ = 2 := by rw [(T n).norm_map_of_nonneg _ (ρ n z).2, hρ, hB]; norm_num)
60    hpoint
61  simpa only [(T _).map.integral_comp_comm (hiρ _)] using h

63/-- The reverse mixture estimate permits the physical Hilbert space to change
64with n; dominated convergence is applied only to the scalar trace error. -/
65theorem reverse_mixture_tendsto
66    (S : ∀ n, HybridToQuantum K (Q n) μ)
67    (ρ : ∀ n, Z → PositiveTraceClass (Q n)) (B : Z → PositiveL1 K μ)
68    (hρ : ∀ n z, ‖(ρ n z).1‖ = 1) (hB : ∀ z, ‖(B z).1‖ = 1)
69    (hiρ : ∀ n, Integrable (fun z => (ρ n z).1) ν)
70    (hiB : Integrable (fun z => (B z).1) ν)
71    (hpoint : ∀ᵐ z ∂ν, Tendsto (fun n => ‖(S n).map (B z).1 - (ρ n z).1‖)
72      atTop (𝓝 0)) :
73    Tendsto (fun n => ‖(S n).map (∫ z, (B z).1 ∂ν) - ∫ z, (ρ n z).1 ∂ν‖)
74      atTop (𝓝 0) := by
75  have h := norm_integral_sub_tendsto_of_pointwise
76    (fun n z => (S n).map (B z).1) (fun n z => (ρ n z).1)
77    (fun n => (S n).map.integrable_comp hiB) hiρ
78    (fun n => Eventually.of_forall fun z => by
79      calc
80        _ ≤ ‖(S n).map (B z).1‖ + ‖(ρ n z).1‖ := norm_sub_le _ _
81        _ = 2 := by rw [(S n).norm_map_of_nonneg _ (B z).2, hB, hρ]; norm_num)
82    hpoint
83  simpa only [(S _).map.integral_comp_comm hiB] using h

85/-- A physical output close to the integrated local model has the comparison
86fidelity limit once the two pointwise LAN directions and seed errors are given.
87No uniform approximation over the unbounded mixing support is assumed. -/
88theorem fidelity_tendsto_of_mixed_mixture
89    (T : ∀ n, QuantumToHybrid (Q n) K μ) (S : ∀ n, HybridToQuantum K (Q n) μ)
90    (ρ σ : ∀ n, PositiveTraceClass (Q n)) (A B : PositiveL1 K μ)
91    (R : ∀ n, Z → PositiveTraceClass (Q n)) (G : Z → PositiveL1 K μ)
92    (hρ : ∀ n, ‖(ρ n).1‖ = 1) (hσ : ∀ n, ‖(σ n).1‖ = 1)
93    (hA : ‖A.1‖ = 1) (hB : ‖B.1‖ = 1)
94    (hR : ∀ n z, ‖(R n z).1‖ = 1) (hG : ∀ z, ‖(G z).1‖ = 1)
95    (hiR : ∀ n, Integrable (fun z => (R n z).1) ν)
96    (hiG : Integrable (fun z => (G z).1) ν)
97    (hmixture : B.1 = ∫ z, (G z).1 ∂ν)
98    (hphysical : Tendsto (fun n => ‖(σ n).1 - ∫ z, (R n z).1 ∂ν‖) atTop (𝓝 0))
99    (hTA : Tendsto (fun n => ‖(T n).map (ρ n).1 - A.1‖) atTop (𝓝 0))
100    (hSA : Tendsto (fun n => ‖(S n).map A.1 - (ρ n).1‖) atTop (𝓝 0))
101    (hTG : ∀ᵐ z ∂ν, Tendsto (fun n => ‖(T n).map (R n z).1 - (G z).1‖)
102      atTop (𝓝 0))
103    (hSG : ∀ᵐ z ∂ν, Tendsto (fun n => ‖(S n).map (G z).1 - (R n z).1‖)
104      atTop (𝓝 0)) :
105    Tendsto (fun n => (ρ n).rootFidelity (σ n)) atTop (𝓝 (A.rootFidelity B)) := by
106  have hforward := forward_mixture_tendsto T R G hR hG hiR hiG hTG
107  have hreverse := reverse_mixture_tendsto S R G hR hG hiR hiG hSG
108  rw [← hmixture] at hforward hreverse
109  apply fidelity_tendsto_of_twoWay_mixed_approximation T S ρ σ A B hρ hσ hA hB hTA
110  · apply squeeze_zero (fun n => norm_nonneg _) _
111      (by simpa using (hphysical.const_mul 2).add hforward)
112    intro n
113    calc
114      _ ≤ ‖(T n).map (σ n).1 - (T n).map (∫ z, (R n z).1 ∂ν)‖ +
115          ‖(T n).map (∫ z, (R n z).1 ∂ν) - B.1‖ := norm_sub_le_norm_sub_add_norm_sub _ _ _
116      _ ≤ 2 * ‖(σ n).1 - ∫ z, (R n z).1 ∂ν‖ +
117          ‖(T n).map (∫ z, (R n z).1 ∂ν) - B.1‖ := by
118        apply add_le_add _ le_rfl
119        rw [← map_sub]
120        exact (T n).norm_le_two _
121  · exact hSA
122  · apply squeeze_zero (fun n => norm_nonneg _) _ (by simpa using hreverse.add hphysical)
123    intro n
124    calc
125      _ ≤ ‖(S n).map B.1 - ∫ z, (R n z).1 ∂ν‖ +
126          ‖(∫ z, (R n z).1 ∂ν) - (σ n).1‖ := norm_sub_le_norm_sub_add_norm_sub _ _ _
127      _ = _ := by rw [norm_sub_rev (∫ z, (R n z).1 ∂ν) (σ n).1]
