17/-- A finite-multimode Fock vector orthogonal to every coherent vector is zero. -/
18theorem coherentVector_total {d : ℕ} (v : Fock d)
19    (h : ∀ z : Fin d → ℂ, ⟪coherentVector z, v⟫_ℂ = 0) : v = 0 := by
20  have hv (z : Fin d → ℂ) : ⟪v, coherentVector z⟫_ℂ = 0 := inner_eq_zero_symm.mp (h z)
21  have hz (n : Fin d → ℕ) : ⟪v,
22      (vectorMixture (numberBasis d) (fun k ↦ ∏ i, Thermal.geometric (1 / 2) (k i))).1
23        (numberBasis d n)⟫_ℂ = 0 := by
24    have hi := (traceClassMatrixCoefficient v (numberBasis d n)).integral_comp_comm
25      (integrable_coherentProjector_gaussian (s := fun _ : Fin d ↦ 1)
26        (fun _ ↦ by norm_num))
27    rw [integral_coherentProjector_gaussian_eq_productThermal
28      (s := fun _ : Fin d ↦ 1) (fun _ ↦ by norm_num)] at hi
29    norm_num only at hi
30    change traceClassMatrixCoefficient v (numberBasis d n)
31      (vectorMixture (numberBasis d) (fun k ↦ ∏ i, Thermal.geometric (1 / 2) (k i))) = 0
32    rw [← hi]
33    have he (z : Fin d → ℂ) :
34        traceClassMatrixCoefficient v (numberBasis d n) (coherentProjector z) = 0 := by
35      change ⟪v, (InnerProductSpace.rankOne ℂ (coherentVector z) (coherentVector z))
36        (numberBasis d n)⟫_ℂ = 0
37      rw [InnerProductSpace.rankOne_apply, inner_smul_right, hv, mul_zero]
38    simp only [he, integral_zero]
39  apply (numberBasis d).repr.injective
40  ext n
41  have hn := hz n
42  rw [InfiniteOccupationStates.vectorMixture_apply_basis (numberBasis d) _
43    (Thermal.multimode_geometric_hasSum
44      (q := fun _ : Fin d ↦ 1 / 2) (fun _ ↦ by norm_num) (fun _ ↦ by norm_num)).summable n,
45    inner_smul_right] at hn
46  have hp : ((∏ i : Fin d, Thermal.geometric (1 / 2) (n i) : ℝ) : ℂ) ≠ 0 := by
47    apply Complex.ofReal_ne_zero.mpr
48    apply ne_of_gt
49    apply Finset.prod_pos
50    intro i _
51    unfold Thermal.geometric
52    positivity
53  have hinner := (mul_eq_zero.mp hn).resolve_left hp
54  simpa only [(numberBasis d).repr_apply_apply, map_zero, lp.coeFn_zero, Pi.zero_apply] using
55    (inner_eq_zero_symm.mp hinner)

57/-- Multimode coherent vectors span a dense complex subspace. -/
58theorem coherentVector_dense_span (d : ℕ) :
59    (Submodule.span ℂ (Set.range (fun z : Fin d → ℂ ↦ coherentVector z))).topologicalClosure = ⊤ := by
60  rw [Submodule.topologicalClosure_eq_top_iff, Submodule.eq_bot_iff]
61  intro v hv
62  apply coherentVector_total
63  intro z
64  exact Submodule.inner_right_of_mem_orthogonal
65    (Submodule.subset_span (Set.mem_range_self z)) hv
