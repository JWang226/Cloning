105private theorem inputSlots_append_iff {n r d : ℕ} (w : Word n d) (z : Word r d) (a : Fin d) :
106    inputSlots n (n + r) (Nat.le_add_right n r) ⊆ letterSlots (Fin.append w z) a ↔
107      w = fun _ => a := by
108  constructor
109  · intro h
110    funext i
111    have hi : Fin.castAdd r i ∈ inputSlots n (n + r) (Nat.le_add_right n r) := by
112      rw [mem_inputSlots]
113      exact i.isLt
114    have hv := mem_letterSlots _ _ _ |>.mp (h hi)
115    simpa using hv
116  · intro h i hi
117    rw [mem_inputSlots] at hi
118    let j : Fin n := ⟨i.val, hi⟩
119    have he : i = Fin.castAdd r j := Fin.ext rfl
120    rw [he, mem_letterSlots]
121    simp [h]
