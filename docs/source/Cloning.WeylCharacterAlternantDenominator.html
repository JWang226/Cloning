12theorem prod_positiveRoot {d : ℕ} {M : Type*} [CommMonoid M] (f : Fin d → Fin d → M) :
13    (∏ r : PositiveRoot d, f r.val.1 r.val.2) =
14      ∏ i : Fin d, ∏ j ∈ Finset.Ioi i, f i j := by
15  rw [← Finset.prod_subtype (Finset.univ.filter (fun r : Fin d × Fin d ↦ r.1 < r.2))
16    (by simp) (fun r ↦ f r.1 r.2)]
17  rw [Finset.prod_filter, Fintype.prod_prod_type]
18  apply Finset.prod_congr rfl
19  intro i hi
20  rw [← Finset.prod_filter]
21  congr 1
22  ext j
23  simp

25def denominator (d : ℕ) : MvPolynomial (Fin d) ℂ :=

28theorem denominator_eq_scalar_vandermonde (d : ℕ) :
29    denominator d = ((-1 : ℂ)^Fintype.card (PositiveRoot d)) •
30      (Matrix.vandermonde (fun i : Fin d ↦ (X i : MvPolynomial (Fin d) ℂ))).det := by
31  rw [Matrix.det_vandermonde, ← prod_positiveRoot]
32  unfold denominator
33  calc
34    _ = ∏ r : PositiveRoot d, (-1 : ℂ) • (X r.val.2 - X r.val.1) := by
35      apply Finset.prod_congr rfl
36      intro r hr
37      simp
38    _ = _ := by rw [Finset.prod_smul]; simp
