5lemma. The finite physical Schur law first receives exponent -N epsilon^2/2;

16theorem tensorYoungPMF_tail_half_le (p : Fin d→ℝ) (hp : ∀a,0≤p a)
17    (hs : ∑a,p a=1) (horder : Antitone p)
18    (ε : ℝ) (hε : 0≤ε) (hε1 : ε≤1) :
19    tailProbability (tensorYoungPMF n d p hp hs) p ε≤
20      ((n*d+1:ℕ):ℝ)^Fintype.card (PositiveRoot d)*
21        (2*d*Real.exp (-(n:ℝ)*ε^2/2)) := by
22  have he := tensorYoungPMF_sum_mul (n:=n) p hp hs (fun mu=>
23    if ∃i,(n:ℝ)*ε≤|((mu i).val:ℝ)-n*p i| then 1 else 0)
24  simp only [mul_ite,mul_one,mul_zero,wordShape_val] at he
25  change (∑mu : Shape d n,
26    if ∃i,(n:ℝ)*ε≤|((mu i).val:ℝ)-n*p i| then
27      (tensorYoungPMF n d p hp hs mu).toReal else 0)≤_
28  rw [he]
29  calc
30    _≤∑w : Fin n→Fin d,
31        ((n*d+1:ℕ):ℝ)^Fintype.card (PositiveRoot d)*
32          (if ∃i,(n:ℝ)*ε≤|(rowCount w i:ℝ)-n*p i| then wordWeight p w else 0) := by
33      apply Finset.sum_le_sum
34      intro w _
35      split_ifs
36      · exact physicalStandardWordWeight_le p hp horder w
37      · simp
38    _=_ := (Finset.mul_sum ..).symm
39    _≤_ := mul_le_mul_of_nonneg_left
40      (max_rowCount_tail_half_le p hp hs ε hε hε1) (by positivity)

42/-- A polynomial envelope with a whole additional exp(-N^(1/3)/4) factor
43left available. This is stronger than the old vanishing-tail envelope. -/
44theorem tensorYoungPMF_tail_shrinking_half_le (n : ℕ) (hn : 1≤n)
45    (p : Fin d→ℝ) (hp : ∀a,0≤p a) (hs : ∑a,p a=1) (horder : Antitone p) :
46    tailProbability (tensorYoungPMF n d p hp hs) p (shrinkingRadius n)≤
47      (((d:ℝ)+1)^Fintype.card (PositiveRoot d)*
48        (((n:ℝ)+1)^Fintype.card (PositiveRoot d)*concentrationEnvelope d n))*
49      Real.exp (-((n:ℝ)^(1/3:ℝ))/4) := by
50  have ht := tensorYoungPMF_tail_half_le (n:=n) p hp hs horder (shrinkingRadius n)
51    (shrinkingRadius_nonneg n) (shrinkingRadius_le_one n hn)
52  have he : -(n:ℝ)*shrinkingRadius n^2/2=-((n:ℝ)^(1/3:ℝ))/2 := by
53    rw [neg_mul,sample_mul_radius_sq n (by omega)]
54  rw [he] at ht
55  let s := Fintype.card (PositiveRoot d)
56  let e := Real.exp (-((n:ℝ)^(1/3:ℝ))/4)
57  have hsplit : Real.exp (-((n:ℝ)^(1/3:ℝ))/2)=e*e := by
58    rw [← Real.exp_add]
59    congr 1
60    ring
61  have hpoly : ((n*d+1:ℕ):ℝ)^s≤((d:ℝ)+1)^s*((n:ℝ)+1)^s := by
62    rw [← mul_pow]
63    apply pow_le_pow_left₀ (by positivity)
64    push_cast
65    nlinarith [(Nat.cast_nonneg n : (0:ℝ)≤n),(Nat.cast_nonneg d : (0:ℝ)≤d)]
66  have hone : (1:ℝ)≤((n:ℝ)+1)^(d*d) :=
67    one_le_pow₀ (by linarith [(Nat.cast_nonneg n : (0:ℝ)≤n)])
68  have hepos : 0≤e := (Real.exp_pos _).le
69  have hbase : ((n*d+1:ℕ):ℝ)^s*(2*d*e)≤
70      ((d:ℝ)+1)^s*((n:ℝ)+1)^s*(((n:ℝ)+1)^(d*d)*(2*d*e)) := by
71    calc
72      _≤(((d:ℝ)+1)^s*((n:ℝ)+1)^s)*(2*d*e) :=
73        mul_le_mul_of_nonneg_right hpoly (by positivity)
74      _≤_ := by
75        have h := mul_le_mul_of_nonneg_left hone
76          (show 0≤(((d:ℝ)+1)^s*((n:ℝ)+1)^s)*(2*d*e) by positivity)
77        nlinarith
78  have hmul := mul_le_mul_of_nonneg_right hbase hepos
79  rw [hsplit] at ht
80  apply ht.trans
81  convert hmul using 1 <;> dsimp [s,e,concentrationEnvelope] <;> ring

83/-- Exact manuscript rate, uniformly over the closed ordered probability
84simplex. Its sample threshold depends only on the ambient dimension d. -/
85theorem tensorYoungPMF_tail_shrinking_uniform_exp (d : ℕ) :
86    ∀ᶠ n in atTop,∀(p : Fin d→ℝ) (hp : ∀a,0≤p a) (hs : ∑a,p a=1),
87      Antitone p →
88      tailProbability (tensorYoungPMF n d p hp hs) p (shrinkingRadius n)≤
89        Real.exp (-((n:ℝ)^(1/3:ℝ))/4) := by
90  have ht := (polynomial_concentrationEnvelope_tendsto_zero d
91    (Fintype.card (PositiveRoot d))).const_mul (((d:ℝ)+1)^Fintype.card (PositiveRoot d))
92  simp only [mul_zero] at ht
93  have hsmall := ht.eventually (Iio_mem_nhds (by norm_num : (0:ℝ)<1))
94  filter_upwards [hsmall,eventually_ge_atTop 1] with n hsmall hn p hp hs horder
95  have h := tensorYoungPMF_tail_shrinking_half_le n hn p hp hs horder
96  exact h.trans (by
97    have hmul := mul_le_mul_of_nonneg_right hsmall.le
98      (Real.exp_pos (-((n:ℝ)^(1/3:ℝ))/4)).le
99    simpa only [one_mul] using hmul)
