15theorem matrixRegister_inner (J : Matrix B A ℂ) (hJ : Jᴴ * J = 1)
16    (x y : Register A) : ⟪matrixRegister J x, matrixRegister J y⟫_ℂ = ⟪x,y⟫_ℂ := by

33def wernerScale (n m s : ℕ) : ℝ := (Nat.choose (n+s) s : ℝ) / Nat.choose (m+s) s

35theorem wernerScale_pos (n m s : ℕ) : 0 < wernerScale n m s := by

40/-- Exact support mass as the quotient of ambient and internal symmetric

42def supportFactor (n m small large : ℕ) : ℝ :=
43  wernerScale n m large / wernerScale n m small

45theorem supportFactor_pos (n m small large : ℕ) : 0 < supportFactor n m small large :=

48/-- The uncompressed output is the actual normalized physical pure-state
49Werner operator, and the compressed output is the smaller physical Werner
50operator times the exact dimension ratio. -/
51theorem pureWernerOutput_compression {L small large : ℕ}
52    (hA : Fintype.card A = small+1) (hB : Fintype.card B = large+1)
53    (J : Matrix B A ℂ) (hJ : Jᴴ * J = 1)
54    (ψ : Register A) (hψ : ‖ψ‖ = 1) (S : Finset (Fin L)) :
55    conjugationLinearMap (tensorMap L J).adjoint
56      (pureWernerOutput hB (matrixRegister J ψ) ((matrixRegister_norm J hJ ψ).trans hψ) S) =
57      (supportFactor S.card L small large : ℂ) • pureWernerOutput hA ψ hψ S := by
58  apply Subtype.ext
59  change operatorConjugation (tensorMap L J).adjoint
60    (pureWernerOperator large (matrixRegister J ψ) S) =
61    (supportFactor S.card L small large : ℂ) • pureWernerOperator small ψ S
62  simp only [pureWernerOperator, map_smul]
63  have he : operatorConjugation (tensorMap L J).adjoint
64      (physicalProjector L * pureSlotsOperator (matrixRegister J ψ) S * physicalProjector L) =
65      physicalProjector L * pureSlotsOperator ψ S * physicalProjector L := by
66    have hh := wernerSandwich_compression J hJ ψ S
67    simpa only [operatorConjugation, ContinuousLinearMap.adjoint_adjoint,
68      ContinuousLinearMap.comp_assoc] using hh
69  rw [he, smul_smul]
70  congr 1
71  change (wernerScale S.card L large : ℂ) =
72    (supportFactor S.card L small large : ℂ) * (wernerScale S.card L small : ℂ)
73  rw [← Complex.ofReal_mul, supportFactor,
74    div_mul_cancel₀ _ (wernerScale_pos S.card L small).ne']
