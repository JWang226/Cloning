14/-- Invariance under all matrix units also gives invariance under their
15adjoints, so the actual orthogonal projection commutes with every generator. -/
16theorem invariant_starProjection_commutes
17    (W : Submodule ℂ (TensorRegister n (Fin d)))
18    (hW : ∀ a b x, x ∈ W → collectiveGenerator n a b x ∈ W)
19    (a b : Fin d) (x : TensorRegister n (Fin d)) :
20    W.starProjection (collectiveGenerator n a b x) =
21      collectiveGenerator n a b (W.starProjection x) := by
22  apply Submodule.eq_starProjection_of_mem_orthogonal
23  · exact hW a b _ (W.starProjection_apply_mem x)
24  · rw [← map_sub]
25    apply (W.mem_orthogonal _).mpr
26    intro y hy
27    rw [← ContinuousLinearMap.adjoint_inner_left, collectiveGenerator_adjoint]
28    exact W.inner_right_of_mem_orthogonal (hW b a y hy)
29      (Submodule.sub_starProjection_mem_orthogonal x)

31/-- A nonzero normalized highest tensor generates an irreducible physical
32matrix-unit module. Every invariant subspace is zero or the full cyclic sector. -/
33theorem cyclicSector_irreducible
34    (Ω : TensorRegister n (Fin d)) (hΩ : ‖Ω‖ = 1)
35    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
36    (W : Submodule ℂ (TensorRegister n (Fin d))) (hWS : W ≤ cyclicSector Ω)
37    (hW : ∀ a b x, x ∈ W → collectiveGenerator n a b x ∈ W) :
38    W = ⊥ ∨ W = cyclicSector Ω := by
39  obtain ⟨c, hc⟩ := cyclicSector_commutant_scalar Ω hΩ hraise W.starProjection.toLinearMap
40    (fun x _ => hWS (W.starProjection_apply_mem x))
41    (fun a b x _ => invariant_starProjection_commutes W hW a b x)
42  by_cases hzero : W = ⊥
43  · exact Or.inl hzero
44  · right
45    obtain ⟨x, hxW, hx0⟩ := W.ne_bot_iff.mp hzero
46    have hcx : c • x = x := (hc x (hWS hxW)).symm.trans
47      (Submodule.starProjection_eq_self_iff.mpr hxW)
48    have hc1 : c = 1 := by
49      have he : (c - 1) • x = 0 := by rw [sub_smul, one_smul, hcx, sub_self]
50      exact sub_eq_zero.mp ((smul_eq_zero.mp he).resolve_right hx0)
51    apply le_antisymm hWS
52    intro y hy
53    have he : W.starProjection y = y := by simpa only [hc1, one_smul] using hc y hy
54    exact Submodule.starProjection_eq_self_iff.mp he

56/-- The constructed highest tensor of every finite partition generates an
57irreducible actual tensor sector, including the one-dimensional empty tensor. -/
58theorem partition_cyclicSector_irreducible {d : ℕ}
59    (mu : Fin d → ℕ) (hmu : Antitone mu)
60    (W : Submodule ℂ (TensorRegister (∑ i, mu i) (Fin d)))
61    (hWS : W ≤ cyclicSector (partitionHighestTensor mu hmu))
62    (hW : ∀ a b x, x ∈ W → collectiveGenerator (∑ i, mu i) a b x ∈ W) :
63    W = ⊥ ∨ W = cyclicSector (partitionHighestTensor mu hmu) :=
