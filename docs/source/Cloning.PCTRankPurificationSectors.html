18    (H K : PhysicalHighestTensor n d) (hHK : ⟪H.vector,K.vector⟫_ℂ=0) : K.sector ≤ H.sectorᗮ := by
19  apply cyclicSector_le_of_mem_of_invariant _
20    (fun a b x hx => generatorInvariant_orthogonal H.sector

23  intro x hx
24  induction hx using Submodule.span_induction with
25  | mem x hx =>
26    obtain ⟨w,rfl⟩ := hx

28    | nil => exact hHK

34def PhysicalHighestTensor.coordinateLift (H : PhysicalHighestTensor n r) (k : ℕ) :

66theorem PhysicalHighestTensor.coordinateLift_canonicalEmbedding
67    (H : PhysicalHighestTensor n r) (k : ℕ) (x : H.CanonicalSector) :
68    (H.coordinateLift k).canonicalEmbedding (rankSectorEmbedding H.weight H.weight_antitone k x) =
69      coordinateTensorEmbedding n r k (H.canonicalEmbedding x) := by
70  let f := (H.coordinateLift k).canonicalEmbedding.toLinearMap.comp
71    (rankSectorEmbedding H.weight H.weight_antitone k).toLinearMap
72  let g := (coordinateTensorEmbedding n r k).toLinearMap.comp H.canonicalEmbedding.toLinearMap

84      coordinateTensorEmbedding n r k (H.canonicalEmbedding _)
