22/-- All classical coordinates, and real and imaginary quantum coordinates,

24def phaseSpaceBox (k s : ℕ) (L : ℝ) : Set (PhaseSpace k s) :=
25  {ξ | (∀ i, |ξ.1 i| ≤ L) ∧ (∀ j, |(ξ.2 j).re| ≤ L ∧ |(ξ.2 j).im| ≤ L)}

89def flatBoxDensity (k s : ℕ) : PhaseDensity k s where

121/-- Exact Jacobian for joint real classical and complex quantum coordinates. -/
122theorem integral_unitBox_smul {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]
123    {L : ℝ} (hL : 0 < L) (f : PhaseSpace k s → E) :
124    (∫ z in phaseSpaceUnitBox k s, f (L • z)) =
125      (L ^ Module.finrank ℝ (PhaseSpace k s))⁻¹ •
126        ∫ z in phaseSpaceBox k s L, f z := by
127  classical
128  have he : (fun z => (phaseSpaceBox k s L).indicator f (L • z)) =
129      (phaseSpaceUnitBox k s).indicator (fun z => f (L • z)) := by
130    funext z
131    by_cases hz : z ∈ phaseSpaceUnitBox k s
132    · simp only [Set.indicator_of_mem hz,
133        Set.indicator_of_mem ((smul_mem_phaseSpaceBox_iff hL z).mpr hz)]
134    · simp only [Set.indicator_of_notMem hz,
135        Set.indicator_of_notMem (mt (smul_mem_phaseSpaceBox_iff hL z).mp hz)]
136  have h := Measure.integral_comp_smul_of_nonneg volume ((phaseSpaceBox k s L).indicator f) L
137    (hR := hL.le)
138  calc
139    _ = ∫ z, (phaseSpaceUnitBox k s).indicator (fun z => f (L • z)) z :=
140      (integral_indicator (phaseSpaceUnitBox_closed k s).measurableSet).symm
141    _ = ∫ z, (phaseSpaceBox k s L).indicator f (L • z) :=
142      (congrArg (fun g : PhaseSpace k s → E => ∫ z, g z) he).symm
143    _ = _ := h
144    _ = _ := congrArg (fun v : E => (L ^ Module.finrank ℝ (PhaseSpace k s))⁻¹ • v)
145      (integral_indicator (phaseSpaceBox_closed k s L).measurableSet)

147theorem phaseSpaceBox_volume_real {L : ℝ} (hL : 0 < L) :
148    volume.real (phaseSpaceBox k s L) =
149      L ^ Module.finrank ℝ (PhaseSpace k s) * volume.real (phaseSpaceUnitBox k s) := by
150  have hv := integral_unitBox_smul hL (fun _ : PhaseSpace k s => (1 : ℝ))
151  simp only [integral_const, Measure.restrict_apply_univ, measureReal_def, smul_eq_mul, mul_one] at hv
