17theorem partitionActionMatrix_unitary_intertwiner_eq_zero
18    (mu nu : Fin d → ℕ) (hmu : Antitone mu) (hnu : Antitone nu) (hne : mu ≠ nu)
19    (M : Matrix (PartitionIndex mu hmu) (PartitionIndex nu hnu) ℂ)
20    (hcomm : ∀ U : unitary (Matrix (Fin d) (Fin d) ℂ),
21      M * partitionActionMatrix nu hnu U = partitionActionMatrix mu hmu U * M) : M=0 := by
22  let B := (partitionBasis mu hmu).toBasis
23  let C := (partitionBasis nu hnu).toBasis
24  let T := (Matrix.toLin C B M).toContinuousLinearMap
25  have hT : ∀ U : unitary (Matrix (Fin d) (Fin d) ℂ),
26      T.comp (partitionTensorAction nu hnu U) = (partitionTensorAction mu hmu U).comp T := by
27    intro U
28    apply ContinuousLinearMap.coe_injective
29    apply (LinearMap.toMatrix C B).injective
30    rw [ContinuousLinearMap.coe_comp,ContinuousLinearMap.coe_comp,
31      LinearMap.toMatrix_comp C C B,LinearMap.toMatrix_comp C B B]
32    simpa only [T,LinearMap.coe_toContinuousLinearMap,LinearMap.toMatrix_toLin] using hcomm U
33  have hn : (fun a => (mu a : ℂ)) ≠ (fun a => (nu a : ℂ)) := by
34    intro hh
35    apply hne
36    funext a
37    exact Nat.cast_injective (congrFun hh a)
38  have ht := cyclicTensorOperator_unitary_intertwiner_eq_zero
39    (partitionHighestTensor mu hmu) (partitionHighestTensor nu hnu)
40    (fun a => (mu a : ℂ)) (fun a => (nu a : ℂ))
41    (partitionHighestTensor_cartan mu hmu) (partitionHighestTensor_cartan nu hnu)
42    (partitionHighestTensor_raising_zero mu hmu) (partitionHighestTensor_raising_zero nu hnu)
43    hn (partitionHighestTensor_norm mu hmu) (partitionHighestTensor_norm nu hnu) T hT
44  have hh := congrArg (fun F => LinearMap.toMatrix C B (F : _ →L[ℂ] _).toLinearMap) ht
45  simpa only [T,LinearMap.coe_toContinuousLinearMap,LinearMap.toMatrix_toLin,
46    ContinuousLinearMap.coe_zero,map_zero] using hh

48def rectangularSandwichCLM {I J : Type*} [Fintype I] [Fintype J] [DecidableEq I] [DecidableEq J]
49    (A : Matrix I I ℂ) (B : Matrix J J ℂ) : Matrix I J ℂ →L[ℂ] Matrix I J ℂ :=

69theorem continuous_mixedSandwich (mu nu : Fin d → ℕ) (hmu : Antitone mu) (hnu : Antitone nu)
70    (M : Matrix (PartitionIndex mu hmu) (PartitionIndex nu hnu) ℂ) :
71    Continuous (fun U : unitary (Matrix (Fin d) (Fin d) ℂ) =>
72      partitionActionMatrix mu hmu U * M * (partitionActionMatrix nu hnu U)ᴴ) := by

86theorem mixedTwirl_invariant (mu nu : Fin d → ℕ) (hmu : Antitone mu) (hnu : Antitone nu)
87    (M : Matrix (PartitionIndex mu hmu) (PartitionIndex nu hnu) ℂ)
88    (V : unitary (Matrix (Fin d) (Fin d) ℂ)) :
89    partitionActionMatrix mu hmu V * mixedTwirl mu nu hmu hnu M *
90      (partitionActionMatrix nu hnu V)ᴴ = mixedTwirl mu nu hmu hnu M := by
91  have hh := (rectangularSandwichCLM (partitionActionMatrix mu hmu V)
92      (partitionActionMatrix nu hnu V)).integral_comp_comm (integrable_mixedSandwich mu nu hmu hnu M)
93  change (∫ U : unitary (Matrix (Fin d) (Fin d) ℂ),
94    partitionActionMatrix mu hmu V * (partitionActionMatrix mu hmu U * M *
95      (partitionActionMatrix nu hnu U)ᴴ) * (partitionActionMatrix nu hnu V)ᴴ ∂unitaryHaar) =
96    partitionActionMatrix mu hmu V * mixedTwirl mu nu hmu hnu M *
97      (partitionActionMatrix nu hnu V)ᴴ at hh
98  rw [← hh]
99  have he (U : unitary (Matrix (Fin d) (Fin d) ℂ)) :
100      partitionActionMatrix mu hmu V * (partitionActionMatrix mu hmu U * M *
101        (partitionActionMatrix nu hnu U)ᴴ) * (partitionActionMatrix nu hnu V)ᴴ =
102      partitionActionMatrix mu hmu (V*U) * M * (partitionActionMatrix nu hnu (V*U))ᴴ := by
103    simp only [Submonoid.coe_mul,partitionActionMatrix_mul,Matrix.conjTranspose_mul,Matrix.mul_assoc]
104  simp_rw [he]
105  exact integral_mul_left_eq_self (μ := unitaryHaar)
106    (fun U : unitary (Matrix (Fin d) (Fin d) ℂ) =>
107      partitionActionMatrix mu hmu U*M*(partitionActionMatrix nu hnu U)ᴴ) V

109theorem mixedTwirl_eq_zero (mu nu : Fin d → ℕ) (hmu : Antitone mu) (hnu : Antitone nu)
110    (hne : mu ≠ nu) (M : Matrix (PartitionIndex mu hmu) (PartitionIndex nu hnu) ℂ) :
111    mixedTwirl mu nu hmu hnu M = 0 := by
112  apply partitionActionMatrix_unitary_intertwiner_eq_zero mu nu hmu hnu hne
113  intro U
114  have hh := congrArg (fun X => X*partitionActionMatrix nu hnu U)
115    (mixedTwirl_invariant mu nu hmu hnu M U)
