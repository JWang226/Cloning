4/-!
5# Trace-norm continuity of actual infinite-dimensional pure states
6 
7Every rank-one operator is trace class with trace norm `‖x‖ * ‖y‖`.
8The standard two-term expansion of the difference of vector projectors gives
9trace-norm continuity, and hence the concrete binomial coefficient vectors
10converge to their coherent Poisson projector in the trace-class Banach space.
11The physical symmetric-tensor occupation embedding is not constructed here.
12-/

33/-- General rank-one operators, including nonpositive ones, are trace class. -/
34theorem isTraceClass_rankOne (x y : H) : IsTraceClass (InnerProductSpace.rankOne ℂ x y) := by
35  by_cases hy : y = 0
36  · subst y
37    simpa using (isTraceClass_zero (H := H))
38  have hy0 : ‖y‖ ≠ 0 := norm_ne_zero_iff.mpr hy
39  have hc : ((‖y‖ ^ 2 : ℝ) : ℂ) ≠ 0 := by exact_mod_cast pow_ne_zero 2 hy0
40  have hprod : IsTraceClass
41      (InnerProductSpace.rankOne ℂ x y * InnerProductSpace.rankOne ℂ y y) := by
42    simpa only [mul_one] using
43      (isTraceClass_mul_mul (A := InnerProductSpace.rankOne ℂ x y) (B := 1)
44        (isTraceClass_rankOne_self y))
45  rw [rankOne_mul_self_right] at hprod
46  have h := isTraceClass_smul (((‖y‖ ^ 2 : ℝ) : ℂ)⁻¹) hprod
47  simpa only [smul_smul, inv_mul_cancel₀ hc, one_smul] using h

49/-- The true trace norm of a rank-one operator equals the product of the
50Hilbert norms, with no finite-dimensional assumption. -/
51theorem traceNorm_rankOne (x y : H) :
52    traceNorm (InnerProductSpace.rankOne ℂ x y) (isTraceClass_rankOne x y) = ‖x‖ * ‖y‖ := by
53  apply le_antisymm
54  · by_cases hy : y = 0
55    · subst y
56      simp [traceNorm_zero]
57    have hysq : 0 < ‖y‖ ^ 2 := sq_pos_of_pos (norm_pos_iff.mpr hy)
58    have hprod : IsTraceClass
59        (InnerProductSpace.rankOne ℂ x y * InnerProductSpace.rankOne ℂ y y * 1) :=
60      isTraceClass_mul_mul (A := InnerProductSpace.rankOne ℂ x y) (B := 1)
61        (isTraceClass_rankOne_self y)
62    have hle := traceNorm_mul_mul_le (isTraceClass_rankOne_self y) hprod
63    have hnorm1 : ‖(1 : H →L[ℂ] H)‖ ≤ 1 :=
64      ContinuousLinearMap.opNorm_le_bound _ zero_le_one (by intro z; simp)
65    have hle' : traceNorm
66        (InnerProductSpace.rankOne ℂ x y * InnerProductSpace.rankOne ℂ y y * 1) hprod ≤
67        ‖x‖ * ‖y‖ * ‖y‖ ^ 2 := by
68      apply hle.trans
69      rw [InnerProductSpace.norm_rankOne, traceNorm_rankOne_self]
70      exact mul_le_of_le_one_right (by positivity) hnorm1
71    have heq : InnerProductSpace.rankOne ℂ x y * InnerProductSpace.rankOne ℂ y y * 1 =
72        ((‖y‖ ^ 2 : ℝ) : ℂ) • InnerProductSpace.rankOne ℂ x y := by
73      rw [mul_one, rankOne_mul_self_right]
74    rw [traceNorm_transport heq hprod, traceNorm_smul _ (isTraceClass_rankOne x y),
75      Complex.norm_of_nonneg (sq_nonneg _)] at hle'
76    nlinarith
77  · simpa only [InnerProductSpace.norm_rankOne] using
78      opNorm_le_traceNorm (isTraceClass_rankOne x y)

80/-- A general rank-one operator bundled in the trace-class Banach space. -/
81def rankOneOperator (x y : H) : TraceClass H :=
82  TraceClass.ofOperator (InnerProductSpace.rankOne ℂ x y) (isTraceClass_rankOne x y)

87theorem vectorProjector_sub_eq (x y : H) :

99/-- The standard pure-state trace-distance bound holds on any complex Hilbert
100space. The norm on the left is the genuine trace norm. -/
101theorem norm_vectorProjector_sub_le (x y : H) :
102    ‖vectorProjector x - vectorProjector y‖ ≤ (‖x‖ + ‖y‖) * ‖x - y‖ := by
