7/-!
8# Infinite mixtures in the actual trace-class Banach space
9 
10The analytic trace and inclusion into bounded operators are continuous linear
11maps. Summable mixtures of unit-vector projectors therefore define genuine
12trace-class operators. Their trace-norm variation is bounded by the ℓ¹ variation
13of the coefficients, linking countable occupation laws to quantum states.
14-/

26theorem norm_trace_le_traceNorm {T : H →L[ℂ] H} (hT : IsTraceClass T) :
27    ‖trace T hT‖ ≤ traceNorm T hT := by
28  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
29  rw [trace_eq_of_hilbertBasis hT b]
30  have hs := summable_trace_diagonal_of_isTraceClass hT b
31  calc
32    ‖∑' i : w, ⟪b i, T (b i)⟫_ℂ‖ ≤ ∑' i : w, ‖⟪b i, T (b i)⟫_ℂ‖ :=
33      norm_tsum_le_tsum_norm hs.norm
34    _ ≤ traceNorm T hT := by
35      simpa using tsum_norm_inner_contraction_le_traceNorm hT
36        (W := (1 : H →L[ℂ] H)) (ContinuousLinearMap.opNorm_le_bound _ zero_le_one (by intro x; simp)) b

38/-- The analytic trace as a continuous complex-linear functional. -/
39def traceCLM : TraceClass H →L[ℂ] ℂ :=
40  LinearMap.mkContinuous
41    { toFun := fun T => trace T.1 T.2
42      map_add' := fun S T => trace_add S.2 T.2
43      map_smul' := fun c T => trace_smul c T.2 }
44    1 (fun T => by simpa using norm_trace_le_traceNorm T.2)

46theorem traceCLM_apply (T : TraceClass H) : traceCLM T = trace T.1 T.2 := rfl

48/-- The trace norm continuously dominates the bounded-operator norm. -/
49def inclusionCLM : TraceClass H →L[ℂ] (H →L[ℂ] H) :=
50  LinearMap.mkContinuous
51    { toFun := fun T => T.1
52      map_add' := fun _ _ => rfl
53      map_smul' := fun _ _ => rfl }
54    1 (fun T => by simpa using opNorm_le_traceNorm T.2)

58/-- A rank-one projector as an element of the trace-class Banach space. -/
59def vectorProjector (x : H) : TraceClass H :=
60  TraceClass.ofOperator (InnerProductSpace.rankOne ℂ x x) (isTraceClass_rankOne_self x)

81/-- The mixture is a series in the trace norm, not just a formal diagonal. -/
82def vectorMixture (x : ι → H) (p : ι → ℝ) : TraceClass H :=
83  ∑' i, (p i : ℂ) • vectorProjector (x i)

85theorem traceCLM_vectorMixture (x : ι → H) (hx : ∀ i, ‖x i‖ = 1)
86    (p : ι → ℝ) (hp : Summable p) :
87    traceCLM (vectorMixture x p) = ((∑' i, p i : ℝ) : ℂ) := by
88  rw [vectorMixture, ContinuousLinearMap.map_tsum _ (summable_weighted_projectors x hx p hp)]
89  simp only [map_smul, traceCLM_vectorProjector, hx, one_pow, Complex.ofReal_one,

93theorem vectorMixture_nonneg (x : ι → H) (hx : ∀ i, ‖x i‖ = 1)
94    (p : ι → ℝ) (hp : Summable p) (hp0 : ∀ i, 0 ≤ p i) :
95    0 ≤ (vectorMixture x p).1 := by
96  change 0 ≤ inclusionCLM (vectorMixture x p)
97  rw [vectorMixture, ContinuousLinearMap.map_tsum _ (summable_weighted_projectors x hx p hp)]
98  apply tsum_nonneg
99  intro i
100  change 0 ≤ (p i : ℂ) • InnerProductSpace.rankOne ℂ (x i) (x i)
101  have hcast : (p i : ℂ) • InnerProductSpace.rankOne ℂ (x i) (x i) =
102      p i • InnerProductSpace.rankOne ℂ (x i) (x i) := by
103    ext y
104    simp only [ContinuousLinearMap.smul_apply, RCLike.real_smul_eq_coe_smul (K := ℂ)]
105    rfl
106  rw [hcast]
107  exact smul_nonneg (hp0 i)
108    ((InnerProductSpace.rankOne ℂ (x i) (x i)).nonneg_iff_isPositive.mpr
109      (InnerProductSpace.isPositive_rankOne_self (x i)))

122theorem vectorMixture_sub (x : ι → H) (hx : ∀ i, ‖x i‖ = 1)
123    (p q : ι → ℝ) (hp : Summable p) (hq : Summable q) :
124    vectorMixture x p - vectorMixture x q = vectorMixture x (p - q) := by

131theorem norm_vectorMixture_le (x : ι → H) (hx : ∀ i, ‖x i‖ = 1)
132    (p : ι → ℝ) (hp : Summable p) :
133    ‖vectorMixture x p‖ ≤ ∑' i, |p i| := by

139theorem norm_vectorMixture_sub_le_l1 (x : ι → H) (hx : ∀ i, ‖x i‖ = 1)
140    (p q : ι → ℝ) (hp : Summable p) (hq : Summable q) :
141    ‖vectorMixture x p - vectorMixture x q‖ ≤ CountableScheffe.l1Distance p q := by
142  rw [vectorMixture_sub x hx p q hp hq]

145theorem vectorMixture_tendsto_of_pointwise (x : ι → H) (hx : ∀ i, ‖x i‖ = 1)
146    (p : ℕ → ι → ℝ) (q : ι → ℝ)
147    (hp : ∀ n i, 0 ≤ p n i) (hq : ∀ i, 0 ≤ q i)
148    (hp_sum : ∀ n, HasSum (p n) 1) (hq_sum : HasSum q 1)
149    (hlim : ∀ i, Tendsto (fun n => p n i) atTop (𝓝 (q i))) :
150    Tendsto (fun n => vectorMixture x (p n)) atTop (𝓝 (vectorMixture x q)) := by
