30/-- A strictly decreasing test weight detects every nonzero sequence whose
31proper partial sums are nonnegative and whose total is zero. -/
32theorem weighted_sum_eq_zero_forces_zero (a w : ℕ → ℝ) (n : ℕ)
33    (hpartial : ∀ k ≤ n, 0 ≤ partialSum a k) (htotal : partialSum a n = 0)
34    (hw : ∀ k, k+1 < n → w (k+1) < w k)
35    (hweighted : ∑ i ∈ Finset.range n, a i * w i = 0) :
36    ∀ i < n, a i = 0 := by
37  have hnonneg (k : ℕ) (hk : k ∈ Finset.range n) :
38      0 ≤ partialSum a (k+1) * (w k - w (k+1)) := by
39    have hkn : k < n := Finset.mem_range.mp hk
40    by_cases hlast : k+1 = n
41    · rw [hlast, htotal, zero_mul]
42    · exact mul_nonneg (hpartial (k+1) (by omega)) (sub_nonneg.mpr (hw k (by omega)).le)
43  have hsum : (∑ k ∈ Finset.range n, partialSum a (k+1) * (w k - w (k+1))) = 0 := by
44    rw [finite_summation_by_parts a w n, htotal, zero_mul, zero_add] at hweighted
45    exact hweighted
46  have hterm := (Finset.sum_eq_zero_iff_of_nonneg hnonneg).mp hsum
47  have hpzero (k : ℕ) (hk : k ≤ n) : partialSum a k = 0 := by
48    by_cases hk0 : k = 0
49    · subst k; exact partialSum_zero a
50    by_cases hkn : k = n
51    · rw [hkn, htotal]
52    have hkpos : 0 < k := by omega
53    have h := hterm (k-1) (Finset.mem_range.mpr (by omega))
54    rw [show k-1+1=k by omega] at h
55    have hw' := hw (k-1) (by omega)
56    rw [show k-1+1=k by omega] at hw'
57    exact (mul_eq_zero.mp h).resolve_right (sub_ne_zero.mpr (ne_of_gt hw'))
58  intro i hi
59  have h := partialSum_succ a i
60  rw [hpzero (i+1) (by omega), hpzero i (by omega), zero_add] at h
61  exact h.symm

63/-- Dominance and the squared-norm eigenvalue determine a strictly dominant
64weight uniquely among dominant weights. -/
65theorem dominant_eq_of_sum_squares (a b : ℕ → ℝ) (n : ℕ)
66    (ha : ∀ k, k+1 < n → a (k+1) < a k)
67    (hb : ∀ k, k+1 < n → b (k+1) ≤ b k)
68    (hprefix : ∀ k ≤ n, partialSum b k ≤ partialSum a k)
69    (htotal : partialSum a n = partialSum b n)
70    (hsq : ∑ i ∈ Finset.range n, a i ^ 2 = ∑ i ∈ Finset.range n, b i ^ 2) :
71    ∀ i < n, a i = b i := by
72  have hp k : partialSum (fun i ↦ a i - b i) k = partialSum a k - partialSum b k := by
73    simp [partialSum, Finset.sum_sub_distrib]
74  have hc := weighted_sum_eq_zero_forces_zero (fun i ↦ a i - b i)
75    (fun i ↦ a i + b i) n
76    (fun k hk ↦ by rw [hp]; exact sub_nonneg.mpr (hprefix k hk))
77    (by rw [hp, htotal, sub_self])
78    (fun k hk ↦ by have := ha k hk; have := hb k hk; linarith)
79    (by
80      simp_rw [show ∀ i, (a i - b i) * (a i + b i) = a i ^ 2 - b i ^ 2 by intro i; ring]
81      rw [Finset.sum_sub_distrib, hsq, sub_self])
82  intro i hi
83  exact sub_eq_zero.mp (hc i hi)

85def zeroExtend {d : ℕ} (a : Fin d → ℝ) (k : ℕ) : ℝ :=

96def DominatedBy {d : ℕ} (b a : Fin d → ℝ) : Prop :=
97  (∀ k ≤ d, partialSum (zeroExtend b) k ≤ partialSum (zeroExtend a) k) ∧
98    (∑ i, b i) = ∑ i, a i

100theorem fin_dominant_eq_of_sum_squares {d : ℕ} (a b : Fin d → ℝ)
101    (ha : StrictAnti a) (hb : Antitone b) (hdom : DominatedBy b a)
