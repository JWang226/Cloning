166def slotInsertion (n r : ℕ)
167    (X : Matrix ((Fin n → A) × (Fin n → A)) ((Fin n → A) × (Fin n → A)) ℂ) :
168    Matrix (Fin (n + r) → A × A) (Fin (n + r) → A × A) ℂ :=
169  fun a c => X
170    ((fun i => (a (Fin.castAdd r i)).1), (fun i => (a (Fin.castAdd r i)).2))
171    ((fun i => (c (Fin.castAdd r i)).1), (fun i => (c (Fin.castAdd r i)).2)) *
172    ∏ j : Fin r, if a (Fin.natAdd n j) = c (Fin.natAdd n j) then 1 else 0

256/-- The actual finite Haar purification channel, followed by the literal
257Werner sandwich and physical partial trace, has the Gaussian product-mixture
258limit. The purification frame is constructed, and Haar randomization is
259removed by environment invariance. The downstream sandwich here is a
260continuous linear map; its global CPTP extension is a separate statement. -/
261theorem physical_pct_gaussian_product_mixture {s : ℕ} (hs : 1 ≤ s)
262    (hcard : Fintype.card (A × A) = s + 1) (ρ : Cloning.MatrixFidelity.State A)
263    (r : ℕ → ℕ) {γ : ℝ} (hγ : 1 < γ)
264    (h : Filter.Tendsto (fun n ↦ ((n + r n : ℕ) : ℝ) / n) Filter.atTop (𝓝 γ)) :
265    ∃ u : OrthonormalBasis (Fin (s + 1)) ℂ (Register (A × A)),
266      u 0 = canonicalPurification ρ.matrix ∧
267      Filter.Tendsto (fun n => ‖physicalCloneTraceCLM n (r n) s
268        ((haarPurificationChannel (A := A) n).toFun (tensorPower n ρ.matrix)) -
269        ∫ z, matrixTensorPower
270          (reducedDensityMatrix (frameParticle u z (n + r n))) (n + r n)
271          ∂MultimodeCoherentGaussianMixture.gaussianProductMeasure
272            (fun _ : Fin s => γ - 1)‖) Filter.atTop (𝓝 0) := by
273  obtain ⟨u, hu, ht⟩ := exists_pct_product_mixture (A := A) (B := A) (s := s) hs hcard
274    (canonicalPurification ρ.matrix) (canonicalPurification_norm ρ)
275    (unitaryHaar (A := A)) (purificationEnvironment (A := A)) (fun n => n + r n)
276    (fun n => Nat.le_add_right n (r n)) hγ h
277  refine ⟨u, hu, ?_⟩
278  have he (n : ℕ) := physical_pct_eq_reducedWerner n (r n) s hcard ρ
279  simpa only [he, randomizedPurificationOutput_eq] using ht
