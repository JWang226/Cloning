33/-- Coordinate phases define a complex-linear isometry, rather than merely a
34coefficient transformation preserving a formally assigned norm. -/
35def phaseIsometry (u : ι → ℂ) (hu : ∀ i, ‖u i‖ = 1) :
36    lp (fun _ : ι => ℂ) 2 →ₗᵢ[ℂ] lp (fun _ : ι => ℂ) 2 where
37  toLinearMap :=
38    { toFun := phaseVector u hu
39      map_add' := by intro x y; ext i; simp [mul_add]
40      map_smul' := by intro c x; ext i; simp [mul_left_comm] }
41  norm_map' := phaseVector_norm u hu

46/-- The phase of zero is chosen to be one, making all coefficients total. -/
47def phase (z : ℂ) : ℂ := if z = 0 then 1 else z / (‖z‖ : ℂ)

49theorem phase_norm (z : ℂ) : ‖phase z‖ = 1 := by

54theorem phase_mul_norm (z : ℂ) : phase z * (‖z‖ : ℂ) = z := by
