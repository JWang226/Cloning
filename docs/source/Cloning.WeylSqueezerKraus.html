27/-- A bounded-observable identity verified on every pure vector extends to all

29theorem channel_pairing_of_pure (Φ : QuantumChannel H K)
30    (A : K →L[ℂ] K) (B : H →L[ℂ] H) (c : ℂ)
31    (hpure : ∀x : H,tracePairing (Φ.toLinearMap (vectorProjector x)) A=
32      c*⟪x,B x⟫_ℂ) (T : TraceClass H) :
33    tracePairing (Φ.toLinearMap T) A=c*tracePairing T B := by
34  have he : (tracePairingCLM.flip A).comp
35      Φ.toPositiveTracePreservingMap.toContinuousLinearMap=c • tracePairingCLM.flip B := by
36    apply traceClass_functional_ext
37    intro x

39      c*tracePairing (vectorProjector x) B
