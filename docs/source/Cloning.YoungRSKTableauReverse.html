11noncomputable def tableauReverse {d : ℕ} : List (RowContent d) → ℕ → Option (List (RowContent d) × Fin d)
12  | [], _ => none
13  | U :: T, 0 =>
14      match rowLast U with
15      | none => none
16      | some x => some (rowDeleteLast U :: T, x)
17  | U :: T, r + 1 =>
18      match tableauReverse T r with
19      | none => none
20      | some (S, y) =>
21          match rowReverseBump U y with
22          | none => none
23          | some x => some (rowReversed U y :: S, x)

25noncomputable def tableauRemovable {d : ℕ} : List (RowContent d) → ℕ → Prop
26  | [], _ => False
27  | U :: T, 0 => 0 < ∑ j, U j ∧ ∀ V ∈ T.head?, (∑ j, V j) < ∑ j, U j
28  | _ :: T, r + 1 => tableauRemovable T r

71theorem tableauReverse_head_prefix {d : ℕ} (U : RowContent d)
72    (T S : List (RowContent d)) (r : ℕ) (x : Fin d)
73    (h : tableauReverse (U :: T) r = some (S, x)) :
74    ∃ W R, S = W :: R ∧ 0 < U x ∧
75      (∀ k, rowPrefix W k ≤ rowPrefix U k) ∧
76      rowPrefix W x + 1 = rowPrefix U x := by
77  cases r with
78  | zero =>
79    cases hb : rowLast U with
80    | none => simp [tableauReverse, hb] at h
81    | some b =>
82      simp only [tableauReverse, hb, Option.some.injEq, Prod.mk.injEq] at h
83      obtain ⟨rfl, rfl⟩ := h
84      have hbpos := ((rowLast_eq_some_iff U b).mp hb).1
85      refine ⟨rowDeleteLast U, T, rfl, hbpos, ?_, ?_⟩
86      · intro k
87        have he := rowRemove_prefix U b k hbpos
88        simp only [rowDeleteLast, hb]
89        omega
90      · simpa only [rowDeleteLast, hb, le_refl, if_true] using rowRemove_prefix U b b hbpos
91  | succ r =>
92    cases ht : tableauReverse T r with
93    | none => simp [tableauReverse, ht] at h
94    | some P =>
95      obtain ⟨R, b⟩ := P
96      cases hb : rowReverseBump U b with
97      | none => simp [tableauReverse, ht, hb] at h
98      | some a =>
99        simp only [tableauReverse, ht, hb, Option.some.injEq, Prod.mk.injEq] at h
100        obtain ⟨rfl, rfl⟩ := h
101        obtain ⟨hab, hapos, _⟩ := (rowReverseBump_eq_some_iff U b a).mp hb
102        refine ⟨rowReversed U b, R, rfl, hapos, ?_, ?_⟩
103        · intro k
104          have he := rowReversed_prefix U b a k hb

108theorem rowInterlaces_mass_le {d : ℕ} (U V : RowContent d)
109    (hUV : ∀ k, rowPrefix V k ≤ rowStrictPrefix U k) :
110    (∑ j, V j) ≤ ∑ j, U j := by
111  cases d with
112  | zero => simp
113  | succ d =>
114    have h := hUV (Fin.last d)
115    have hv : rowPrefix V (Fin.last d) = ∑ j, V j := by
116      simp [rowPrefix, Fin.le_last]
117    rw [hv] at h

134theorem rowStrictPrefix_eq_mass_of_last {d : ℕ} (U : RowContent d) (b k : Fin d)
135    (hb : rowLast U = some b) (hbk : b < k) : rowStrictPrefix U k = ∑ j, U j := by

143/-- Removing the final entry at an outer corner preserves its inequalities
144with every shorter lower row. -/
145theorem rowRemoveLast_interlaces {d : ℕ} (U V : RowContent d) (b : Fin d)
146    (hUV : ∀ k, rowPrefix V k ≤ rowStrictPrefix U k)
147    (hmass : (∑ j, V j) < ∑ j, U j) (hb : rowLast U = some b) :
148    ∀ k, rowPrefix V k ≤ rowStrictPrefix (rowDeleteLast U) k := by
149  have hbpos := ((rowLast_eq_some_iff U b).mp hb).1
150  intro k
151  simp only [rowDeleteLast, hb]
152  have he := rowRemove_strictPrefix U b k hbpos
153  by_cases hbk : b < k
154  · rw [if_pos hbk, rowStrictPrefix_eq_mass_of_last U b k hb hbk] at he
155    have hbound : rowPrefix V k ≤ ∑ j, V j :=
156      Finset.sum_le_sum_of_subset (Finset.filter_subset _ _)
157    omega
158  · rw [if_neg hbk, add_zero] at he
159    rw [he]
160    exact hUV k

216/-- A genuine removable corner admits reverse deletion, and all strict
217column inequalities are preserved throughout the reverse path. -/
218theorem tableauReverse_exists_pairwise {d : ℕ} (T : List (RowContent d))
219    (hT : List.Pairwise RowInterlaces T) (r : ℕ) (hr : tableauRemovable T r) :
220    ∃ S x, tableauReverse T r = some (S, x) ∧ List.Pairwise RowInterlaces S := by
221  induction T generalizing r with
222  | nil => simp [tableauRemovable] at hr
223  | cons U T ih =>
224    obtain ⟨hU, hT⟩ := List.pairwise_cons.mp hT
225    cases r with
226    | zero =>
227      obtain ⟨hUpos, hnext⟩ := hr
228      obtain ⟨b, hb⟩ := rowLast_exists_of_mass U hUpos
229      refine ⟨rowDeleteLast U :: T, b, by simp [tableauReverse, hb], ?_⟩
230      apply List.pairwise_cons.mpr
231      refine ⟨?_, hT⟩
232      intro V hV
233      have hmass : (∑ j, V j) < ∑ j, U j := by
234        cases T with
235        | nil => simp at hV
236        | cons W R =>
237          have hWU := hnext W (by simp)
238          rcases List.mem_cons.mp hV with rfl | hVR
239          · exact hWU
240          · exact (rowInterlaces_mass_le W V ((List.pairwise_cons.mp hT).1 V hVR)).trans_lt hWU
241      exact rowRemoveLast_interlaces U V b (hU V hV) hmass hb
242    | succ r =>
243      cases T with
244      | nil => simp [tableauRemovable] at hr
245      | cons V R =>
246        obtain ⟨S, b, hS, hSpair⟩ := ih hT r hr
247        obtain ⟨W, R', heq, hbpos, hprefix, hdrop⟩ :=
248          tableauReverse_head_prefix V R S r b hS
249        obtain ⟨a, ha⟩ := rowReverseBump_exists_of_prefix U V b
250          (hU V List.mem_cons_self) hbpos
251        refine ⟨rowReversed U b :: S, a, by simp [tableauReverse, hS, ha], ?_⟩
252        rw [heq] at hSpair ⊢
253        apply pairwise_interlaces_cons _ hSpair
254        exact rowReversed_interlaces_of_prefix U V W b a
255          (hU V List.mem_cons_self) hbpos ha hprefix hdrop.le

257theorem tableauReverse_pairwise {d : ℕ} (T S : List (RowContent d))
258    (hT : List.Pairwise RowInterlaces T) (r : ℕ) (hr : tableauRemovable T r) (x : Fin d)
259    (h : tableauReverse T r = some (S, x)) : List.Pairwise RowInterlaces S := by

262  have heS := congrArg Prod.fst he
