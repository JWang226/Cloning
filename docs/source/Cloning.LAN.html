20/-- The channel-composition estimate in Lemma `LAN-fidelity-transfer`.
21`dist` represents trace distance with the normalization used in the paper;
22`FB` and `FG` represent its root fidelities. -/
23theorem fidelity_transfer
24    {A B G H : Type*} [PseudoMetricSpace A] [PseudoMetricSpace B]
25    [PseudoMetricSpace H]
26    (FB : B → B → ℝ) (FG : H → H → ℝ)
27    (S : G → A) (M : A → B) (T : B → H)
28    (hM : ∀ x y, dist (M x) (M y) ≤ dist x y)
29    (hT : ∀ x y, dist (T x) (T y) ≤ dist x y)
30    (hdata : ∀ x y, FB x y ≤ FG (T x) (T y))
31    (hcontinuity : ∀ x x' y y', FG x y ≤ FG x' y' +
32      Real.sqrt (dist x x') + Real.sqrt (dist y y'))
33    (ρ : A) (σ : B) (Φ : G) (Ψ : H) (δ η : ℝ)
34    (hδ : dist (S Φ) ρ ≤ δ) (hη : dist (T σ) Ψ ≤ η) :
35    FB (M ρ) σ ≤ FG (T (M (S Φ))) Ψ + Real.sqrt δ + Real.sqrt η := by
36  have hd : dist (T (M ρ)) (T (M (S Φ))) ≤ δ :=
37    (hT _ _).trans ((hM _ _).trans (by simpa only [dist_comm] using hδ))
38  have hc := hcontinuity (T (M ρ)) (T (M (S Φ))) (T σ) Ψ
39  have hd' := Real.sqrt_le_sqrt hd
40  have hη' := Real.sqrt_le_sqrt hη
41  have hm := hdata (M ρ) σ
42  linarith

44/-- Vanishing trace-distance errors give a vanishing uniform transfer error. -/
45theorem transfer_error_tendsto {δ η : ℕ → ℝ}
46    (hδ : Tendsto δ atTop (𝓝 0)) (hη : Tendsto η atTop (𝓝 0)) :
47    Tendsto (fun n => Real.sqrt (δ n) + Real.sqrt (η n)) atTop (𝓝 0) := by
48  simpa using (Real.continuous_sqrt.continuousAt.tendsto.comp hδ).add
49    (Real.continuous_sqrt.continuousAt.tendsto.comp hη)

51/-- A genuine worst-case minimax value; channel choices are indexed by `I`. -/
52noncomputable def minimaxValue {I P : Type*} (payoff : I → P → ℝ) : ℝ :=
53  ⨆ i, ⨅ p, payoff i p

55/-- The optimized limiting-model average against a normalized prior. -/
56noncomputable def bayesValue {J Θ : Type*} [MeasurableSpace Θ]
57    (ν : Measure Θ) (payoff : J → Θ → ℝ) : ℝ :=
58  ⨆ j, ∫ θ, payoff j θ ∂ν

60/-- A single admissible channel with a uniform fidelity lower bound gives
61the same lower bound on the minimax value. The upper fidelity bound ensures
62that the real supremum is taken over a bounded family. -/
63theorem candidate_le_minimaxValue
64    {I P : Type*} [Nonempty P] (payoff : I → P → ℝ)
65    (candidate : I) (a : ℝ)
66    (hupper : ∀ i p, payoff i p ≤ 1)
67    (hlower : ∀ p, a ≤ payoff candidate p)
68    (hnonneg : ∀ i p, 0 ≤ payoff i p) :
69    a ≤ minimaxValue payoff := by
70  classical
71  have hbelow (i : I) : BddBelow (Set.range (payoff i)) := by
72    refine ⟨0, ?_⟩
73    rintro x ⟨p, rfl⟩
74    exact hnonneg i p
75  have hbdd : BddAbove (Set.range (fun i => ⨅ p, payoff i p)) := by
76    refine ⟨1, ?_⟩
77    rintro x ⟨i, rfl⟩
78    exact (ciInf_le (hbelow i) (Classical.arbitrary P)).trans (hupper i _)
79  exact (le_ciInf hlower).trans (le_ciSup hbdd candidate)

81/-- Restricting a global worst case to a local chart, transferring channels,
82and integrating against a probability prior transfers an upper bound on the
83Gaussian Bayes payoff to the finite-sample minimax value.
84 
85Neither a maximizing channel nor an attained worst-case parameter is needed.
86The local chart need not be injective. -/
87theorem minimax_le_average_bound
88    {I J P Θ : Type*} [Nonempty I] [MeasurableSpace Θ]
89    (ν : Measure Θ) [IsProbabilityMeasure ν]
90    (payoff : I → P → ℝ) (gaussian : J → Θ → ℝ)
91    (localChart : Θ → P) (transfer : I → J) (ε B : ℝ)
92    (hnonneg : ∀ i p, 0 ≤ payoff i p)
93    (htransfer : ∀ i θ,
94      payoff i (localChart θ) ≤ gaussian (transfer i) θ + ε)
95    (hintegrable : ∀ j, Integrable (gaussian j) ν)
96    (hbound : ∀ j, (∫ θ, gaussian j θ ∂ν) ≤ B) :
97    minimaxValue payoff ≤ B + ε := by
98  unfold minimaxValue
99  apply ciSup_le
100  intro i
101  have hbelow : BddBelow (Set.range (payoff i)) := by
102    refine ⟨0, ?_⟩
103    rintro x ⟨p, rfl⟩
104    exact hnonneg i p
105  have hpoint : ∀ θ, (⨅ p, payoff i p) ≤ gaussian (transfer i) θ + ε :=
106    fun θ => (ciInf_le hbelow (localChart θ)).trans (htransfer i θ)
107  have havg := integral_mono (integrable_const (⨅ p, payoff i p))
108    ((hintegrable (transfer i)).add (integrable_const ε)) hpoint
109  simp only [Pi.add_apply] at havg
110  rw [integral_add (hintegrable (transfer i)) (integrable_const ε)] at havg
111  simp only [integral_const, probReal_univ, one_smul] at havg
112  have hb := hbound (transfer i)
113  linarith

115/-- In particular, LAN transfer bounds the global minimax value by the
116supremum over all limiting-model channels, including those not obtained by
117transferring a finite-sample channel. -/
118theorem minimax_le_bayesValue
119    {I J P Θ : Type*} [Nonempty I] [MeasurableSpace Θ]
120    (ν : Measure Θ) [IsProbabilityMeasure ν]
121    (payoff : I → P → ℝ) (gaussian : J → Θ → ℝ)
122    (localChart : Θ → P) (transfer : I → J) (ε : ℝ)
123    (hnonneg : ∀ i p, 0 ≤ payoff i p)
124    (htransfer : ∀ i θ,
125      payoff i (localChart θ) ≤ gaussian (transfer i) θ + ε)
126    (hintegrable : ∀ j, Integrable (gaussian j) ν)
127    (hleone : ∀ j θ, gaussian j θ ≤ 1) :
128    minimaxValue payoff ≤ bayesValue ν gaussian + ε := by
129  have hbdd : BddAbove (Set.range (fun j => ∫ θ, gaussian j θ ∂ν)) := by
130    refine ⟨1, ?_⟩
131    rintro x ⟨j, rfl⟩
132    have h := integral_mono (hintegrable j) (integrable_const (1 : ℝ)) (hleone j)
133    simpa only [integral_const, probReal_univ, one_smul] using h
134  exact minimax_le_average_bound ν payoff gaussian localChart transfer ε _
135    hnonneg htransfer hintegrable (fun j => le_ciSup hbdd j)

137/-- The integrated finite-sample LAN converse. The only link required
138between finite and Gaussian channel classes is closure under composition
139with the two LAN maps (`hcompose`). The Gaussian supremum may include
140additional channels. -/
141theorem lan_minimax_le_bayesValue
142    {A B G H I J P Θ : Type*}
143    [PseudoMetricSpace A] [PseudoMetricSpace B] [PseudoMetricSpace H]
144    [Nonempty I] [MeasurableSpace Θ]
145    (ν : Measure Θ) [IsProbabilityMeasure ν]
146    (FB : B → B → ℝ) (FG : H → H → ℝ)
147    (S : G → A) (M : I → A → B) (T : B → H)
148    (C : J → G → H) (transferred : I → J)
149    (hcompose : ∀ i x, C (transferred i) x = T (M i (S x)))
150    (hM : ∀ i x y, dist (M i x) (M i y) ≤ dist x y)
151    (hT : ∀ x y, dist (T x) (T y) ≤ dist x y)
152    (hdata : ∀ x y, FB x y ≤ FG (T x) (T y))
153    (hcontinuity : ∀ x x' y y', FG x y ≤ FG x' y' +
154      Real.sqrt (dist x x') + Real.sqrt (dist y y'))
155    (hFBnonneg : ∀ x y, 0 ≤ FB x y) (hFGleone : ∀ x y, FG x y ≤ 1)
156    (ρ : P → A) (σ : P → B) (Φ : Θ → G) (Ψ : Θ → H)
157    (localChart : Θ → P) (δ η : ℝ)
158    (hδ : ∀ θ, dist (S (Φ θ)) (ρ (localChart θ)) ≤ δ)
159    (hη : ∀ θ, dist (T (σ (localChart θ))) (Ψ θ) ≤ η)
160    (hintegrable : ∀ j, Integrable (fun θ => FG (C j (Φ θ)) (Ψ θ)) ν) :
161    minimaxValue (fun i p => FB (M i (ρ p)) (σ p)) ≤
162      bayesValue ν (fun j θ => FG (C j (Φ θ)) (Ψ θ)) +
163        (Real.sqrt δ + Real.sqrt η) := by
164  apply minimax_le_bayesValue ν _ _ localChart transferred _
165    (fun i p => hFBnonneg _ _)
166  · intro i θ
167    rw [hcompose]
168    have h := fidelity_transfer FB FG S (M i) T (hM i) hT hdata hcontinuity
169      (ρ (localChart θ)) (σ (localChart θ)) (Φ θ) (Ψ θ) δ η (hδ θ) (hη θ)
170    simpa only [add_assoc] using h
171  · exact hintegrable
172  · exact fun j θ => hFGleone _ _

174/-- The order of limits in the converse: first let sample size tend to
175infinity for a fixed local window, then let that window grow. No diagonal
176uniformity in the window is assumed. -/
177theorem two_scale_eventually_upper
178    (u b : ℕ → ℝ) (error : ℕ → ℕ → ℝ) (V : ℝ)
179    (hbound : ∀ L, ∀ᶠ n in atTop, u n ≤ b L + error L n)
180    (herror : ∀ L, Tendsto (error L) atTop (𝓝 0))
181    (hb : Tendsto b atTop (𝓝 V)) :
182    ∀ ε > 0, ∀ᶠ n in atTop, u n < V + ε := by
183  intro ε hε
184  obtain ⟨L, hL⟩ := (hb.eventually_lt_const (show V < V + ε / 2 by linarith)).exists
185  have he := (herror L).eventually_lt_const (show (0 : ℝ) < ε / 2 by linarith)
186  filter_upwards [hbound L, he] with n hn hen
187  linarith

189/-- The paper's `limsup` converse, with the lower boundedness of fidelity
190made explicit so that the real-valued `limsup` is well behaved. -/
191theorem two_scale_limsup_le
192    (u b : ℕ → ℝ) (error : ℕ → ℕ → ℝ) (V : ℝ)
193    (hnonneg : ∀ n, 0 ≤ u n)
194    (hbound : ∀ L, ∀ᶠ n in atTop, u n ≤ b L + error L n)
195    (herror : ∀ L, Tendsto (error L) atTop (𝓝 0))
196    (hb : Tendsto b atTop (𝓝 V)) : limsup u atTop ≤ V := by
197  apply le_of_forall_pos_le_add
198  intro ε hε
199  exact limsup_le_of_le (isCoboundedUnder_le_of_le atTop hnonneg)
200    ((two_scale_eventually_upper u b error V hbound herror hb ε hε).mono
201      fun _ h => le_of_lt h)

203/-- Combining the two-scale converse with an achievability lower bound
204proves ordinary convergence of the optimal finite-sample fidelities. -/
205theorem minimax_converges
206    (u b : ℕ → ℝ) (error : ℕ → ℕ → ℝ) (V : ℝ)
207    (hbound : ∀ L, ∀ᶠ n in atTop, u n ≤ b L + error L n)
208    (herror : ∀ L, Tendsto (error L) atTop (𝓝 0))
209    (hb : Tendsto b atTop (𝓝 V))
210    (hachieve : ∀ ε > 0, ∀ᶠ n in atTop, V - ε < u n) :
211    Tendsto u atTop (𝓝 V) := by
212  apply Metric.tendsto_nhds.mpr
213  intro ε hε
214  have hu := two_scale_eventually_upper u b error V hbound herror hb ε hε
215  filter_upwards [hu, hachieve ε hε] with n hn hn'
216  rw [Real.dist_eq, abs_lt]
217  constructor <;> linarith

219/-- A channel-level achievability bound and the two-scale minimax converse
220combine to give convergence of the genuine optimized worst-case payoff.
221The finite-sample admissible-channel type may depend on sample size. -/
222theorem minimax_converges_of_candidates
223    {I : ℕ → Type*} {P : Type*} [Nonempty P]
224    (payoff : (n : ℕ) → I n → P → ℝ)
225    (candidate : (n : ℕ) → I n) (a b : ℕ → ℝ)
226    (error : ℕ → ℕ → ℝ) (V : ℝ)
227    (hnonneg : ∀ n i p, 0 ≤ payoff n i p)
228    (hupper : ∀ n i p, payoff n i p ≤ 1)
229    (hcandidate : ∀ n p, a n ≤ payoff n (candidate n) p)
230    (ha : Tendsto a atTop (𝓝 V))
231    (hbound : ∀ L, ∀ᶠ n in atTop,
232      minimaxValue (payoff n) ≤ b L + error L n)
233    (herror : ∀ L, Tendsto (error L) atTop (𝓝 0))
234    (hb : Tendsto b atTop (𝓝 V)) :
235    Tendsto (fun n => minimaxValue (payoff n)) atTop (𝓝 V) := by
236  apply minimax_converges _ b error V hbound herror hb
237  intro ε hε
238  have ha' := ha.eventually_const_lt (show V - ε < V by linarith)
239  filter_upwards [ha'] with n hn
240  exact hn.trans_le (candidate_le_minimaxValue (payoff n) (candidate n)
241    (a n) (hupper n) (hcandidate n) (hnonneg n))

243/-- Extension from the dense interior of a compact spectral set to its
244boundary, applied to a continuous Gaussian optimum. Use the subtype of the
245compact set as `P` and its relative interior as `D`. -/
246theorem upper_bound_on_dense_set
247    {P : Type*} [TopologicalSpace P] (D : Set P) (hD : Dense D)
248    (F : P → ℝ) (hF : Continuous F) (a : ℝ)
249    (hbound : ∀ p ∈ D, a ≤ F p) : ∀ p, a ≤ F p := by
250  exact hD.induction hbound (isClosed_le continuous_const hF)
