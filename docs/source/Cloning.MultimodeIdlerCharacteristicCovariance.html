27/-- Characteristic evaluation of the actual Kraus channel equals evaluation
28of its actual Stinespring output on the second-register Weyl operator. -/
29theorem channel_characteristic_joint (q : Fin d → ℝ)
30    (hq0 : ∀ i, 0 ≤ q i) (hq1 : ∀ i, q i < 1)
31    (T : TraceClass (Fock d)) (b : Fin d → ℂ) :
32    tracePairing ((channel q hq0 hq1).toLinearMap T) (displacement b) =
33      tracePairing (conjugationLinearMap (appendedIsometry q hq0 hq1).toContinuousLinearMap T)
34        (displacement (Fin.append 0 b)) := by
35  rw [tracePairing_conjugation]
36  suffices hpure : ∀ x : Fock d,
37      tracePairing ((channel q hq0 hq1).toLinearMap (vectorProjector x)) (displacement b) =
38        1 * ⟪x, (appendedIsometry q hq0 hq1).toContinuousLinearMap.adjoint
39          (displacement (Fin.append 0 b) (appendedIsometry q hq0 hq1 x))⟫_ℂ by
40    simpa only [one_mul] using Cloning.WeylSqueezer.channel_pairing_of_pure
41      (channel q hq0 hq1) (displacement b)
42      ((appendedIsometry q hq0 hq1).toContinuousLinearMap.adjoint.comp
43        ((displacement (Fin.append 0 b)).comp
44          (appendedIsometry q hq0 hq1).toContinuousLinearMap)) 1 hpure T
45  intro x
46  have hs := (channel_hasSum q hq0 hq1 (vectorProjector x)).mapL
47    (tracePairingCLM.flip (displacement b))
48  have ht := idlerSlice_inner_hasSum (appendedIsometry q hq0 hq1 x)
49    (displacement (Fin.append 0 b) (appendedIsometry q hq0 hq1 x))
50  simp only [idlerSlice_displacement, idlerSlice_appendedIsometry] at ht
51  have he (m : Occupation d) :
52      tracePairing (krausTerm (kraus q hq0 hq1 m) (vectorProjector x)) (displacement b) =
53        ⟪kraus q hq0 hq1 m x, displacement b (kraus q hq0 hq1 m x)⟫_ℂ := by
54    rw [krausTerm_vectorProjector]
55    exact tracePairing_rankOneOperator _ _ _
56  simp only [ContinuousLinearMap.flip_apply, tracePairingCLM_apply, he] at hs
57  simpa only [one_mul, ContinuousLinearMap.comp_apply,
58    ContinuousLinearMap.adjoint_inner_right, LinearIsometry.coe_toContinuousLinearMap]
59    using hs.unique ht

61private theorem conjugation_comp
62    {H K L : Type*} [NormedAddCommGroup H] [InnerProductSpace ℂ H] [CompleteSpace H]
63    [NormedAddCommGroup K] [InnerProductSpace ℂ K] [CompleteSpace K]
64    [NormedAddCommGroup L] [InnerProductSpace ℂ L] [CompleteSpace L]
65    (V : K →L[ℂ] L) (W : H →L[ℂ] K) (T : TraceClass H) :
66    conjugationLinearMap (V.comp W) T = conjugationLinearMap V (conjugationLinearMap W T) := by
67  apply Subtype.ext
68  apply ContinuousLinearMap.ext

118lemma kraus_vacuum (q : Fin d → ℝ)
119    (hq0 : ∀ i, 0 ≤ q i) (hq1 : ∀ i, q i < 1) (m : Occupation d) :
120    kraus q hq0 hq1 m (numberBasis d 0) =

131/-- Its vacuum output is literally the normalized product thermal state. -/
132theorem channel_vacuum (q : Fin d → ℝ)
133    (hq0 : ∀ i, 0 ≤ q i) (hq1 : ∀ i, q i < 1) :
134    (channel q hq0 hq1).toLinearMap (coherentProjector (0 : Fin d → ℂ)) =
135      vectorMixture (Cloning.MultimodeCoherentGaussianMixture.numberBasis d)
136        (Cloning.ThermalWitness.productGeometric q) := by
