27theorem physicalCharacterPolynomial_symmetric
28    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℂ)
29    (hweight : ∀ a, collectiveGenerator n a a Ω = mu a • Ω)
30    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
31    (σ : Equiv.Perm (Fin d)) :
32    rename σ (physicalCharacterPolynomial Ω) = physicalCharacterPolynomial Ω := by
33  apply MvPolynomial.funext
34  intro D
35  rw [eval_rename, eval_physicalCharacterPolynomial Ω mu hweight hraise,
36    eval_physicalCharacterPolynomial Ω mu hweight hraise,
37    diagonal_permutation_conjugate, cyclicTensorOperator_mul, cyclicTensorOperator_mul]
38  let A := cyclicTensorOperator Ω mu hweight hraise (σ.permMatrix ℂ)
39  let B := cyclicTensorOperator Ω mu hweight hraise (Matrix.diagonal D)
40  let C := cyclicTensorOperator Ω mu hweight hraise (σ⁻¹.permMatrix ℂ)
41  have he : C * A = 1 := by
42    dsimp only [C, A]
43    rw [← cyclicTensorOperator_mul, ← Matrix.permMatrix_mul, mul_inv_cancel,
44      Matrix.permMatrix_one, cyclicTensorOperator_one]
45  change LinearMap.trace ℂ _ (A.toLinearMap * B.toLinearMap * C.toLinearMap) =
46    LinearMap.trace ℂ _ B.toLinearMap
47  rw [LinearMap.trace_mul_cycle]
48  change LinearMap.trace ℂ _ ((C * A) * B).toLinearMap = _
49  rw [he, one_mul]
