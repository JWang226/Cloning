15theorem numberLadder_inner_skew (a : Fin d → ℂ) (k l : Fin d → ℕ) :
16    ⟪numberLadder a k, numberBasis d l⟫_ℂ =
17      -⟪numberBasis d k, numberLadder a l⟫_ℂ := by
18  have h := (displacement_number_hasDerivAt a k 0).inner ℂ
19    (displacement_number_hasDerivAt a l 0)
20  have he (t : ℝ) :
21      ⟪displacement (t • a) (numberBasis d k), displacement (t • a) (numberBasis d l)⟫_ℂ =
22        ⟪numberBasis d k, numberBasis d l⟫_ℂ :=
23    (weylUnitary (t • a)).inner_map_map _ _
24  simp only [he, zero_smul, displacement_zero, ContinuousLinearMap.id_apply] at h
25  have hz := h.unique (hasDerivAt_const 0 ⟪numberBasis d k, numberBasis d l⟫_ℂ)

28theorem number_lowering_entry (k l : Fin d → ℕ) (i : Fin d) :
29    (Real.sqrt (k i : ℝ) : ℂ) * (if l = Function.update k i (k i-1) then 1 else 0) =
30      (Real.sqrt (l i+1 : ℝ) : ℂ) * (if k = Function.update l i (l i+1) then 1 else 0) := by
31  by_cases h : k = Function.update l i (l i+1)
32  · have hi : k i = l i+1 := by rw [h, Function.update_self]
33    have hl : l = Function.update k i (k i-1) := by
34      funext j
35      by_cases hji : j = i
36      · subst j; simp only [Function.update_self, hi, Nat.add_sub_cancel]
37      · rw [Function.update_of_ne hji, h, Function.update_of_ne hji]
38    rw [if_pos h, if_pos hl, mul_one, mul_one, hi]
39    push_cast
