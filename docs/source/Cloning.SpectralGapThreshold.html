60theorem gapSet_threshold_singleton (d : ℕ) :
61    gapSet d (threshold d) = {endpointSpectrum d} := by
62  ext p
63  constructor
64  · intro hp
65    have hle (i : Fin (d+1)) : (endpointSpectrum d).eigenvalue i ≤ p.eigenvalue i := by
66      change arithmeticVector d (threshold d) i ≤ p.eigenvalue i
67      rw [arithmeticVector_threshold]
68      exact gapSet_coordinate_lower p hp i
69    have he : ∀ i, (endpointSpectrum d).eigenvalue i = p.eigenvalue i := by
70      have hsum : (∑ i, (endpointSpectrum d).eigenvalue i) = ∑ i, p.eigenvalue i := by
71        rw [p.normalized,(endpointSpectrum d).normalized]
72      exact fun i => (Finset.sum_eq_sum_iff_of_le (s := Finset.univ) (fun i _ => hle i)).mp hsum i (Finset.mem_univ i)
73    exact SimpleSpectrum.eigenvalue_injective (funext (fun i => (he i).symm))
74  · rintro rfl
