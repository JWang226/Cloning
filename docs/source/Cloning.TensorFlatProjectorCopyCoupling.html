16theorem sum_by_label (labels : I → A) (f : A → ℝ) :
17    ∑ i, f (labels i) = ∑ a, (labelMultiplicity labels a : ℝ) * f a := by
18  calc
19    _ = ∑ i : I, ∑ a : A, if labels i=a then f a else 0 := by simp
20    _ = ∑ a : A, ∑ i : I, if labels i=a then f a else 0 := Finset.sum_comm
21    _ = _ := by simp [labelMultiplicity, Finset.sum_ite, mul_comm]

23theorem sum_uniform_copies (labels : I → A) (f : A → ℝ)
24    (hf : ∀ a, labelMultiplicity labels a=0 → f a=0) :
25    ∑ i, f (labels i)/(labelMultiplicity labels (labels i) : ℝ) = ∑ a, f a := by
26  rw [sum_by_label labels (fun a => f a/(labelMultiplicity labels a : ℝ))]

28  intro a _
29  by_cases ha : labelMultiplicity labels a=0

33def uniformCopyCoupling (l : I → A) (r : J → B) (K : A → B → ℝ) (i : I) (j : J) : ℝ :=
34  K (l i) (r j) / ((labelMultiplicity l (l i) : ℝ)*(labelMultiplicity r (r j) : ℝ))

36theorem labelMultiplicity_pos_at (l : I → A) (i : I) : 0 < labelMultiplicity l (l i) := by

40theorem uniformCopyCoupling_row (l : I → A) (r : J → B) (K : A → B → ℝ)
41    (v : A → ℝ) (w : B → ℝ) (hK : ∀ a b, 0 ≤ K a b)
42    (hrow : ∀ a, ∑ b, K a b = (labelMultiplicity l a : ℝ)*v a)
43    (hcol : ∀ b, ∑ a, K a b = (labelMultiplicity r b : ℝ)*w b) (i : I) :
44    ∑ j, uniformCopyCoupling l r K i j = v (l i) := by
45  have hz (b : B) (hb : labelMultiplicity r b=0) : K (l i) b=0 := by
46    have he := Finset.single_le_sum (fun a _ => hK a b) (Finset.mem_univ (l i))
47    rw [hcol,hb,Nat.cast_zero,zero_mul] at he
48    exact le_antisymm he (hK _ _)
49  have he : (∑ j, uniformCopyCoupling l r K i j) =
50      (∑ j, K (l i) (r j)/(labelMultiplicity r (r j) : ℝ))/(labelMultiplicity l (l i) : ℝ) := by
51    rw [Finset.sum_div]
52    apply Finset.sum_congr rfl
53    intro j _
54    unfold uniformCopyCoupling
55    ring
56  rw [he,sum_uniform_copies r _ hz,hrow]
57  exact mul_div_cancel_left₀ _ (Nat.cast_ne_zero.mpr (labelMultiplicity_pos_at l i).ne')

59theorem uniformCopyCoupling_col (l : I → A) (r : J → B) (K : A → B → ℝ)
60    (v : A → ℝ) (w : B → ℝ) (hK : ∀ a b, 0 ≤ K a b)
61    (hrow : ∀ a, ∑ b, K a b = (labelMultiplicity l a : ℝ)*v a)
62    (hcol : ∀ b, ∑ a, K a b = (labelMultiplicity r b : ℝ)*w b) (j : J) :
63    ∑ i, uniformCopyCoupling l r K i j = w (r j) := by
64  have he := uniformCopyCoupling_row r l (fun b a => K a b) w v
65    (fun b a => hK a b) hcol hrow j
66  simpa only [uniformCopyCoupling, mul_comm] using he

68theorem uniformCopyCoupling_observable (l : I → A) (r : J → B) (K : A → B → ℝ)
69    (hzeroL : ∀ a b, labelMultiplicity l a=0 → K a b=0)
70    (hzeroR : ∀ a b, labelMultiplicity r b=0 → K a b=0) (f : A → B → ℝ) :
71    ∑ i : I, ∑ j : J, uniformCopyCoupling l r K i j*f (l i) (r j) =
72      ∑ a : A, ∑ b : B, K a b*f a b := by
73  have hi (i : I) : (∑ j : J, uniformCopyCoupling l r K i j*f (l i) (r j)) =
74      ∑ b : B, (K (l i) b*f (l i) b)/(labelMultiplicity l (l i) : ℝ) := by
75    have he : (∑ j : J, uniformCopyCoupling l r K i j*f (l i) (r j)) =
76        ∑ j : J, ((K (l i) (r j)*f (l i) (r j))/(labelMultiplicity l (l i) : ℝ)) /
77          (labelMultiplicity r (r j) : ℝ) := by
78      apply Finset.sum_congr rfl
79      intro j _
80      unfold uniformCopyCoupling
81      ring
82    rw [he]
83    exact sum_uniform_copies r (fun b => K (l i) b*f (l i) b/(labelMultiplicity l (l i) : ℝ))
84      (fun b hb => by simp only [hzeroR (l i) b hb, zero_mul, zero_div])
85  simp_rw [hi]
86  rw [Finset.sum_comm]
87  calc
88    _ = ∑ b : B, ∑ a : A, K a b*f a b := by
89      apply Finset.sum_congr rfl
90      intro b _
91      exact sum_uniform_copies l (fun a => K a b*f a b)
92        (fun a ha => by simp only [hzeroL a b ha, zero_mul])
93    _ = _ := Finset.sum_comm
