20/-- Choose a Taylor order first, then any sufficiently large fixed cutoff.
21The actual physical exponential approaches that oscillator polynomial to
22arbitrary precision, for every convergent displacement sequence. -/
23theorem exists_order_eventually_exponential_near_polynomial
24    (mu : ℕ → Fin d → ℕ) (hmu : ∀ N, Antitone (mu N))
25    (δ : ℕ → ℝ) (hδ : Tendsto δ atTop atTop)
26    (hgap : ∀ᶠ N in atTop, ∀ a : PositiveRoot d, δ N ≤ rootGap (mu N) a)
27    (z : ℕ → PositiveRoot d → ℂ) (z₀ : PositiveRoot d → ℂ)
28    (hz : Tendsto z atTop (𝓝 z₀))
29    (w : List (PositiveRoot d)) (ε : ℝ) (hε : 0 < ε) :
30    ∃ M : ℕ, ∀ m ≥ M, ∀ Q : ℕ, loweringHeight w+m*d ≤ Q →
31      ∀ᶠ N in atTop,
32        ‖cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q
33            (exp (rootGenerator (n := ∑ j, mu N j) (mu N) (z N))
34              (normalizedLoweringWord (partitionHighestTensor (mu N) (hmu N)) (mu N) w)) -
35          oscillatorTaylor Q z₀ w m‖ < ε := by
36  let R := loweringHeight w
37  let B := (∑ a, ‖z₀ a‖)+1
38  have hB : 0 ≤ B := by dsimp [B]; positivity
39  have hzsum : Tendsto (fun N => ∑ a, ‖z N a‖) atTop (𝓝 (∑ a, ‖z₀ a‖)) :=
40    tendsto_finset_sum _ (fun a _ => (tendsto_pi_nhds.mp hz a).norm)
41  have hzB : ∀ᶠ N in atTop, (∑ a, ‖z N a‖) ≤ B :=
42    (hzsum.eventually_lt_const (by dsimp [B]; linarith)).mono (fun _ h => h.le)
43  obtain ⟨M,hM⟩ := eventually_atTop.mp
44    ((uniformTaylorBound_tendsto B hB R d).eventually_lt_const (show 0 < ε/4 by linarith))
45  refine ⟨M, ?_⟩
46  intro m hm Q hQ
47  have hsmall := hM m hm
48  have hnorm : ∀ᶠ N in atTop,
49      ‖normalizedLoweringWord (partitionHighestTensor (mu N) (hmu N)) (mu N) w‖ ≤ 2 :=
50    ((partition_normalizedWord_norm_tendsto mu hmu δ hδ hgap w).eventually_lt_const
51      (by norm_num : (1 : ℝ)<2)).mono (fun _ h => h.le)
52  have hpoly := cutoffEmbedding_taylor_tendsto mu hmu δ hδ hgap z z₀ hz R Q m hQ w le_rfl
53  have hpolyerr := (tendsto_iff_norm_sub_tendsto_zero.mp hpoly).eventually_lt_const
54    (show 0 < ε/2 by linarith)
55  have hlarge := hδ.eventually (eventually_ge_atTop (2*(((R+(m+1)*d : ℕ) : ℝ)+1)))
56  filter_upwards [hzB,hnorm,hpolyerr,hgap,hlarge,
57    cutoffEmbedding_eventually_contraction mu hmu δ hδ hgap Q] with N hzN hnN hpN hgN hLN hV
58  let x := normalizedLoweringWord (partitionHighestTensor (mu N) (hmu N)) (mu N) w
59  let G := rootGenerator (n := ∑ j, mu N j) (mu N) (z N)
60  let T := ∑ j ∈ Finset.range (m+1), ((j.factorial : ℝ)⁻¹) • ((G^j) x)
61  have hx : x ∈ cyclicCutoff (partitionHighestTensor (mu N) (hmu N)) (R : ℤ) :=
62    normalizedLoweringWord_mem_cyclicCutoff _ _ _ (by exact_mod_cast (le_rfl : loweringHeight w ≤ R))
63  have hrem : ‖exp G x - T‖ < ε/2 := by
64    have hr := exp_rootGenerator_taylor_remainder
65      (partitionHighestTensor (mu N) (hmu N)) (mu N)
66      (partitionHighestTensor_cartan (mu N) (hmu N))
67      (partitionHighestTensor_raising_zero (mu N) (hmu N)) (z N) R m
68      (fun a => hLN.trans (hgN a)) hx
69    have hb : ‖exp G x - T‖ ≤ uniformTaylorBound B R d m * 2 := by
70      apply hr.trans
71      unfold uniformTaylorBound
72      rw [div_mul_eq_mul_div]
73      apply div_le_div_of_nonneg_right _ (Nat.cast_nonneg _)
74      apply mul_le_mul _ hnN (norm_nonneg _) (by positivity)
75      apply pow_le_pow_left₀ (by positivity)
76      exact mul_le_mul_of_nonneg_right
77        (mul_le_mul_of_nonneg_left hzN (by norm_num)) (Real.sqrt_nonneg _)
78    linarith
79  let V := cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q
80  have hV' : ∀ y, ‖V y‖ ≤ ‖y‖ := hV
81  change ‖V T - oscillatorTaylor Q z₀ w m‖ < ε/2 at hpN
82  have hVT : ‖V (exp G x) - V T‖ ≤ ‖exp G x - T‖ := by
83    rw [← V.map_sub]
84    exact hV' (exp G x - T)
85  calc
86    ‖V (exp G x) - oscillatorTaylor Q z₀ w m‖ ≤
87        ‖V (exp G x) - V T‖ + ‖V T - oscillatorTaylor Q z₀ w m‖ :=
88      norm_sub_le_norm_sub_add_norm_sub _ _ _
89    _ ≤ ‖exp G x - T‖ + ‖V T - oscillatorTaylor Q z₀ w m‖ := add_le_add hVT le_rfl
90    _ < ε := by linarith
