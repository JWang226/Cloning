13/-- The full cyclic sector generated by actual lowering operators. -/
14def cyclicSector (Ω : TensorRegister n (Fin d)) : Submodule ℂ (TensorRegister n (Fin d)) :=
15  Submodule.span ℂ (Set.range (loweringWord Ω))

17theorem loweringWord_mem_cyclicSector (Ω : TensorRegister n (Fin d))

19  Submodule.subset_span (Set.mem_range_self w)

30/-- The full cyclic space is invariant under every literal matrix-unit generator. -/
31theorem cyclicSector_generator_invariant
32    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℂ)
33    (hweight : ∀ a, collectiveGenerator n a a Ω = mu a • Ω)
34    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
35    (a b : Fin d) {x : TensorRegister n (Fin d)} (hx : x ∈ cyclicSector Ω) :

44/-- A vector killed by every raising operator is orthogonal to every nonempty

46theorem inner_loweringWord_cons_eq_zero

54/-- The joint raising kernel within a cyclic highest sector is precisely its
55highest line. No representation classification or independence premise occurs. -/
56theorem cyclicSector_raising_kernel
57    (Ω : TensorRegister n (Fin d)) (hΩ : ‖Ω‖ = 1)
58    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
59    {x : TensorRegister n (Fin d)} (hx : x ∈ cyclicSector Ω)
60    (hxraise : ∀ a b, a < b → collectiveGenerator n a b x = 0) :
61    x = ⟪Ω, x⟫_ℂ • Ω := by
62  let y := x - ⟪Ω, x⟫_ℂ • Ω
63  have hy : y ∈ cyclicSector Ω := (cyclicSector Ω).sub_mem hx
64    ((cyclicSector Ω).smul_mem _ (highest_mem_cyclicSector Ω))
65  have hyraise : ∀ a b, a < b → collectiveGenerator n a b y = 0 := by
66    intro a b hab
67    simp only [y, map_sub, map_smul, hxraise a b hab, hraise a b hab, smul_zero, sub_zero]
68  have hΩinner : ⟪Ω, Ω⟫_ℂ = 1 := by
69    rw [inner_self_eq_norm_sq_to_K, hΩ]
70    norm_num
71  have hΩy : ⟪Ω, y⟫_ℂ = 0 := by
72    simp only [y, inner_sub_right, inner_smul_right, hΩinner, mul_one, sub_self]
73  have horth : ∀ z ∈ cyclicSector Ω, ⟪z, y⟫_ℂ = 0 := by
74    intro z hz
75    induction hz using Submodule.span_induction with
76    | mem z hz =>
77      obtain ⟨w, rfl⟩ := hz
78      cases w with
79      | nil => exact hΩy
80      | cons a w => exact inner_loweringWord_cons_eq_zero Ω y hyraise a w
81    | zero => simp
82    | add u v hu hv ihu ihv => simp [inner_add_left, ihu, ihv]
83    | smul c z hz ih => simp [inner_smul_left, ih]
84  have hyzero : y = 0 := (inner_self_eq_zero).mp (horth y hy)
85  exact sub_eq_zero.mp hyzero

87theorem cyclicSector_raising_iff
88    (Ω : TensorRegister n (Fin d)) (hΩ : ‖Ω‖ = 1)
89    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
90    {x : TensorRegister n (Fin d)} (hx : x ∈ cyclicSector Ω) :
91    (∀ a b, a < b → collectiveGenerator n a b x = 0) ↔
92      x ∈ Submodule.span ℂ {Ω} := by
93  constructor
94  · intro h
95    rw [cyclicSector_raising_kernel Ω hΩ hraise hx h]
96    exact Submodule.smul_mem _ _ (Submodule.subset_span (Set.mem_singleton Ω))
97  · intro h a b hab
98    obtain ⟨c, rfl⟩ := Submodule.mem_span_singleton.mp h
99    simp only [map_smul, hraise a b hab, smul_zero]

101/-- Every endomorphism of the physical cyclic sector commuting with all actual
102matrix units is scalar on that sector. -/
103theorem cyclicSector_commutant_scalar
104    (Ω : TensorRegister n (Fin d)) (hΩ : ‖Ω‖ = 1)
105    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
106    (T : TensorRegister n (Fin d) →ₗ[ℂ] TensorRegister n (Fin d))
107    (hT : ∀ x ∈ cyclicSector Ω, T x ∈ cyclicSector Ω)
108    (hcomm : ∀ a b x, x ∈ cyclicSector Ω →
109      T (collectiveGenerator n a b x) = collectiveGenerator n a b (T x)) :
110    ∃ c : ℂ, ∀ x ∈ cyclicSector Ω, T x = c • x := by
111  let c := ⟪Ω, T Ω⟫_ℂ
112  have hTΩ : T Ω = c • Ω := by
113    apply cyclicSector_raising_kernel Ω hΩ hraise (hT Ω (highest_mem_cyclicSector Ω))
114    intro a b hab
115    rw [← hcomm a b Ω (highest_mem_cyclicSector Ω), hraise a b hab, map_zero]
116  have hword : ∀ w, T (loweringWord Ω w) = c • loweringWord Ω w := by
117    intro w
118    induction w with
119    | nil => exact hTΩ
120    | cons a w ih =>
121      rw [loweringWord, hcomm a.val.2 a.val.1 _ (loweringWord_mem_cyclicSector Ω w), ih,
122        map_smul]
123  refine ⟨c, ?_⟩
124  intro x hx
125  induction hx using Submodule.span_induction with
126  | mem x hx => obtain ⟨w, rfl⟩ := hx; exact hword w
127  | zero => simp
128  | add x y hx hy ihx ihy => simp [map_add, ihx, ihy, smul_add]
129  | smul a x hx ih => simp [map_smul, ih, smul_smul, mul_comm]
