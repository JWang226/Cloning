20def principalSpecialization (d : ℕ) : MvPolynomial (Fin d) ℂ →+* Polynomial ℂ :=
21  MvPolynomial.eval₂Hom Polynomial.C (fun i ↦ Polynomial.X ^ i.val)

27/-- The geometric quotient of `t^u-t^v` by `1-t`, for `u≤v`. -/
28def powerDifferenceQuotient (u v : ℕ) : Polynomial ℂ :=
29  Polynomial.X ^ u * ∑ k ∈ Finset.range (v-u), Polynomial.X ^ k

31theorem power_difference_factor {u v : ℕ} (huv : u ≤ v) :
32    (Polynomial.X : Polynomial ℂ) ^ u - Polynomial.X ^ v =
33      (1 - Polynomial.X) * powerDifferenceQuotient u v := by
34  have hgeom := geom_sum_mul (Polynomial.X : Polynomial ℂ) (v-u)
35  have hp : (Polynomial.X : Polynomial ℂ) ^ u * Polynomial.X ^ (v-u) = Polynomial.X ^ v := by
36    rw [← pow_add, Nat.add_sub_of_le huv]
37  unfold powerDifferenceQuotient
38  linear_combination (Polynomial.X : Polynomial ℂ) ^ u * hgeom + hp

44/-- The principal specialization of the alternant has no permutation sign. -/
45theorem principalSpecialization_alternant {d : ℕ} (a : Fin d → ℕ) :
46    principalSpecialization d (alternant a) =
47      ∏ r : PositiveRoot d, ((Polynomial.X : Polynomial ℂ) ^ a r.val.2 - Polynomial.X ^ a r.val.1) := by
48  rw [alternant, (principalSpecialization d).map_det]
49  have hm : (principalSpecialization d).mapMatrix (fun i j : Fin d ↦ MvPolynomial.X i ^ a j) =
50      (Matrix.vandermonde (fun j : Fin d ↦ (Polynomial.X : Polynomial ℂ) ^ a j)).transpose := by
51    apply Matrix.ext
52    intro i j
53    simp only [RingHom.mapMatrix_apply, Matrix.map_apply, map_pow, principalSpecialization_X,
54      Matrix.transpose_apply, Matrix.vandermonde_apply]
55    rw [← pow_mul, ← pow_mul, Nat.mul_comm]
56  rw [hm, Matrix.det_transpose, Matrix.det_vandermonde, ← prod_positiveRoot]

64/-- The factor left after removing every root's common first-order zero. -/
65def alternantQuotient {d : ℕ} (a : Fin d → ℕ) : Polynomial ℂ :=
66  ∏ r : PositiveRoot d, powerDifferenceQuotient (a r.val.2) (a r.val.1)

68def denominatorQuotient (d : ℕ) : Polynomial ℂ :=
69  ∏ r : PositiveRoot d, powerDifferenceQuotient r.val.1.val r.val.2.val

71theorem principalSpecialization_alternant_factor {d : ℕ} (a : Fin d → ℕ) (ha : StrictAnti a) :
72    principalSpecialization d (alternant a) =
73      (1 - Polynomial.X : Polynomial ℂ) ^ Fintype.card (PositiveRoot d) * alternantQuotient a := by
74  rw [principalSpecialization_alternant]
75  have hfac (r : PositiveRoot d) := power_difference_factor (ha r.property).le
76  simp_rw [hfac]
77  simp [alternantQuotient, Finset.prod_mul_distrib]

98/-- Cancel the shared zero as a nonzero polynomial, before specialization at one. -/
99theorem principalSpecialization_quotient_identity {d : ℕ}
100    (χ : MvPolynomial (Fin d) ℂ) (a : Fin d → ℕ) (ha : StrictAnti a)
101    (hχ : χ * denominator d = alternant a) :
102    principalSpecialization d χ * denominatorQuotient d = alternantQuotient a := by
103  have he := congrArg (principalSpecialization d) hχ
104  rw [map_mul, principalSpecialization_denominator_factor,
105    principalSpecialization_alternant_factor a ha] at he
106  have hX : (1 - Polynomial.X : Polynomial ℂ) ≠ 0 := by
107    intro h
108    have hh := congrArg (Polynomial.eval 0) h
109    simpa using hh
110  apply mul_left_cancel₀ (pow_ne_zero (Fintype.card (PositiveRoot d)) hX)
111  simpa only [mul_left_comm, mul_assoc] using he

113/-- The finite principal-specialization dimension formula for a strict alternant. -/
114theorem eval_one_eq_root_gap_product {d : ℕ}
115    (χ : MvPolynomial (Fin d) ℂ) (a : Fin d → ℕ) (ha : StrictAnti a)
116    (hχ : χ * denominator d = alternant a) :
117    MvPolynomial.eval (fun _ ↦ 1) χ =
118      ∏ r : PositiveRoot d, (((a r.val.1 - a r.val.2 : ℕ) : ℂ) /
119        ((r.val.2.val - r.val.1.val : ℕ) : ℂ)) := by
120  have he := congrArg (Polynomial.eval 1)
121    (principalSpecialization_quotient_identity χ a ha hχ)
122  simp only [Polynomial.eval_mul, eval_one_principalSpecialization,
123    denominatorQuotient, alternantQuotient, Polynomial.eval_prod,
124    eval_one_powerDifferenceQuotient] at he
125  rw [Finset.prod_div_distrib]
126  apply (eq_div_iff ?_).mpr he
127  apply Finset.prod_ne_zero_iff.mpr
128  intro r _
