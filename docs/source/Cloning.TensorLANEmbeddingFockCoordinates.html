50/-- The positive geometric weights and every occupation vector are transported

52theorem fockReindex_productGeometric (e : Fin a ≃ Fin b) (q : Fin a → ℝ)
53    (hq0 : ∀ i, 0 ≤ q i) (hq1 : ∀ i, q i < 1) :
54    conjugationLinearMap (fockReindex e).toLinearIsometry.toContinuousLinearMap
55      (vectorMixture (numberBasis a) (ThermalWitness.productGeometric q)) =
56    vectorMixture (numberBasis b) (ThermalWitness.productGeometric (fun j => q (e.symm j))) := by
57  let Φ := QuantumChannel.ofIsometry (fockReindex e).toLinearIsometry
58  have hs := summable_weighted_projectors (numberBasis a) (numberBasis a).orthonormal.norm_eq_one
59    (ThermalWitness.productGeometric q) (Thermal.multimode_geometric_hasSum hq0 hq1).summable
60  change Φ.toPositiveTracePreservingMap.toContinuousLinearMap
61    (∑' k, (ThermalWitness.productGeometric q k : ℂ) • vectorProjector (numberBasis a k)) = _
62  rw [ContinuousLinearMap.map_tsum _ hs]
63  simp only [map_smul]
64  change (∑' k, (ThermalWitness.productGeometric q k : ℂ) •
65    conjugationLinearMap (fockReindex e).toLinearIsometry.toContinuousLinearMap (vectorProjector (numberBasis a k))) = _
66  simp only [conjugationLinearMap_vectorProjector, LinearIsometry.coe_toContinuousLinearMap,
67    LinearIsometryEquiv.coe_toLinearIsometry, fockReindex_numberBasis]
68  rw [← (modeOccupationEquiv e).tsum_eq]
69  unfold vectorMixture
70  apply tsum_congr

73  congr 2
