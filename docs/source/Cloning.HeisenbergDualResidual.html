18/-- A positive map with identically zero trace is zero on every complex input. -/
19theorem positiveMap_eq_zero_of_trace_zero (Φ : TraceClass H →ₗ[ℂ] TraceClass K)
20    (hpos : ∀ T, 0 ≤ T.1 → 0 ≤ (Φ T).1)
21    (htrace : ∀ T, traceCLM (Φ T) = 0) : Φ = 0 := by
22  ext T
23  apply norm_eq_zero.mp

29/-- Positive rescaling preserves complete positivity. -/
30theorem IsCompletelyPositive.real_smul
31    {Φ : TraceClass H →ₗ[ℂ] TraceClass K} (hΦ : IsCompletelyPositive Φ)
32    {c : ℝ} (hc : 0 ≤ c) : IsCompletelyPositive ((c : ℂ) • Φ) := by
33  intro n A hA x
34  have h := hΦ n A hA x

37  simpa only [inner_smul_right, Finset.mul_sum] using mul_nonneg hc' h

39/-- The normalization is the genuine original map divided by its success
40probability; its positivity, complete positivity and exact trace law are proved. -/
41def normalizedQuantumChannel
42    (Φ : TraceClass H →ₗ[ℂ] TraceClass K)
43    (hpos : ∀ T, 0 ≤ T.1 → 0 ≤ (Φ T).1)
44    (hCP : IsCompletelyPositive Φ) (c : ℝ) (hc : 0 < c)
45    (htrace : ∀ T, traceCLM (Φ T) = (c : ℂ) * traceCLM T) :
46    QuantumChannel H K where
47  toLinearMap := ((c⁻¹ : ℝ) : ℂ) • Φ
48  map_nonneg := by
49    intro T hT
50    change 0 ≤ (((c⁻¹ : ℝ) : ℂ) • (Φ T).1)
51    have heq : (((c⁻¹ : ℝ) : ℂ) • (Φ T).1) = c⁻¹ • (Φ T).1 := by
52      ext x
53      simp only [ContinuousLinearMap.smul_apply, RCLike.real_smul_eq_coe_smul (K := ℂ)]
54      rfl
55    rw [heq]
56    exact smul_nonneg (inv_nonneg.mpr hc.le) (hpos T hT)
57  trace_preserving := by
58    intro T
59    change traceCLM (((c⁻¹ : ℝ) : ℂ) • Φ T) = traceCLM T
60    rw [map_smul, htrace, smul_eq_mul, ← mul_assoc,
61      ← Complex.ofReal_mul, inv_mul_cancel₀ hc.ne', Complex.ofReal_one, one_mul]
62  completelyPositive := hCP.real_smul (inv_nonneg.mpr hc.le)

64@[simp] theorem normalizedQuantumChannel_scale
65    (Φ : TraceClass H →ₗ[ℂ] TraceClass K)
66    (hpos : ∀ T, 0 ≤ T.1 → 0 ≤ (Φ T).1)
67    (hCP : IsCompletelyPositive Φ) (c : ℝ) (hc : 0 < c)

125/-- Every covariant completely positive trace-nonincreasing map is either
126zero or a scalar multiple of an actual CPTP map. The scalar and normalized
127channel are derived from the given physical map. -/
128theorem covariant_cpTNI_zero_or_scaled_channel
129    (Γ : TraceClass (Fock d) →ₗ[ℂ] TraceClass (Fock d))
130    (hpos : ∀ T, 0 ≤ T.1 → 0 ≤ (Γ T).1)
131    (hCP : IsCompletelyPositive Γ)
132    (htrace : ∀ T, 0 ≤ T.1 → (traceCLM (Γ T)).re ≤ (traceCLM T).re)
133    (r : ℝ)
134    (hcov : ∀ a T, Γ (displacementTraceMap a T) =
135      displacementTraceMap (r • a) (Γ T)) :
136    Γ = 0 ∨ ∃ (c : ℝ) (Φ : QuantumChannel (Fock d) (Fock d)),
137      0 < c ∧ c ≤ 1 ∧ Γ = (c : ℂ) • Φ.toLinearMap ∧
138      ∀ a T, Φ.toLinearMap (displacementTraceMap a T) =
139        displacementTraceMap (r • a) (Φ.toLinearMap T) := by
140  let Γc := toContinuousLinearMapOfTraceBound Γ hpos (c := 1) zero_le_one
141    (fun T hT => by simpa only [one_mul] using htrace T hT)
142  have hb := covariantTraceMass_mem_unitInterval Γc hpos htrace
143  have hscale := covariant_trace_scaling Γc r hcov hpos
144  by_cases hc : covariantTraceMass Γc = 0
145  · left
146    apply positiveMap_eq_zero_of_trace_zero Γ hpos
147    intro T
148    simpa only [hc, Complex.ofReal_zero, zero_mul] using hscale T
149  · right
150    have hcpos : 0 < covariantTraceMass Γc := lt_of_le_of_ne hb.1 (Ne.symm hc)
151    let Φ := normalizedQuantumChannel Γ hpos hCP (covariantTraceMass Γc) hcpos hscale
152    refine ⟨covariantTraceMass Γc, Φ, hcpos, hb.2, ?_, ?_⟩
153    · exact (normalizedQuantumChannel_scale Γ hpos hCP _ hcpos hscale).symm
154    · intro a T
155      change (((covariantTraceMass Γc)⁻¹ : ℝ) : ℂ) • Γ (displacementTraceMap a T) =
156        displacementTraceMap (r • a) ((((covariantTraceMass Γc)⁻¹ : ℝ) : ℂ) • Γ T)
