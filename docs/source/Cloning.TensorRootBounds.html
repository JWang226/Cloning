15theorem creator_norm_sq_le_of_raw_commutator
16    (F A : H →L[ℂ] H)
17    (hadj : ∀ x y, ⟪F x, y⟫_ℂ = ⟪x, A y⟫_ℂ)
18    (B : ℝ) (x : H) (hdef : ‖A (F x) - F (A x)‖ ≤ B * ‖x‖) :
19    ‖F x‖ ^ 2 ≤ ‖A x‖ ^ 2 + B * ‖x‖ ^ 2 := by
20  have hF : (⟪x, A (F x)⟫_ℂ).re = ‖F x‖ ^ 2 := by
21    rw [← hadj]
22    exact (norm_sq_eq_re_inner (𝕜 := ℂ) (F x)).symm
23  have hA : (⟪x, F (A x)⟫_ℂ).re = ‖A x‖ ^ 2 := by
24    calc
25      _ = (⟪F (A x), x⟫_ℂ).re := inner_re_symm (𝕜 := ℂ) x (F (A x))
26      _ = _ := by rw [hadj]; exact (norm_sq_eq_re_inner (𝕜 := ℂ) (A x)).symm
27  have hh : (⟪x, A (F x) - F (A x)⟫_ℂ).re ≤ B * ‖x‖ ^ 2 := by
28    calc
29      _ ≤ ‖⟪x, A (F x) - F (A x)⟫_ℂ‖ := Complex.re_le_norm _
30      _ ≤ ‖x‖ * ‖A (F x) - F (A x)‖ := norm_inner_le_norm _ _
31      _ ≤ ‖x‖ * (B * ‖x‖) := mul_le_mul_of_nonneg_left hdef (norm_nonneg _)
32      _ = _ := by ring
33  simp only [inner_sub_right, Complex.sub_re, hF, hA] at hh
34  linarith

36theorem creator_norm_sq_le_raw_cutoff
37    (F A : H →L[ℂ] H) (K : ℕ → Submodule ℂ H)
38    (hadj : ∀ x y, ⟪F x, y⟫_ℂ = ⟪x, A y⟫_ℂ)
39    (hzero : ∀ x ∈ K 0, A x = 0)
40    (hlower : ∀ r x, x ∈ K (r + 1) → A x ∈ K r)
41    (L : ℕ) (B : ℝ) (hB : 0 ≤ B)
42    (hcomm : ∀ r, r ≤ L → ∀ x ∈ K r, ‖A (F x) - F (A x)‖ ≤ B * ‖x‖)
43    (r : ℕ) (hr : r ≤ L) (x : H) (hx : x ∈ K r) :
44    ‖F x‖ ^ 2 ≤ ((r : ℝ) + 1) * B * ‖x‖ ^ 2 := by
45  induction r generalizing x with
46  | zero =>
47      have hh := creator_norm_sq_le_of_raw_commutator F A hadj B x (hcomm 0 hr x hx)
48      simpa [hzero x hx] using hh
49  | succ r ih =>
50      have hr' : r ≤ L := by omega
51      let C : ℝ := Real.sqrt (((r : ℝ) + 1) * B)
52      have hC : 0 ≤ C := Real.sqrt_nonneg _
53      have hCsq : C ^ 2 = ((r : ℝ) + 1) * B := Real.sq_sqrt (by positivity)
54      have hlocal : ∀ y ∈ K r, ‖F y‖ ≤ C * ‖y‖ := by
55        intro y hy
56        have hybound := ih hr' y hy
57        have hpos : 0 ≤ C * ‖y‖ := mul_nonneg hC (norm_nonneg _)
58        have heq : (C * ‖y‖) ^ 2 = ((r : ℝ) + 1) * B * ‖y‖ ^ 2 := by
59          rw [mul_pow, hCsq]
60        nlinarith [norm_nonneg (F y)]
61      have hAbound := PBW.annihilator_norm_le_from_local_creator F A (K r) C hC hadj hlocal
62        x (hlower r x hx)
63      have hAsq : ‖A x‖ ^ 2 ≤ ((r : ℝ) + 1) * B * ‖x‖ ^ 2 := by
64        have hh := (sq_le_sq₀ (norm_nonneg (A x)) (mul_nonneg hC (norm_nonneg x))).mpr hAbound
65        simpa [mul_pow, hCsq] using hh
66      have hh := creator_norm_sq_le_of_raw_commutator F A hadj B x (hcomm (r + 1) hr x hx)
67      simp only [Nat.cast_add, Nat.cast_one]
68      nlinarith
69end RawRecurrence

85theorem root_commutator_norm_le
86    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
87    (hweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
88    (r : ℕ) {x : TensorRegister n (Fin d)} (hx : x ∈ cyclicCutoff Ω r) (a b : Fin d) :
89    ‖collectiveGenerator n a b (collectiveGenerator n b a x) -
90      collectiveGenerator n b a (collectiveGenerator n a b x)‖ ≤
91      (|(mu a : ℝ) - mu b| + 2 * r) * ‖x‖ := by
92  rw [root_diagonal_commutator]
93  let y := collectiveGenerator n a a x - collectiveGenerator n b b x
94  have he : y = (y - ((mu a : ℂ) - mu b) • x) + ((mu a : ℂ) - mu b) • x := by module
95  calc
96    ‖y‖ = ‖(y - ((mu a : ℂ) - mu b) • x) + ((mu a : ℂ) - mu b) • x‖ := congrArg norm he
97    _ ≤ ‖y - ((mu a : ℂ) - mu b) • x‖ + ‖((mu a : ℂ) - mu b) • x‖ := norm_add_le _ _
98    _ ≤ (2 * r : ℝ) * ‖x‖ + ‖((mu a : ℂ) - mu b) • x‖ :=
99      add_le_add (cartan_difference_defect_norm_le Ω mu hweight r hx a b) le_rfl
100    _ = _ := by
101      rw [norm_smul]
102      simp only [← Complex.ofReal_natCast, ← Complex.ofReal_sub, Complex.norm_real, Real.norm_eq_abs]
103      ring

107theorem lowering_norm_sq_le_cyclicCutoff
108    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
109    (hweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)

112    (hx : x ∈ cyclicCutoff Ω r) :
113    ‖collectiveGenerator n b a x‖ ^ 2 ≤
114      ((r : ℝ) + 1) * (|(mu a : ℝ) - mu b| + 2 * r) * ‖x‖ ^ 2 := by

136/-- The same local estimate holds for every off-diagonal raw root, in either orientation. -/
137theorem root_norm_le_cyclicCutoff
138    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
139    (hweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
140    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
141    (a b : Fin d) (hab : a ≠ b) (r : ℕ) {x : TensorRegister n (Fin d)}
142    (hx : x ∈ cyclicCutoff Ω r) :
143    ‖collectiveGenerator n a b x‖ ≤
144      Real.sqrt (((r : ℝ) + 1) * (|(mu a : ℝ) - mu b| + 2 * r)) * ‖x‖ := by
145  rcases lt_or_gt_of_ne hab with hab | hba
146  · apply PBW.annihilator_norm_le_from_local_creator
147      (collectiveGenerator n b a) (collectiveGenerator n a b) (cyclicCutoff Ω r)
148      _ (Real.sqrt_nonneg _) (collectiveGenerator_inner_adjoint a b)
149      (fun y hy => lowering_norm_le_cyclicCutoff Ω mu hweight hraise a b hab r hy) x
150    exact PBW.annihilator_mem_same_cutoff (collectiveGenerator n a b)
151      (fun t => cyclicCutoff Ω (t : ℤ))
152      (fun i j hij => cyclicCutoff_monotone Ω (Int.ofNat_le.mpr hij))
153      (fun y hy => raising_zero_cyclicCutoff Ω (fun i => (mu i : ℂ)) hweight hraise a b hab hy)
154      (fun t y hy => raising_lowers_cyclicCutoff Ω (fun i => (mu i : ℂ)) hweight hraise a b hab t hy)
