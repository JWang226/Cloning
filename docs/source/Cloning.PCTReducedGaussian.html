50/-- The positive diagonal Schmidt coefficient matrix. -/
51def schmidtCoefficients (p : A → ℝ) : Matrix A A ℂ :=
52  Matrix.diagonal (fun a => (Real.sqrt (p a) : ℂ))

54theorem schmidtCoefficients_star (p : A → ℝ) :
55    (schmidtCoefficients p)ᴴ = schmidtCoefficients p := by
56  simp [schmidtCoefficients, Matrix.diagonal_conjTranspose]

58theorem schmidtCoefficients_square (p : A → ℝ) (hp : ∀ a, 0 ≤ p a) :

72/-- The derivative of partial trace along a purification tangent matrix. -/
73def partialTraceDifferential (p : A → ℝ) (Z : Matrix A A ℂ) : Matrix A A ℂ :=
74  Z * schmidtCoefficients p + schmidtCoefficients p * Zᴴ

76theorem partialTraceDifferential_apply (p : A → ℝ) (Z : Matrix A A ℂ) (a c : A) :
77    partialTraceDifferential p Z a c =

93/-- The tangent orthogonality condition is exactly the vanishing of the

95theorem schmidt_inner_tangent (p : A → ℝ) (Z : Matrix A A ℂ) :

102theorem partialTraceDifferential_trace_zero (p : A → ℝ) (Z : Matrix A A ℂ)
103    (horth : ⟪coefficientVector (schmidtCoefficients p), coefficientVector Z⟫_ℂ = 0) :
104    Matrix.trace (partialTraceDifferential p Z) = 0 := by
105  rw [schmidt_inner_tangent] at horth
106  have hre := congrArg Complex.re horth
107  simp only [Complex.re_sum, Complex.mul_re, Complex.ofReal_re, Complex.ofReal_im,
108    zero_mul, sub_zero, Complex.zero_re] at hre
109  simp only [Matrix.trace, Matrix.diag, partialTraceDifferential_diagonal,
110    Complex.ofReal_mul, Complex.ofReal_ofNat, Finset.mul_sum]
111  simp_rw [mul_assoc]
112  rw [← Finset.mul_sum]
113  have hs : (∑ a, (Real.sqrt (p a) : ℂ) * ((Z a a).re : ℂ)) = 0 := by
114    exact_mod_cast hre

130/-- The normalized physical tangent vector, with its exact quadratic denominator. -/
131def normalizedTangentVector (p : A → ℝ) (Z : Matrix A A ℂ) (e t : ℝ) : Register (A × A) :=
132  ((Real.sqrt (1 + t ^ 2 * e) : ℂ)⁻¹) •
133    coefficientVector (schmidtCoefficients p + t • Z)

135theorem normalizedTangent_reduced_exact (p : A → ℝ) (hp : ∀ a, 0 ≤ p a)
136    (Z : Matrix A A ℂ) (e : ℝ) (he : 0 ≤ e) (t : ℝ) :
137    reducedDensityMatrix (normalizedTangentVector p Z e t) =
138      (1 + t ^ 2 * e)⁻¹ •
139        (Matrix.diagonal (fun a => (p a : ℂ)) +
140          t • partialTraceDifferential p Z + t ^ 2 • (Z * Zᴴ)) := by
141  have hd : 0 ≤ 1 + t ^ 2 * e := by positivity
142  rw [normalizedTangentVector, reducedDensityMatrix_smul]
143  have hc : ((Real.sqrt (1 + t ^ 2 * e) : ℂ)⁻¹) *
144      star ((Real.sqrt (1 + t ^ 2 * e) : ℂ)⁻¹) = ((1 + t ^ 2 * e)⁻¹ : ℝ) := by
145    simp only [map_inv₀, Complex.star_def, Complex.conj_ofReal, ← mul_inv]
146    rw [← Complex.ofReal_mul, Real.mul_self_sqrt hd, Complex.ofReal_inv]
