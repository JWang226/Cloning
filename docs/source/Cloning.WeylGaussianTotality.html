15/-- The translated vacuum projections jointly separate vectors in every regular representation. -/
16theorem RegularWeyl.eq_zero_of_projection_translates_zero (v : H)
17    (hv : ∀ a : Fin d → ℂ, W.gaussianProjection (W.operator a v)=0) : v=0 := by
18  apply W.eq_zero_of_gaussian_averages_zero v
19  intro t ht
20  let s : ℝ := (t-1/2)⁻¹
21  have hs : 0<s := inv_pos.mpr (sub_pos.mpr ht)
22  have h := W.gaussianProjection_twirl (t:=fun _ => s) (fun _ => hs) v
23  simp only [hv, map_zero, smul_zero, integral_zero] at h
24  have hc : (((∏ _i : Fin d, Real.pi/s : ℝ) : ℂ)) ≠ 0 := by
25    apply Complex.ofReal_ne_zero.mpr
26    apply Finset.prod_ne_zero_iff.mpr
27    intro i _
28    exact (div_pos Real.pi_pos hs).ne'
29  have hi := (smul_eq_zero.mp h.symm).resolve_left hc
30  have hp : (((Real.pi^d : ℝ) : ℂ)) ≠ 0 :=
31    Complex.ofReal_ne_zero.mpr (pow_ne_zero _ Real.pi_ne_zero)
32  have he (a : Fin d → ℂ) :
33      (weylGaussianWeight a : ℂ)*(gaussianWeight (fun _ : Fin d => s⁻¹) a : ℂ) =
34        (((Real.pi^d : ℝ) : ℂ))⁻¹*(gaussianWeight (fun _ : Fin d => t) a : ℂ) := by
35    rw [weylGaussianWeight_eq_gaussianWeight, Complex.ofReal_div]
36    have hsum : ((fun _ : Fin d => (1/2:ℝ))+(fun _ : Fin d => s⁻¹))=(fun _ => t) := by
37      funext i
38      simp only [Pi.add_apply, s, inv_inv]
39      ring
40    have hm := gaussianWeight_mul (fun _ : Fin d => (1/2:ℝ)) (fun _ => s⁻¹) a
41    rw [hsum] at hm
42    calc
43      _ = (((Real.pi^d : ℝ) : ℂ))⁻¹ *
44          ((gaussianWeight (fun _ : Fin d => (1/2:ℝ)) a : ℂ) *
45            (gaussianWeight (fun _ : Fin d => s⁻¹) a : ℂ)) := by ring
46      _ = _ := by rw [← Complex.ofReal_mul, hm]
47  simp_rw [he, mul_smul] at hi
48  rw [integral_smul] at hi
49  exact (smul_eq_zero.mp hi).resolve_left (inv_ne_zero hp)

51/-- Orthogonality to all displaced vectors in the Gaussian projection range forces zero. -/
52theorem RegularWeyl.gaussianProjection_translates_total (v : H)
53    (hv : ∀ (a : Fin d → ℂ) (x : H), ⟪W.operator a (W.gaussianProjection x),v⟫_ℂ=0) : v=0 := by
54  apply W.eq_zero_of_projection_translates_zero v
55  intro a
56  apply ext_inner_left ℂ
57  intro x
58  rw [inner_zero_right]
59  have hh := hv (-a) x
60  rw [W.inner_operator, neg_neg, W.gaussianProjection_symmetric] at hh
61  exact hh
