94theorem rootGenerator_norm_le (z : PositiveRoot d → ℂ) (R : ℕ)
95    (hgap : ∀ a : PositiveRoot d, 2 * ((R : ℝ)+1) ≤ (mu a.val.1 : ℝ) - mu a.val.2)
96    {x : TensorRegister n (Fin d)} (hx : x ∈ cyclicCutoff Ω R) :
97    ‖rootGenerator (n := n) mu z x‖ ≤
98      (2 * ∑ a, ‖z a‖) * Real.sqrt (((R : ℝ)+1)*2) * ‖x‖ := by
99  simp only [rootGenerator, ContinuousLinearMap.sum_apply, ContinuousLinearMap.sub_apply,
100    ContinuousLinearMap.smul_apply]
101  calc
102    _ ≤ ∑ a : PositiveRoot d, ‖z a • normalizedCreator n mu a.val.1 a.val.2 x -
103        star (z a) • normalizedAnnihilator n mu a.val.1 a.val.2 x‖ := norm_sum_le _ _
104    _ ≤ ∑ a : PositiveRoot d, (‖z a‖ + ‖z a‖) *
105        (Real.sqrt (((R : ℝ)+1)*2) * ‖x‖) := by
106      apply Finset.sum_le_sum

108      apply (norm_sub_le _ _).trans

129theorem rootGenerator_pow_norm_le (z : PositiveRoot d → ℂ) (R M : ℕ)
130    (hgap : ∀ a : PositiveRoot d, 2 * (((R+M*d : ℕ) : ℝ)+1) ≤ (mu a.val.1 : ℝ) - mu a.val.2)
131    {x : TensorRegister n (Fin d)} (hx : x ∈ cyclicCutoff Ω R) :
132    ‖(rootGenerator (n := n) mu z ^ M) x‖ ≤
133      ((2 * ∑ a, ‖z a‖) * Real.sqrt ((((R+M*d : ℕ) : ℝ)+1)*2)) ^ M * ‖x‖ := by
134  let C := (2 * ∑ a, ‖z a‖) * Real.sqrt ((((R+M*d : ℕ) : ℝ)+1)*2)
135  have hC : 0 ≤ C := by dsimp [C]; positivity
136  have hpow : ∀ m ≤ M, ‖(rootGenerator (n := n) mu z ^ m) x‖ ≤ C ^ m * ‖x‖ := by
137    intro m hm
138    induction m with
139    | zero => simp
140    | succ m ih =>
141      have hmem := rootGenerator_pow_mem Ω mu hweight hraise z R m hx
142      have hmem' : (rootGenerator (n := n) mu z ^ m) x ∈ cyclicCutoff Ω ((R+M*d : ℕ) : ℤ) :=
143        cyclicCutoff_nat_monotone Ω (Nat.add_le_add_left (Nat.mul_le_mul_right d (by omega : m ≤ M)) R) hmem
144      rw [pow_succ', ContinuousLinearMap.mul_apply]
145      calc
146        _ ≤ C * ‖(rootGenerator (n := n) mu z ^ m) x‖ := rootGenerator_norm_le Ω mu hweight hraise z _ hgap hmem'
147        _ ≤ C * (C ^ m * ‖x‖) := mul_le_mul_of_nonneg_left (ih (by omega)) hC
148        _ = C ^ (m+1) * ‖x‖ := by rw [pow_succ']; ring
149  exact hpow M le_rfl
