21/-- Orthogonal projection onto a finite set of actual Hilbert-basis vectors. -/
22def basisProjection {ι : Type*} (b : HilbertBasis ι ℂ H) (s : Finset ι) : H →L[ℂ] H :=
23  ∑ i ∈ s, InnerProductSpace.rankOne ℂ (b i) (b i)

56lemma sandwichCLM_vectorProjector (P : H →L[ℂ] H) (hP : IsSelfAdjoint P) (x : H) :
57    sandwichCLM P P (vectorProjector x) = vectorProjector (P x) := by
58  apply Subtype.ext
59  change P * InnerProductSpace.rankOne ℂ x x * P = InnerProductSpace.rankOne ℂ (P x) (P x)
60  simp only [ContinuousLinearMap.mul_def, InnerProductSpace.comp_rankOne,
61    InnerProductSpace.rankOne_comp]
62  rw [← ContinuousLinearMap.star_eq_adjoint, hP.star_eq]

64lemma norm_sandwich_projection_le {P : H →L[ℂ] H} (hP : IsStarProjection P)
65    (A : TraceClass H) : ‖sandwichCLM P P A‖ ≤ ‖A‖ := by
66  change traceNorm (P * A.1 * P) _ ≤ traceNorm A.1 A.2
67  calc
68    traceNorm (P * A.1 * P) _ ≤ ‖P‖ * traceNorm A.1 A.2 * ‖P‖ :=
69      traceNorm_mul_mul_le A.2 _
70    _ ≤ 1 * traceNorm A.1 A.2 * 1 :=
71      mul_le_mul (mul_le_mul_of_nonneg_right hP.norm_le (traceNorm_nonneg _ _))
72        hP.norm_le (norm_nonneg _) (mul_nonneg zero_le_one (traceNorm_nonneg _ _))
73    _ = traceNorm A.1 A.2 := by ring

75/-- Strong convergence of orthogonal projections gives trace-norm convergence
76of their compressions on every positive trace-class input. -/
77theorem sandwich_tendsto_of_strong_projections {α : Type*} {l : Filter α}
78    (P : α → H →L[ℂ] H) (hP : ∀ a, IsStarProjection (P a))
79    (hstrong : ∀ x : H, Tendsto (fun a => P a x) l (𝓝 x))
80    (A : TraceClass H) (hA : 0 ≤ A.1) :
81    Tendsto (fun a => sandwichCLM (P a) (P a) A) l (𝓝 A) := by
82  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
83  let v : w → H := fun i => CFC.sqrt A.1 (b i)
84  have hseries : HasSum (fun i => vectorProjector (v i)) A :=
85    positive_rankOne_series hA A.2 b
86  have hmass : Summable (fun i => ‖vectorProjector (v i)‖) :=
87    (positive_rankOne_mass hA A.2 b).summable
88  have hconv : Tendsto (fun a => ∑' i, sandwichCLM (P a) (P a) (vectorProjector (v i))) l
89      (𝓝 (∑' i, vectorProjector (v i))) := by
90    apply tendsto_tsum_of_dominated_convergence hmass
91    · intro i
92      simp only [sandwichCLM_vectorProjector _ (hP _).isSelfAdjoint]
93      exact vectorProjector_tendsto (hstrong (v i))
94    · exact Eventually.of_forall (fun a i => norm_sandwich_projection_le (hP a) _)
95  have hsum (a : α) :
96      (∑' i, sandwichCLM (P a) (P a) (vectorProjector (v i))) = sandwichCLM (P a) (P a) A :=
97    (hseries.mapL (sandwichCLM (P a) (P a))).tsum_eq
98  simpa only [hsum, hseries.tsum_eq] using hconv

100/-- Compression and replacement converge in the actual trace norm, with no
101tail or finite-rank approximation premise. -/
102theorem projectionReplacement_tendsto_of_strong_projections {α : Type*} {l : Filter α}
103    (P : α → H →L[ℂ] H) (hP : ∀ a, IsStarProjection (P a))
104    (hstrong : ∀ x : H, Tendsto (fun a => P a x) l (𝓝 x))
105    (σ : DensityState H) (A : TraceClass H) (hA : 0 ≤ A.1) :
106    Tendsto (fun a => projectionReplacement (P a) (hP a) σ A) l (𝓝 A) := by
107  have hc := sandwich_tendsto_of_strong_projections P hP hstrong A hA
108  have ht : Tendsto (fun a => traceCLM A - traceCLM (sandwichCLM (P a) (P a) A)) l
109      (𝓝 (0 : ℂ)) := by
110    have htr := traceCLM.continuous.tendsto A |>.comp hc
111    simpa using htr.const_sub (traceCLM A)
112  have heq (a : α) : traceCLM (sandwichCLM (1 - P a) (1 - P a) A) =
113      traceCLM A - traceCLM (sandwichCLM (P a) (P a) A) := by
114    linear_combination traceCLM_projection_split (hP a) A
115  have hresult := hc.add (ht.smul_const (TraceClass.ofOperator σ.op σ.traceClass))
116  simpa only [zero_smul, add_zero, projectionReplacement_apply, heq] using hresult

127/-- Contractivity turns pointwise approximation into uniform approximation on
128any trace-norm compact family of positive operators. -/
129theorem PositiveTracePreservingMap.tendstoUniformlyOn_of_isCompact
130    {α : Type*} {l : Filter α} (Φ : α → PositiveTracePreservingMap H H)
131    {S : Set (TraceClass H)} (hS : IsCompact S) (hpos : ∀ A ∈ S, 0 ≤ A.1)
132    (hpoint : ∀ A ∈ S, Tendsto (fun a => Φ a A) l (𝓝 A)) :
133    TendstoUniformlyOn (fun a A => Φ a A) (fun A => A) l S := by
134  classical
135  rw [Metric.tendstoUniformlyOn_iff]
136  intro ε hε
137  obtain ⟨t, htS, ht, hcover⟩ := hS.finite_cover_balls (show 0 < ε / 3 by positivity)
138  have hevent : ∀ᶠ a in l, ∀ B ∈ t, dist (Φ a B) B < ε / 3 := by
139    rw [ht.eventually_all]
140    intro B hB
141    exact (Metric.tendsto_nhds.mp (hpoint B (htS hB))) (ε / 3) (by positivity)
142  filter_upwards [hevent] with a ha
143  intro A hA
144  obtain ⟨B, hB, hAB⟩ : ∃ B ∈ t, dist A B < ε / 3 := by
145    simpa only [Set.mem_iUnion, Metric.mem_ball, exists_prop] using hcover hA
146  have hcon : dist (Φ a B) (Φ a A) ≤ dist B A := by
147    simpa only [dist_eq_norm] using (Φ a).norm_map_sub_le B A
148      (hpos B (htS hB)) (hpos A hA)
149  have hnet := ha B hB
150  have htri := dist_triangle A B (Φ a A)
151  have htri' := dist_triangle B (Φ a B) (Φ a A)
152  rw [dist_comm B A] at hcon
153  rw [dist_comm (Φ a B) B] at hnet
154  linarith

175/-- A finite cutoff with a pure replacement vector included in its retained subspace. -/
176def finiteBasisCutoff {ι : Type*} [DecidableEq ι] (b : HilbertBasis ι ℂ H)
177    (i₀ : ι) (s : Finset ι) : PositiveTracePreservingMap H H :=
178  projectionReplacement (basisProjection b (insert i₀ s))
179    (basisProjection_isStarProjection b (insert i₀ s))
180    (DensityState.pure (b i₀) (b.orthonormal.norm_eq_one i₀))

182/-- Every output of the finite cutoff is supported on the explicit finite basis span. -/
183lemma finiteBasisCutoff_supported {ι : Type*} [DecidableEq ι] (b : HilbertBasis ι ℂ H)
184    (i₀ : ι) (s : Finset ι) (A : TraceClass H) :
185    basisProjection b (insert i₀ s) * (finiteBasisCutoff b i₀ s A).1 *
186        basisProjection b (insert i₀ s) = (finiteBasisCutoff b i₀ s A).1 := by
187  apply projectionReplacement_supported
188  have h := sandwichCLM_vectorProjector (basisProjection b (insert i₀ s))
189    (basisProjection_isStarProjection b (insert i₀ s)).isSelfAdjoint (b i₀)
190  rw [basisProjection_apply_basis_of_mem b _ (Finset.mem_insert_self _ _)] at h
191  exact congrArg Subtype.val h

193lemma finiteBasisCutoff_mem_span {ι : Type*} [DecidableEq ι] (b : HilbertBasis ι ℂ H)
194    (i₀ : ι) (s : Finset ι) (A : TraceClass H) (x : H) :
195    (finiteBasisCutoff b i₀ s A).1 x ∈
196      Submodule.span ℂ (b '' ((insert i₀ s : Finset ι) : Set ι)) := by
197  have h := congrArg (fun T : H →L[ℂ] H => T x) (finiteBasisCutoff_supported b i₀ s A)
198  change basisProjection b (insert i₀ s)
199    ((finiteBasisCutoff b i₀ s A).1 (basisProjection b (insert i₀ s) x)) =
200      (finiteBasisCutoff b i₀ s A).1 x at h
201  rw [← h]
202  exact basisProjection_mem_span b _ _

204/-- Every output operator has finite-dimensional range, with no finite-dimensional

206theorem finiteBasisCutoff_finiteDimensional_range {ι : Type*} [DecidableEq ι]
207    (b : HilbertBasis ι ℂ H) (i₀ : ι) (s : Finset ι) (A : TraceClass H) :
208    FiniteDimensional ℂ (LinearMap.range (finiteBasisCutoff b i₀ s A).1.toLinearMap) := by
209  let U := Submodule.span ℂ (b '' ((insert i₀ s : Finset ι) : Set ι))
210  letI instFiniteDimensionalCutoffSpan : FiniteDimensional ℂ U :=
211    FiniteDimensional.span_of_finite ℂ ((insert i₀ s).finite_toSet.image b)
212  apply Submodule.finiteDimensional_of_le (S₂ := U)
213  rintro _ ⟨x, rfl⟩
214  exact finiteBasisCutoff_mem_span b i₀ s A x
