13private theorem occupationHeight_mono {k l : PositiveRoot d → ℕ} (h : ∀ a, k a ≤ l a) :

19def occupationSplitLeft (r : HeightOccupation d R) (s : OccupationSplit r) : HeightOccupation d R :=

41theorem exists_occupationSplitPair (r j k : HeightOccupation d R)
42    (h : ∀ a, j.val a + k.val a = r.val a) :

54/-- The height cutoff contains every component of a compatible occupation split. -/
55theorem sum_occupationSplits (r : HeightOccupation d R)
56    (f : HeightOccupation d R → HeightOccupation d R → ℂ)
57    (hz : ∀ j k, (¬ ∀ a, j.val a + k.val a = r.val a) → f j k = 0) :
58    (∑ j, ∑ k, f j k) = ∑ s : OccupationSplit r,
59      f (occupationSplitLeft r s) (occupationSplitRight r s) := by
60  classical
61  let F : HeightOccupation d R × HeightOccupation d R → ℂ := fun x => f x.1 x.2
62  have hs : (∑ x ∈ Finset.univ.image (occupationSplitPair r), F x) = ∑ x, F x := by
63    apply Finset.sum_subset (Finset.subset_univ _)
64    intro x _ hx
65    apply hz x.1 x.2
66    intro he
67    obtain ⟨s,hs⟩ := exists_occupationSplitPair r x.1 x.2 he
68    exact hx (Finset.mem_image.mpr ⟨s,Finset.mem_univ _,hs⟩)
69  rw [Finset.sum_image (fun a _ b _ h => occupationSplitPair_injective r h)] at hs
70  simpa only [Fintype.sum_prod_type, F, occupationSplitPair] using hs.symm

147theorem cutoffSplitAmplitude_thermal_matrix
148    (t q : PositiveRoot d → ℝ) (ht0 : ∀ a, 0 ≤ t a) (ht1 : ∀ a, t a ≤ 1)
149    (r l : HeightOccupation d R) :
