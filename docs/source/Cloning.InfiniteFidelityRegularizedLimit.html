3/-! The literal finite inverse-moment hypothesis of the manuscript: a finite
4right limit of the bounded regularized inverse moments. No separate uniform
5bound or monotonicity hypothesis on those moments is required. -/

14/-- Weighted root fidelity for arbitrary positive trace-class operators,
15with the inverse trace defined by its finite right limit at zero. -/
16theorem fidelity_sq_le_of_regularized_inverse_limit {A B W : H→L[ℂ] H}
17    (hA : 0≤A) (hB : 0≤B) (hW : 0≤W)
18    (hTA : IsTraceClass A) (hTB : IsTraceClass B) {M : ℝ}
19    (hInv : Tendsto (fun ε : ℝ=>
20      (trace (B*CFC.rpow (regularizedWeight W ε) (-1))
21        (isTraceClass_mul_mul (A:=1)
22          (B:=CFC.rpow (regularizedWeight W ε) (-1)) hTB)).re)
23      (𝓝[>] (0:ℝ)) (𝓝 M)) :
24    fidelity A B hA hB hTA hTB^2≤
25      (trace (A*W) (isTraceClass_mul_mul (A:=1) (B:=W) hTA)).re*M := by
26  have hid : Tendsto (fun ε : ℝ=>ε) (𝓝[>] (0:ℝ)) (𝓝 0) := nhdsWithin_le_nhds
27  have htrace : Tendsto (fun ε : ℝ=>
28      (trace (A*regularizedWeight W ε)
29        (isTraceClass_mul_mul (A:=1) (B:=regularizedWeight W ε) hTA)).re)
30      (𝓝[>] (0:ℝ))
31      (𝓝 ((trace (A*W) (isTraceClass_mul_mul (A:=1) (B:=W) hTA)).re)) := by
32    have h := (tendsto_const_nhds (x :=
33      (trace (A*W) (isTraceClass_mul_mul (A:=1) (B:=W) hTA)).re)).add
34      (hid.mul_const (trace A hTA).re)
35    simp only [zero_mul,add_zero] at h
36    convert h using 1
37    ext ε
38    rw [trace_regularizedWeight hTA ε]
39    simp only [Complex.add_re,Complex.mul_re,Complex.ofReal_re,Complex.ofReal_im,
40      zero_mul,sub_zero]
41  apply le_of_tendsto_of_tendsto tendsto_const_nhds (htrace.mul hInv)
42  filter_upwards [self_mem_nhdsWithin] with ε hε
43  exact fidelity_sq_le_positive_weight hA hB hTA hTB
44    (regularizedWeight_strictlyPositive hW hε)

46end Cloning.InfiniteFidelity
