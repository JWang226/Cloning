47theorem tensorJoin_inner (x x' : TensorRegister n A) (y y' : TensorRegister m A) :
48    ⟪tensorJoin x y, tensorJoin x' y'⟫_ℂ = ⟪x, x'⟫_ℂ * ⟪y, y'⟫_ℂ := by
49  simp only [lp.inner_eq_tsum, tsum_fintype, RCLike.inner_apply, tensorJoin_apply,
50    map_mul]
51  change (∑ w, (fun p : (Fin n → A) × (Fin m → A) =>
52    (x' p.1 * y' p.2) * (star (x p.1) * star (y p.2)))
53    (splitWordsEquiv n m w)) = _
54  rw [(splitWordsEquiv (A := A) n m).sum_comp (fun p =>
55    (x' p.1 * y' p.2) * (star (x p.1) * star (y p.2)))]
56  simp only [Fintype.sum_prod_type]
57  simp only [Finset.sum_mul, Finset.mul_sum]
58  rw [Finset.sum_comm (f := fun v w =>
59    x' v * y' w * (star (x v) * star (y w)))]
60  apply Finset.sum_congr rfl
61  intro v hv
62  apply Finset.sum_congr rfl
63  intro w hw
64  simp only [starRingEnd_apply]
65  ring

67theorem tensorJoin_norm (x : TensorRegister n A) (y : TensorRegister m A) :
68    ‖tensorJoin x y‖ = ‖x‖ * ‖y‖ := by
69  have he := congrArg Complex.re (tensorJoin_inner x x y y)
70  rw [Complex.mul_re] at he
71  change (RCLike.re : ℂ → ℝ) ⟪tensorJoin x y, tensorJoin x y⟫_ℂ =
72    (RCLike.re : ℂ → ℝ) ⟪x, x⟫_ℂ * (RCLike.re : ℂ → ℝ) ⟪y, y⟫_ℂ -
73    (RCLike.im : ℂ → ℝ) ⟪x, x⟫_ℂ * (RCLike.im : ℂ → ℝ) ⟪y, y⟫_ℂ at he
74  simp only [inner_self_eq_norm_sq, inner_self_im, mul_zero, sub_zero] at he
75  have hx := norm_nonneg x
76  have hy := norm_nonneg y
77  have hz := norm_nonneg (tensorJoin x y)
78  nlinarith [mul_nonneg hx hy]

122/-- The actual collective matrix-unit generator obeys the tensor product rule. -/
123theorem collectiveGenerator_tensorJoin (a b : A)
124    (x : TensorRegister n A) (y : TensorRegister m A) :
125    collectiveGenerator (n + m) a b (tensorJoin x y) =
126      tensorJoin (collectiveGenerator n a b x) y +
127      tensorJoin x (collectiveGenerator m a b y) := by
128  ext w
129  simp only [collectiveGenerator_apply, tensorJoin_apply, lp.coeFn_add, Pi.add_apply,
130    Fin.sum_univ_add, Finset.sum_mul, Finset.mul_sum]
131  congr 1
132  · apply Finset.sum_congr rfl
133    intro t ht
134    rw [restrict_update_left, restrict_update_left_other]
135    split_ifs <;> simp
136  · apply Finset.sum_congr rfl
137    intro t ht
138    rw [restrict_update_right, restrict_update_right_other]
