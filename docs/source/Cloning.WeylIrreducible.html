14lemma weighted_weyl_kernel (a z w : ℂ) :
15    (weylGaussianWeight a : ℂ) * ⟪coherentVector z, displacement a (coherentVector w)⟫_ℂ =
16      (⟪coherentVector z, coherentVector w⟫_ℂ / (Real.pi : ℂ)) *
17        Complex.exp (-((‖a‖ ^ 2 : ℝ) : ℂ) + starRingEnd ℂ z * a + (-w) * starRingEnd ℂ a) := by
18  rw [displacement_coherentVector, inner_smul_right]
19  simp only [weylGaussianWeight, Complex.ofReal_div, Complex.ofReal_exp,
20    displacementPhase, inner_coherentVector]
21  have h : ((-‖a‖ ^ 2 / 2 : ℝ) : ℂ) +
22      ((a * starRingEnd ℂ w - starRingEnd ℂ a * w) / 2) +
23      (((-(‖z‖ ^ 2 + ‖a + w‖ ^ 2) / 2 : ℝ) : ℂ) + starRingEnd ℂ z * (a + w)) =
24      (((-(‖z‖ ^ 2 + ‖w‖ ^ 2) / 2 : ℝ) : ℂ) + starRingEnd ℂ z * w) +
25      (-((‖a‖ ^ 2 : ℝ) : ℂ) + starRingEnd ℂ z * a + (-w) * starRingEnd ℂ a) := by
26    push_cast
27    simp only [norm_sq_cast, map_add]
28    ring
29  field_simp
30  simp only [← Complex.exp_add]
31  congr 1
32  push_cast at h ⊢
33  linear_combination h

35lemma integral_weighted_weyl_kernel (z w : ℂ) :
36    (∫ a : ℂ, (weylGaussianWeight a : ℂ) *
37      ⟪coherentVector z, displacement a (coherentVector w)⟫_ℂ) =
38      ⟪coherentVector z, coherentVector 0⟫_ℂ *
39        ⟪coherentVector 0, coherentVector w⟫_ℂ := by
40  simp_rw [weighted_weyl_kernel]
41  rw [integral_const_mul, integral_complex_gaussian_linear]
42  have hp : (Real.pi : ℂ) ≠ 0 := Complex.ofReal_ne_zero.mpr Real.pi_ne_zero
43  rw [div_mul_eq_mul_div]
44  rw [show ⟪coherentVector z, coherentVector w⟫_ℂ *
45      ((Real.pi : ℂ) * Complex.exp (starRingEnd ℂ z * -w)) / Real.pi =
46      ⟪coherentVector z, coherentVector w⟫_ℂ * Complex.exp (starRingEnd ℂ z * -w) by
47        field_simp]
48  simp only [inner_coherentVector_factorized]
49  simp only [norm_zero, sq, neg_zero, zero_div, Real.exp_zero, Complex.ofReal_one,
50    map_zero, mul_zero, zero_mul, Complex.exp_zero, mul_one, one_mul]
51  rw [mul_assoc, ← Complex.exp_add]
52  have he : starRingEnd ℂ z * w + starRingEnd ℂ z * -w = 0 := by ring
53  rw [he, Complex.exp_zero, mul_one]

55/-- The vacuum projection is a strong Gaussian integral of the actual Weyl
56operators, with the normalization `exp(-|a|²/2)/π`. -/
57theorem weylGaussianAverage_eq_vacuumProjector :
58    weylGaussianAverage =
59      InnerProductSpace.rankOne ℂ (coherentVector 0) (coherentVector 0) := by
60  apply continuousLinearMap_ext_coherent
61  intro w
62  apply sub_eq_zero.mp
63  apply coherentVector_total
64  intro z
65  rw [inner_sub_right]
66  apply sub_eq_zero.mpr
67  have hi := (innerSL ℂ (coherentVector z)).integral_comp_comm
68    (integrable_weighted_displacement (coherentVector w))
69  simp only [innerSL_apply_apply, inner_smul_right] at hi
70  rw [weylGaussianAverage_apply, ← hi, integral_weighted_weyl_kernel,
71    InnerProductSpace.rankOne_apply, inner_smul_right]
72  exact mul_comm _ _

74/-- Every bounded operator in the Weyl commutant commutes with the Gaussian
75average, by functoriality of the actual vector-valued Bochner integral. -/
76lemma commutes_weylGaussianAverage (T : Fock →L[ℂ] Fock)
77    (hT : ∀ a : ℂ, T.comp (displacement a) = (displacement a).comp T) :
78    T.comp weylGaussianAverage = weylGaussianAverage.comp T := by
79  apply ContinuousLinearMap.ext
80  intro v
81  change T (weylGaussianAverage v) = weylGaussianAverage (T v)
82  rw [weylGaussianAverage_apply, weylGaussianAverage_apply,
83    ← T.integral_comp_comm (integrable_weighted_displacement v)]
84  congr 1
85  funext a
86  rw [map_smul]
87  congr 1
88  exact congrArg (fun L : Fock →L[ℂ] Fock ↦ L v) (hT a)

90/-- The constructed one-mode Weyl representation has scalar commutant.
91The proof derives the vacuum projection from Weyl operators, then uses the
92proved cyclicity of the vacuum. -/
93theorem displacement_commutant_scalar (T : Fock →L[ℂ] Fock)
94    (hT : ∀ a : ℂ, T.comp (displacement a) = (displacement a).comp T) :
95    T = ⟪coherentVector 0, T (coherentVector 0)⟫_ℂ • ContinuousLinearMap.id ℂ Fock := by
96  have hP := commutes_weylGaussianAverage T hT
97  rw [weylGaussianAverage_eq_vacuumProjector] at hP
98  have hv := congrArg (fun L : Fock →L[ℂ] Fock ↦ L (coherentVector 0)) hP
99  simp only [ContinuousLinearMap.comp_apply, InnerProductSpace.rankOne_apply,
100    inner_self_eq_norm_sq_to_K, coherentVector_norm] at hv
101  norm_num at hv
102  apply continuousLinearMap_ext_coherent
103  intro z
104  have hz := congrArg (fun L : Fock →L[ℂ] Fock ↦ L (coherentVector 0)) (hT z)
105  simp only [ContinuousLinearMap.comp_apply, displacement_vacuum] at hz
106  conv_lhs => rw [hz, hv, map_smul, displacement_vacuum]
107  rfl

109/-- Existence formulation of the exact scalar-commutant conclusion. -/
110theorem exists_scalar_of_commutes_displacement (T : Fock →L[ℂ] Fock)
111    (hT : ∀ a : ℂ, T.comp (displacement a) = (displacement a).comp T) :
112    ∃ c : ℂ, T = c • ContinuousLinearMap.id ℂ Fock :=
113  ⟨_, displacement_commutant_scalar T hT⟩
