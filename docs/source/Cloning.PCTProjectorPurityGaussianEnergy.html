21/-- Exact one-coordinate energy moment, throughout its true integrability
22range, including negative exponential weights. -/
23theorem integral_exp_norm_sq {δ : ℝ} (hδ : 0<δ) (t : ℝ) (ht : t*δ<1) :
24    (∫z : ℂ,Real.exp (t*‖z‖^2) ∂gaussianMeasure δ)=(1-t*δ)⁻¹ := by
25  have hq : 0<δ⁻¹-t := by
26    have := (lt_div_iff₀ hδ).2 ht
27    simpa only [one_div] using sub_pos.mpr this
28  rw [integral_gaussianMeasure_real δ hδ]
29  have he (z : ℂ) : gaussianDensity δ z*Real.exp (t*‖z‖^2)=
30      (Real.pi*δ)⁻¹*Real.exp (-(δ⁻¹-t)*‖z‖^2) := by
31    unfold gaussianDensity
32    rw [div_eq_mul_inv,mul_right_comm,←Real.exp_add,mul_comm]
33    congr 1
34    congr 1
35    ring
36  simp_rw [he]
37  rw [integral_const_mul,Cloning.MultimodeCoherent.integral_gaussian_single hq]
38  have hd : δ≠0 := hδ.ne'
39  have ht0 : 1-t*δ≠0 := (sub_pos.mpr ht).ne'
40  have hq0 : δ⁻¹-t≠0 := hq.ne'
41  field_simp
42  <;> ring

51theorem integral_exp_energy (s : ℕ) {δ : ℝ} (hδ : 0<δ) (t : ℝ) (ht : t*δ<1) :

66/-- The two-variable domination used for projector purity is integrable. -/
67theorem integrable_exp_energy_pair (s : ℕ) {δ : ℝ} (hδ : 0<δ) (t : ℝ) (ht : t*δ<1) :
68    Integrable (fun zw : (Fin s → ℂ) × (Fin s → ℂ) =>
