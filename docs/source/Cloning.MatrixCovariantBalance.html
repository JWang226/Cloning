26/-- Schur's lemma for the concrete matrices of an irreducible complex
27representation: every commuting matrix is scalar. -/
28theorem commuting_matrix_is_scalar
29    (ρ : Representation ℂ G (A → ℂ)) [Representation.IsIrreducible ρ]
30    (M : Matrix A A ℂ)
31    (hcomm : ∀ g, M * LinearMap.toMatrix' (ρ g) = LinearMap.toMatrix' (ρ g) * M) :
32    ∃ c : ℂ, M = c • (1 : Matrix A A ℂ) := by
33  let f : Representation.IntertwiningMap ρ ρ :=
34    { toLinearMap := Matrix.toLin' M
35      isIntertwining' := fun g => by
36        have h := congrArg Matrix.toLin' (hcomm g)
37        simpa only [Matrix.toLin'_mul, Matrix.toLin'_toMatrix'] using h }
38  obtain ⟨c, hc⟩ :=
39    (Representation.IsIrreducible.algebraMap_intertwiningMap_bijective_of_isAlgClosed
40      (ρ := ρ)).surjective f
41  refine ⟨c, ?_⟩
42  have hlin := congrArg (fun h : Representation.IntertwiningMap ρ ρ => h.toLinearMap) hc
43  have hmat := congrArg LinearMap.toMatrix' hlin
44  have hId : (1 : Representation.IntertwiningMap ρ ρ).toLinearMap = LinearMap.id := rfl
45  simpa [f, Representation.IntertwiningMap.algebraMap_apply, hId] using hmat.symm

47/-- The scalar in Schur's lemma is determined by the normalized trace. -/
48theorem commuting_matrix_eq_trace_smul
49    (ρ : Representation ℂ G (A → ℂ)) [Representation.IsIrreducible ρ]
50    (M : Matrix A A ℂ) (hA : 0 < Fintype.card A)
51    (hcomm : ∀ g, M * LinearMap.toMatrix' (ρ g) = LinearMap.toMatrix' (ρ g) * M) :
52    M = (Matrix.trace M / (Fintype.card A : ℂ)) • (1 : Matrix A A ℂ) := by
53  obtain ⟨c, hc⟩ := commuting_matrix_is_scalar ρ M hcomm
54  have htr : Matrix.trace M = c * (Fintype.card A : ℂ) := by
55    rw [hc, Matrix.trace_smul, Matrix.trace_one]
56    rfl
57  have hcard : (Fintype.card A : ℂ) ≠ 0 := Nat.cast_ne_zero.mpr (Nat.ne_of_gt hA)
58  have hscalar : c = Matrix.trace M / (Fintype.card A : ℂ) :=
59    (eq_div_iff hcard).mpr htr.symm
60  exact hc.trans (congrArg (fun z : ℂ => z • (1 : Matrix A A ℂ)) hscalar)

74theorem trace_partialTrace_isometry (V : Matrix (A × B) C ℂ)

80/-- A commuting partial trace is balanced by irreducibility, with its
81constant fixed by the actual matrix dimensions and the isometry identity. -/
82theorem balanced_partialTrace_of_commuting
83    (ρ : Representation ℂ G (A → ℂ)) [Representation.IsIrreducible ρ]
84    (V : Matrix (A × B) C ℂ) (hA : 0 < Fintype.card A)
85    (hV : V.conjTranspose * V = 1)
86    (hcomm : ∀ g,
87      Cloning.Compression.partialTrace (V * V.conjTranspose) * LinearMap.toMatrix' (ρ g) =
88        LinearMap.toMatrix' (ρ g) * Cloning.Compression.partialTrace (V * V.conjTranspose)) :
89    Cloning.Compression.partialTrace (V * V.conjTranspose) =
90      ((Fintype.card C : ℝ) / (Fintype.card A : ℝ)) • (1 : Matrix A A ℂ) := by
91  rw [commuting_matrix_eq_trace_smul ρ _ hA hcomm, trace_partialTrace_isometry V hV]
92  ext i j
93  simp only [Matrix.smul_apply, Complex.real_smul, smul_eq_mul, Complex.ofReal_div,
94    Complex.ofReal_natCast]

96/-- Unitary conjugation invariance gives the commuting operator needed by
97Schur's lemma. -/
98theorem commutes_of_conjugation_invariant (U M : Matrix A A ℂ)
99    (hU : U.conjTranspose * U = 1) (hM : U * M * U.conjTranspose = M) :
100    M * U = U * M := by
101  have h := congrArg (fun X : Matrix A A ℂ => X * U) hM
102  simp only [Matrix.mul_assoc, hU, Matrix.mul_one] at h
103  exact h.symm

106/-- The balancing identity follows from actual unitary invariance and
107irreducibility; scalarity is derived using Schur's lemma. -/
108theorem balanced_partialTrace_of_invariant
109    (ρ : Representation ℂ G (A → ℂ)) [Representation.IsIrreducible ρ]
110    (V : Matrix (A × B) C ℂ) (hA : 0 < Fintype.card A)
111    (hV : V.conjTranspose * V = 1)
112    (hunitary : ∀ g, (LinearMap.toMatrix' (ρ g)).conjTranspose * LinearMap.toMatrix' (ρ g) = 1)
113    (hinvariant : ∀ g,
114      LinearMap.toMatrix' (ρ g) * Cloning.Compression.partialTrace (V * V.conjTranspose) *
115        (LinearMap.toMatrix' (ρ g)).conjTranspose =
116          Cloning.Compression.partialTrace (V * V.conjTranspose)) :
117    Cloning.Compression.partialTrace (V * V.conjTranspose) =
118      ((Fintype.card C : ℝ) / (Fintype.card A : ℝ)) • (1 : Matrix A A ℂ) := by
119  apply balanced_partialTrace_of_commuting ρ V hA hV
120  intro g
121  exact commutes_of_conjugation_invariant _ _ (hunitary g) (hinvariant g)

123/-- Balanced partial trace from an actual isometric intertwiner of unitary
124representations. Irreducibility is required only on the retained factor.
125All partial-trace covariance, Schur scalarity, and dimension normalization
126steps are proved; there is no balancing assumption. -/
127theorem balanced_partialTrace_of_intertwining
128    (ρ : Representation ℂ G (A → ℂ))
129    (σ : Representation ℂ G (B → ℂ))
130    (τ : Representation ℂ G (C → ℂ))
131    [Representation.IsIrreducible ρ]
132    (V : Matrix (A × B) C ℂ) (hA : 0 < Fintype.card A)
133    (hV : V.conjTranspose * V = 1)
134    (hρ : ∀ g, (LinearMap.toMatrix' (ρ g)).conjTranspose * LinearMap.toMatrix' (ρ g) = 1)
135    (hσ : ∀ g, (LinearMap.toMatrix' (σ g)).conjTranspose * LinearMap.toMatrix' (σ g) = 1)
136    (hτ : ∀ g, LinearMap.toMatrix' (τ g) * (LinearMap.toMatrix' (τ g)).conjTranspose = 1)
137    (hintertwine : ∀ g, (LinearMap.toMatrix' (ρ g) ⊗ₖ LinearMap.toMatrix' (σ g)) * V =
138      V * LinearMap.toMatrix' (τ g)) :
139    Cloning.Compression.partialTrace (V * V.conjTranspose) =
140      ((Fintype.card C : ℝ) / (Fintype.card A : ℝ)) • (1 : Matrix A A ℂ) := by
141  apply balanced_partialTrace_of_commuting ρ V hA hV
142  intro g
143  exact Cloning.Compression.partialTrace_intertwiner_commutes
144    (LinearMap.toMatrix' (ρ g)) (LinearMap.toMatrix' (σ g))
145    (LinearMap.toMatrix' (τ g)) V (hρ g) (hσ g) (hτ g) (hintertwine g)

147/-- The Cartan-sector channel constructed from irreducible unitary
148representation data and an isometric intertwiner. Its complete positivity
149and trace preservation are obtained without a partial-trace balance premise. -/
150def cartanChannel_of_irreducible_intertwiner
151    (ρ : Representation ℂ G (A → ℂ))
152    (σ : Representation ℂ G (B → ℂ))
153    (τ : Representation ℂ G (C → ℂ))
154    [Representation.IsIrreducible ρ]
155    (V : Matrix (A × B) C ℂ)
156    (hA : 0 < Fintype.card A) (hC : 0 < Fintype.card C)
157    (hV : V.conjTranspose * V = 1)
158    (hρ : ∀ g, (LinearMap.toMatrix' (ρ g)).conjTranspose * LinearMap.toMatrix' (ρ g) = 1)
159    (hσ : ∀ g, (LinearMap.toMatrix' (σ g)).conjTranspose * LinearMap.toMatrix' (σ g) = 1)
160    (hτ : ∀ g, LinearMap.toMatrix' (τ g) * (LinearMap.toMatrix' (τ g)).conjTranspose = 1)
161    (hintertwine : ∀ g, (LinearMap.toMatrix' (ρ g) ⊗ₖ LinearMap.toMatrix' (σ g)) * V =
162      V * LinearMap.toMatrix' (τ g)) : Cloning.Channels.MatrixChannel A C :=
163  Cloning.CartanChannel.cartanChannel V (Fintype.card A : ℝ) (Fintype.card C : ℝ)
164    (Nat.cast_pos.mpr hA) (Nat.cast_pos.mpr hC)
165    (balanced_partialTrace_of_intertwining ρ σ τ V hA hV hρ hσ hτ hintertwine)
