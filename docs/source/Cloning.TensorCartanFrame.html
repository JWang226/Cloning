38theorem productSectorFrame_orthonormal : Orthonormal ℂ (productSectorFrame P Q bP bQ) := by
39  rw [orthonormal_iff_ite]
40  intro i j
41  change ⟪tensorJoin (bP i.1 : TensorRegister n (Fin d)) (bQ i.2 : TensorRegister m (Fin d)),
42    tensorJoin (bP j.1 : TensorRegister n (Fin d)) (bQ j.2 : TensorRegister m (Fin d))⟫_ℂ = _
43  rw [tensorJoin_inner]
44  change ⟪bP i.1, bP j.1⟫_ℂ * ⟪bQ i.2, bQ j.2⟫_ℂ = _
45  rw [orthonormal_iff_ite.mp bP.orthonormal, orthonormal_iff_ite.mp bQ.orthonormal]
46  by_cases h1 : i.1 = j.1 <;> by_cases h2 : i.2 = j.2 <;>
47    simp [h1, h2, Prod.ext_iff]

49theorem tensorJoin_basis_expansion (x : P) (y : Q) :
50    tensorJoin (x : TensorRegister n (Fin d)) (y : TensorRegister m (Fin d)) =
51      ∑ i : I, ∑ j : J, (bP.repr x i * bQ.repr y j) •
52        tensorJoin (bP i) (bQ j) := by
53  have hx := congrArg (fun z : P => (z : TensorRegister n (Fin d))) (bP.sum_repr x)
54  have hy := congrArg (fun z : Q => (z : TensorRegister m (Fin d))) (bQ.sum_repr y)
55  simp only [Submodule.coe_sum, Submodule.coe_smul] at hx hy
56  conv_lhs => rw [← hx, ← hy]
57  simp only [tensorJoin_sum_left, tensorJoin_sum_right, tensorJoin_smul_left,
58    tensorJoin_smul_right, Finset.smul_sum, smul_smul]
59  rw [Finset.sum_comm]
60  simp only [mul_comm]

62/-- Every literal product of sector vectors lies in the span of the product
63orthonormal frame, by finite expansions in the two actual sector bases. -/
64theorem productSectorFrame_spans :
65    ⊤ ≤ Submodule.span ℂ (Set.range (productSectorFrame P Q bP bQ)) := by
66  let S := Submodule.span ℂ (Set.range (productSectorFrame P Q bP bQ))
67  let T := S.map (tensorProductSector P Q).subtype
68  have hT : tensorProductSector P Q ≤ T := by
69    apply Submodule.span_le.mpr
70    rintro z ⟨x, hx, y, hy, rfl⟩
71    rw [tensorJoin_basis_expansion P Q bP bQ ⟨x, hx⟩ ⟨y, hy⟩]
72    apply T.sum_mem
73    intro i hi
74    apply T.sum_mem
75    intro j hj
76    apply T.smul_mem
77    exact ⟨productSectorFrame P Q bP bQ (i, j),
78      Submodule.subset_span (Set.mem_range_self (i, j)), rfl⟩
79  intro x _
80  obtain ⟨y, hy, he⟩ := hT x.property
81  have hxy : y = x := Subtype.ext he
82  rw [← hxy]
83  exact hy
