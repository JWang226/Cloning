5/-!
6# The root hyperplane and its actual coordinate transport
7 
8The affine lattice and the real sum-zero hyperplane are constructed as
9subtypes, with explicit coordinate equivalences.  Probability distributions,
10cells and densities are transported along those equivalences.  This exposes
11the dependent last coordinate rather than dropping it from the state space.
12-/

21/-- The actual root hyperplane, with the metric and inner product induced
22from the ambient Euclidean space of `d+1` coordinates. -/
23abbrev rootSpace (d : ℕ) : Submodule ℝ (EuclideanSpace ℝ (Fin (d + 1))) where
24  carrier := {x | ∑ i, x i = 0}
25  zero_mem' := by simp
26  add_mem' := by
27    intro x y hx hy
28    change (∑ i, (x + y) i) = 0
29    change (∑ i, x i) = 0 at hx
30    change (∑ i, y i) = 0 at hy
31    simp only [PiLp.add_apply, Finset.sum_add_distrib, hx, hy, add_zero]
32  smul_mem' := by
33    intro c x hx
34    change (∑ i, (c • x) i) = 0
35    change (∑ i, x i) = 0 at hx
36    simp only [PiLp.smul_apply, smul_eq_mul, ← Finset.mul_sum, hx, mul_zero]

38/-- Coordinates along `e_i-e_last` as an explicit linear equivalence. -/
39def coordinates (d : ℕ) : (Fin d → ℝ) ≃ₗ[ℝ] rootSpace d where
40  toFun x := ⟨WithLp.toLp 2 (Fin.snoc (α := fun _ : Fin (d + 1) ↦ ℝ) x (-∑ i, x i)), by
41    change (∑ i, Fin.snoc (α := fun _ : Fin (d + 1) ↦ ℝ) x (-∑ j, x j) i) = 0
42    rw [Fin.sum_univ_castSucc]
43    simp⟩
44  invFun x i := x.1 i.castSucc
45  left_inv x := by ext i; simp
46  right_inv x := by
47    apply Subtype.ext
48    ext i
49    change Fin.snoc (α := fun _ : Fin (d + 1) ↦ ℝ) (fun j : Fin d ↦ x.1 j.castSucc) (-∑ j : Fin d, x.1 j.castSucc) i = x.1 i
50    refine Fin.lastCases ?_ (fun j ↦ ?_) i
51    · rw [Fin.snoc_last]
52      have hx := x.2
53      change (∑ i, x.1 i) = 0 at hx
54      rw [Fin.sum_univ_castSucc] at hx
55      linarith
56    · simp
57  map_add' x y := by
58    apply Subtype.ext
59    ext i
60    change Fin.snoc (α := fun _ : Fin (d + 1) ↦ ℝ) (x + y) (-∑ j, (x + y) j) i =
61      Fin.snoc (α := fun _ : Fin (d + 1) ↦ ℝ) x (-∑ j, x j) i + Fin.snoc (α := fun _ : Fin (d + 1) ↦ ℝ) y (-∑ j, y j) i
62    refine Fin.lastCases ?_ (fun j ↦ ?_) i
63    · simp [Finset.sum_add_distrib, add_comm]
64    · simp
65  map_smul' c x := by
66    apply Subtype.ext
67    ext i
68    change Fin.snoc (α := fun _ : Fin (d + 1) ↦ ℝ) (c • x) (-∑ j, (c • x) j) i = c * Fin.snoc (α := fun _ : Fin (d + 1) ↦ ℝ) x (-∑ j, x j) i
69    refine Fin.lastCases ?_ (fun j ↦ ?_) i
70    · simp [← Finset.mul_sum]
71    · simp

97/-- Every affine label is uniquely determined by its first `d` coordinates. -/
98def latticeCoordinates (d : ℕ) (n : ℤ) : (Fin d → ℤ) ≃ Lattice d n where
99  toFun z := ⟨YoungCompatibility.complete n z, YoungCompatibility.complete_sum n z⟩
100  invFun μ i := μ.1 i.castSucc
101  left_inv z := by ext i; simp
102  right_inv μ := by
103    apply Subtype.ext
104    ext i
105    refine Fin.lastCases ?_ (fun j ↦ ?_) i
106    · change YoungCompatibility.complete n (fun j ↦ μ.1 j.castSucc) (Fin.last d) = μ.1 (Fin.last d)
107      rw [YoungCompatibility.complete_last]
108      unfold Rounding.completeLast
109      have hμ := μ.2
110      rw [Fin.sum_univ_castSucc] at hμ
111      change n - (∑ j : Fin d, μ.1 j.castSucc) = μ.1 (Fin.last d)
112      omega
113    · simp

179/-- Scaled and translated interpolation on the full root hyperplane. The law
180is indexed by the actual affine lattice; `a` is the recentering displacement
181in the hyperplane and `h>0` is the cell scale. -/
182def interpolate (d : ℕ) (n : ℤ) (h : ℝ) (a : rootSpace d)
183    (P : Lattice d n → ℝ) (x : rootSpace d) : ℝ :=
184  YoungRounding.affineDensity h ((coordinates d).symm a)
185    (YoungRounding.interpolate (fun z ↦ P (latticeCoordinates d n z))) ((coordinates d).symm x)

187theorem interpolate_integral (d : ℕ) (n : ℤ) (h : ℝ) (hh : 0 < h) (a : rootSpace d)
188    (P : Lattice d n → ℝ) (hP : Summable P) :
189    (∫ x, interpolate d n h a P x ∂coordinateMeasure d) = ∑' μ, P μ := by
190  have hp : Summable (fun z ↦ P (latticeCoordinates d n z)) :=
191    (latticeCoordinates d n).summable_iff.mpr hP

193  rw [integral_coordinates, YoungRounding.integral_affineDensity h hh,

216/-- Positive affine density transport preserves Hellinger affinity exactly,
217including its Jacobian on the density, at every positive scale. -/
218theorem affineDensity_affinity (d : ℕ) (h : ℝ) (hh : 0 < h) (a : Fin d → ℝ)
219    (f g : (Fin d → ℝ) → ℝ) :
220    (∫ x, Real.sqrt (YoungRounding.affineDensity h a f x) *
221      Real.sqrt (YoungRounding.affineDensity h a g x)) = ∫ x, Real.sqrt (f x) * Real.sqrt (g x) := by
222  have hscale : 0 ≤ (h ^ Fintype.card (Fin d))⁻¹ := inv_nonneg.mpr (pow_nonneg hh.le _)
223  have heq : (fun x ↦ Real.sqrt (YoungRounding.affineDensity h a f x) *
224      Real.sqrt (YoungRounding.affineDensity h a g x)) =
225      YoungRounding.affineDensity h a (fun x ↦ Real.sqrt (f x) * Real.sqrt (g x)) := by
226    funext x
227    simp only [YoungRounding.affineDensity, Real.sqrt_mul hscale]
228    rw [show Real.sqrt (h ^ Fintype.card (Fin d))⁻¹ * Real.sqrt (f (h⁻¹ • (x - a))) *
229          (Real.sqrt (h ^ Fintype.card (Fin d))⁻¹ * Real.sqrt (g (h⁻¹ • (x - a)))) =
230        (Real.sqrt (h ^ Fintype.card (Fin d))⁻¹) ^ 2 *
231          (Real.sqrt (f (h⁻¹ • (x - a))) * Real.sqrt (g (h⁻¹ • (x - a)))) by ring]
232    rw [Real.sq_sqrt hscale]
233  rw [heq, YoungRounding.integral_affineDensity h hh]

235/-- Exact Hellinger-affinity preservation on the actual affine lattice. -/
236theorem interpolate_affinity (d : ℕ) (n : ℤ) (h : ℝ) (hh : 0 < h) (a : rootSpace d)
237    (P Q : Lattice d n → ℝ) (hP0 : ∀ μ, 0 ≤ P μ) (hQ0 : ∀ μ, 0 ≤ Q μ)
238    (hP : Summable P) (hQ : Summable Q) :
239    (∫ x, Real.sqrt (interpolate d n h a P x) * Real.sqrt (interpolate d n h a Q x)
240      ∂coordinateMeasure d) = ∑' μ, Real.sqrt (P μ) * Real.sqrt (Q μ) := by
241  have hp : Summable (fun z ↦ P (latticeCoordinates d n z)) :=
242    (latticeCoordinates d n).summable_iff.mpr hP
243  have hq : Summable (fun z ↦ Q (latticeCoordinates d n z)) :=
244    (latticeCoordinates d n).summable_iff.mpr hQ
245  unfold interpolate
246  rw [integral_coordinates d (fun y ↦
247    Real.sqrt (YoungRounding.affineDensity h ((coordinates d).symm a)
248        (YoungRounding.interpolate (fun z ↦ P (latticeCoordinates d n z))) y) *
249      Real.sqrt (YoungRounding.affineDensity h ((coordinates d).symm a)
250        (YoungRounding.interpolate (fun z ↦ Q (latticeCoordinates d n z))) y)),
251    affineDensity_affinity d h hh,
252    YoungRounding.interpolate_affinity (fun z ↦ hP0 _) (fun z ↦ hQ0 _) hp hq]
253  exact (latticeCoordinates d n).tsum_eq (fun μ ↦ Real.sqrt (P μ) * Real.sqrt (Q μ))
