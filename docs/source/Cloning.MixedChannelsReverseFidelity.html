16private theorem block_adjoint {C : Fin 2 → Fin 2 → TraceClass K}
17    (hC : BlockPositive C) (h0 : 0 ≤ (C 0 0).1) (h1 : 0 ≤ (C 1 1).1) :
18    C 1 0 = TraceClass.adjoint (C 0 1) := by
19  have hsum := IsSelfAdjoint.of_nonneg (hC.scalarCompression ![1, 1])
20  have him := IsSelfAdjoint.of_nonneg (hC.scalarCompression ![1, Complex.I])
21  have hd0 := (IsSelfAdjoint.of_nonneg h0).star_eq
22  have hd1 := (IsSelfAdjoint.of_nonneg h1).star_eq
23  simp only [Fin.sum_univ_two, Matrix.cons_val_zero, Matrix.cons_val_one,
24    star_one, one_mul, mul_one, one_smul] at hsum him
25  change IsSelfAdjoint ((C 0 0).1 + (C 0 1).1 + ((C 1 0).1 + (C 1 1).1)) at hsum
26  have hs := hsum.star_eq
27  simp only [star_add, hd0, hd1] at hs
28  change IsSelfAdjoint ((C 0 0).1 + Complex.I • (C 0 1).1 +
29    ((star Complex.I) • (C 1 0).1 + ((star Complex.I) * Complex.I) • (C 1 1).1)) at him
30  simp only [Complex.star_def, Complex.conj_I, neg_mul, Complex.I_mul_I, neg_neg,
31    one_smul, neg_smul] at him
32  have hi := him.star_eq
33  simp only [star_add, star_smul, star_neg, Complex.star_def, Complex.conj_I,
34    hd0, hd1] at hi
35  apply Subtype.ext
36  change (C 1 0).1 = star (C 0 1).1
37  have hs' : star (C 0 1).1 + star (C 1 0).1 = (C 0 1).1 + (C 1 0).1 := by
38    apply add_left_cancel (a := (C 0 0).1 + (C 1 1).1)
39    convert hs using 1 <;> module
40  have hi' : -Complex.I • star (C 0 1).1 + Complex.I • star (C 1 0).1 =
41      Complex.I • (C 0 1).1 - Complex.I • (C 1 0).1 := by
42    apply add_left_cancel (a := (C 0 0).1 + (C 1 1).1)
43    convert hi using 1 <;> module
44  have he := congrArg (fun x : K →L[ℂ] K => Complex.I • x) hi'
45  simp only [smul_add, smul_sub, smul_smul, mul_neg, Complex.I_mul_I,
46    neg_neg, one_smul, neg_one_smul] at he
47  have he' : star (C 0 1).1 - star (C 1 0).1 = -(C 0 1).1 + (C 1 0).1 := by
48    simpa only [sub_eq_add_neg, neg_neg] using he
49  calc
50    (C 1 0).1 = (1 / 2 : ℂ) • (((C 0 1).1 + (C 1 0).1) +
51        (-(C 0 1).1 + (C 1 0).1)) := by module
52    _ = (1 / 2 : ℂ) • ((star (C 0 1).1 + star (C 1 0).1) +
53        (star (C 0 1).1 - star (C 1 0).1)) := by rw [hs', he']
54    _ = star (C 0 1).1 := by module

82private theorem simple_data_processing (S : HybridToQuantum H K μ)
83    (f g : SimpleFunc Ω (TraceClass H)) (hf : Integrable f μ) (hg : Integrable g μ)
84    (hfp : ∀ y, 0 ≤ (f y).1) (hgp : ∀ y, 0 ≤ (g y).1) :
85    (positiveSimple f hf hfp).rootFidelity (positiveSimple g hg hgp) ≤
86      (S.mapPositive (positiveSimple f hf hfp)).rootFidelity
87        (S.mapPositive (positiveSimple g hg hgp)) := by
88  let p := f.pair g
89  let b (i j : Fin 2) := p.map (fun q => fidelityBlock q.1 q.2 (fibreWitness q) i j)
90  have hbi (i j : Fin 2) : Integrable (b i j) μ := by
91    apply SimpleFunc.FinMeasSupp.integrable
92    apply (SimpleFunc.integrable_iff_finMeasSupp.mp (SimpleFunc.integrable_pair hf hg)).map
93    simp only [Prod.fst_zero, Prod.snd_zero, fibreWitness_zero, fidelityBlock]
94    split_ifs <;> try rfl
95    apply Subtype.ext
96    simp [TraceClass.adjoint]
97  let C (i j : Fin 2) := (hbi i j).toL1 (b i j)
98  have hC : ∀ᵐ y ∂μ, ∀ i j : Fin 2,
99      C i j y = fidelityBlock (f y) (g y) (fibreWitness (f y, g y)) i j := by
100    exact ae_all_iff.mpr fun i => ae_all_iff.mpr fun j => (hbi i j).coeFn_toL1
101  have hCp : L1BlockPositive C := by
102    filter_upwards [hC] with y hy
103    simpa only [hy] using fibreWitness_block (f y) (g y) (hfp y) (hgp y)
104  have h00 : C 0 0 = hf.toL1 f := by
105    apply Lp.ext
106    filter_upwards [hC, hf.coeFn_toL1] with y hy hf'
107    rw [hy, hf']
108    simp [fidelityBlock]
109  have h11 : C 1 1 = hg.toL1 g := by
110    apply Lp.ext
111    filter_upwards [hC, hg.coeFn_toL1] with y hy hg'
112    rw [hy, hg']
113    simp [fidelityBlock]
114  have hout := S.completelyPositive 2 C hCp
115  have hdiag0 : 0 ≤ (S.map (C 0 0)).1 := by
116    rw [h00]
117    exact (S.mapPositive (positiveSimple f hf hfp)).2
118  have hdiag1 : 0 ≤ (S.map (C 1 1)).1 := by
119    rw [h11]
120    exact (S.mapPositive (positiveSimple g hg hgp)).2
121  have hadj := block_adjoint hout hdiag0 hdiag1
122  have hblock : BlockPositive (fidelityBlock (S.map (C 0 0)) (S.map (C 1 1))
123      (S.map (C 0 1))) := by
124    have he : fidelityBlock (S.map (C 0 0)) (S.map (C 1 1)) (S.map (C 0 1)) =
125        (fun i j => S.map (C i j)) := by
126      funext i j
127      fin_cases i <;> fin_cases j <;> simp [fidelityBlock, hadj]
128    rwa [he]
129  have hbound := trace_re_le_fidelity_of_block _ _ _ hdiag0 hdiag1
130    (fidelityBlock_quadratic _ _ _ hblock)
131  have ht : (traceCLM (S.map (C 0 1))).re =
132      (positiveSimple f hf hfp).rootFidelity (positiveSimple g hg hgp) := by
133    rw [S.tracePreserving, integratedTrace_apply, PositiveL1.rootFidelity_eq_integral]
134    calc
135      _ = ∫ y, (traceCLM (C 0 1 y)).re ∂μ :=
136        (Complex.reCLM.integral_comp_comm
137          (traceCLM.integrable_comp (L1.integrable_coeFn (C 0 1)))).symm
138      _ = _ := ?_
139    apply integral_congr_ae
140    filter_upwards [hC, hf.coeFn_toL1, hg.coeFn_toL1] with y hy hf' hg'
141    change (traceCLM (C 0 1 y)).re = extendedRootFidelity
142      ((hf.toL1 f) y, (hg.toL1 g) y)
143    rw [hy, hf', hg']
144    simpa [fidelityBlock] using fibreWitness_trace (f y) (g y) (hfp y) (hgp y)
145  rw [ht] at hbound
146  simpa only [h00, h11] using hbound

153/-- Root fidelity cannot decrease under any actual classical–quantum to quantum
154channel. The measure, Hilbert spaces, and integrable positive fields are arbitrary;
155there is no measurable optimizer, rank, separability, or finiteness premise. -/
156theorem HybridToQuantum.fidelity_data_processing (S : HybridToQuantum H K μ)
157    (A B : PositiveL1 H μ) :
158    A.rootFidelity B ≤ (S.mapPositive A).rootFidelity (S.mapPositive B) := by
159  obtain ⟨f, hf, hfp, hft⟩ := A.exists_simple_approx
160  obtain ⟨g, hg, hgp, hgt⟩ := B.exists_simple_approx
161  let a n := positiveSimple (f n) (hf n) (hfp n)
162  let b n := positiveSimple (g n) (hg n) (hgp n)
163  have ha : Tendsto a atTop (𝓝 A) := tendsto_subtype_rng.mpr hft
164  have hb : Tendsto b atTop (𝓝 B) := tendsto_subtype_rng.mpr hgt
165  have hleft := PositiveL1.continuous_rootFidelity.tendsto (A, B) |>.comp
166    (ha.prodMk_nhds hb)
167  have hright := PositiveTraceClass.continuous_rootFidelity.tendsto
168    (S.mapPositive A, S.mapPositive B) |>.comp
169      (((S.continuous_mapPositive.tendsto A).comp ha).prodMk_nhds
170        ((S.continuous_mapPositive.tendsto B).comp hb))
171  exact le_of_tendsto_of_tendsto hleft hright (Eventually.of_forall fun n =>
172    simple_data_processing S (f n) (g n) (hf n) (hg n) (hfp n) (hgp n))
