16theorem gaussian_shift_weighted_mass_le
17    (R : PositiveField (H := H) (volume : Measure (Fin k → ℝ)))
18    (a : Fin k → ℝ) (ha : ∀ i, 0 < a i) {g : ℝ} (hg : 1 ≤ g) :
19    (∫ h : Fin k → ℝ, GaussianAffinity.productDensity a h *
20      ∫ v : Fin k → ℝ, GaussianAffinity.productWitness a (fun i => a i / g)
21        (fun i => v i + Real.sqrt g * h i) * ‖(R.value v).1‖) ≤
22      Thermal.classicalBase g ^ ((k : ℝ) / 2) * R.mass := by
23  let G := GaussianAffinity.productDensity a
24  let χ := GaussianAffinity.productWitness a (fun i => a i / g)
25  let B : (Fin k → ℝ) → ℝ := fun v => ‖(R.value v).1‖
26  let F : (Fin k → ℝ) × (Fin k → ℝ) → ℝ := fun z =>
27    G z.1 * B z.2 * χ (fun i => z.2 i + Real.sqrt g * z.1 i)
28  have hg0 : 0 < g := lt_of_lt_of_le zero_lt_one hg
29  have hχp (v : Fin k → ℝ) : 0 < χ v :=
30    GaussianAffinity.productWitness_pos a _ ha (fun i => div_pos (ha i) hg0) v
31  have hχb (v : Fin k → ℝ) : ‖χ v‖ ≤
32      ∏ i, Real.sqrt (Real.sqrt (a i) / Real.sqrt (a i / g)) := by
33    rw [Real.norm_eq_abs, abs_of_pos (hχp v)]
34    apply GaussianAffinity.productWitness_le a _ ha (fun i => div_pos (ha i) hg0)
35    intro i
36    exact (div_le_iff₀ hg0).mpr (by nlinarith [ha i])
37  have hχc : Continuous χ := GaussianAffinity.continuous_productWitness a _ ha
38    (fun i => div_pos (ha i) hg0)
39  have hF : Integrable F (volume.prod volume) := by
40    apply ((GaussianAffinity.integrable_productDensity a ha).mul_prod R.integrable.norm).mul_bdd
41      ((hχc.comp (by fun_prop)).aestronglyMeasurable)
42    exact Eventually.of_forall fun z => hχb _
43  have he (v : Fin k → ℝ) :
44      (∫ h : Fin k → ℝ, F (h, v)) = B v *
45        (∫ h : Fin k → ℝ, G h * χ (fun i => v i + Real.sqrt g * h i)) := by
46    rw [← integral_const_mul]
47    apply integral_congr_ae
48    exact Eventually.of_forall fun h => by dsimp [F]; ring
49  calc
50    _ = ∫ h : Fin k → ℝ, ∫ v : Fin k → ℝ, F (h, v) := by
51      apply integral_congr_ae
52      exact Eventually.of_forall fun h => by
53        dsimp only
54        rw [← integral_const_mul]
55        apply integral_congr_ae
56        exact Eventually.of_forall fun v => by dsimp [F, G, B, χ]; ring
57    _ = ∫ v : Fin k → ℝ, ∫ h : Fin k → ℝ, F (h, v) := integral_integral_swap hF
58    _ ≤ ∫ v : Fin k → ℝ, B v * Thermal.classicalBase g ^ ((k : ℝ) / 2) := by
59      apply integral_mono hF.integral_prod_right (R.integrable.norm.mul_const _)
60      intro v
61      dsimp only
62      rw [he]
63      apply mul_le_mul_of_nonneg_left _ (norm_nonneg _)
64      simpa only [Fintype.card_fin] using
65        GaussianAffinity.integral_product_translated_witness_le a ha hg v
66    _ = _ := by rw [integral_mul_const]; exact mul_comm _ _
