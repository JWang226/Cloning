14/-- Spectral functions act by their eigenvalue on a rectangular isometric
15block, even when the ambient positive matrix has a kernel. -/
16theorem momentSpectral_mul_isometry (H : Matrix A A ℂ) (hH : H.PosSemidef)
17    (E : Matrix A B ℂ) (hE : Eᴴ*E=1) (c : ℝ)
18    (hHE : H*E=(c : ℂ) • E) (f : ℝ → ℂ) :
19    momentSpectral H hH f*E=f c • E := by
20  have hP : H*(E*Eᴴ)=(c : ℂ) • (E*Eᴴ) := by
21    rw [← Matrix.mul_assoc,hHE,Matrix.smul_mul]
22  have hh := congrArg (fun M : Matrix A A ℂ => M*E)
23    (momentSpectral_left_eigenrelation H hH f (E*Eᴴ) c hP)
24  simpa only [Matrix.mul_assoc,hE,Matrix.mul_one,Matrix.smul_mul] using hh

26theorem momentNormalizer_mul_isometry (H : Matrix A A ℂ) (hH : H.PosSemidef)
27    (E : Matrix A B ℂ) (hE : Eᴴ*E=1) (c : ℝ) (hHE : H*E=(c : ℂ) • E) :
28    momentNormalizer H hH*E=((Real.sqrt c)⁻¹ : ℂ) • E :=
29  momentSpectral_mul_isometry H hH E hE c hHE (fun t => ((Real.sqrt t)⁻¹ : ℂ))

31theorem momentSupport_mul_isometry (H : Matrix A A ℂ) (hH : H.PosSemidef)
32    (E : Matrix A B ℂ) (hE : Eᴴ*E=1) (c : ℝ) (hc : 0<c)
33    (hHE : H*E=(c : ℂ) • E) : momentSupport H hH*E=E := by
34  have hh := momentSpectral_mul_isometry H hH E hE c hHE (fun t => if t=0 then 0 else 1)
35  simpa only [momentSupport,momentSpectral,if_neg hc.ne',one_smul] using hh

37/-- Positive scalar block weights give literal eigenrelations for the full
38ambient matrix, without any spanning requirement on the ambient blocks. -/
39theorem sum_weighted_projectors_mul {J : I → Type*} [∀ i,Fintype (J i)] [∀ i,DecidableEq (J i)]
40    (E : ∀ i,Matrix A (J i) ℂ) (hE : ∀ i,(E i)ᴴ*E i=1)
41    (horth : ∀ i j,i≠j → (E i)ᴴ*E j=0) (c : I → ℂ) (j : I) :
42    (∑ i,c i • (E i*(E i)ᴴ))*E j=c j • E j := by
43  rw [Matrix.sum_mul]
44  rw [Finset.sum_eq_single j]
45  · rw [Matrix.smul_mul,Matrix.mul_assoc,hE,Matrix.mul_one]
46  · intro i hi hij
47    rw [Matrix.smul_mul,Matrix.mul_assoc,horth i j hij,Matrix.mul_zero,smul_zero]
48  · simp

50/-- Transport a complete input block resolution through any rectangular
51map; no orthogonality on the larger ambient space is assumed here. -/
52theorem conjugated_identity_resolution {J : I → Type*} [∀ i,Fintype (J i)] [∀ i,DecidableEq (J i)]
53    (F : ∀ i,Matrix B (J i) ℂ) (hF : ∑ i,F i*(F i)ᴴ=1) (K : Matrix A B ℂ) :
54    K*Kᴴ=∑ i,(K*F i)*(K*F i)ᴴ := by
55  calc
56    _ = K*(∑ i,F i*(F i)ᴴ)*Kᴴ := by rw [hF,Matrix.mul_one]
57    _ = _ := by
58      rw [Matrix.mul_sum,Matrix.sum_mul]
59      apply Finset.sum_congr rfl
60      intro i hi
61      simp only [Matrix.conjTranspose_mul,Matrix.mul_assoc]

63/-- Rectangular conjugation as a continuous linear map, for exact Bochner
64transport of the actual sector Haar integrals. -/
65def rectangularSandwichCLM (E : Matrix A B ℂ) : Matrix B B ℂ →L[ℂ] Matrix A A ℂ :=
66  LinearMap.toContinuousLinearMap
67    { toFun := fun X => E*X*Eᴴ
68      map_add' := by intro X Y; simp only [Matrix.mul_add,Matrix.add_mul]
69      map_smul' := by intro c X; simp only [Matrix.mul_smul,Matrix.smul_mul,RingHom.id_apply] }
