17/-- Integration against the expanding Gaussian prior is literal integration
18against the fixed product Gaussian density after dilation. -/
19theorem integral_expandingGaussianPrior {E : Type*}
20    [NormedAddCommGroup E] [NormedSpace ℝ E]
21    (n : ℕ) (f : (Fin d → ℂ) → E) (hf : Continuous f) :
22    (∫ a, f a ∂expandingGaussianPrior d n) =
23      ∫ z, priorDensity z • f (((n : ℝ) + 1) • z) := by
24  rw [expandingGaussianPrior, integral_map (by fun_prop) hf.aestronglyMeasurable]
25  rw [gaussianPrior, integral_withDensity_eq_integral_toReal_smul
26    continuous_priorDensity.measurable.ennreal_ofReal
27    (Eventually.of_forall (fun _ => ENNReal.ofReal_lt_top))]
28  simp only [ENNReal.toReal_ofReal (priorDensity_pos _).le]

30/-- A fixed displacement of an expanding prior is a vanishing displacement
31of its fixed reference density; this is an exact change of variables. -/
32theorem integral_expandingGaussianPrior_shift {E : Type*}
33    [NormedAddCommGroup E] [NormedSpace ℝ E]
34    (n : ℕ) (b : Fin d → ℂ) (f : (Fin d → ℂ) → E) (hf : Continuous f) :
35    (∫ a, f (a + b) ∂expandingGaussianPrior d n) =
36      ∫ z, priorDensity (-(((n : ℝ) + 1)⁻¹ • b) + z) •
37        f (((n : ℝ) + 1) • z) := by
38  rw [integral_expandingGaussianPrior n (fun a => f (a + b))
39    (hf.comp (continuous_id.add continuous_const))]
40  rw [← integral_add_left_eq_self
41    (fun z => priorDensity z • f (((n : ℝ) + 1) • z + b))

43  congr 1
44  funext z
45  congr 1
46  congr 1

51/-- A general L¹ density perturbation estimate with an exact uniform norm
52bound. This will be applied to the actual translated-channel family. -/
53theorem weighted_integral_difference_le {E : Type*}
54    [NormedAddCommGroup E] [NormedSpace ℝ E]
55    (f : (Fin d → ℂ) → E) (hf : Continuous f) (C : ℝ)
56    (hn : ∀ a, ‖f a‖ = C) (p q : (Fin d → ℂ) → ℝ)
57    (hp : Integrable p) (hq : Integrable q) :
58    ‖(∫ a, p a • f a) - (∫ a, q a • f a)‖ ≤ C * ∫ a, |p a - q a| := by
59  have hi (r : (Fin d → ℂ) → ℝ) (hr : Integrable r) :
60      Integrable (fun a => r a • f a) := by
61    apply (hr.norm.mul_const C).mono' (hr.aestronglyMeasurable.smul hf.aestronglyMeasurable)
62    exact Eventually.of_forall (fun a => by simp [norm_smul, hn])
63  rw [← integral_sub (hi p hp) (hi q hq)]
64  calc
65    _ ≤ ∫ a, ‖p a • f a - q a • f a‖ := norm_integral_le_integral_norm _
66    _ = _ := by
67      simp only [← sub_smul, norm_smul, hn, Real.norm_eq_abs]
68      rw [integral_mul_const]
69      ring

76/-- Explicit covariance-error bound, valid at each finite averaging scale. -/
77theorem gaussianFoelnerChannel_covariance_bound (gain : ℝ)
78    (Φ : QuantumChannel (Fock d) (Fock d)) (b : Fin d → ℂ)
79    (A : TraceClass (Fock d)) (hA : 0 ≤ A.1) (n : ℕ) :
80    ‖(gaussianFoelnerChannel gain Φ n).toLinearMap (displacementTraceMap b A) -
81      displacementTraceMap (gain • b) ((gaussianFoelnerChannel gain Φ n).toLinearMap A)‖ ≤
82      ‖A‖ * ∫ z, |priorDensity (-(((n : ℝ) + 1)⁻¹ • b) + z) - priorDensity z| := by
83  let f : (Fin d → ℂ) → TraceClass (Fock d) :=
84    fun a => (translatedChannel gain Φ a).toLinearMap A
85  have hf : Continuous f := continuous_translatedChannel gain Φ A
86  change ‖(covariantAverage (expandingGaussianPrior d n) gain Φ).toLinearMap
87    (displacementTraceMap b A) - displacementTraceMap (gain • b)
88      ((covariantAverage (expandingGaussianPrior d n) gain Φ).toLinearMap A)‖ ≤ _
89  rw [covariantAverage_shift, ← map_sub, displacementTraceMap_norm]
90  change ‖(∫ a, f (a + b) ∂expandingGaussianPrior d n) -
91    (∫ a, f a ∂expandingGaussianPrior d n)‖ ≤ _
92  rw [integral_expandingGaussianPrior_shift n b f hf, integral_expandingGaussianPrior n f hf]
93  apply weighted_integral_difference_le
94  · exact hf.comp (by fun_prop)
95  · intro a
96    exact (translatedChannel gain Φ (((n : ℝ) + 1) • a)).toPositiveTracePreservingMap.norm_map_of_nonneg A hA
97  · exact (measurePreserving_add_left volume (-(((n : ℝ) + 1)⁻¹ • b))).integrable_comp_of_integrable
98      integrable_priorDensity
99  · exact integrable_priorDensity

101/-- The explicitly constructed Gaussian averages become displacement
102covariant in trace norm, for every fixed positive trace-class input and every
103phase-space displacement. No covariance or prior-limit assumption remains. -/
104theorem gaussianFoelnerChannel_covariance_tendsto (gain : ℝ)
105    (Φ : QuantumChannel (Fock d) (Fock d)) (b : Fin d → ℂ)
106    (A : TraceClass (Fock d)) (hA : 0 ≤ A.1) :

108      displacementTraceMap (gain • b) ((gaussianFoelnerChannel gain Φ n).toLinearMap A)‖)

127/-- The same vanishing defect holds on every trace-class input, including
128nonself-adjoint operators, as required for the common CP-limit extraction. -/
129theorem gaussianFoelnerChannel_covariance_tendsto_all (gain : ℝ)
130    (Φ : QuantumChannel (Fock d) (Fock d)) (b : Fin d → ℂ)
131    (A : TraceClass (Fock d)) :
132    Tendsto (fun n => ‖(gaussianFoelnerChannel gain Φ n).toLinearMap (displacementTraceMap b A) -
133      displacementTraceMap (gain • b) ((gaussianFoelnerChannel gain Φ n).toLinearMap A)‖)
134      atTop (𝓝 0) := by
135  let D := gaussianFoelnerDefect gain Φ b
136  have hpos (B : TraceClass (Fock d)) (hB : 0 ≤ B.1) :
137      Tendsto (fun n => D n B) atTop (𝓝 0) :=
138    tendsto_zero_iff_norm_tendsto_zero.mpr (gaussianFoelnerChannel_covariance_tendsto gain Φ b B hB)
139  have hself (B : TraceClass (Fock d)) (hB : IsSelfAdjoint B.1) :
140      Tendsto (fun n => D n B) atTop (𝓝 0) := by
141    have h := (hpos _ (TraceClass.positivePart_nonneg B hB)).sub
142      (hpos _ (TraceClass.negativePart_nonneg B hB))
143    simpa only [← map_sub, TraceClass.positivePart_sub_negativePart B hB, sub_zero] using h
144  have h := (hself _ (TraceClass.realComponent_isSelfAdjoint A)).add
145    ((hself _ (TraceClass.imaginaryComponent_isSelfAdjoint A)).const_smul Complex.I)
146  have h' : Tendsto (fun n => D n A) atTop (𝓝 0) := by
147    simpa only [← map_smul, ← map_add, TraceClass.realComponent_add_I_smul_imaginaryComponent A,
148      smul_zero, add_zero] using h
149  exact tendsto_zero_iff_norm_tendsto_zero.mp h'
