23/-- If the physical errors lie between these envelopes after taking large
24sample sizes, their normalized small-gain coefficient is exactly a. The
25statement allows all parameters in the index type to vary uniformly. -/
26theorem error_envelopes_first_order {ι : Type*} (a b : ℕ) (δ₀ : ℝ) (hδ₀ : 0<δ₀)
27    (E : ℝ→ℕ→ι→ℝ)
28    (henv : ∀δ,0<δ→δ<δ₀→∀ε>0,∀ᶠ n in atTop,∀i,
29      lowerEnvelope a δ-ε≤E δ n i ∧ E δ n i≤upperEnvelope a b δ+ε) :
30    ∀η>0,∀ᶠ δ in 𝓝[>] (0:ℝ),∀ᶠ n in atTop,∀i,
31      |E δ n i/δ-(a:ℝ)|≤η := by
32  intro η hη
33  have hfilter : (𝓝[>] (0:ℝ))≤𝓝[≠] (0:ℝ) :=
34    nhdsWithin_mono 0 (fun δ hδ=>ne_of_gt hδ)
35  have hl := (lowerEnvelope_div_tendsto a).mono_left hfilter
36  have hu := (upperEnvelope_div_tendsto a b).mono_left hfilter
37  have hlo : ∀ᶠ δ in 𝓝[>] (0:ℝ),(a:ℝ)-η/2≤lowerEnvelope a δ/δ :=
38    hl.eventually (eventually_ge_nhds (by linarith))
39  have hup : ∀ᶠ δ in 𝓝[>] (0:ℝ),upperEnvelope a b δ/δ≤(a:ℝ)+η/2 :=
40    hu.eventually (eventually_le_nhds (by linarith))
41  have hsmall : ∀ᶠ δ in 𝓝[>] (0:ℝ),δ<δ₀ :=
42    (eventually_lt_nhds hδ₀).filter_mono nhdsWithin_le_nhds
43  filter_upwards [self_mem_nhdsWithin,hsmall,hlo,hup] with δ hδ hs hl hu
44  have hd : 0<δ := hδ
45  filter_upwards [henv δ hd hs (δ*(η/2)) (by positivity)] with n hn i
46  have ha := div_le_div_of_nonneg_right (hn i).1 hd.le
47  have hb := div_le_div_of_nonneg_right (hn i).2 hd.le
48  have he : (δ*(η/2))/δ=η/2 := mul_div_cancel_left₀ _ hd.ne'
49  simp only [sub_div,add_div,he] at ha hb
50  exact abs_le.mpr ⟨by linarith,by linarith⟩
