29/-- Compressing the literal tensor power to a physical Schur copy gives its

31theorem matrixTensorPower_canonical_block (H : PhysicalHighestTensor n d)
32    (X : Matrix (Fin d) (Fin d) ℂ) :
33    conjugationLinearMap H.canonicalEmbedding.toContinuousLinearMap.adjoint
34      (matrixTensorPower X n) = canonicalTensorWeight H X := by

36  apply ContinuousLinearMap.ext
