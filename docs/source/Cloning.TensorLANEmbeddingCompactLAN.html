16/-- Unconditional existence of the literal compact-window physical LAN maps
17for every strictly ordered positive spectrum and every allowed coordinate
18frame and mode enumeration. -/
19theorem nonempty_physicalCompactWindowLAN (p : SimpleSpectrum (k+1))
20    (b : OrthonormalBasis (Fin (k+1)) ℝ (EuclideanSpace ℝ (Fin (k+1))))
21    (hb : b 0 = sqrtSpectrum p.eigenvalue) (e : Fin s ≃ PairIndex (k+1)) :
22    Nonempty (CompactWindowLAN p b e) := by
23  have hex (j : ℕ) := exists_cutoff_uniform_physical_protocol p b hb e (j : ℝ)
24    (1/((j : ℝ)+1)) (by positivity)
25  choose Q hQ using hex
26  let cutoff (j : ℕ) := max j (Q j)
27  let P (N j : ℕ) : Prop := ∀ θ : Parameters k, ‖θ‖ ≤ (j : ℝ) → (∑ a, θ.1 a = 0) →
28    ‖(physicalCoordinateForward p b hb e N (cutoff j)).map (chartTensor p θ N)-(model p b e θ).1‖ < 1/((j : ℝ)+1) ∧
29    ‖(physicalCoordinateReverse p b hb e N (cutoff j)).map (model p b e θ).1-chartTensor p θ N‖ < 1/((j : ℝ)+1)
30  have hP (j : ℕ) : ∀ᶠ N in atTop, P N j := hQ j (cutoff j) (le_max_right _ _)
31  obtain ⟨ell,hellMono,hell,hellBound,hvalid⟩ := exists_common_diverging_cutoff P hP
32  refine ⟨{ forward := fun N => physicalCoordinateForward p b hb e N (cutoff (ell N))
33            reverse := fun N => physicalCoordinateReverse p b hb e N (cutoff (ell N))
34            approximation := ?_ }⟩
35  intro K hK hzero
36  obtain ⟨B,hB,hKB⟩ := hK.isBounded.exists_pos_norm_le
37  obtain ⟨j,hj⟩ := exists_nat_ge B
38  refine ⟨fun N => 1/((ell N : ℝ)+1), ?_, ?_⟩
39  · exact (tendsto_one_div_add_atTop_nhds_zero_nat (𝕜 := ℝ)).comp hell
40  · filter_upwards [hvalid, hell.eventually (eventually_ge_atTop j)] with N hN hjN θ hθ
41    have hn : ‖θ‖ ≤ (ell N : ℝ) := (hKB θ hθ).trans (hj.trans (by exact_mod_cast hjN))
42    exact ⟨(hN (ell N) le_rfl θ hn (hzero θ hθ)).1.le,
43      (hN (ell N) le_rfl θ hn (hzero θ hθ)).2.le⟩

46def physicalCompactWindowLAN (p : SimpleSpectrum (k+1))
