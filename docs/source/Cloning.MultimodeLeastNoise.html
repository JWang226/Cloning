17theorem moment_le_of_diagonal {s : ℕ} (σ : TraceClass (Fock s)) (hσ : 0 ≤ σ.1)
18    (q : Fin s → ℝ) (hq0 : ∀ i, 0 ≤ q i) (hq1 : ∀ i, q i < 1)
19    (w : Fin s → ℕ → ℝ) (hw0 : ∀ i k, 0 ≤ w i k) (hw : ∀ i, Antitone (w i))
20    (W : Fock s →L[ℂ] Fock s)
21    (hW : ∀ k, W (numberBasis s k) = ((∏ i, w i (k i) : ℝ) : ℂ) • numberBasis s k) :
22    (tracePairing ((channel q hq0 hq1).toLinearMap σ) W).re ≤
23      (trace σ.1 σ.2).re *
24        (tracePairing (vectorMixture (numberBasis s) (productGeometric q)) W).re := by
25  rw [tracePairing_re_eq_number_moment (numberBasis s) _ W _ hW,
26    diagonalMixture_tracePairing (numberBasis s) _ (productGeometric_hasSum hq0 hq1).summable
27      W _ hW]
28  simp_rw [channel_diagonal_re]
29  exact BosonicStochasticOrder.mixtureLaw_antitone_moment_le_geometric hq0 hq1 hw0 hw
30    (fun l => (σ.1.nonneg_iff_isPositive.mp hσ).re_inner_nonneg_right (numberBasis s l))
31    (trace_real_hasSum_basis (numberBasis s) σ)

47def productWeight {s : ℕ} (w : Fin s → ℕ → ℝ) (k : Occupation s) : ℝ :=
48  ∏ i, w i (k i)

50theorem productWeight_nonneg {s : ℕ} {w : Fin s → ℕ → ℝ}
51    (hw0 : ∀ i k, 0 ≤ w i k) (k : Occupation s) : 0 ≤ productWeight w k :=
52  Finset.prod_nonneg (fun i _ => hw0 i (k i))

54theorem productWeight_le_zero {s : ℕ} {w : Fin s → ℕ → ℝ}
55    (hw0 : ∀ i k, 0 ≤ w i k) (hw : ∀ i, Antitone (w i)) (k : Occupation s) :
56    productWeight w k ≤ productWeight w 0 := by
57  apply Finset.prod_le_prod
58  · intro i hi; exact hw0 i (k i)
59  · intro i hi; exact hw i (Nat.zero_le (k i))

61private theorem coefficient_bound {s : ℕ} (w : Fin s → ℕ → ℝ)

63    (x : Fock s) (k : Occupation s) :
64    ‖(productWeight w k : ℂ) * x k‖ ^ 2 ≤

71/-- Pointwise multiplication is an actual square-summable vector, since the
72nonnegative antitone product is automatically bounded by its vacuum value. -/
73def productDiagonalVector {s : ℕ} (w : Fin s → ℕ → ℝ)
74    (hw0 : ∀ i k, 0 ≤ w i k) (hw : ∀ i, Antitone (w i)) (x : Fock s) : Fock s := by
75  have hx : Summable (fun k : Occupation s => ‖x k‖ ^ (2 : ℕ)) := by
76    simpa only [ENNReal.toReal_ofNat, Real.rpow_two] using
77      (lp.memℓp x).summable (by norm_num)
78  have hs : Summable (fun k : Occupation s => ‖(productWeight w k : ℂ) * x k‖ ^ (2 : ℕ)) :=
79    (hx.mul_left (productWeight w 0 ^ 2)).of_nonneg_of_le
80      (fun k => sq_nonneg _) (coefficient_bound w hw0 hw x)
81  refine ⟨fun k => (productWeight w k : ℂ) * x k, memℓp_gen ?_⟩
82  simpa only [ENNReal.toReal_ofNat, Real.rpow_two] using hs

88theorem productDiagonalVector_norm_le {s : ℕ} (w : Fin s → ℕ → ℝ)
89    (hw0 : ∀ i k, 0 ≤ w i k) (hw : ∀ i, Antitone (w i)) (x : Fock s) :
90    ‖productDiagonalVector w hw0 hw x‖ ≤ productWeight w 0 * ‖x‖ := by
91  apply lp.norm_le_of_tsum_le (by norm_num)
92    (mul_nonneg (productWeight_nonneg hw0 0) (norm_nonneg x))
93  simp only [ENNReal.toReal_ofNat, Real.rpow_two, productDiagonalVector_apply, mul_pow]
94  have hx : Summable (fun k : Occupation s => ‖x k‖ ^ (2 : ℕ)) := by
95    simpa only [ENNReal.toReal_ofNat, Real.rpow_two] using
96      (lp.memℓp x).summable (by norm_num)
97  have hy : Summable (fun k : Occupation s => ‖(productWeight w k : ℂ) * x k‖ ^ (2 : ℕ)) := by
98    simpa only [ENNReal.toReal_ofNat, Real.rpow_two, productDiagonalVector_apply] using
99      (lp.memℓp (productDiagonalVector w hw0 hw x)).summable (by norm_num)
100  have heq : (∑' k : Occupation s, ‖x k‖ ^ (2 : ℕ)) = ‖x‖ ^ 2 := by
101    simpa only [ENNReal.toReal_ofNat, Real.rpow_two] using
102      (lp.hasSum_norm (show 0 < (2 : ENNReal).toReal by norm_num) x).tsum_eq
103  calc
104    _ ≤ ∑' k : Occupation s, productWeight w 0 ^ 2 * ‖x k‖ ^ (2 : ℕ) :=
105      hy.tsum_le_tsum (coefficient_bound w hw0 hw x) (hx.mul_left _)
106    _ = _ := by rw [tsum_mul_left, heq]

108/-- The physical product of bounded decreasing number functions, as a genuine

110def productObservable {s : ℕ} (w : Fin s → ℕ → ℝ)
111    (hw0 : ∀ i k, 0 ≤ w i k) (hw : ∀ i, Antitone (w i)) : Fock s →L[ℂ] Fock s :=
112  LinearMap.mkContinuous
113    { toFun := productDiagonalVector w hw0 hw
114      map_add' := by intro x y; ext k; simp [mul_add]
115      map_smul' := by intro c x; ext k; simp [mul_left_comm] }
