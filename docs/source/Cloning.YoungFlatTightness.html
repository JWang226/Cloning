112theorem flatDensity_tight (d : ℕ) (ε : ℝ) (hε : 0 < ε) :
113    ∃ K : Set (rootSpace d), IsCompact K ∧
114      ∀ N : ℕ, 0 < N → (∫ x in Kᶜ, flatDensity d N x) < ε := by
115  let C : ℝ := 2*((d+1 : ℕ) : ℝ)+2*((d+1 : ℕ) : ℝ)^3
116  let R : ℝ := Real.sqrt (C/ε+1)
117  have hC : 0 ≤ C := by dsimp [C]; positivity
118  have hR : 0 < R := by dsimp [R]; positivity
119  have hsq : R^2=C/ε+1 := Real.sq_sqrt (by positivity)
120  refine ⟨Metric.closedBall 0 R, isCompact_closedBall _ _, fun N hN ↦ ?_⟩
121  refine (flatDensity_tail_le d N hN R hR).trans_lt ?_
122  change C/R^2 < ε
123  apply (div_lt_iff₀ (sq_pos_of_pos hR)).mpr
124  rw [hsq]
125  have he : ε*(C/ε)=C := mul_div_cancel₀ C hε.ne'
126  nlinarith
