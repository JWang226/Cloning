69theorem inverseDimension_error_le_forward (r k N : ℕ) (hr : 0<r) (hN : 1≤N)
70    (hsmall : dimensionErrorEnvelope r k N≤1/2) (mu : Shape r N)
71    (ht : ¬∃i,(N:ℝ)*shrinkingRadius N≤|shapeRows mu i-N*flatSpectrum r i|) :
72    |inverseNormalizedDimension r k N mu-1|≤2*|normalizedDimension r k N mu-1| := by
73  let R := dimensionRatio r k (shapeRows mu)
74  let D := leadingConstant r k*(N:ℝ)^(r*k)
75  have hR : 0<R := dimensionRatio_pos r k (shapeRows mu) (shapeRows_nonneg mu)
76  have hn : (0:ℝ)<N := by exact_mod_cast (show 0<N by omega)
77  have hD : 0<D := mul_pos (leadingConstant_pos r k hr) (pow_pos hn _)
78  have he : |R/D-1|≤1/2 := (normalizedDimension_typical_error r k N hr hN mu ht).trans hsmall
79  have hl : (1/2:ℝ)≤R/D := by have hh := (abs_le.mp he).1; linarith
80  have hl' : D≤2*R := by have hh := (le_div_iff₀ hD).mp hl; linarith
81  have hi : D/R≤2 := (div_le_iff₀ hR).mpr hl'
82  change |D/R-1|≤2*|R/D-1|
83  have ha : D/R-1=-(D/R)*(R/D-1) := by field_simp; ring
84  rw [ha,abs_mul,abs_neg,abs_of_pos (div_pos hD hR)]
85  exact mul_le_mul_of_nonneg_right hi (abs_nonneg _)
