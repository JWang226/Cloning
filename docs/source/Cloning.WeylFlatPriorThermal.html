18/-- The ordinary flat-box average of the physical orbit root fidelity, with
19coordinates scaled from the unit box to the box of radius `n+1`. -/
20def flatBoxPayoff (gain : ℝ) (Φ : QuantumChannel (Fock d) (Fock d))
21    (A B : PositiveTraceClass (Fock d)) (n : ℕ) : ℝ :=
22  (volume.real (phaseSpaceUnitBox d))⁻¹ * ∫ z in phaseSpaceUnitBox d,
23    orbitPayoff gain Φ A B (((n : ℝ) + 1) • z)

58/-- The optimal flat-box root-fidelity payoff, taking the supremum over actual
59CPTP channels independently at every box radius. -/
60def flatBoxOptimalPayoff (gain : ℝ) (A B : PositiveTraceClass (Fock d)) (n : ℕ) : ℝ :=
61  sSup (Set.range (fun Φ : QuantumChannel (Fock d) (Fock d) => flatBoxPayoff gain Φ A B n))

63lemma flatBoxOptimalPayoff_nonneg (gain : ℝ) (A B : PositiveTraceClass (Fock d)) (n : ℕ) :
64    0 ≤ flatBoxOptimalPayoff gain A B n := by
65  have hb : BddAbove (Set.range (fun Φ : QuantumChannel (Fock d) (Fock d) =>
66      flatBoxPayoff gain Φ A B n)) := by
67    refine ⟨Real.sqrt ‖A.1‖ * Real.sqrt ‖B.1‖, ?_⟩
68    rintro _ ⟨Φ, rfl⟩
69    exact flatBoxPayoff_le_sqrt gain Φ A B n
70  exact (flatBoxPayoff_nonneg gain (displacementChannel 0) A B n).trans
71    (le_csSup hb (Set.mem_range_self (displacementChannel 0)))

96/-- The actual flat-prior compact-limit reduction. It applies to a different
97competitor at every scale, as required by a channel supremum inside the prior
98limit. The thermal inverse moment and compactness are conclusions of the
99concrete product witness theorems, rather than extra hypotheses. -/
100theorem exists_flatBox_thermal_payoff_limit (gain : ℝ)
101    (Φ : ℕ → QuantumChannel (Fock d) (Fock d))
102    (A : PositiveTraceClass (Fock d)) {q x : Fin d → ℝ}
103    (hq0 : ∀ i, 0 < q i) (hqx : ∀ i, q i < x i) (hx1 : ∀ i, x i < 1) :
104    ∃ Ψ : TraceClass (Fock d) →ₗ[ℂ] TraceClass (Fock d),
105      IsCompletelyPositive Ψ ∧
106      (∀ C, 0 ≤ C.1 → (traceCLM (Ψ C)).re ≤ (traceCLM C).re) ∧
107      (∀ (b : Fin d → ℂ) C,
108        Ψ (displacementTraceMap b C) = displacementTraceMap (gain • b) (Ψ C)) ∧
109      ∃ φ : ℕ → ℕ, StrictMono φ ∧
110        (∀ C (O : (Fock d) →L[ℂ] (Fock d)), IsCompactOperator O →
111          Tendsto (fun n => tracePairing
112            (((flatBoxDensity d).foelnerChannel gain (Φ (φ n)) (φ n)).toLinearMap C) O)
113              atTop (𝓝 (tracePairing (Ψ C) O))) ∧
114        (∀ ε > 0, ∀ᶠ n in atTop,
115          flatBoxPayoff gain (Φ (φ n)) A
116            (thermalPositive q (fun i => (hq0 i).le) (fun i => (hqx i).trans (hx1 i))) (φ n) ^ 2 ≤
117          (tracePairing (Ψ A.1) (productWitnessOperator (numberBasis d) q x)).re *
118            (∏ i, Cloning.Thermal.fidelity (q i) (x i)) + ε) := by
119  obtain ⟨Ψ, hp, ht, hc, φ, hφ, _, hcompact⟩ :=
120    (flatBoxDensity d).exists_foelner_sequence_covariant_limit gain Φ id tendsto_id
121  refine ⟨Ψ, hp, ht, hc, φ, hφ, hcompact, ?_⟩
122  have hM : 0 ≤ ∏ i, Cloning.Thermal.fidelity (q i) (x i) := Finset.prod_nonneg (fun i _ =>
123    (Cloning.Thermal.fidelity_pos (hq0 i).le ((hqx i).trans (hx1 i))
124      ((hq0 i).trans (hqx i)).le (hx1 i)).le)
125  have h := (flatBoxDensity d).orbitPayoff_eventually_sq_le_compact_limit gain Φ id A
126    (thermalPositive q (fun i => (hq0 i).le) (fun i => (hqx i).trans (hx1 i)))
127    (productWitnessOperator (numberBasis d) q x)
128    (productWitnessOperator_nonneg _ hq0 hqx hx1) hM
129    (fun ε hε => product_thermal_regularized_inverse_moment _ hq0 hqx hx1 hε)
130    Ψ φ (hcompact A.1 _ (productWitnessOperator_compact _ hq0 hqx hx1))
131  simpa only [flatBoxPayoff_eq_integral] using h

133/-- Once the universal covariant oscillator moment is supplied, the literal
134flat-box quantum converse follows uniformly over all original channels, in
135the correct order of quantifiers. No averaging, prior, inverse-moment, or
136compact-limit assumption is left in this implication. -/
