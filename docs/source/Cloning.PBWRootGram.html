4/-!
5# Root-height filtration to normalized PBW Gram estimates
6 
7This connects the two different cutoffs needed by the manuscript proof.
8Annihilators lower the *fine* simple-root-height cutoff by at least one, while
9creators raise it by at most the maximal positive-root height `D`.
10The normal-ordering proof uses the grouped cutoff `K (D*r)`. The creator bound
11on that grouped cutoff is derived on the fine filtration first.
12 
13The result needs no separately postulated bounded-root-operator estimate:
14small canonical-commutator defects, adjointness, and root-height shifts imply
15the normalized factorial Gram estimate for every bounded-length word.
16-/

26/-- Fine-cutoff strict lowering implies preservation of each fine cutoff. -/
27theorem annihilator_mem_same_cutoff (A : H →L[ℂ] H)
28    (K : ℕ → Submodule ℂ H) (hK : Monotone K)
29    (hzero : ∀ x ∈ K 0, A x = 0)
30    (hlower : ∀ r x, x ∈ K (r + 1) → A x ∈ K r)
31    (r : ℕ) (x : H) (hx : x ∈ K r) : A x ∈ K r := by
32  cases r with
33  | zero => rw [hzero x hx]; exact (K 0).zero_mem
34  | succ r => exact hK (Nat.le_succ r) (hlower r x hx)

36/-- Root-height form of the fixed-word PBW Gram estimate. The normalized root
37operator bound is a conclusion of `creator_norm_le_cutoff`, rather than an
38additional premise. `D` can be `d-1` for the positive roots of `U(d)`. -/
39theorem normalized_gram_error_le_of_root_filtration
40    (F A : ι → H →L[ℂ] H) (K : ℕ → Submodule ℂ H) (hK : Monotone K)
41    (Ω : H) (hΩmem : Ω ∈ K 0) (hnorm : ⟪Ω, Ω⟫_ℂ = 1)
42    (hadj : ∀ i x y, ⟪F i x, y⟫_ℂ = ⟪x, A i y⟫_ℂ)
43    (hzero : ∀ i x, x ∈ K 0 → A i x = 0)
44    (hlower : ∀ i r x, x ∈ K (r + 1) → A i x ∈ K r)
45    (D L : ℕ)
46    (hraise : ∀ i r x, x ∈ K r → F i x ∈ K (r + D))
47    (ε : ℝ) (hε : 0 ≤ ε) (hεone : ε ≤ 1)
48    (hcomm : ∀ i j r, r ≤ L * D → ∀ x ∈ K r,
49      ‖commutatorDefect F A i j x‖ ≤ ε * ‖x‖)
50    (u v : List ι) (hu : u.length ≤ L) (hv : v.length ≤ L) :
51    ‖⟪normalizedWord F u Ω, normalizedWord F v Ω⟫_ℂ -
52      (if u.Perm v then 1 else 0 : ℂ)‖ ≤
53      (gramConstant L u.length : ℝ) *
54        (Real.sqrt (((L * D : ℕ) : ℝ) + 1) * Real.sqrt 2) ^ (2 * L) * ε := by
55  let G : ℕ → Submodule ℂ H := fun r => K (r * D)
56  let M : ℝ := Real.sqrt ((((L * D : ℕ) : ℝ) + 1) * 2)
57  have hM : 1 ≤ M := by
58    apply Real.one_le_sqrt.mpr
59    have : 0 ≤ ((L * D : ℕ) : ℝ) := Nat.cast_nonneg _
60    linarith
61  have hG : Monotone G := fun a b hab => hK (Nat.mul_le_mul_right D hab)
62  have hF : ∀ i r x, x ∈ G r → F i x ∈ G (r + 1) := by
63    intro i r x hx
64    simpa [G, Nat.add_mul] using hraise i (r * D) x hx
65  have hA : ∀ i r x, x ∈ G r → A i x ∈ G (r + 1) := by
66    intro i r x hx
67    apply hG (Nat.le_succ r)
68    exact annihilator_mem_same_cutoff (A i) K hK (hzero i) (hlower i) (r * D) x hx
69  have hdiag : ∀ i r, r ≤ L * D → ∀ x ∈ K r,
70      ‖A i (F i x) - F i (A i x) - x‖ ≤ 1 * ‖x‖ := by
71    intro i r hr x hx
72    have hh := hcomm i i r hr x hx
73    simp only [commutatorDefect, if_true] at hh
74    exact hh.trans (mul_le_mul_of_nonneg_right hεone (norm_nonneg _))
75  have hbound : ∀ i r, r ≤ L → ∀ x ∈ G r, ‖F i x‖ ≤ M * ‖x‖ := by
76    intro i r hr x hx
77    have hh := creator_norm_le_cutoff (F i) (A i) K (hadj i) (hzero i) (hlower i)
78      (L * D) 1 (by norm_num) (hdiag i) (r * D) (Nat.mul_le_mul_right D hr) x hx
79    simpa [M, show (1 : ℝ) + 1 = 2 by norm_num] using hh
80  have hcommG : ∀ i j r, r ≤ L → ∀ x ∈ G r,
81      ‖commutatorDefect F A i j x‖ ≤ ε * ‖x‖ := by
82    intro i j r hr x hx
83    exact hcomm i j (r * D) (Nat.mul_le_mul_right D hr) x hx
84  have hh := normalized_gram_error_le_of_cutoff F A G hG Ω
85    (by simpa [G] using hΩmem) hnorm (fun i => hzero i Ω hΩmem) hadj hF hA
86    L M ε hM hε hbound hcommG u v hu hv
87  simpa [M, Real.sqrt_mul (by positivity : (0:ℝ) ≤ ((L * D : ℕ) : ℝ) + 1)] using hh
