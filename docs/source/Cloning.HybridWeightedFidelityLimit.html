63/-- The manuscript's weighted-hybrid bound with its precise finite-limit
64hypothesis and arbitrary positive trace-class T. The integrability assumptions
65express the finite RHS. The statement also permits unnormalized g. -/
66theorem rootFidelity_sq_le_weighted_productPositive_of_inverse_limit
67    (R : PositiveField (H:=H) μ)
68    (g : Ω→ℝ) (hg : Integrable g μ) (hgn : ∀x,0≤g x) (T : PositiveTraceClass H)
69    {W : H→L[ℂ] H} (hW : 0≤W) {M : ℝ}
70    (hInv : Tendsto (regularizedInverseMoment T W) (𝓝[>] (0:ℝ)) (𝓝 M))
71    (χ : Ω→ℝ) (hχ : ∀x,0<χ x)
72    (hweighted : Integrable (fun x=>χ x*witnessMoment (R.value x).1 W) μ)
73    (hclassical : Integrable (fun x=>g x/χ x) μ) :
74    R.rootFidelity (productPositive g hg hgn T)^2≤
75      (∫x,χ x*witnessMoment (R.value x).1 W ∂μ)*
76      (∫x,g x/χ x ∂μ)*M := by
77  have hM := regularizedInverseMoment_limit_nonneg T W hInv
78  let f : Ω→ℝ := fun x=>χ x*witnessMoment (R.value x).1 W
79  let k : Ω→ℝ := fun x=>g x/χ x
80  have hfn (x : Ω) : 0≤f x := mul_nonneg (hχ x).le (witnessMoment_nonneg _ hW)
81  have hkn (x : Ω) : 0≤k x := div_nonneg (hgn x) (hχ x).le
82  have hip := integrable_sqrt_mul_sqrt hweighted hclassical hfn hkn
83  have hpoint (x : Ω) :
84      PositiveTraceClass.rootFidelity (R.value x) ((productPositive g hg hgn T).value x)≤
85        Real.sqrt (f x)*Real.sqrt (k x)*Real.sqrt M := by
86    have hsq := rootFidelity_productPositive_sq_le R g hg hgn T hW hInv x
87    have heq : f x*k x*M=witnessMoment (R.value x).1 W*(g x*M) := by
88      dsimp [f,k]
89      field_simp [ne_of_gt (hχ x)]
90    have hsqrt : (Real.sqrt (f x)*Real.sqrt (k x)*Real.sqrt M)^2=
91        witnessMoment (R.value x).1 W*(g x*M) := by
92      rw [mul_pow,mul_pow,Real.sq_sqrt (hfn x),Real.sq_sqrt (hkn x),Real.sq_sqrt hM]
93      exact heq
94    have hpos := mul_nonneg (mul_nonneg (Real.sqrt_nonneg (f x))
95      (Real.sqrt_nonneg (k x))) (Real.sqrt_nonneg M)
96    nlinarith [PositiveTraceClass.rootFidelity_nonneg (R.value x) ((productPositive g hg hgn T).value x)]
97  have hroot : R.rootFidelity (productPositive g hg hgn T)≤
98      Real.sqrt (∫x,f x ∂μ)*Real.sqrt (∫x,k x ∂μ)*Real.sqrt M := by
99    calc
100      _≤∫x,Real.sqrt (f x)*Real.sqrt (k x)*Real.sqrt M ∂μ :=
101        integral_mono (R.integrable_rootFidelity _) (hip.mul_const _) hpoint
102      _=(∫x,Real.sqrt (f x)*Real.sqrt (k x) ∂μ)*Real.sqrt M := integral_mul_const _ _
103      _≤_ := mul_le_mul_of_nonneg_right
104        (integral_sqrt_mul_sqrt_le hweighted hclassical hfn hkn) (Real.sqrt_nonneg M)
105  have hfint : 0≤∫x,f x ∂μ := integral_nonneg (fun x=>hfn x)
106  have hkint : 0≤∫x,k x ∂μ := integral_nonneg (fun x=>hkn x)
107  have hsquared := (sq_le_sq₀ (R.rootFidelity_nonneg _) (by positivity)).mpr hroot
108  simpa only [mul_pow,Real.sq_sqrt hfint,Real.sq_sqrt hkint,Real.sq_sqrt hM] using hsquared
