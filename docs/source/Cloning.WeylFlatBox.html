12/-- The unit box in all real and imaginary phase-space coordinates. -/
13def phaseSpaceUnitBox (d : ℕ) : Set (Fin d → ℂ) :=
14  {z | ∀ i, |(z i).re| ≤ 1 ∧ |(z i).im| ≤ 1}

67lemma smul_mem_phaseSpaceBox_iff {L : ℝ} (hL : 0 < L) (z : Fin d → ℂ) :
68    L • z ∈ phaseSpaceBox d L ↔ z ∈ phaseSpaceUnitBox d := by
69  have he (t : ℝ) : |L * t| ≤ L ↔ |t| ≤ 1 := by
70    rw [abs_mul, abs_of_pos hL]
71    simpa only [mul_one] using (mul_le_mul_iff_right₀ hL : L * |t| ≤ L * 1 ↔ |t| ≤ 1)
72  simp only [phaseSpaceBox, phaseSpaceUnitBox, Set.mem_setOf_eq, Pi.smul_apply,
73    Complex.smul_re, Complex.smul_im, smul_eq_mul, he]

75/-- Exact Lebesgue change of variables between unit-box coordinates and the
76physical expanding box. The Jacobian is the real phase-space dimension. -/
77lemma integral_unitBox_smul {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]
78    {L : ℝ} (hL : 0 < L) (f : (Fin d → ℂ) → E) :
79    (∫ z in phaseSpaceUnitBox d, f (L • z)) =
80      (L ^ Module.finrank ℝ (Fin d → ℂ))⁻¹ • ∫ z in phaseSpaceBox d L, f z := by
81  classical
82  have he : (fun z => (phaseSpaceBox d L).indicator f (L • z)) =
83      (phaseSpaceUnitBox d).indicator (fun z => f (L • z)) := by
84    funext z
85    by_cases hz : z ∈ phaseSpaceUnitBox d
86    · simp only [Set.indicator_of_mem hz,
87        Set.indicator_of_mem ((smul_mem_phaseSpaceBox_iff hL z).mpr hz)]
88    · simp only [Set.indicator_of_notMem hz,
89        Set.indicator_of_notMem (mt (smul_mem_phaseSpaceBox_iff hL z).mp hz)]
90  have h := Measure.integral_comp_smul_of_nonneg volume ((phaseSpaceBox d L).indicator f) L
91    (hR := hL.le)
92  calc
93    _ = ∫ z, (phaseSpaceUnitBox d).indicator (fun z => f (L • z)) z :=
94      (integral_indicator (phaseSpaceUnitBox_closed d).measurableSet).symm
95    _ = ∫ z, (phaseSpaceBox d L).indicator f (L • z) :=
96      (congrArg (fun g : (Fin d → ℂ) → E => ∫ z, g z) he).symm
97    _ = _ := h
98    _ = _ := congrArg (fun v : E => (L ^ Module.finrank ℝ (Fin d → ℂ))⁻¹ • v)
99      (integral_indicator (phaseSpaceBox_closed d L).measurableSet)

101/-- The normalized unit-box dilation is exactly the normalized Lebesgue
102integral on the physical box, including zero quantum modes. -/
103theorem normalized_unitBox_smul_eq {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]
104    {L : ℝ} (hL : 0 < L) (f : (Fin d → ℂ) → E) :
105    (volume.real (phaseSpaceUnitBox d))⁻¹ • (∫ z in phaseSpaceUnitBox d, f (L • z)) =
106      (volume.real (phaseSpaceBox d L))⁻¹ • ∫ z in phaseSpaceBox d L, f z := by
107  have hv := integral_unitBox_smul hL (fun _ : Fin d → ℂ => (1 : ℝ))
108  simp only [integral_const, Measure.restrict_apply_univ, measureReal_def,
109    smul_eq_mul, mul_one] at hv
110  have hc : (L ^ Module.finrank ℝ (Fin d → ℂ)) ≠ 0 := pow_ne_zero _ hL.ne'
111  rw [integral_unitBox_smul hL f, smul_smul]
112  congr 1
113  change (volume.real (phaseSpaceUnitBox d))⁻¹ * (L ^ Module.finrank ℝ (Fin d → ℂ))⁻¹ = _
114  change volume.real (phaseSpaceUnitBox d) =
115    (L ^ Module.finrank ℝ (Fin d → ℂ))⁻¹ * volume.real (phaseSpaceBox d L) at hv
116  rw [hv, mul_inv_rev, inv_inv]
117  rw [mul_assoc, mul_inv_cancel₀ hc, mul_one]
