17/-- Schrödinger covariance implies covariance of the reconstructed normal
18Heisenberg dual, with the exact physical displacement conventions. -/
19theorem heisenbergDual_weyl_covariance
20    (Φ : TraceClass (Fock d) →L[ℂ] TraceClass (Fock d)) (r : ℝ)
21    (hΦ : ∀ a T, Φ (displacementTraceMap a T) =
22      displacementTraceMap (r • a) (Φ T))
23    (a : Fin d → ℂ) (A : Fock d →L[ℂ] Fock d) :
24    ((displacement a).comp (heisenbergDual Φ A)).comp (displacement (-a)) =
25      heisenbergDual Φ
26        (((displacement (r • a)).comp A).comp (displacement (-(r • a)))) := by
27  have hcov : ∀ T, Φ (sandwichCLM (displacement (-a))
28      (star (displacement (-a))) T) =
29      sandwichCLM (displacement (r • (-a)))
30        (star (displacement (r • (-a)))) (Φ T) := by
31    intro T
32    simpa only [displacementTraceMap, ContinuousLinearMap.star_eq_adjoint,
33      displacement_adjoint] using hΦ (-a) T
34  have h := heisenbergDual_covariance Φ (displacement (-a))
35    (displacement (r • (-a))) hcov A
36  simpa only [ContinuousLinearMap.star_eq_adjoint, displacement_adjoint,
37    neg_neg, smul_neg, ContinuousLinearMap.mul_def] using h

53    (Φ : QuantumChannel (Fock d) (Fock d)) (r : ℝ)
