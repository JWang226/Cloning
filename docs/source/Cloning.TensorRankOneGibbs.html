15theorem top_occupation_eq_smul_basis (a : Fin d) (Ω : TensorRegister n (Fin d))
16    (hΩ : collectiveGenerator n a a Ω=(n:ℂ) • Ω) :
17    Ω=Ω (fun _=>a) • registerBasis (Fin n → Fin d) (fun _=>a) := by
18  ext w
19  have he := congrArg (fun x : TensorRegister n (Fin d) => x w) hΩ
20  simp only [collectiveGenerator_diagonal,lp.coeFn_smul,Pi.smul_apply,smul_eq_mul] at he
21  by_cases hw : w=(fun _=>a)
22  · subst w
23    simp [registerBasis_apply,lp.single_apply,Pi.single_apply]
24  · have hne : occupancy w a ≠ n := by
25      intro hn
26      apply hw
27      have hall : ∀ i : Fin n, w i=a := by
28        simpa only [Finset.card_univ,Fintype.card_fin,Finset.mem_univ,true_implies,occupancy] using
29          (Finset.card_filter_eq_iff (s := (Finset.univ : Finset (Fin n)))
30            (p := fun i=>w i=a)).mp (by simpa only [Finset.card_univ,Fintype.card_fin] using hn)
31      exact funext hall
32    have hn : (occupancy w a : ℂ) ≠ n := by exact_mod_cast hne
33    have hz : Ω w=0 := (mul_eq_mul_right_iff.mp he).resolve_left hn
34    simp [registerBasis_apply,lp.single_apply,Pi.single_apply,hw,hz]

36theorem tensorOperator_coordinate_projection (a : Fin d) (x : TensorRegister n (Fin d)) :
37    tensorOperator n (Matrix.diagonal (fun b => if b=a then (1:ℂ) else 0)) x =
38      x (fun _=>a) • registerBasis (Fin n → Fin d) (fun _=>a) := by
39  ext w
40  rw [tensorOperator_diagonal_apply]
41  by_cases hw : w=(fun _=>a)
42  · subst w
43    simp [registerBasis_apply,lp.single_apply,Pi.single_apply]
44  · have ht : ∃ t, w t≠a := by
45      by_contra hh
46      push_neg at hh
47      exact hw (funext hh)
48    obtain ⟨t,ht⟩ := ht

53theorem tensorOperator_coordinate_projection_highest (a : Fin d)
54    (Ω : TensorRegister n (Fin d)) (hΩnorm : ‖Ω‖=1)
55    (hΩ : collectiveGenerator n a a Ω=(n:ℂ) • Ω) (x : TensorRegister n (Fin d)) :
56    tensorOperator n (Matrix.diagonal (fun b => if b=a then (1:ℂ) else 0)) x =
57      ⟪Ω,x⟫_ℂ • Ω := by
58  let e := registerBasis (Fin n → Fin d) (fun _=>a)
59  let c := Ω (fun _=>a)
60  have he : Ω=c • e := top_occupation_eq_smul_basis a Ω hΩ
61  have hc : star c*c=1 := by
62    have hh : ⟪Ω,Ω⟫_ℂ=1 := by simp [inner_self_eq_norm_sq_to_K,hΩnorm]
63    rw [he,inner_smul_left,inner_smul_right] at hh
64    simpa only [e,inner_self_eq_norm_sq_to_K,
65      (registerBasis (Fin n → Fin d)).orthonormal.1,one_pow,
66      RCLike.ofReal_one,mul_one,RCLike.star_def] using hh
67  rw [tensorOperator_coordinate_projection,he,inner_smul_left]
68  have hi : ⟪e,x⟫_ℂ=x (fun _=>a) := by
69    simpa only [e,registerBasis_apply] using register_inner_single (fun _ : Fin n=>a) x
70  rw [hi,smul_smul]
71  have hh : (star c*x (fun _=>a))*c=x (fun _=>a) := by
72    calc _ = (star c*c)*x (fun _=>a) := by ring
73         _ = _ := by rw [hc,one_mul]
74  exact (congrArg (fun z : ℂ => z • e) hh).symm

77theorem sectorGibbsDensity_coordinate (a : Fin d)
78    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
79    (hweight : ∀ b,collectiveGenerator n b b Ω=(mu b:ℂ) • Ω)
80    (hraise : ∀ b c,b<c→collectiveGenerator n b c Ω=0)
81    (hΩnorm : ‖Ω‖=1) (ha : mu a=n) :
82    sectorGibbsDensity Ω mu hweight hraise (fun b=>if b=a then 1 else 0)=
83      vectorProjector (⟨Ω,highest_mem_cyclicSector Ω⟩ : cyclicSector Ω) := by
84  let v : cyclicSector Ω := ⟨Ω,highest_mem_cyclicSector Ω⟩
85  have hv : ‖v‖=1 := hΩnorm
86  have hw : sectorGibbsWeight Ω mu hweight hraise (fun b=>if b=a then 1 else 0)=
87      vectorProjector v := by

91    apply Subtype.ext

94    simp only [apply_ite,Complex.ofReal_one,Complex.ofReal_zero]
