38/-- The actual positive Werner operator dominates its pure-product vacuum
39component by the finite symmetric-dimension ratio. -/
40theorem wernerOutput_vacuum_lower {L s : ℕ} (S : Finset (Fin L)) :
41    (wernerScale S.card L s : ℂ) •
42      (vectorProjector (lp.single 2 (fun _ : Fin L => (0:Fin (s+1))) 1)).1 ≤
43      (wernerOutput (s:=s) S).1 := by
44  let q0 : Occupation L (s+1) := label (fun _ => 0)
45  let c : Occupation L (s+1) → ℝ := fun q =>
46    ((q.val 0).choose S.card : ℝ)/(L+s).choose (S.card+s)
47  have hq : q0.val 0=L := by simp [q0,label,profile]
48  have hc : c q0=wernerScale S.card L s := by
49    dsimp only [c]
50    rw [hq,wernerScale_eq_vacuum S.card L s (by simpa using Finset.card_le_univ S)]
51    rfl
52  have hpos (q : Occupation L (s+1)) :
53      0 ≤ (c q : ℂ) • (vectorProjector (column q)).1 := by
54    change 0 ≤ (c q : ℂ) • InnerProductSpace.rankOne ℂ (column q) (column q)
55    rw [Complex.coe_smul]
56    exact smul_nonneg (by dsimp [c]; positivity)
57      ((InnerProductSpace.rankOne ℂ (column q) (column q)).nonneg_iff_isPositive.mpr
58        (InnerProductSpace.isPositive_rankOne_self (column q)))
59  have hs := Finset.single_le_sum (fun q (_ : q∈Finset.univ) => hpos q)
60    (Finset.mem_univ q0)
61  change (c q0 : ℂ) • (vectorProjector (column q0)).1 ≤
62    ∑ q, (c q : ℂ) • (vectorProjector (column q)).1 at hs
63  rw [hc] at hs
64  change (wernerScale S.card L s : ℂ) •
65    (vectorProjector (column (label (fun _ => 0)))).1 ≤ _ at hs
66  rw [zero_column] at hs
67  change _ ≤ inclusionCLM (wernerOutput (s:=s) S)
68  simpa only [wernerOutput,map_sum,map_smul] using hs
