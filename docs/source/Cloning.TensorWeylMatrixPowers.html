41theorem numberCoefficients_eq_cutoff_sum (Q : ℕ)
42    (c : NumberCoefficients (Fintype.card (PositiveRoot d))) (hc : c ∈ numberHeightCutoff d Q) :
43    c = ∑ i : CutoffIndex d Q,
44      numberCoordinates Q c i • Finsupp.single (cutoffNumberOccupation d Q i) 1 := by
45  classical
46  ext k
47  simp only [Finsupp.finset_sum_apply, Finsupp.smul_apply, Finsupp.single_apply, smul_eq_mul,
48    numberCoordinates]
49  by_cases hk : rootNumberHeight d k ≤ Q
50  · obtain ⟨i, hi⟩ := exists_cutoffNumberOccupation k hk
51    change cutoffNumberOccupation d Q i = k at hi
52    subst k
53    simp only [mul_ite, mul_one, mul_zero]
54    have he (j : CutoffIndex d Q) :
55        cutoffNumberOccupation d Q j = cutoffNumberOccupation d Q i ↔ j = i :=
56      (cutoffNumberOccupation_injective d Q).eq_iff
57    simp only [he, Finset.sum_ite_eq', Finset.mem_univ, ↓reduceIte]
58  · have hz : c k = 0 := (Finsupp.mem_supported' ℂ c).mp hc k hk
59    rw [hz]
60    symm
61    apply Finset.sum_eq_zero
62    intro i _
63    have hi : cutoffNumberOccupation d Q i ≠ k := by
64      intro hi
65      apply hk
66      rw [← hi]
67      exact cutoffNumberOccupation_height Q i
68    simp [hi]

101theorem numberCoordinates_iterate (Q R : ℕ) (z : PositiveRoot d → ℂ)
102    (c : NumberCoefficients (Fintype.card (PositiveRoot d))) (hc : c ∈ numberHeightCutoff d R) :
103    ∀ m : ℕ, R+m*d ≤ Q →
104      numberCoordinates Q (numberIterate (rootFockAmplitude z) c m) =
105        ((oscillatorCutoffMatrix Q z)^m).mulVec (numberCoordinates Q c) := by
106  intro m
107  induction m with
108  | zero => intro _; simp [numberIterate_zero]
109  | succ m ih =>
110    intro hQ
111    have hm : R+m*d ≤ Q := (Nat.add_le_add_left
112      (Nat.mul_le_mul_right d (Nat.le_succ m)) R).trans hQ
113    rw [numberIterate_succ, numberCoordinates_generator Q z _
114      (numberHeightCutoff_mono hm (numberIterate_mem z c hc m)), ih hm,
115      pow_succ', Matrix.mulVec_mulVec]

127/-- All intermediate ladder states fit into the chosen height cutoff. Hence
128the finite oscillator matrix power equals the actual infinite-Fock vector. -/
129theorem oscillatorCutoffMatrix_power_eq_numberPower
130    (Q m : ℕ) (z : PositiveRoot d → ℂ) (w : List (PositiveRoot d))
131    (hQ : loweringHeight w+m*d ≤ Q) :
132    (∑ i : CutoffIndex d Q,
133      (((oscillatorCutoffMatrix Q z)^m).mulVec (limitingWordCoordinates Q w) i) •
134        cutoffNumberFrame d Q i) =
135      numberPower (rootFockAmplitude z) (wordFockOccupation w) m := by
136  rw [← numberCoordinates_word, ← numberCoordinates_iterate Q (loweringHeight w) z _
137    (wordCoefficients_mem w) m hQ]
138  exact (numberVector_eq_cutoff_sum Q _
139    (numberHeightCutoff_mono hQ (numberIterate_mem z _ (wordCoefficients_mem w) m))).symm
