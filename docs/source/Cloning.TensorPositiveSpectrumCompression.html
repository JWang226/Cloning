32/-- Literal zero extension of the diagonal tensor power along the coordinate
33isometry. This establishes the rank identity before taking Schur blocks. -/
34theorem tensorOperator_padSpectrum (p : Fin r → ℝ) :
35    tensorOperator n (Matrix.diagonal (fun a => (padSpectrum p k a : ℂ))) =
36      (coordinateTensorEmbedding n r k).toContinuousLinearMap.comp
37        ((tensorOperator n (Matrix.diagonal (fun a => (p a : ℂ)))).comp
38          (coordinateTensorEmbedding n r k).toContinuousLinearMap.adjoint) := by
39  ext y w
40  rw [tensorOperator_diagonal_apply]
41  by_cases hw : ∀t,(w t).val<r
42  · obtain ⟨v,rfl⟩ := (coordinateWordEmbedding_range w).mpr hw
43    change _ = coordinateTensorEmbedding n r k
44      (tensorOperator n (Matrix.diagonal (fun a => (p a : ℂ)))
45        ((coordinateTensorEmbedding n r k).toContinuousLinearMap.adjoint y))
46          (coordinateWordEmbedding n r k v)
47    rw [coordinateTensorEmbedding,registerEmbedding_apply_image,
48      tensorOperator_diagonal_apply,registerEmbedding_adjoint_apply]
49    simp only [coordinateWordEmbedding,Function.Embedding.coeFn_mk,padSpectrum_left]
50  · have hz : (∏t : Fin n,(padSpectrum p k (w t) : ℂ))=0 := by
51      push_neg at hw
52      obtain ⟨t,ht⟩ := hw
53      apply Finset.prod_eq_zero (Finset.mem_univ t)
54      have hright : ∃a : Fin k, Fin.natAdd r a=w t := by
55        exact ⟨⟨(w t).val-r,by have := (w t).isLt; omega⟩,Fin.ext (by dsimp; omega)⟩
56      obtain ⟨a,ha⟩ := hright
57      rw [← ha,padSpectrum_right,Complex.ofReal_zero]
58    rw [hz,zero_mul]
59    exact (registerEmbedding_apply_not_mem _ _ w
60      (fun he => hw ((coordinateWordEmbedding_range w).mp he))).symm

62/-- The literal rank identity restricted to the cyclic sector generated by an

64theorem coordinateSectorEmbedding_gibbs
65    (Ω : TensorRegister n (Fin r)) (mu : Fin r → ℕ)
66    (hweight : ∀a,collectiveGenerator n a a Ω=(mu a : ℂ) • Ω)
67    (hraise : ∀a b,a<b → collectiveGenerator n a b Ω=0) (p : Fin r → ℝ) :
68    sectorGibbsOperator (coordinateTensorEmbedding n r k Ω) (padPartition mu k)
69      (coordinateTensorEmbedding_highest Ω mu hweight hraise).1
70      (coordinateTensorEmbedding_highest Ω mu hweight hraise).2 (padSpectrum p k) =
71    (coordinateSectorEmbedding (k := k) Ω).toContinuousLinearMap.comp
72      ((sectorGibbsOperator Ω mu hweight hraise p).comp
73        (coordinateSectorEmbedding (k := k) Ω).toContinuousLinearMap.adjoint) := by
74  apply ContinuousLinearMap.ext
75  intro x
76  apply Subtype.ext
77  change tensorOperator n (Matrix.diagonal (fun a => (padSpectrum p k a : ℂ)))
78      (x : TensorRegister n (Fin (r+k))) =
79    coordinateTensorEmbedding n r k

81        (((coordinateSectorEmbedding (k := k) Ω).toContinuousLinearMap.adjoint x : cyclicSector Ω) :

90      (partitionHighestTensor (padPartition mu k) (padPartition_antitone mu hmu k))

101    (partitionHighestTensor (padPartition mu k) (padPartition_antitone mu hmu k))
