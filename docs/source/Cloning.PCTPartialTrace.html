59/-- The literal finite Kraus sum with Kraus maps `I ⊗ ⟨b|`. -/
60def partialTraceLinear : TraceClass (Register (A × B)) →ₗ[ℂ] TraceClass (Register A) :=

63theorem partialTraceLinear_coefficient (X : TraceClass (Register (A × B))) (a c : A) :

74theorem traceCLM_register (X : TraceClass (Register A)) :
75    traceCLM X = ∑ a : A, X.1 (lp.single 2 a 1) a := by
76  letI : FiniteDimensional ℂ (Register A) :=
77    (registerBasis A).toOrthonormalBasis.toBasis.finiteDimensional_of_finite
78  change trace X.1 X.2 = _
79  rw [trace_eq_linearMap_trace_of_finiteDimensional]
80  simpa only [HilbertBasis.coe_toOrthonormalBasis, ContinuousLinearMap.coe_coe,
81    registerBasis_apply, register_inner_single] using
82    LinearMap.trace_eq_sum_inner X.1.toLinearMap (registerBasis A).toOrthonormalBasis

95theorem partialTraceLinear_completelyPositive :

109/-- The actual partial trace, including complete positivity at every ancilla
110size and exact trace preservation on all complex trace-class inputs. -/
111def partialTraceChannel : QuantumChannel (Register (A × B)) (Register A) where
112  toLinearMap := partialTraceLinear
113  map_nonneg := partialTraceLinear_nonneg
114  trace_preserving := partialTraceLinear_trace
115  completelyPositive := partialTraceLinear_completelyPositive
