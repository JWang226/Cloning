18theorem physicalWordHeight_update (w : Fin n → Fin d) (t : Fin n) (a : Fin d) :
19    physicalWordHeight (Function.update w t a) + (w t).val = physicalWordHeight w + a.val := by
20  have he := Finset.sum_congr (s₁ := (Finset.univ : Finset (Fin n))) rfl
21    (fun u _ => show (Function.update w t a u).val + (if u = t then (w t).val else 0) =
22      (w u).val + (if u = t then a.val else 0) from by
23        by_cases h : u = t
24        · subst u; simp [Nat.add_comm]
25        · simp [h])
26  simpa only [physicalWordHeight, Finset.sum_add_distrib, Finset.sum_ite_eq',
27    Finset.mem_univ, if_true] using he

79theorem highest_weight_antitone (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
80    (hweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
81    (hraise : ∀ a b, a < b → collectiveGenerator n a b Ω = 0)
82    (hΩ : ‖Ω‖ = 1) : Antitone mu := by
83  intro a b hab
84  rcases eq_or_lt_of_le hab with rfl | hab
85  · rfl
86  · have hc := root_diagonal_commutator a b Ω
87    rw [hraise a b hab, map_zero, sub_zero, hweight a, hweight b, ← sub_smul] at hc
88    have hi : ⟪collectiveGenerator n b a Ω, collectiveGenerator n b a Ω⟫_ℂ =
89        (mu a : ℂ) - mu b := by
90      rw [collectiveGenerator_inner_adjoint, hc, inner_smul_right,
91        inner_self_eq_norm_sq_to_K, hΩ]
