16def localSpectrum (p h : Fin d → ℝ) (N : ℕ) : Fin d → ℝ :=

23/-- Every bounded sequence of local spectral parameters has the same base

28    Tendsto (fun k => localSpectrum p (h k) (n k) a) atTop (𝓝 (p a)) := by
