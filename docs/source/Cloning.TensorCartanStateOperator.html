38theorem basis_ofMatrix_compression (bH : OrthonormalBasis I ℂ H)

100theorem physicalCartanChannel_operator
101    (mu nu : Fin d → ℕ) (hmu : Antitone mu) (hnu : Antitone nu)
102    (A : TraceClass (cyclicSector (partitionHighestTensor mu hmu))) :
103    ((physicalCartanChannel mu nu hmu hnu).toLinearMap A).1 =
104      (((partitionDimension mu hmu : ℝ) /
105        (partitionDimension _ (sumPartition_antitone mu nu hmu hnu) : ℝ)) : ℂ) •
106      (cartanInclusion mu nu hmu hnu).toContinuousLinearMap.adjoint.comp
107        ((cartanProductOperator mu nu hmu hnu A.1).comp
108          (cartanInclusion mu nu hmu hnu).toContinuousLinearMap) := by
109  let M := matrixOf (partitionBasis mu hmu) A.1
110  have hA : partitionMatrixOperator mu hmu M = A := partitionMatrixOperator_matrixOf _ _ A
111  have hc := congrArg Subtype.val
112    (physicalCartanChannel_partitionMatrixOperator mu nu hmu hnu M)
113  rw [hA, partitionMatrixOperator_eq_matrixLift] at hc
114  change ((physicalCartanChannel mu nu hmu hnu).toLinearMap A).1 =
115    ofMatrix (partitionBasis _ (sumPartition_antitone mu nu hmu hnu))
116      (Cloning.Compression.sectorMap _ _ (cartanMatrix mu nu hmu hnu) M) at hc
117  have hcomp := basis_ofMatrix_compression
118    (partitionBasis _ (sumPartition_antitone mu nu hmu hnu))
119    (cartanProductBasis mu nu hmu hnu)
120    (cartanInclusion mu nu hmu hnu).toContinuousLinearMap
121    (M ⊗ₖ (1 : Matrix (PartitionIndex nu hnu) _ ℂ))
122  have hs := basis_ofMatrix_smul
123    (partitionBasis _ (sumPartition_antitone mu nu hmu hnu))
124    ((((partitionDimension mu hmu : ℝ) /
125      (partitionDimension _ (sumPartition_antitone mu nu hmu hnu) : ℝ)) : ℂ))
126    ((cartanMatrix mu nu hmu hnu)ᴴ * (M ⊗ₖ (1 : Matrix (PartitionIndex nu hnu) _ ℂ)) *
127      cartanMatrix mu nu hmu hnu)
128  have hfinal := hs.trans (congrArg (fun
129    (T : cyclicSector (partitionHighestTensor _ (sumPartition_antitone mu nu hmu hnu)) →L[ℂ]
130      cyclicSector (partitionHighestTensor _ (sumPartition_antitone mu nu hmu hnu))) =>
131    ((((partitionDimension mu hmu : ℝ) /
132      (partitionDimension _ (sumPartition_antitone mu nu hmu hnu) : ℝ)) : ℂ)) • T) hcomp)
133  have hr : Cloning.Compression.sectorMap (partitionDimension mu hmu : ℝ)
134      (partitionDimension _ (sumPartition_antitone mu nu hmu hnu) : ℝ)
135      (cartanMatrix mu nu hmu hnu) M =
136      ((((partitionDimension mu hmu : ℝ) /
137        (partitionDimension _ (sumPartition_antitone mu nu hmu hnu) : ℝ)) : ℂ)) •
138      ((cartanMatrix mu nu hmu hnu)ᴴ * (M ⊗ₖ (1 : Matrix (PartitionIndex nu hnu) _ ℂ)) *
139        cartanMatrix mu nu hmu hnu) := by

144      ((cartanMatrix mu nu hmu hnu)ᴴ * (M ⊗ₖ (1 : Matrix (PartitionIndex nu hnu) _ ℂ)) *
145        cartanMatrix mu nu hmu hnu)

147    (sumPartition_antitone mu nu hmu hnu))) hr).trans hfinal)
