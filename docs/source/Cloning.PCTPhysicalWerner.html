22def physicalSymmetric (L : ℕ) : Submodule ℂ (Register (Fin L → C)) where
23  carrier := {v | ∀ (w : Fin L → C) (σ : Equiv.Perm (Fin L)), v (w ∘ σ) = v w}
24  zero_mem' := by intro w σ; rfl
25  add_mem' := by intro v u hv hu w σ; exact congrArg₂ (· + ·) (hv w σ) (hu w σ)
26  smul_mem' := by intro a v hv w σ; exact congrArg (a • ·) (hv w σ)

41theorem tensorFrame_permute {d L : ℕ} {u : Fin d → Register C} (hu : Orthonormal ℂ u)
42    (σ : Equiv.Perm (Fin L)) (x : TensorSpace L d) :
43    tensorFrame hu L (permuteRegister σ x) = permuteRegister σ (tensorFrame hu L x) := by
44  ext v
45  simp only [tensorFrame_apply, permuteRegister_apply]
46  calc
47    _ = ∑ w : Word L d, x w * ∏ i, u (w (σ.symm i)) (v i) := by
48      have he := (wordPermutation L d σ.symm).sum_comp
49        (fun w => x (w ∘ σ) * ∏ i, u (w i) (v i))
50      have hc (w : Word L d) : (w ∘ σ.symm) ∘ σ = w := by funext i; simp
51      simpa only [wordPermutation, Equiv.coe_fn_mk, hc, Function.comp_apply] using he.symm
52    _ = _ := by
53      apply Finset.sum_congr rfl
54      intro w hw
55      congr 1
56      have he := Equiv.prod_comp σ (fun i => u (w (σ.symm i)) (v i))
57      simpa only [Equiv.symm_apply_apply, Function.comp_apply] using he.symm

72theorem physicalSymmetric_map_tensorFrame {d : ℕ}
73    (u : OrthonormalBasis (Fin d) ℂ (Register C)) (L : ℕ) :
74    (symmetricSubspace L d).map (tensorFrame u.orthonormal L).toLinearMap =
75      physicalSymmetric (C := C) L := by
76  apply le_antisymm
77  · rintro y ⟨x, hx, rfl⟩
78    intro w σ
79    have hp : permuteRegister σ x = x := by ext v; exact hx v σ
80    have he := tensorFrame_permute u.orthonormal σ x
81    rw [hp] at he
82    exact (congrArg (fun v : Register (Fin L → C) => v w) he).symm
83  · intro y hy
84    obtain ⟨x, rfl⟩ := tensorFrame_surjective (L := L) u y
85    refine ⟨x, ?_, rfl⟩
86    intro w σ
87    have hp : permuteRegister σ (tensorFrame u.orthonormal L x) =

118/-- A pure product on the selected slots tensored with full identities on all
119remaining slots, specified in the actual computational register. -/
120def pureSlotsMatrix {L : ℕ} (ψ : Register C) (S : Finset (Fin L)) :
121    Matrix (Fin L → C) (Fin L → C) ℂ := fun a c =>
122  ∏ i, if i ∈ S then ψ (a i) * star (ψ (c i)) else if a i = c i then 1 else 0

145/-- The actual framed Werner output is the canonical physical Werner
146symmetric sandwich on an arbitrary purification, with the full environment
147identity. The projector and input operator on the right are frame-independent. -/
148theorem wernerOutput_physical_purification {L s : ℕ}
149    (u : OrthonormalBasis (Fin (s + 1)) ℂ (Register C)) (S : Finset (Fin L)) :
150    ((QuantumChannel.ofIsometry (tensorFrame u.orthonormal L)).toLinearMap
151      (wernerOutput (s := s) S)).1 =
152      (((Nat.choose (S.card + s) s : ℝ) / Nat.choose (L + s) s : ℝ) : ℂ) •
153        (physicalProjector L * pureSlotsOperator (u 0) S * physicalProjector L) := by
154  change operatorConjugation (tensorFrame u.orthonormal L).toContinuousLinearMap
155    (wernerOutput (s := s) S).1 = _
156  rw [wernerOutput_op, wernerOutputOperator, occupation_card, occupation_card]
157  change operatorConjugation (tensorFrame u.orthonormal L).toContinuousLinearMap (_ • _) = _
158  rw [map_smul]
159  congr 1
160  apply ContinuousLinearMap.ext
161  intro x
162  change tensorFrame u.orthonormal L (projector L (s + 1)
163    (fixedSlotProjector 0 S (projector L (s + 1)
164      ((tensorFrame u.orthonormal L).toContinuousLinearMap.adjoint x)))) = _
165  rw [tensorFrame_physicalProjector, ← tensorFrame_adjoint_physicalProjector]
166  have he := congrArg (fun T : Register (Fin L → C) →L[ℂ] Register (Fin L → C) =>
167    physicalProjector L (T (physicalProjector L x))) (tensorFrame_fixedSlot_physical u 0 S)
