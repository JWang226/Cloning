25/-- Finite tensor products of convergent complex series, with absolute
26summability supplied by finite-dimensionality of the scalar field. -/
27theorem hasSum_fin_product_complex (s : ℕ) (f : Fin s → ℕ → ℂ) (a : Fin s → ℂ)
28    (hs : ∀ i, HasSum (f i) (a i)) :
29    HasSum (fun k : Fin s → ℕ ↦ ∏ i, f i (k i)) (∏ i, a i) := by
30  induction s with
31  | zero => simp
32  | succ s ih =>
33    have ht : HasSum (fun k : Fin s → ℕ ↦ ∏ i : Fin s, f i.succ (k i))
34        (∏ i : Fin s, a i.succ) := ih (fun i ↦ f i.succ) (fun i ↦ a i.succ)
35          (fun i ↦ hs i.succ)
36    have hsum : Summable (fun k : ℕ × (Fin s → ℕ) ↦
37        f 0 k.1 * ∏ i : Fin s, f i.succ (k.2 i)) :=
38      summable_mul_of_summable_norm (f := f 0)
39        (g := fun k : Fin s → ℕ ↦ ∏ i : Fin s, f i.succ (k i))
40        (hs 0).summable.norm ht.summable.norm
41    have hm : HasSum (fun k : ℕ × (Fin s → ℕ) ↦
42        f 0 k.1 * ∏ i : Fin s, f i.succ (k.2 i)) (a 0 * ∏ i : Fin s, a i.succ) :=
43      HasSum.mul (f := f 0) (g := fun k : Fin s → ℕ ↦ ∏ i : Fin s, f i.succ (k i))
44        (s := a 0) (t := ∏ i : Fin s, a i.succ) (hs 0) ht hsum
45    rw [← (Fin.consEquiv (fun _ : Fin (s + 1) ↦ ℕ)).hasSum_iff]
46    simpa [Fin.prod_univ_succ, Fin.consEquiv, Function.comp_def] using hm

112/-- Norm continuity holds jointly in all amplitudes, including zero amplitudes
113and the empty tuple of modes. -/
114theorem continuous_coherentVector (s : ℕ) : Continuous (@coherentVector s) := by
115  apply continuous_iff_continuousAt.mpr
116  intro z
117  have hn (w : Fin s → ℂ) : ‖coherentVector w - coherentVector z‖ ^ 2 =
118      2 - 2 * (⟪coherentVector w, coherentVector z⟫_ℂ).re := by
119    have h := norm_sub_sq (𝕜 := ℂ) (coherentVector w) (coherentVector z)
120    simp only [coherentVector_norm, one_pow] at h
121    change ‖coherentVector w - coherentVector z‖ ^ 2 =
122      1 - 2 * (⟪coherentVector w, coherentVector z⟫_ℂ).re + 1 at h
123    linarith
124  have hs : Continuous (fun w : Fin s → ℂ ↦ ‖coherentVector w - coherentVector z‖ ^ 2) := by
125    simp_rw [hn, inner_coherentVector]
126    fun_prop
127  have ht : Tendsto (fun w : Fin s → ℂ ↦ ‖coherentVector w - coherentVector z‖ ^ 2)
128      (𝓝 z) (𝓝 (0 : ℝ)) := by
129    simpa only [sub_self, norm_zero, zero_pow (by norm_num : (2 : ℕ) ≠ 0)] using hs.tendsto z
130  apply tendsto_iff_norm_sub_tendsto_zero.mpr
131  have hroot := (Real.continuous_sqrt.tendsto 0).comp ht
132  simpa only [Function.comp_def, Real.sqrt_sq (norm_nonneg _), Real.sqrt_zero] using hroot
