68/-- For a positive rank-boundary coefficient, the two envelopes bracket a
69small target error at gains (1±η)ε/a. This is an iterated-limit certificate. -/
70theorem envelopes_target_bracket (a b : ℕ) (ha : 0<a) {η : ℝ}
71    (hη : 0<η) (hη1 : η<1) :
72    ∀ᶠ ε in 𝓝[>] (0:ℝ),
73      upperEnvelope a b ((1-η)*ε/a)<ε ∧
74      ε<lowerEnvelope a ((1+η)*ε/a) := by
75  have ha0 : (a:ℝ)≠0 := by exact_mod_cast Nat.ne_of_gt ha
76  have hl := scaled_ratio_tendsto
77    (show lowerEnvelope a 0=0 by simp [lowerEnvelope,supportLimit])
78    (lowerEnvelope_hasDerivAt_zero a) ((1+η)/a)
79  have hu := scaled_ratio_tendsto
80    (show upperEnvelope a b 0=0 by simp [upperEnvelope,supportLimit,internalError])
81    (upperEnvelope_hasDerivAt_zero a b) ((1-η)/a)
82  have hal : (a:ℝ)*((1+η)/a)=1+η := by field_simp
83  have hau : (a:ℝ)*((1-η)/a)=1-η := by field_simp
84  rw [hal] at hl
85  rw [hau] at hu
86  have hmono : 𝓝[>] (0:ℝ)≤𝓝[≠] (0:ℝ) :=
87    nhdsWithin_mono _ (by intro x hx; simpa using ne_of_gt hx)
88  have hl' := (hl.mono_left hmono).eventually (Ioi_mem_nhds (show 1<1+η by linarith))
89  have hu' := (hu.mono_left hmono).eventually (Iio_mem_nhds (show 1-η<1 by linarith))
90  filter_upwards [hl',hu',self_mem_nhdsWithin] with ε hlε huε hε
91  have hε0 : 0<ε := hε
92  constructor
93  · have h := (div_lt_one hε0).mp huε
94    simpa [div_eq_mul_inv,mul_assoc,mul_left_comm,mul_comm] using h
95  · have h := (one_lt_div hε0).mp hlε
96    simpa [div_eq_mul_inv,mul_assoc,mul_left_comm,mul_comm] using h
