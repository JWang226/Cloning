9/-!
10# Concrete asymptotics of the crossing-root Weyl dimension product
11 
12For an `r`-row label in ambient dimension `r+k`, the manuscript's explicit
13dimension-ratio expression is the product of `(rowᵢ+j-i)/(j-i)` over crossing
14roots. This file defines that actual finite product and derives its uniform
15relative-error bound directly from the row lengths. No dimension-ratio
16convergence or product approximation is assumed.
17 
18The identification of this explicit Weyl product with dimensions of the
19particular representation spaces is a separate representation-theoretic
20statement. This file proves the analytic input once that formula is used.
21-/

29/-- The positive-root gap `j-i`, using zero-based indices on either side
30of the rank boundary. Ambient dimension is `r+k`. -/
31def crossingGap {r k : ℕ} (a : Fin r × Fin k) : ℝ :=
32  (r : ℝ) + (a.2.val : ℝ) - (a.1.val : ℝ)
33 
34theorem crossingGap_pos {r k : ℕ} (a : Fin r × Fin k) : 0 < crossingGap a := by
35  have hi : (a.1.val : ℝ) < r := by exact_mod_cast a.1.isLt
36  have hj : 0 ≤ (a.2.val : ℝ) := Nat.cast_nonneg _
37  dsimp [crossingGap]
38  linarith
39 
40theorem crossingGap_le {r k : ℕ} (a : Fin r × Fin k) :
41    crossingGap a ≤ (r : ℝ) + k := by
42  have hj : (a.2.val : ℝ) < k := by exact_mod_cast a.2.isLt
43  have hi : 0 ≤ (a.1.val : ℝ) := Nat.cast_nonneg _
44  dsimp [crossingGap]
45  linarith

47/-- The manuscript's concrete crossing-root dimension-ratio product. -/
48def dimensionRatio (r k : ℕ) (row : Fin r → ℝ) : ℝ :=
49  ∏ a : Fin r × Fin k, (row a.1 + crossingGap a) / crossingGap a
50 
51/-- The exact leading coefficient `c_{r,r+k}`. -/
52def leadingConstant (r k : ℕ) : ℝ :=
53  ∏ a : Fin r × Fin k, 1 / ((r : ℝ) * crossingGap a)
54 
55theorem leadingConstant_pos (r k : ℕ) (hr : 0 < r) : 0 < leadingConstant r k := by
56  apply Finset.prod_pos
57  intro a _
58  exact one_div_pos.mpr (mul_pos (Nat.cast_pos.mpr hr) (crossingGap_pos a))

60/-- An exact rescaling identity, before taking any limit. -/
61theorem dimensionRatio_factorization (r k : ℕ) (row : Fin r → ℝ) (N : ℝ)
62    (hr : 0 < r) (hN : 0 < N) :
63    dimensionRatio r k row = leadingConstant r k * N ^ (r * k) *
64      ∏ a : Fin r × Fin k, ((r : ℝ) * (row a.1 + crossingGap a) / N) := by
65  have hfactor (a : Fin r × Fin k) :
66      (row a.1 + crossingGap a) / crossingGap a =
67        ((1 / ((r : ℝ) * crossingGap a)) * N) *
68          ((r : ℝ) * (row a.1 + crossingGap a) / N) := by
69    have hr' : (r : ℝ) ≠ 0 := (Nat.cast_pos.mpr hr).ne'
70    have hg : crossingGap a ≠ 0 := (crossingGap_pos a).ne'
71    field_simp
72  unfold dimensionRatio
73  simp_rw [hfactor]
74  rw [Finset.prod_mul_distrib, Finset.prod_mul_distrib]
75  simp only [leadingConstant, Finset.prod_const, Finset.card_univ,
76    Fintype.card_prod, Fintype.card_fin]

78/-- The per-root rescaled factor is uniformly close to one whenever the
79row proportions are uniformly close to the flat spectrum. -/
80theorem rescaled_factor_error (r k : ℕ) (row : Fin r → ℝ) (N δ : ℝ)
81    (hr : 0 < r) (hN : 0 < N)
82    (hclose : ∀ i, |row i / N - 1 / (r : ℝ)| ≤ δ) (a : Fin r × Fin k) :
83    |(r : ℝ) * (row a.1 + crossingGap a) / N - 1| ≤
84      (r : ℝ) * δ + (r : ℝ) * ((r : ℝ) + k) / N := by
85  have hr' : 0 < (r : ℝ) := Nat.cast_pos.mpr hr
86  have heq : (r : ℝ) * (row a.1 + crossingGap a) / N - 1 =
87      (r : ℝ) * (row a.1 / N - 1 / (r : ℝ)) + (r : ℝ) * crossingGap a / N := by
88    field_simp
89    ring
90  rw [heq]
91  calc
92    _ ≤ |(r : ℝ) * (row a.1 / N - 1 / (r : ℝ))| +
93        |(r : ℝ) * crossingGap a / N| := abs_add_le _ _
94    _ = (r : ℝ) * |row a.1 / N - 1 / (r : ℝ)| + (r : ℝ) * crossingGap a / N := by
95      rw [abs_mul, abs_of_pos hr', abs_of_nonneg
96        (div_nonneg (mul_nonneg hr'.le (crossingGap_pos a).le) hN.le)]
97    _ ≤ (r : ℝ) * δ + (r : ℝ) * ((r : ℝ) + k) / N := by
98      apply add_le_add (mul_le_mul_of_nonneg_left (hclose a.1) hr'.le)
99      exact div_le_div_of_nonneg_right
100        (mul_le_mul_of_nonneg_left (crossingGap_le a) hr'.le) hN.le

102/-- A uniform quantitative estimate for the actual Weyl ratio product.
103The right-hand side tends to zero whenever `δ → 0` and `N → ∞`.
104In particular, this proves uniform leading-order asymptotics on the
105manuscript's typical row sets without assuming a product approximation. -/
106theorem dimensionRatio_relative_error (r k : ℕ) (row : Fin r → ℝ) (N δ : ℝ)
107    (hr : 0 < r) (hN : 0 < N)
108    (hclose : ∀ i, |row i / N - 1 / (r : ℝ)| ≤ δ) :
109    |dimensionRatio r k row / (leadingConstant r k * N ^ (r * k)) - 1| ≤
110      Real.exp ((r * k : ℕ) * ((r : ℝ) * δ + (r : ℝ) * ((r : ℝ) + k) / N)) - 1 := by
111  let f : Fin r × Fin k → ℝ := fun a => (r : ℝ) * (row a.1 + crossingGap a) / N
112  have hden : leadingConstant r k * N ^ (r * k) ≠ 0 :=
113    (mul_pos (leadingConstant_pos r k hr) (pow_pos hN _)).ne'
114  rw [dimensionRatio_factorization r k row N hr hN, mul_div_cancel_left₀ _ hden]
115  have hprod := Finset.norm_prod_one_add_sub_one_le Finset.univ (fun a => f a - 1)
116  have hsum : (∑ a, |f a - 1|) ≤
117      (r * k : ℕ) * ((r : ℝ) * δ + (r : ℝ) * ((r : ℝ) + k) / N) := by
118    calc
119      _ ≤ ∑ _a : Fin r × Fin k,
120          ((r : ℝ) * δ + (r : ℝ) * ((r : ℝ) + k) / N) := by
121        exact Finset.sum_le_sum (fun a _ => rescaled_factor_error r k row N δ hr hN hclose a)
122      _ = _ := by simp [Fintype.card_prod]; ring
123  have hp : |(∏ a, f a) - 1| ≤ Real.exp (∑ a, |f a - 1|) - 1 := by
124    have h1 (a) : 1 + (f a - 1) = f a := by ring
125    simpa only [h1, Real.norm_eq_abs] using hprod
126  exact hp.trans (sub_le_sub_right (Real.exp_le_exp.mpr hsum) 1)

128/-- Nonnegative rows give a strictly positive dimension-ratio product. -/
129theorem dimensionRatio_pos (r k : ℕ) (row : Fin r → ℝ)
130    (hrow : ∀ i, 0 ≤ row i) : 0 < dimensionRatio r k row := by
131  apply Finset.prod_pos
132  intro a _
133  exact div_pos (add_pos_of_nonneg_of_pos (hrow a.1) (crossingGap_pos a))
134    (crossingGap_pos a)

136/-- The explicit error envelope vanishes. This bound is independent of
137the selected row vector, so it controls all typical labels simultaneously. -/
138theorem error_envelope_tendsto_zero (r k : ℕ) (N δ : ℕ → ℝ)
139    (hN : Tendsto N atTop atTop) (hδ : Tendsto δ atTop (𝓝 0)) :
140    Tendsto (fun n =>
141      Real.exp ((r * k : ℕ) * ((r : ℝ) * δ n + (r : ℝ) * ((r : ℝ) + k) / N n)) - 1)
142      atTop (𝓝 0) := by
143  have hinv : Tendsto (fun n => (N n)⁻¹) atTop (𝓝 (0 : ℝ)) :=
144    tendsto_inv_atTop_zero.comp hN
145  have harg : Tendsto
146      (fun n => (r * k : ℕ) * ((r : ℝ) * δ n + (r : ℝ) * ((r : ℝ) + k) / N n))
147      atTop (𝓝 (0 : ℝ)) := by
148    simpa only [div_eq_mul_inv, mul_zero, add_zero] using
149      (((tendsto_const_nhds (x := (r : ℝ))).mul hδ).add
150        ((tendsto_const_nhds (x := (r : ℝ) * ((r : ℝ) + k))).mul hinv)).const_mul
151          ((r * k : ℕ) : ℝ)
152  simpa only [Real.exp_zero, sub_self] using
153    ((Real.continuous_exp.tendsto 0).comp harg).sub_const 1

155/-- Leading-order normalization for the concrete Weyl ratio, derived
156from row proximity and the proved quantitative estimate. -/
157theorem normalized_dimensionRatio_tendsto_one (r k : ℕ) (hr : 0 < r)
158    (row : ℕ → Fin r → ℝ) (N δ : ℕ → ℝ)
159    (hNpos : ∀ n, 0 < N n) (hN : Tendsto N atTop atTop)
160    (hδ : Tendsto δ atTop (𝓝 0))
161    (hclose : ∀ n i, |row n i / N n - 1 / (r : ℝ)| ≤ δ n) :
162    Tendsto (fun n => dimensionRatio r k (row n) /
163      (leadingConstant r k * (N n) ^ (r * k))) atTop (𝓝 1) := by
164  have hzero : Tendsto (fun n => dimensionRatio r k (row n) /
165      (leadingConstant r k * (N n) ^ (r * k)) - 1) atTop (𝓝 0) := by
166    apply squeeze_zero_norm
167      (fun n => by simpa only [Real.norm_eq_abs] using
168        dimensionRatio_relative_error r k (row n) (N n) (δ n) hr (hNpos n) (hclose n))
169    exact error_envelope_tendsto_zero r k N δ hN hδ
170  simpa only [sub_add_cancel, zero_add] using hzero.add_const 1

172/-- Exact scalar identity used to compare two independently rescaled
173dimension products. -/
174theorem ratio_rescaling (X Y c N M : ℝ) (a : ℕ)
175    (hY : Y ≠ 0) (hc : c ≠ 0) (hN : N ≠ 0) (hM : M ≠ 0) :
176    X / Y = ((X / (c * N ^ a)) / (Y / (c * M ^ a))) * (N / M) ^ a := by
177  rw [div_pow]

180/-- The concrete crossing-root products at sample sizes with ratio `γ`
181have ratio limit `γ^(-r*k)`. Both normalization limits are derived above;
182neither product limit is a hypothesis. -/
183theorem dimensionRatio_quotient_tendsto (r k : ℕ) (hr : 0 < r)
184    (rowIn rowOut : ℕ → Fin r → ℝ) (N M δ ε : ℕ → ℝ) (γ : ℝ)
185    (hNpos : ∀ n, 0 < N n) (hMpos : ∀ n, 0 < M n)
186    (hN : Tendsto N atTop atTop) (hγ : 0 < γ)
187    (hratio : Tendsto (fun n => M n / N n) atTop (𝓝 γ))
188    (hδ : Tendsto δ atTop (𝓝 0)) (hε : Tendsto ε atTop (𝓝 0))
189    (hcloseIn : ∀ n i, |rowIn n i / N n - 1 / (r : ℝ)| ≤ δ n)
190    (hcloseOut : ∀ n i, |rowOut n i / M n - 1 / (r : ℝ)| ≤ ε n)
191    (hout : ∀ n i, 0 ≤ rowOut n i) :
192    Tendsto (fun n => dimensionRatio r k (rowIn n) / dimensionRatio r k (rowOut n))
193      atTop (𝓝 ((γ⁻¹) ^ (r * k))) := by
194  have hM : Tendsto M atTop atTop := by
195    have h := Filter.Tendsto.pos_mul_atTop hγ hratio hN
196    apply h.congr'
197    exact Filter.Eventually.of_forall (fun n => div_mul_cancel₀ _ (hNpos n).ne')
198  have hin := normalized_dimensionRatio_tendsto_one r k hr rowIn N δ hNpos hN hδ hcloseIn
199  have houtlim := normalized_dimensionRatio_tendsto_one r k hr rowOut M ε hMpos hM hε hcloseOut
200  have hquot := hin.div houtlim one_ne_zero
201  have hscale : Tendsto (fun n => N n / M n) atTop (𝓝 γ⁻¹) := by
202    simpa only [inv_div] using hratio.inv₀ hγ.ne'
203  have h := hquot.mul (hscale.pow (r * k))
204  simp only [div_self one_ne_zero, one_mul] at h
205  apply h.congr'
206  apply Filter.Eventually.of_forall
207  intro n
208  exact (ratio_rescaling _ _ _ _ _ _ (dimensionRatio_pos r k (rowOut n) (hout n)).ne'
209    (leadingConstant_pos r k hr).ne' (hNpos n).ne' (hMpos n).ne').symm

211/-- The explicit support-sector factor tends to the manuscript's
212`γ^(-r(d-r)/2)`, with `d = r+k`. This is the analytic dimension-ratio input
213to the projector-state cloning theorem. -/
214theorem projector_sector_factor_tendsto (r k : ℕ) (hr : 0 < r)
215    (rowIn rowOut : ℕ → Fin r → ℝ) (N M δ ε : ℕ → ℝ) (γ : ℝ)
216    (hNpos : ∀ n, 0 < N n) (hMpos : ∀ n, 0 < M n)
217    (hN : Tendsto N atTop atTop) (hγ : 0 < γ)
218    (hratio : Tendsto (fun n => M n / N n) atTop (𝓝 γ))
219    (hδ : Tendsto δ atTop (𝓝 0)) (hε : Tendsto ε atTop (𝓝 0))
220    (hcloseIn : ∀ n i, |rowIn n i / N n - 1 / (r : ℝ)| ≤ δ n)
221    (hcloseOut : ∀ n i, |rowOut n i / M n - 1 / (r : ℝ)| ≤ ε n)
222    (hout : ∀ n i, 0 ≤ rowOut n i) :
223    Tendsto (fun n => Real.sqrt (dimensionRatio r k (rowIn n) / dimensionRatio r k (rowOut n)))
224      atTop (𝓝 (γ ^ (-((r * k : ℕ) : ℝ) / 2))) := by
225  have h := (dimensionRatio_quotient_tendsto r k hr rowIn rowOut N M δ ε γ
226    hNpos hMpos hN hγ hratio hδ hε hcloseIn hcloseOut hout).sqrt
227  have hvalue : Real.sqrt ((γ⁻¹) ^ (r * k)) = γ ^ (-((r * k : ℕ) : ℝ) / 2) := by
228    rw [Real.sqrt_eq_rpow, ← Real.rpow_natCast, ← Real.rpow_mul (inv_nonneg.mpr hγ.le),
229      ← Real.rpow_neg_eq_inv_rpow]
230    congr 1
231    ring
232  rw [hvalue] at h
233  exact h
