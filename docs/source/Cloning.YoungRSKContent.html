21/-- Strict columns force row `i` to contain only letters at least `i`. -/
22theorem row_array_triangular_of_interlacing {d : ℕ} (A : Fin d → RowContent d)
23    (hA : ∀ i j, i < j → RowInterlaces (A i) (A j)) :
24    ∀ i j, j < i → A i j = 0 := by
25  suffices h : ∀ n, ∀ i : Fin d, i.val = n → ∀ j, j < i → A i j = 0 by
26    intro i j hj
27    exact h i.val i rfl j hj
28  intro n
29  induction n with
30  | zero => intro i hi j hj; have : j.val < i.val := hj; omega
31  | succ n ih =>
32    intro i hi j hj
33    have hnd : n < d := by omega
34    let prev : Fin d := ⟨n, hnd⟩
35    have hp : prev < i := by change n < i.val; omega
36    have hs : rowStrictPrefix (A prev) j = 0 := by
37      apply Finset.sum_eq_zero
38      intro k hk
39      have hk : k < j := (Finset.mem_filter.mp hk).2
40      exact ih prev rfl k (by change k.val < n; have : j.val < i.val := hj; omega)
41    have hbound := (row_entry_le_prefix (A i) j).trans (hA prev i hp j)
42    rw [hs] at hbound
43    exact Nat.eq_zero_of_le_zero hbound

129/-- Any interlacing count-row tableau with bounded row sizes gives one of the
130literal finite arrays appearing in the Schur sum. -/
131def rowsSemistandardContent {d N : ℕ} {μ : Fin d → ℕ}
132    (T : List (RowContent d)) (hT : T.length = d)
133    (hμ : tableauShape T = List.ofFn μ) (hp : List.Pairwise RowInterlaces T)
134    (hN : ∀ i, μ i ≤ N) : SemistandardContent N μ := by
135  have hmass := rowsArray_shape hT hμ
136  have hb : ∀ i j, rowsArray T hT i j ≤ N := by
137    intro i j
138    apply le_trans (Finset.single_le_sum (fun _ _ => Nat.zero_le _) (Finset.mem_univ j))
139    exact (hmass i).le.trans (hN i)
140  refine ⟨rowsContent T hT hb, ?_, ?_, ?_⟩
141  · exact hmass
142  · exact row_array_triangular_of_interlacing _ (rowsArray_pairwise hT hp)
143  · exact fun i j k hij => rowsArray_pairwise hT hp i j hij k

159/-- Genuine RSK insertion on the exact bounded arrays of the Schur sum. -/
160def contentInsert {d N : ℕ} {μ : Fin d → ℕ} (A : SemistandardContent N μ)
161    (hN : ∀ i, μ i ≤ N) (x : Fin d) :
162    SemistandardContent (N + 1) (addBox μ (contentInsertCorner A x)) :=
163  rowsSemistandardContent (tableauInsert (contentRows A.val) x).1
164    (tableauInsert_fixed_length _ x (contentRows_length _))
165    (by rw [tableauInsert_shape, contentRows_shape]
166        exact shapeAdd_ofFn μ (contentInsertCorner A x))
167    (tableauInsert_pairwise _ (contentRows_pairwise A) x)
168    (fun i => by have := hN i; simp only [addBox]; split_ifs <;> omega)
