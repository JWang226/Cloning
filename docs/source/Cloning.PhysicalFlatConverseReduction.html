85theorem payoff_le_schurCopyMatrices (r k : ℕ) (hr : 0 < r) (n m : ℕ)

87    (U : unitary (Matrix (Fin (r+k)) (Fin (r+k)) ℂ)) :

90      Cloning.MatrixFidelity.fidelity

93  letI : Nonempty (Fin (r+k)) := ⟨⟨0,Nat.lt_of_lt_of_le hr (Nat.le_add_right r k)⟩⟩

95    ⟨Φ.toLinearMap (flatInflatedInput n r k),
