23/-- Actual physical local rotations converge to actual Weyl displacements on
24every retained occupation-frame vector, uniformly after a sufficiently large
25fixed compression cutoff. All coefficient and scale adapters are discharged. -/
26theorem exists_cutoff_eventually_physical_orbital_frame
27    (R : ℕ) (i : CutoffIndex d R) (ε : ℝ) (hε : 0 < ε) :
28    ∃ Q₀ : ℕ, R ≤ Q₀ ∧ ∀ Q ≥ Q₀, ∀ᶠ N in atTop,
29      ‖cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q
30          (tensorOperator (∑ j, mu N j)
31            (exp (sampleScale (∑ j, mu N j) • orbitalGenerator p (z N)))
32            (cutoffFrame (partitionHighestTensor (mu N) (hmu N)) (mu N) R i)) -
33        displacement (rootFockAmplitude z₀) (cutoffNumberFrame d R i)‖ < ε := by
34  let z' N := scaledRootParameter p (mu N) (sampleScale (∑ j, mu N j)) (z N)
35  have hz' : Tendsto z' atTop (𝓝 z₀) :=
36    scaledRootParameter_tendsto p hp mu hmu (fun N => ∑ j, mu N j) hfreq z z₀ hz
37  obtain ⟨Q₀,hRQ,hQ₀⟩ := exists_cutoff_eventually_frame_displacement
38    mu hmu δ hδ hgap z' z₀ hz' R i ε hε
39  refine ⟨Q₀,hRQ,?_⟩
40  intro Q hQ
41  filter_upwards [hQ₀ Q hQ,hgap,hδ.eventually (eventually_gt_atTop 0)] with N hN hg hpos
42  have hg' : ∀ a : PositiveRoot d, 0 < (mu N a.val.1 : ℝ)-mu N a.val.2 :=
43    fun a => hpos.trans_le (hg a)
44  simpa only [tensorOperator_local_orbital p (mu N) hg', z'] using hN
