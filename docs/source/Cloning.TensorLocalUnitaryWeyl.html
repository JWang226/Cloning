17theorem oscillatorTaylor_eq_numberTaylor (Q m : ℕ) (z : PositiveRoot d → ℂ)
18    (w : List (PositiveRoot d)) (hQ : loweringHeight w+m*d ≤ Q) :
19    oscillatorTaylor Q z w m = numberTaylor (rootFockAmplitude z) (wordFockOccupation w) m := by
20  unfold oscillatorTaylor numberTaylor

26      (loweringHeight w)).trans hQ)

39/-- Actual normalized PBW input, actual physical exponential, actual Fock

42theorem exists_cutoff_eventually_word_displacement
43    (w : List (PositiveRoot d)) (ε : ℝ) (hε : 0 < ε) :
44    ∃ Q₀ : ℕ, ∀ Q ≥ Q₀, ∀ᶠ N in atTop,
45      ‖cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q
46          (exp (rootGenerator (n := ∑ j, mu N j) (mu N) (z N))
47            (normalizedLoweringWord (partitionHighestTensor (mu N) (hmu N)) (mu N) w)) -
48        displacement (rootFockAmplitude z₀) (wordNumberVector w)‖ < ε := by
49  obtain ⟨M,hM⟩ := exists_order_eventually_exponential_near_polynomial
50    mu hmu δ hδ hgap z z₀ hz w (ε/2) (by linarith)
51  have ht := (tendsto_iff_norm_sub_tendsto_zero.mp
52    (numberTaylor_tendsto_displacement (rootFockAmplitude z₀) (wordFockOccupation w))).eventually_lt_const
53      (show 0 < ε/2 by linarith)
54  obtain ⟨m,hm,hmt⟩ := ((eventually_ge_atTop M).and ht).exists
55  refine ⟨loweringHeight w+m*d, ?_⟩
56  intro Q hQ
57  filter_upwards [hM m hm Q hQ] with N hN
58  rw [oscillatorTaylor_eq_numberTaylor Q m z₀ w hQ] at hN
59  have hh := norm_sub_le_norm_sub_add_norm_sub
60    (cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q
61      (exp (rootGenerator (n := ∑ j, mu N j) (mu N) (z N))
62        (normalizedLoweringWord (partitionHighestTensor (mu N) (hmu N)) (mu N) w)))
63    (numberTaylor (rootFockAmplitude z₀) (wordFockOccupation w) m)
64    (displacement (rootFockAmplitude z₀) (wordNumberVector w))

71theorem exists_cutoff_eventually_frame_displacement
72    (R : ℕ) (i : CutoffIndex d R) (ε : ℝ) (hε : 0 < ε) :
73    ∃ Q₀ : ℕ, R ≤ Q₀ ∧ ∀ Q ≥ Q₀, ∀ᶠ N in atTop,
74      ‖cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q
75          (exp (rootGenerator (n := ∑ j, mu N j) (mu N) (z N))
76            (cutoffFrame (partitionHighestTensor (mu N) (hmu N)) (mu N) R i)) -
77        displacement (rootFockAmplitude z₀) (cutoffNumberFrame d R i)‖ < ε := by
78  obtain ⟨Q₀,hQ₀⟩ := exists_cutoff_eventually_word_displacement
79    mu hmu δ hδ hgap z z₀ hz (cutoffWord d R i) (ε/2) (by linarith)
80  refine ⟨max R Q₀,le_max_left _ _,?_⟩
81  intro Q hQ
82  have hc := (partition_cutoffFrame_close mu hmu δ hδ hgap R i).eventually_lt_const
83    (show 0 < ε/2 by linarith)
84  filter_upwards [hQ₀ Q ((le_max_right _ _).trans hQ),hc,
85    cutoffEmbedding_eventually_contraction mu hmu δ hδ hgap Q] with N hN hcN hV
86  let V := cutoffEmbedding (partitionHighestTensor (mu N) (hmu N)) (mu N) Q
87  let G := rootGenerator (n := ∑ j, mu N j) (mu N) (z N)
88  let x := cutoffFrame (partitionHighestTensor (mu N) (hmu N)) (mu N) R i
89  let y := cutoffRawFrame (partitionHighestTensor (mu N) (hmu N)) (mu N) R i
90  have hclose : ‖V (exp G x)-V (exp G y)‖ < ε/2 := by
91    rw [← V.map_sub, ← (exp G).map_sub]
92    exact ((hV _).trans_eq (rootGenerator_exp_norm (mu N) (z N) (x-y))).trans_lt hcN
93  rw [← cutoffNumberFrame_eq_word] at hN
94  have hh := norm_sub_le_norm_sub_add_norm_sub (V (exp G x)) (V (exp G y))
95    (displacement (rootFockAmplitude z₀) (cutoffNumberFrame d R i))
96  change ‖V (exp G y)-displacement (rootFockAmplitude z₀) (cutoffNumberFrame d R i)‖ < ε/2 at hN
