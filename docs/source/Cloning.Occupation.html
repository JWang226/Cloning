6/-!
7# Exact Cartan occupation coefficient calculations
8 
9The scalar coefficients in `eq:cartan-coefficients` of `cloning.tex` have
10binomial squares. We prove their normalization, their thermal moments, and
11the corresponding identities for any finite collection of roots. These
12are exact finite identities, with no assumptions about limiting quantum
13channels. The identification of the representation-theoretic coefficients
14with this formula remains a separate obligation.
15-/

24/-- Square of the one-root Cartan splitting coefficient. -/
25def splitWeight (r a : ℕ) (s : ℝ) : ℝ :=
26  (r.choose a : ℝ) * s ^ a * (1 - s) ^ (r - a)
27 
28theorem splitWeight_nonneg (r a : ℕ) {s : ℝ} (hs0 : 0 ≤ s) (hs1 : s ≤ 1) :
29    0 ≤ splitWeight r a s := by
30  unfold splitWeight
31  exact mul_nonneg (mul_nonneg (Nat.cast_nonneg _) (pow_nonneg hs0 _))
32    (pow_nonneg (sub_nonneg.mpr hs1) _)

34/-- Exact finite thermal moment used in `eq:sector-output-fixed-matrix`. -/
35theorem splitWeight_moment (r : ℕ) (s q : ℝ) :
36    (∑ a ∈ range (r + 1), splitWeight r a s * q ^ a) =
37      (1 - s + s * q) ^ r := by
38  rw [show 1 - s + s * q = s * q + (1 - s) by ring, add_pow]
39  apply sum_congr rfl
40  intro a ha
41  simp only [splitWeight, mul_pow]
42  ring

44/-- The one-root squared splitting coefficients sum to one. -/
45theorem splitWeight_sum (r : ℕ) (s : ℝ) :
46    (∑ a ∈ range (r + 1), splitWeight r a s) = 1 := by
47  simpa using splitWeight_moment r s 1

49theorem splitWeight_moment_fin (r : ℕ) (s q : ℝ) :
50    (∑ a : Fin (r + 1), splitWeight r a s * q ^ (a : ℕ)) =
51      (1 - s + s * q) ^ r := by
52  rw [Fin.sum_univ_eq_sum_range (fun a => splitWeight r a s * q ^ a)]
53  exact splitWeight_moment r s q
54 
55theorem splitWeight_sum_fin (r : ℕ) (s : ℝ) :
56    (∑ a : Fin (r + 1), splitWeight r a s) = 1 := by
57  rw [Fin.sum_univ_eq_sum_range (fun a => splitWeight r a s)]
58  exact splitWeight_sum r s

60/-- The nonnegative Cartan coefficient for one root. -/
61def splitAmplitude (r a : ℕ) (s : ℝ) : ℝ := Real.sqrt (splitWeight r a s)

63theorem splitAmplitude_sq (r a : ℕ) {s : ℝ} (hs0 : 0 ≤ s) (hs1 : s ≤ 1) :
64    splitAmplitude r a s ^ 2 = splitWeight r a s :=
65  Real.sq_sqrt (splitWeight_nonneg r a hs0 hs1)

67theorem splitAmplitude_sq_sum (r : ℕ) {s : ℝ} (hs0 : 0 ≤ s) (hs1 : s ≤ 1) :
68    (∑ a : Fin (r + 1), splitAmplitude r a s ^ 2) = 1 := by
69  simp_rw [splitAmplitude_sq _ _ hs0 hs1]
70  exact splitWeight_sum_fin r s

76/-- Multi-root splitting coefficient, with `a i ≤ r i` encoded in its type. -/
77def cartanAmplitude (r : ι → ℕ) (s : ι → ℝ) (a : ∀ i, Fin (r i + 1)) : ℝ :=
78  ∏ i, splitAmplitude (r i) (a i) (s i)

100/-- The multi-root finite thermal moment appearing in the sector-output calculation. -/
101theorem cartanAmplitude_thermal_moment (r : ι → ℕ) (s q : ι → ℝ)
102    (hs0 : ∀ i, 0 ≤ s i) (hs1 : ∀ i, s i ≤ 1) :
103    (∑ a : ∀ i, Fin (r i + 1),

105      ∏ i, (1 - s i + s i * q i) ^ r i := by

110  intro i hi

115/-- Finite negative-binomial convolution. The specialization `a = D - 2`,
116`b = n`, `k = m - n` normalizes Werner's occupation law after grouping
117occupation vectors by their total excitation number. -/
118theorem choose_convolution (a b k : ℕ) :
119    (∑ t ∈ range (k + 1), (t + a).choose a * (k - t + b).choose b) =
120      (k + a + b + 1).choose (a + b + 1) := by
121  induction a generalizing k with
122  | zero =>
123    simp only [Nat.add_zero, Nat.choose_zero_right, one_mul, Nat.zero_add]
124    rw [← Finset.sum_flip]
125    calc
126      _ = ∑ t ∈ range (k + 1), (t + b).choose b := by
127        apply sum_congr rfl
128        intro t ht
129        rw [Nat.sub_sub_self (Nat.lt_succ_iff.mp (mem_range.mp ht))]
130      _ = _ := Nat.sum_range_add_choose k b
131  | succ a ih =>
132    induction k with
133    | zero => simp
134    | succ k hk =>
135      have hsplit :
136          (∑ t ∈ range (k + 1 + 1), (t + (a + 1)).choose (a + 1) *
137            (k + 1 - t + b).choose b) =
138          (∑ t ∈ range (k + 1 + 1), (t + a).choose a *
139            (k + 1 - t + b).choose b) +
140          (∑ t ∈ range (k + 1 + 1), (t + a).choose (a + 1) *
141            (k + 1 - t + b).choose b) := by
142        simp_rw [← Nat.add_assoc, Nat.choose_succ_succ, add_mul, sum_add_distrib]
143      rw [hsplit, ih]
144      have hshift :
145          (∑ t ∈ range (k + 1 + 1), (t + a).choose (a + 1) *
146            (k + 1 - t + b).choose b) =
147          (∑ t ∈ range (k + 1), (t + (a + 1)).choose (a + 1) *
148            (k - t + b).choose b) := by
149        rw [sum_range_succ']
150        simp only [Nat.zero_add, Nat.choose_succ_self, zero_mul, add_zero]
151        apply sum_congr rfl
152        intro t ht
153        simp [Nat.add_comm, Nat.add_left_comm]
154      rw [hshift, hk]
155      simpa [Nat.add_assoc, Nat.add_comm, Nat.add_left_comm] using
156        (Nat.choose_succ_succ (k + a + b + 2) (a + b + 1)).symm

158/-- Exact finite normalization numerator for Werner's occupation law,
159including the number of occupation vectors with a given total `t`. -/
160theorem werner_occupation_numerator (n m D : ℕ) (hnm : n ≤ m) (hD : 2 ≤ D) :
161    (∑ t ∈ range (m - n + 1),
162      (t + D - 2).choose (D - 2) * (m - t).choose n) =
163      (m + D - 1).choose (n + D - 1) := by
164  have htop : m - n + (D - 2) + n + 1 = m + D - 1 := by omega
165  have hbottom : D - 2 + n + 1 = n + D - 1 := by omega
166  rw [← htop, ← hbottom, ← choose_convolution]
167  apply sum_congr rfl
168  intro t ht
169  have ht' : t ≤ m - n := Nat.lt_succ_iff.mp (mem_range.mp ht)
170  congr 2 <;> omega

172/-- The eigenvalue attached to any occupation vector of total size `t`
173in `eq:occupation-law`. -/
174def wernerWeight (n m D t : ℕ) : ℝ :=
175  ((m - t).choose n : ℝ) / ((m + D - 1).choose (n + D - 1) : ℝ)
176 
177/-- Grouping Werner eigenvalues by total excitation gives total probability
178one, for every `m ≥ n` and ambient dimension `D ≥ 2`. -/
179theorem werner_occupation_normalization (n m D : ℕ) (hnm : n ≤ m) (hD : 2 ≤ D) :
180    (∑ t ∈ range (m - n + 1),
181      ((t + D - 2).choose (D - 2) : ℝ) * wernerWeight n m D t) = 1 := by
182  have hpos : 0 < (m + D - 1).choose (n + D - 1) :=
183    Nat.choose_pos (by omega)
184  have hne : ((m + D - 1).choose (n + D - 1) : ℝ) ≠ 0 :=
185    ne_of_gt (Nat.cast_pos.mpr hpos)
186  simp only [wernerWeight, div_eq_mul_inv, ← mul_assoc, ← sum_mul]
187  have hsum :
188      (∑ t ∈ range (m - n + 1),
189        ((t + D - 2).choose (D - 2) : ℝ) * ((m - t).choose n : ℝ)) =
190        ((m + D - 1).choose (n + D - 1) : ℝ) := by
191    exact_mod_cast werner_occupation_numerator n m D hnm hD
192  rw [hsum, mul_inv_cancel₀ hne]
