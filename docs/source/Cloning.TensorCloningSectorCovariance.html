47theorem transitionFidelity_unitary (n m d : ℕ) (p : Fin d → ℝ) (hp : ∀ a, 0 < p a)
48    (U : Matrix (Fin d) (Fin d) ℂ) (hU : Uᴴ * U = 1)
49    (i : SchurCopy n d) (j : SchurCopy m d)
50    (hc : PartitionCompatible ((recursivePhysicalDecomposition n d).get i).weight
51      ((recursivePhysicalDecomposition m d).get j).weight) :
52    transitionFidelity n m d p hp U i j = transitionFidelity n m d p hp 1 i j := by
53  let A := (recursivePhysicalDecomposition n d).get i
54  let B := (recursivePhysicalDecomposition m d).get j
55  let C := partitionTransitionChannel A.weight B.weight A.weight_antitone B.weight_antitone
56  let V := QuantumChannel.ofIsometry (copySectorUnitary B U hU).toLinearIsometry
57  have he : (copySectorState A U p hp).map C.toPositiveTracePreservingMap =
58      ((copySectorState A 1 p hp).map C.toPositiveTracePreservingMap).map V.toPositiveTracePreservingMap := by
59    rw [copySectorState_unitary A U hU p hp]
60    apply Subtype.ext
61    exact partitionTransitionChannel_covariant A.weight B.weight A.weight_antitone B.weight_antitone hc U hU
62      (copySectorState A 1 p hp).1
63  change ((copySectorState A U p hp).map C.toPositiveTracePreservingMap).rootFidelity
64    (copySectorState B U p hp) =
65    ((copySectorState A 1 p hp).map C.toPositiveTracePreservingMap).rootFidelity (copySectorState B 1 p hp)
66  rw [he, copySectorState_unitary B U hU p hp]
67  exact rootFidelity_map_isometryEquiv (copySectorUnitary B U hU) _ _
