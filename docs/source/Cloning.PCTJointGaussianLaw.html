35def orbitalVariance (p : A → ℝ) (e : Fin d ≃ OrbitalPair A) (v : ℝ) (k : Fin d) : ℝ :=

107    (u : OrthonormalBasis (Fin (s + 1)) ℂ (Register (A × A)))
108    (p : A → ℝ) (hp : ∀ a, 0 ≤ p a)

113    (∫ z, Complex.exp (Complex.I * ((∑ i, t i * tangentClassical u p z i : ℝ) : ℂ) +

162/-- The joint pushforward of the physical tangent coordinates is exactly the
163product of its classical marginal and independent circular Gaussian orbital
164coordinates. No independence or law-identification premise occurs. -/
165theorem jointTangent_map_eq_prod
166    (u : OrthonormalBasis (Fin (s + 1)) ℂ (Register (A × A)))
167    (p : A → ℝ) (hp : ∀ a, 0 < p a)
168    (hu : u 0 = coefficientVector (schmidtCoefficients p))
169    (hgap : ∀ i j, i < j → 0 < p i - p j)
170    (e : Fin d ≃ OrbitalPair A) {v : ℝ} (hv : 0 < v) :
171    (gaussianProductMeasure (fun _ : Fin s => v)).map (jointTangent u p e) =
172      (classicalTangentLaw u p v).prod (gaussianProductMeasure (orbitalVariance p e v)) := by
173  letI := gaussianProductMeasure_probability (fun _ : Fin s => hv)
174  letI := gaussianProductMeasure_probability (orbitalVariance_pos p hp hgap e hv)
175  letI := classicalTangentLaw_probability u p hv
176  apply charFunDual_eq_prod_iff.mp
177  intro L
178  rw [classical_charFunDual u p (fun a => (hp a).le) hu hv,
179    orbital_charFunDual (orbitalVariance_pos p hp hgap e hv), charFunDual_apply,
180    integral_map (continuous_jointTangent u p e).aemeasurable (by fun_prop)]
181  simp only [jointTangent, joint_dual_phase]
182  exact joint_characteristic_reindexed u p (fun a => (hp a).le) hu hgap e hv
183    (fun i => (L.comp (.inl ℝ (A → ℝ) (Fin d → ℂ))) (Pi.single i 1))
184    (orbitalDualTest (L.comp (.inr ℝ (A → ℝ) (Fin d → ℂ))))

188theorem integral_jointTangent {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]
189    (u : OrthonormalBasis (Fin (s + 1)) ℂ (Register (A × A)))
190    (p : A → ℝ) (hp : ∀ a, 0 < p a)
191    (hu : u 0 = coefficientVector (schmidtCoefficients p))
192    (hgap : ∀ i j, i < j → 0 < p i - p j)
193    (e : Fin d ≃ OrbitalPair A) {v : ℝ} (hv : 0 < v)
194    (f : (A → ℝ) × (Fin d → ℂ) → E)
195    (hf : Integrable f ((classicalTangentLaw u p v).prod
196      (gaussianProductMeasure (orbitalVariance p e v)))) :
197    (∫ z, f (jointTangent u p e z) ∂gaussianProductMeasure (fun _ : Fin s => v)) =
198      ∫ x, ∫ a, f (x,a) ∂gaussianProductMeasure (orbitalVariance p e v)
199        ∂classicalTangentLaw u p v := by
200  letI := gaussianProductMeasure_probability (orbitalVariance_pos p hp hgap e hv)
201  letI := classicalTangentLaw_probability u p hv
202  have hm := jointTangent_map_eq_prod u p hp hu hgap e hv
203  rw [← integral_prod f hf, ← hm]
204  exact (integral_map (continuous_jointTangent u p e).aemeasurable
205    (by rw [hm]; exact hf.aestronglyMeasurable)).symm
