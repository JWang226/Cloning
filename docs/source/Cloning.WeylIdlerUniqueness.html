19lemma norm_tracePairing_displacement_le (T : TraceClass (Fock d)) (a : Fin d → ℂ) :
20    ‖tracePairing T (displacement a)‖ ≤ ‖T‖ := by
21  have hT : ‖tracePairing T‖ ≤ ‖T‖ :=
22    ((tracePairingCLM (H := Fock d)).le_opNorm T).trans (by
23      simpa using mul_le_mul_of_nonneg_right
24        (norm_tracePairingCLM_le (H := Fock d)) (norm_nonneg T))

26    _ ≤ ‖tracePairing T‖ * ‖displacement a‖ := (tracePairing T).le_opNorm _
27    _ ≤ ‖T‖ * 1 := mul_le_mul hT (displacement_opNorm_le a) (norm_nonneg _) (norm_nonneg _)
28    _ = _ := mul_one _

130/-- Vanishing Weyl characteristic function forces an actual trace-class
131operator to vanish. -/
132theorem traceClass_eq_zero_of_characteristic_zero (T : TraceClass (Fock d))
133    (hT : ∀ a : Fin d → ℂ, tracePairing T (displacement a) = 0) : T = 0 := by
134  have hcoeff (a b : Fin d → ℂ) : ⟪coherentVector b, T.1 (coherentVector a)⟫_ℂ = 0 := by
135    let S := sandwichCLM (displacement (-b)) (displacement a) T
136    have hS (z : Fin d → ℂ) : tracePairing S (displacement z) = 0 := by
137      rw [show S = sandwichCLM (displacement (-b)) (displacement a) T from rfl,
138        tracePairing_sandwich]
139      simp only [ContinuousLinearMap.mul_def, displacement_comp,
140        ContinuousLinearMap.smul_comp, smul_smul, map_smul, hT, smul_zero]
141    have h := integral_weighted_characteristic S
142    simp only [hS, mul_zero, integral_zero] at h
143    rw [show S = sandwichCLM (displacement (-b)) (displacement a) T from rfl,
144      tracePairing_sandwich, displaced_vacuum_dyad, tracePairing_rankOne] at h
145    exact h.symm

147  apply continuousLinearMap_ext_coherent
148  intro a

152/-- The actual Weyl characteristic function is injective on the entire
153complex trace class, so it distinguishes arbitrary mixed states. -/
154theorem characteristic_injective :
155    Function.Injective (fun T : TraceClass (Fock d) => fun a => tracePairing T (displacement a)) := by
156  intro S T h
157  apply sub_eq_zero.mp
158  apply traceClass_eq_zero_of_characteristic_zero
159  intro a
160  change ((tracePairingCLM (H := Fock d)).flip (displacement a)) (S - T) = 0
161  rw [map_sub]
162  exact sub_eq_zero.mpr (congrFun h a)
