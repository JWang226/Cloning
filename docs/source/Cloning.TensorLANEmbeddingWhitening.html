18def rootWhitening (p : Fin (d+1) → ℝ) (hp : ∀ a, 0 < p a)
19    (b : OrthonormalBasis (Fin (d+1)) ℝ (EuclideanSpace ℝ (Fin (d+1))))
20    (hb : b 0 = sqrtSpectrum p) : rootSpace d ≃L[ℝ] (Fin d → ℝ) :=
21  LinearEquiv.toContinuousLinearEquiv
22  { toFun := fun x => whiten p b x.1.ofLp
23    invFun := fun y => ⟨WithLp.toLp 2 (unwhiten p b y), unwhiten_sum_zero p b hb y⟩
24    left_inv := fun x => by
25      apply Subtype.ext
26      exact congrArg (WithLp.toLp 2) (unwhiten_whiten p hp b hb x.1.ofLp x.2)
27    right_inv := whiten_unwhiten p hp b
28    map_add' := by
29      intro x y
30      funext i
31      simp [whiten, mul_add, add_div, Finset.sum_add_distrib]
32    map_smul' := by
33      intro c x
34      funext i
35      simp [whiten, Finset.mul_sum, mul_div_assoc, mul_assoc, mul_left_comm] }

44theorem rootWhitening_norm_sq (p : Fin (d+1) → ℝ) (hp : ∀ a, 0 < p a)

67/-- Push a density to coordinate Lebesgue measure, including its actual
68positive Jacobian. -/
69def whiteningDensity (f : rootSpace d → ℝ) (y : Fin d → ℝ) : ℝ :=
70  (whiteningJacobian W : ℝ) * f (W.symm y)

72theorem whiteningDensity_integral (f : rootSpace d → ℝ) :
73    (∫ y, whiteningDensity W f y) = ∫ x, f x := by
74  have h := integral_map_equiv (μ := (volume : Measure (rootSpace d)))
75    W.toHomeomorph.toMeasurableEquiv (fun y => f (W.symm y))
76  change (∫ y, f (W.symm y) ∂((volume : Measure (rootSpace d)).map W)) =
77    (∫ x, f (W.symm (W x))) at h
78  rw [rootWhitening_map_volume, integral_smul_nnreal_measure] at h
79  simpa only [whiteningDensity, integral_const_mul, NNReal.smul_def,
80    ContinuousLinearEquiv.symm_apply_apply] using h
