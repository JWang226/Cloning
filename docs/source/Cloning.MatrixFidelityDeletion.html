25/-- Root fidelity is subadditive in a positive matrix argument. -/
26theorem fidelity_add_le {G B : Matrix n n ℂ} (hG : G.PosSemidef)
27    (hB : B.PosSemidef) (T : Matrix n n ℂ) :
28    fidelity (G + B) T ≤ fidelity G T + fidelity B T := by
29  unfold fidelity
30  rw [Matrix.mul_add, Matrix.add_mul]
31  exact trace_sqrt_add_le (sandwich_posSemidef hG T) (sandwich_posSemidef hB T)

33/-- Removing a positive summand changes root fidelity by at most the
34square root of the product of its mass and the target mass. -/
35theorem fidelity_deletion_bound {G B T : Matrix n n ℂ}
36    (hG : G.PosSemidef) (hB : B.PosSemidef) (hT : T.PosSemidef) :
37    |fidelity (G + B) T - fidelity G T| ≤
38      Real.sqrt ((Matrix.trace B).re * (Matrix.trace T).re) := by
39  have horder : G ≤ G + B := by
40    change ((G + B) - G).PosSemidef
41    simpa using hB
42  have hmono := fidelity_mono_left horder T
43  rw [abs_of_nonneg (sub_nonneg.mpr hmono)]
44  calc
45    fidelity (G + B) T - fidelity G T ≤ fidelity B T := by
46      linarith [fidelity_add_le hG hB T]
47    _ ≤ Real.sqrt ((Matrix.trace B).re * (Matrix.trace T).re) :=
48      fidelity_le_sqrt_trace_mul_trace hB hT

50/-- For a subnormalized target, only the removed mass enters the estimate. -/
51theorem fidelity_deletion_bound_trace_le_one {G B T : Matrix n n ℂ}
52    (hG : G.PosSemidef) (hB : B.PosSemidef) (hT : T.PosSemidef)
53    (htrace : (Matrix.trace T).re ≤ 1) :
54    |fidelity (G + B) T - fidelity G T| ≤ Real.sqrt (Matrix.trace B).re := by
55  apply (fidelity_deletion_bound hG hB hT).trans
56  apply Real.sqrt_le_sqrt
57  simpa using mul_le_mul_of_nonneg_left htrace (trace_re_nonneg hB)

59/-- Deletion stated as positive domination of the retained matrix. -/
60theorem fidelity_retained_bound {X G T : Matrix n n ℂ}
61    (hG : G.PosSemidef) (hdiff : (X - G).PosSemidef) (hT : T.PosSemidef)
62    (htrace : (Matrix.trace T).re ≤ 1) :
63    |fidelity X T - fidelity G T| ≤
64      Real.sqrt ((Matrix.trace X).re - (Matrix.trace G).re) := by
65  have h := fidelity_deletion_bound_trace_le_one hG hdiff hT htrace
66  have hsum : G + (X - G) = X := by rw [add_comm G, sub_add_cancel]
67  simpa only [hsum, Matrix.trace_sub, Complex.sub_re] using h

76/-- The exact lifted deletion estimate before canonical sector normalization. -/
77theorem lifted_deletion_raw_bound
78    (X G ρ : ∀ i, Matrix (d i) (d i) ℂ)
79    (τ : ∀ i, Matrix (s i) (s i) ℂ) (p : ι → ℝ)
80    (hX : ∀ i, (X i).PosSemidef) (hG : ∀ i, (G i).PosSemidef)
81    (hρ : ∀ i, (ρ i).PosSemidef) (hτ : ∀ i, (τ i).PosSemidef)
82    (hp : ∀ i, 0 ≤ p i) (hdiff : ∀ i, (X i - G i).PosSemidef)
83    (hρtrace : ∀ i, (Matrix.trace (ρ i)).re = 1)
84    (hτtrace : ∀ i, (Matrix.trace (τ i)).re = 1)
85    (hpsum : ∑ i, p i = 1) :
86    |fidelity (Matrix.blockDiagonal' (fun i ↦ X i ⊗ₖ τ i))
87        (Matrix.blockDiagonal' (fun i ↦ (p i • ρ i) ⊗ₖ τ i)) -
88      fidelity (Matrix.blockDiagonal' (fun i ↦ G i ⊗ₖ τ i))
89        (Matrix.blockDiagonal' (fun i ↦ (p i • ρ i) ⊗ₖ τ i))| ≤
90      Real.sqrt (∑ i, ((Matrix.trace (X i)).re - (Matrix.trace (G i)).re)) := by
91  have hmass i : 0 ≤ (Matrix.trace (X i)).re - (Matrix.trace (G i)).re := by
92    simpa only [Matrix.trace_sub, Complex.sub_re] using trace_re_nonneg (hdiff i)
93  have hlocal i : |fidelity (X i) (ρ i) - fidelity (G i) (ρ i)| ≤
94      Real.sqrt ((Matrix.trace (X i)).re - (Matrix.trace (G i)).re) :=
95    fidelity_retained_bound (hG i) (hdiff i) (hρ i) (hρtrace i).le
96  rw [fidelity_lifted_blocks X ρ τ p hX hρ hτ hp hτtrace,
97    fidelity_lifted_blocks G ρ τ p hG hρ hτ hp hτtrace,
98    ← Finset.sum_sub_distrib]
99  simp only [← mul_sub]
100  calc
101    |∑ i, Real.sqrt (p i) * (fidelity (X i) (ρ i) - fidelity (G i) (ρ i))| ≤
102        ∑ i, |Real.sqrt (p i) * (fidelity (X i) (ρ i) - fidelity (G i) (ρ i))| :=
103      Finset.abs_sum_le_sum_abs _ _
104    _ ≤ ∑ i, Real.sqrt (p i) *
105        Real.sqrt ((Matrix.trace (X i)).re - (Matrix.trace (G i)).re) := by
106      apply Finset.sum_le_sum
107      intro i _
108      rw [abs_mul, abs_of_nonneg (Real.sqrt_nonneg _)]
109      exact mul_le_mul_of_nonneg_left (hlocal i) (Real.sqrt_nonneg _)
110    _ ≤ Real.sqrt (∑ i, p i) *
111        Real.sqrt (∑ i, ((Matrix.trace (X i)).re - (Matrix.trace (G i)).re)) :=
112      Real.sum_sqrt_mul_sqrt_le Finset.univ hp hmass
113    _ = Real.sqrt (∑ i, ((Matrix.trace (X i)).re - (Matrix.trace (G i)).re)) := by
114      rw [hpsum, Real.sqrt_one, one_mul]

116/-- The manuscript's retained-block quantum estimate with canonical block
117weights and sector states. Every fidelity here is the concrete matrix root
118fidelity; no quantum-continuity assumption remains in the statement. -/
119theorem lifted_deletion_bound
120    (X G ρ : ∀ i, Matrix (d i) (d i) ℂ)
121    (τ : ∀ i, Matrix (s i) (s i) ℂ) (p : ι → ℝ)
122    (hX : ∀ i, (X i).PosSemidef) (hG : ∀ i, (G i).PosSemidef)
123    (hρ : ∀ i, (ρ i).PosSemidef) (hτ : ∀ i, (τ i).PosSemidef)
124    (hp : ∀ i, 0 ≤ p i) (hdiff : ∀ i, (X i - G i).PosSemidef)
125    (hρtrace : ∀ i, (Matrix.trace (ρ i)).re = 1)
126    (hτtrace : ∀ i, (Matrix.trace (τ i)).re = 1)
127    (hpsum : ∑ i, p i = 1) :
128    |fidelity (Matrix.blockDiagonal' (fun i ↦ X i ⊗ₖ τ i))
129        (Matrix.blockDiagonal' (fun i ↦ (p i • ρ i) ⊗ₖ τ i)) -
130      ∑ i, Real.sqrt (Matrix.trace (G i)).re * Real.sqrt (p i) *
131        fidelity (normalizedBlock (G i)) (ρ i)| ≤
132      Real.sqrt (∑ i, ((Matrix.trace (X i)).re - (Matrix.trace (G i)).re)) := by
133  rw [← fidelity_lifted_blocks_normalized G ρ τ p hG hρ hτ hp hτtrace]
