74lemma translationError_le_two (b : PhaseSpace k s) : p.translationError b ≤ 2 := by
75  have hi : Integrable (fun z => p.value (b + z)) :=
76    (measurePreserving_add_left volume b).integrable_comp_of_integrable p.integrable
77  have h := integral_mono (hi.sub p.integrable).abs (hi.add p.integrable) (fun z => by
78    exact (abs_sub (p.value (b + z)) (p.value z)).trans_eq
79      (by rw [abs_of_nonneg (p.nonneg _), abs_of_nonneg (p.nonneg _)]; rfl))
80  change (∫ z, |p.value (b + z) - p.value z|) ≤ ∫ z, p.value (b + z) + p.value z at h
81  rw [integral_add hi p.integrable, integral_add_left_eq_self, p.integral_one] at h
82  norm_num at h
83  exact h

85lemma continuous_translationError : Continuous p.translationError := by
86  let f : Lp ℝ 1 (volume : Measure (PhaseSpace k s)) := p.integrable.toL1 p.value
87  let shifts : C(PhaseSpace k s, C(PhaseSpace k s, PhaseSpace k s)) :=
88    (ContinuousMap.mk (fun z : PhaseSpace k s × PhaseSpace k s => z.1 + z.2) continuous_add).curry
89  have hm (b : PhaseSpace k s) : MeasurePreserving (shifts b) volume volume :=
90    measurePreserving_add_left volume b
91  have hc : Continuous (fun b => Lp.compMeasurePreserving (shifts b) (hm b) f) :=
92    continuous_const.compMeasurePreservingLp shifts.continuous hm (by norm_num)
93  have he (b : PhaseSpace k s) : ‖Lp.compMeasurePreserving (shifts b) (hm b) f - f‖ =
94      p.translationError b := by
95    have hi := (hm b).integrable_comp_of_integrable p.integrable
96    change ‖hi.toL1 (fun z => p.value (b + z)) - p.integrable.toL1 p.value‖ = _
97    rw [← Integrable.toL1_sub, L1.norm_of_fun_eq_integral_norm]
98    simp only [Pi.sub_apply, Real.norm_eq_abs, translationError]
99  exact (hc.sub continuous_const).norm.congr (fun b => he b)
