4/-!
5# Actual Hilbert-space number-amplitude vectors
6 
7Square roots of normalized nonnegative number laws define unit vectors in the
8complex Hilbert space `ℓ²`. The concrete binomial and Poisson laws therefore
9give unit number-amplitude vectors, and the former converge in Hilbert norm
10to the latter. This uses the physical choice of excitation direction that
11makes number amplitudes nonnegative; an occupation/Fock-space identification
12and the associated quantum channel remain separate constructions.
13-/

24/-- The unit Hilbert vector with square-root probability amplitudes. -/
25def amplitudeVector (p : ι → ℝ) (hp : ∀ i, 0 ≤ p i) (hs : HasSum p 1) :
26    lp (fun _ : ι => ℂ) 2 := by
27  refine ⟨fun i => (Real.sqrt (p i) : ℂ), memℓp_gen ?_⟩
28  simpa only [ENNReal.toReal_ofNat, Real.rpow_two,
29    Complex.norm_of_nonneg (Real.sqrt_nonneg _), Real.sq_sqrt (hp _)] using hs.summable
30 
31@[simp] theorem amplitudeVector_apply (p : ι → ℝ) (hp : ∀ i, 0 ≤ p i)
32    (hs : HasSum p 1) (i : ι) : amplitudeVector p hp hs i = (Real.sqrt (p i) : ℂ) := rfl

34theorem amplitudeVector_norm_sq (p : ι → ℝ) (hp : ∀ i, 0 ≤ p i)
35    (hs : HasSum p 1) : ‖amplitudeVector p hp hs‖ ^ 2 = 1 := by
36  calc
37    _ = ∑' i, ‖amplitudeVector p hp hs i‖ ^ (2 : ℕ) := by
38      simpa only [ENNReal.toReal_ofNat, Real.rpow_two] using
39        (lp.norm_rpow_eq_tsum (p := 2) (by norm_num) (amplitudeVector p hp hs))
40    _ = ∑' i, p i := by
41      apply tsum_congr
42      intro i
43      rw [amplitudeVector_apply, Complex.norm_of_nonneg (Real.sqrt_nonneg _),
44        Real.sq_sqrt (hp i)]
45    _ = 1 := hs.tsum_eq
46 
47theorem amplitudeVector_norm (p : ι → ℝ) (hp : ∀ i, 0 ≤ p i)
48    (hs : HasSum p 1) : ‖amplitudeVector p hp hs‖ = 1 := by
49  have h := amplitudeVector_norm_sq p hp hs
50  nlinarith [norm_nonneg (amplitudeVector p hp hs)]

52/-- Hilbert distance is exactly the square-root coefficient error. -/
53theorem amplitudeVector_sub_norm (p q : ι → ℝ) (hp : ∀ i, 0 ≤ p i)
54    (hq : ∀ i, 0 ≤ q i) (hs : HasSum p 1) (ht : HasSum q 1) :
55    ‖amplitudeVector p hp hs - amplitudeVector q hq ht‖ =
56      Real.sqrt (∑' i, (Real.sqrt (p i) - Real.sqrt (q i)) ^ 2) := by
57  have hsq : ‖amplitudeVector p hp hs - amplitudeVector q hq ht‖ ^ 2 =
58      ∑' i, (Real.sqrt (p i) - Real.sqrt (q i)) ^ 2 := by
59    calc
60      _ = ∑' i, ‖(amplitudeVector p hp hs - amplitudeVector q hq ht) i‖ ^ (2 : ℕ) := by
61        simpa only [ENNReal.toReal_ofNat, Real.rpow_two] using
62          (lp.norm_rpow_eq_tsum (p := 2) (by norm_num)
63            (amplitudeVector p hp hs - amplitudeVector q hq ht))
64      _ = _ := by
65        apply tsum_congr
66        intro i
67        simp only [lp.coeFn_sub, Pi.sub_apply, amplitudeVector_apply,
68          ← Complex.ofReal_sub, Complex.norm_real, Real.norm_eq_abs, sq_abs]
69  rw [← hsq, Real.sqrt_sq (norm_nonneg _)]

71/-- A normalized pointwise probability limit gives an actual complex Hilbert
72vector limit, without a finite-support assumption. -/
73theorem amplitudeVector_tendsto (p : ℕ → ι → ℝ) (q : ι → ℝ)
74    (hp : ∀ n i, 0 ≤ p n i) (hq : ∀ i, 0 ≤ q i)
75    (hs : ∀ n, HasSum (p n) 1) (ht : HasSum q 1)
76    (hpoint : ∀ i, Tendsto (fun n => p n i) atTop (𝓝 (q i))) :
77    Tendsto (fun n => amplitudeVector (p n) (hp n) (hs n)) atTop
78      (𝓝 (amplitudeVector q hq ht)) := by
79  rw [tendsto_iff_dist_tendsto_zero]
80  simp_rw [dist_eq_norm, amplitudeVector_sub_norm]
81  simpa using (Real.continuous_sqrt.tendsto 0).comp
82    (CountableScheffe.sqrt_amplitudes_tendsto_of_pointwise p q hp hq hs ht hpoint)

86/-- The padded binomial number-amplitude vector. -/
87def productVector (t : ℝ) (ht : 0 ≤ t) (L : ℕ) : lp (fun _ : ℕ => ℂ) 2 :=
88  amplitudeVector (binomialWeight t L) (binomialWeight_nonneg ht L)
89    (binomialWeight_hasSum ht L)

91/-- The coherent Poisson number-amplitude vector. -/
92def coherentVector (t : ℝ) (ht : 0 ≤ t) : lp (fun _ : ℕ => ℂ) 2 :=
93  amplitudeVector (poissonWeight t) (poissonWeight_nonneg ht) (poissonWeight_hasSum ht)

95/-- The Poisson amplitude has the usual coherent-vector exponential and
96factorial coefficients. -/
97theorem coherentVector_apply (t : ℝ) (ht : 0 ≤ t) (j : ℕ) :
98    coherentVector t ht j =
99      ((Real.exp (-t / 2) * Real.sqrt t ^ j / Real.sqrt (j.factorial : ℝ) : ℝ) : ℂ) := by
100  change (Real.sqrt (poissonWeight t j) : ℂ) = _
101  congr 1
102  unfold poissonWeight
103  rw [Real.sqrt_div (by positivity), Real.sqrt_mul (Real.exp_nonneg _),
104    ← Real.exp_half, Thermal.sqrt_nat_pow ht]

112/-- Concrete coherent-product convergence in the infinite-dimensional number
113coefficient Hilbert space, including zero amplitude. -/
114theorem productVector_tendsto_coherentVector (t : ℝ) (ht : 0 ≤ t) :
115    Tendsto (productVector t ht) atTop (𝓝 (coherentVector t ht)) :=
