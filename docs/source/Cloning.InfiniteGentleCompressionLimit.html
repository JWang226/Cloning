36/-- Finite-corner convergence and exact total mass exclude escape of positive
37mass to infinity. The proof quantitatively controls both off-diagonal tails. -/
38theorem tendsto_of_positive_unit_corners (b : HilbertBasis ι ℂ H)
39    (A : α → TraceClass H) (B : TraceClass H)
40    (hA : ∀ n, 0 ≤ (A n).1) (hB : 0 ≤ B.1)
41    (hAn : ∀ n, ‖A n‖ = 1) (hBn : ‖B‖ = 1)
42    (hcorner : ∀ s : Finset ι,
43      Tendsto (fun n ↦ sandwichCLM (basisProjection b s) (basisProjection b s) (A n)) l
44        (𝓝 (sandwichCLM (basisProjection b s) (basisProjection b s) B))) :
45    Tendsto A l (𝓝 B) := by
46  let C (s : Finset ι) (T : TraceClass H) := sandwichCLM (basisProjection b s) (basisProjection b s) T
47  have hBC : Tendsto (fun s : Finset ι ↦ C s B) atTop (𝓝 B) :=
48    sandwich_tendsto_of_strong_projections _ (basisProjection_isStarProjection b)
49      (basisProjection_tendsto b) B hB
50  have hmass : Tendsto (fun s : Finset ι ↦ 1 - ‖C s B‖) atTop (𝓝 0) := by
51    simpa only [hBn, sub_self] using hBC.norm.const_sub 1
52  have hbound : Tendsto (fun s : Finset ι ↦
53      2 * Real.sqrt (1-‖C s B‖) + ‖C s B-B‖) atTop (𝓝 0) := by
54    simpa only [sub_self, norm_zero, Real.sqrt_zero, mul_zero, zero_add] using
55      ((Real.continuous_sqrt.continuousAt.tendsto.comp hmass).const_mul 2).add
56        (hBC.sub_const B).norm
57  apply tendsto_iff_norm_sub_tendsto_zero.mpr
58  apply Metric.tendsto_nhds.mpr
59  intro ε hε
60  obtain ⟨s, hs⟩ := (hbound.eventually (gt_mem_nhds hε)).exists
61  have hc : Tendsto (fun n ↦ C s (A n)) l (𝓝 (C s B)) := hcorner s
62  have hn : Tendsto (fun n ↦ 2 * Real.sqrt (1-‖C s (A n)‖) +
63      ‖C s (A n)-C s B‖ + ‖C s B-B‖) l
64      (𝓝 (2 * Real.sqrt (1-‖C s B‖) + ‖C s B-B‖)) := by
65    have hroot := (Real.continuous_sqrt.continuousAt.tendsto.comp (hc.norm.const_sub 1)).const_mul 2
66    simpa only [sub_self, norm_zero, add_zero] using
67      (hroot.add (hc.sub_const (C s B)).norm).add_const ‖C s B-B‖
68  filter_upwards [hn.eventually (gt_mem_nhds hs)] with n hn
69  rw [Real.dist_eq, sub_zero, abs_of_nonneg (norm_nonneg _)]
70  apply lt_of_le_of_lt _ hn
71  have hgentle := gentle_compression_unit (basisProjection_isStarProjection b s) (A n) (hA n) (hAn n)
72  have htri := norm_sub_le_norm_sub_add_norm_sub (A n) (C s (A n)) B
73  have htri' := norm_sub_le_norm_sub_add_norm_sub (C s (A n)) (C s B) B
74  change ‖A n - C s (A n)‖ ≤ _ at hgentle
75  linarith
