15/-- Within the ordered coproduct the left occupation determines both words. -/
16theorem loweringSplits_left_counts_injective (w : List (PositiveRoot d))
17    (hw : RootOrdered w)
18    (ab cd : List (PositiveRoot d) × List (PositiveRoot d))
19    (hab : ab∈loweringSplits w) (hcd : cd∈loweringSplits w)
20    (hcounts : (fun a => ab.1.count a) = (fun a => cd.1.count a)) : ab=cd := by
21  obtain ⟨hao,hbo⟩ := loweringSplits_ordered w hw ab.1 ab.2 hab
22  obtain ⟨hco,hdo⟩ := loweringSplits_ordered w hw cd.1 cd.2 hcd
23  have hright : (fun a => ab.2.count a) = (fun a => cd.2.count a) := by
24    funext a
25    have ha := (loweringSplits_grading w ab.1 ab.2 hab).2.2 a
26    have hc := (loweringSplits_grading w cd.1 cd.2 hcd).2.2 a
27    have h := congrFun hcounts a
28    omega
29  apply Prod.ext
30  · rw [← canonicalWord_counts_of_ordered ab.1 hao,
31      ← canonicalWord_counts_of_ordered cd.1 hco,hcounts]
32  · rw [← canonicalWord_counts_of_ordered ab.2 hbo,
33      ← canonicalWord_counts_of_ordered cd.2 hdo,hright]

35/-- Exact repetition count of a word pair in the ordered coproduct. -/
36theorem loweringSplits_count_eq_occupationChoices
37    (w : List (PositiveRoot d)) (hw : RootOrdered w)
38    (u v : List (PositiveRoot d)) (huv : (u,v)∈loweringSplits w) :
39    (loweringSplits w).count (u,v) = occupationChoices w (fun a => u.count a) := by
40  classical
41  have he : (loweringSplits w).count (u,v) =
42      ((loweringSplits w).map (fun ab => fun a => ab.1.count a)).count
43        (fun a => u.count a) := by
44    simp only [List.count,List.countP_map]
45    apply List.countP_congr
46    intro ab hab
47    simp only [Function.comp_def,beq_iff_eq]
48    exact ⟨fun h => by cases h; rfl,
49      fun h => loweringSplits_left_counts_injective w hw ab (u,v) hab huv h⟩
50  rw [he,loweringSplits_count_left_occupations]
51  rfl

53/-- Multiplicity times the exact finite Cartan coefficient is exactly the
54product-binomial amplitude at the finite root fractions. -/
55theorem occupationChoices_mul_normalizedSplitCoefficient
56    (mu nu : Fin d → ℕ)
57    (hmu : ∀ a : PositiveRoot d, 0 < rootGap mu a)
58    (hnu : ∀ a : PositiveRoot d, 0 < rootGap nu a)
59    (w u v : List (PositiveRoot d)) (h : (u,v)∈loweringSplits w) :
60    (occupationChoices w (fun a => u.count a) : ℂ) *
61      normalizedSplitCoefficient mu nu w u v =
62      ((∏a : PositiveRoot d, Cloning.Occupation.splitAmplitude
63        (w.count a) (u.count a) (rootFraction mu nu a) : ℝ) : ℂ) := by
64  have hcounts := (loweringSplits_grading w u v h).2.2
65  have hreverse : rootFraction nu mu = fun a => 1-rootFraction mu nu a := by
66    funext a
67    exact rootFraction_reverse mu nu a (ne_of_gt (add_pos (hmu a) (hnu a)))
68  rw [normalizedSplitCoefficient_eq_amplitudes mu nu hmu hnu w u v h,hreverse]
69  rw [← occupationSplitMatrixElement_eq_choices (rootFraction mu nu) w u v hcounts]
70  exact occupationSplitMatrixElement_eq_product_amplitude (rootFraction mu nu)
71    (fun a => (rootFraction_mem_Icc mu nu a (hmu a) (hnu a)).1)
72    (fun a => (rootFraction_mem_Icc mu nu a (hmu a) (hnu a)).2) w u v hcounts

74/-- Grouping repeated root assignments gives the exact product amplitudes,
75with no limiting statement and no conditions on the vector-valued summands. -/
76theorem cartan_split_sum_eq_product_amplitude {H : Type*}
77    [AddCommMonoid H] [Module ℂ H]
78    (mu nu : Fin d → ℕ)
79    (hmu : ∀ a : PositiveRoot d, 0 < rootGap mu a)
80    (hnu : ∀ a : PositiveRoot d, 0 < rootGap nu a)
81    (w : List (PositiveRoot d)) (hw : RootOrdered w)
82    (F : (List (PositiveRoot d) × List (PositiveRoot d)) → H) :
83    ((loweringSplits w).map (fun uv =>
84      normalizedSplitCoefficient mu nu w uv.1 uv.2 • F uv)).sum =
85    ∑uv∈(loweringSplits w).toFinset,
86      ((∏a : PositiveRoot d, Cloning.Occupation.splitAmplitude
87        (w.count a) (uv.1.count a) (rootFraction mu nu a) : ℝ) : ℂ) • F uv := by
88  rw [Finset.sum_list_map_count]
89  apply Finset.sum_congr rfl
90  intro uv huv
91  rcases uv with ⟨u,v⟩
92  have hmem := List.mem_toFinset.mp huv
93  have hcount := loweringSplits_count_eq_occupationChoices w hw u v hmem
94  simp only [List.count, Bool.beq_eq_decide_eq] at hcount ⊢
95  rw [hcount]
96  rw [← Nat.cast_smul_eq_nsmul ℂ,smul_smul]
97  have hc := occupationChoices_mul_normalizedSplitCoefficient mu nu hmu hnu w u v hmem
98  simpa only [List.count, Bool.beq_eq_decide_eq] using
99    congrArg (fun z : ℂ => z • F (u,v)) hc

101/-- The finite support of an ordered coproduct contains exactly one word pair
102for each left suboccupation. Repeated assignments affect only its coefficient. -/
103theorem loweringSplits_existsUnique_suboccupation
104    (w : List (PositiveRoot d)) (hw : RootOrdered w)
105    (k : PositiveRoot d → ℕ) (hk : ∀a, k a≤w.count a) :
106    ∃! uv : List (PositiveRoot d) × List (PositiveRoot d),
107      uv∈loweringSplits w ∧ (fun a => uv.1.count a)=k := by
108  classical
109  have hp : 0<((loweringSplits w).map (fun uv => fun a => uv.1.count a)).count k := by
110    rw [loweringSplits_count_left_occupations]
111    exact Finset.prod_pos (fun a _ => Nat.choose_pos (hk a))
112  obtain ⟨uv,huv,hcounts⟩ := List.mem_map.mp (List.count_pos_iff.mp hp)
113  refine ⟨uv,⟨huv,hcounts⟩,?_⟩
114  intro ab hab
115  exact loweringSplits_left_counts_injective w hw ab uv hab.1 huv
116    (hab.2.trans hcounts.symm)

118/-- The unique split attached to a suboccupation consists of its canonical
119word and the canonical complementary word. -/
120theorem canonical_suboccupation_mem_loweringSplits
121    (w : List (PositiveRoot d)) (hw : RootOrdered w)
122    (k : PositiveRoot d → ℕ) (hk : ∀a, k a≤w.count a) :
123    (canonicalWord k,canonicalWord (fun a => w.count a-k a))∈loweringSplits w := by
124  obtain ⟨⟨u,v⟩,⟨huv,hcounts⟩,_⟩ := loweringSplits_existsUnique_suboccupation w hw k hk
125  obtain ⟨huo,hvo⟩ := loweringSplits_ordered w hw u v huv
126  have hu : canonicalWord k=u := by
127    rw [← hcounts]
128    exact canonicalWord_counts_of_ordered u huo
129  have hvcounts : (fun a => v.count a)=(fun a => w.count a-k a) := by
130    funext a
131    have hc := (loweringSplits_grading w u v huv).2.2 a
132    have he := congrFun hcounts a
133    dsimp only at he
134    omega
135  have hv : canonicalWord (fun a => w.count a-k a)=v := by
136    rw [← hvcounts]
137    exact canonicalWord_counts_of_ordered v hvo
138  simpa only [hu,hv] using huv
