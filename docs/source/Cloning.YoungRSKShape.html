9noncomputable def shapeRemovable : List ℕ → ℕ → Prop

14theorem shapeAdd_injective_of_length {M K : List ℕ} (h : M.length = K.length)
15    (r : ℕ) (he : shapeAdd M r = shapeAdd K r) : M = K := by
16  induction M generalizing K r with
17  | nil => simpa using h.symm
18  | cons m M ih =>
19    cases K with
20    | nil => simp at h
21    | cons k K =>
22      have hlen : M.length = K.length := by simpa using h
23      cases r with
24      | zero =>
25        simp only [shapeAdd, List.cons.injEq] at he
26        obtain ⟨hm, hM⟩ := he
27        have : m = k := by omega
28        simp [this, hM]
29      | succ r =>
30        simp only [shapeAdd, List.cons.injEq] at he
31        exact congrArg₂ List.cons he.1 (ih hlen r he.2)

33theorem shapeRemovable_shapeAdd (M : List ℕ)
34    (hM : List.Pairwise (fun a b => b ≤ a) M) (r : ℕ) (hr : r < M.length) :
35    shapeRemovable (shapeAdd M r) r := by
36  induction M generalizing r with
37  | nil => simp at hr
38  | cons m M ih =>
39    obtain ⟨hm, hM⟩ := List.pairwise_cons.mp hM
40    cases r with
41    | zero =>

46      have := hm n hmem

48    | succ r =>
