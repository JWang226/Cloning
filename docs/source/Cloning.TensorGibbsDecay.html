21theorem exists_root_height_decay (p : Fin d → ℝ) (hp : ∀ a, 0 < p a)
22    (hord : StrictAnti p) :
23    ∃ θ : ℝ, 0 < θ ∧ θ < 1 ∧ ∀ r : PositiveRoot d,
24      rootBoltzmann p r < θ ^ r.height := by
25  let t : ℕ → ℝ := fun N => 1 - 1 / ((N : ℝ) + 1)
26  have ht : Tendsto t atTop (𝓝 1) := by
27    simpa only [sub_zero] using tendsto_const_nhds.sub
28      (tendsto_one_div_add_atTop_nhds_zero_nat (𝕜 := ℝ))
29  have hr (r : PositiveRoot d) : ∀ᶠ N : ℕ in atTop, rootBoltzmann p r < t N ^ r.height :=
30    (ht.pow r.height).eventually (eventually_gt_nhds (by
31      simpa only [one_pow] using rootBoltzmann_lt_one p hp hord r))
32  have hall : ∀ᶠ N : ℕ in atTop, ∀ r : PositiveRoot d, rootBoltzmann p r < t N ^ r.height :=
33    Filter.eventually_all.mpr hr
34  obtain ⟨N, hN, hNr⟩ := ((eventually_ge_atTop 1).and hall).exists
35  have hn : (0 : ℝ) < N := by exact_mod_cast (lt_of_lt_of_le Nat.zero_lt_one hN)
36  have hpos : 0 < 1 / ((N : ℝ) + 1) := by positivity
37  have hlt : 1 / ((N : ℝ) + 1) < 1 := (div_lt_one (by positivity)).mpr (by linarith)
38  exact ⟨t N, by dsimp [t]; linarith, by dsimp [t]; linarith, hNr⟩

47theorem exists_eventual_root_height_decay (pN : ℕ → Fin d → ℝ) (p : Fin d → ℝ)
48    (hp : ∀ a, 0 < p a) (hord : StrictAnti p)

50    ∃ θ : ℝ, 0 < θ ∧ θ < 1 ∧

52      ∀ᶠ N : ℕ in atTop, ∀ r : PositiveRoot d, rootBoltzmann (pN N) r ≤ θ ^ r.height := by
