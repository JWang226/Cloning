28theorem infinite_complex_unit_circle : Set.Infinite {z : ℂ | ‖z‖ = 1} := by
29  let f : ℝ → ℂ := fun t => ⟨t, Real.sqrt (1 - t ^ 2)⟩
30  have hi : Set.InjOn f (Set.Ioo (-1 : ℝ) 1) := by
31    intro x _ y _ h
32    exact congrArg Complex.re h
33  apply ((Set.Ioo_infinite (by norm_num : (-1 : ℝ) < 1)).image hi).mono
34  rintro z ⟨t, ht, rfl⟩
35  change ‖f t‖ = 1
36  have ht2 : 0 ≤ 1 - t ^ 2 := by nlinarith [ht.1, ht.2]
37  have hn : ‖f t‖ ^ 2 = 1 := by
38    rw [← Complex.normSq_eq_norm_sq]
39    change t * t + Real.sqrt (1 - t ^ 2) * Real.sqrt (1 - t ^ 2) = 1
40    rw [Real.mul_self_sqrt ht2]
41    ring
42  nlinarith [norm_nonneg (f t)]

44theorem tensorPower_diagonal (n : ℕ) (z : A → ℂ) :
45    tensorPower n (Matrix.diagonal z) = Matrix.diagonal (fun a => ∏ i, z (a i)) := by
46  ext a c
47  by_cases h : a = c
48  · subst c

51    rw [Matrix.diagonal_apply_ne _ h]

53    exact Matrix.diagonal_apply_ne _ hi

75/-- Polynomial extension of the diagonal unitary commutation identities. -/
76theorem commute_diagonal_of_commute_unitary (n : ℕ)
77    (R : Matrix ((Fin n → A) × (Fin n → A)) ((Fin n → A) × (Fin n → A)) ℂ)
78    (hR : ∀ V : unitary (Matrix A A ℂ), Commute R (leftTensor n (V : Matrix A A ℂ)))
79    (z : A → ℂ) : Commute R (leftTensor n (Matrix.diagonal z)) := by
80  rw [leftTensor_diagonal]
81  show R * Matrix.diagonal _ = Matrix.diagonal _ * R
82  ext ab cd
83  rw [Matrix.mul_diagonal, Matrix.diagonal_mul]
84  let p : MvPolynomial A ℂ := MvPolynomial.C (R ab cd) *
85    ∏ i : Fin n, MvPolynomial.X (cd.1 i)
86  let q : MvPolynomial A ℂ := (∏ i : Fin n, MvPolynomial.X (ab.1 i)) *
87    MvPolynomial.C (R ab cd)
88  have hpq : p = q := by
89    apply MvPolynomial.funext_set (fun _ => {w : ℂ | ‖w‖ = 1})
90      (fun _ => infinite_complex_unit_circle)
91    intro w hw
92    have hw' : ∀ a, ‖w a‖ = 1 := fun a => hw a (Set.mem_univ a)
93    have hc := (hR (diagonalUnitary w hw')).eq
94    change R * leftTensor n (Matrix.diagonal w) = leftTensor n (Matrix.diagonal w) * R at hc
95    rw [leftTensor_diagonal] at hc
96    have he := congrArg (fun M => M ab cd) hc
97    simpa [p, q, MvPolynomial.eval_prod, Matrix.mul_diagonal, Matrix.diagonal_mul] using he
98  have he := congrArg (MvPolynomial.eval z) hpq
99  simpa [p, q, MvPolynomial.eval_prod] using he

106theorem commute_hermitian_of_commute_unitary (n : ℕ)
107    (R : Matrix ((Fin n → A) × (Fin n → A)) ((Fin n → A) × (Fin n → A)) ℂ)

112      (hX.eigenvectorUnitary : Matrix A A ℂ)ᴴ := hX.spectral_theorem

194/-- Exact finite-sample random-purification identity for all positive input
195matrices. Both the CPTP map and its Haar moment are explicit constructions. -/
196theorem haarPurificationChannel_tensorPower (n : ℕ) (ρ : Matrix A A ℂ)
197    (hρ : ρ.PosSemidef) :
198    (haarPurificationChannel (A := A) n).toFun (tensorPower n ρ) =
199      ∫ U : unitary (Matrix A A ℂ),
200        entangledMoment n (purificationCoefficients ρ U) ∂unitaryHaar := by
201  rw [haarPurificationChannel_apply]
202  have hsquare : tensorPower n ρ = tensorPower n (CFC.sqrt ρ) * tensorPower n (CFC.sqrt ρ) := by
203    rw [← tensorPower_mul, Cloning.MatrixFidelity.sqrt_mul_self hρ]
204  rw [hsquare, purificationMap_square (haarMoment_posSemidef n) _
205    (haarMoment_commute_hermitian n _ (Cloning.MatrixFidelity.sqrt_posSemidef ρ).isHermitian)]
206  have hs : (leftTensor n (CFC.sqrt ρ))ᴴ = leftTensor n (CFC.sqrt ρ) := by
207    rw [← leftTensor_star, Cloning.MatrixFidelity.sqrt_conjTranspose]
208  have h := (matrixSandwichCLM (leftTensor n (CFC.sqrt ρ))).integral_comp_comm
209    (integrable_haar_entangledMoment (A := A) n)
210  change (∫ U : unitary (Matrix A A ℂ), leftTensor n (CFC.sqrt ρ) *
211      entangledMoment n (U : Matrix A A ℂ) * (leftTensor n (CFC.sqrt ρ))ᴴ ∂unitaryHaar) =
212    leftTensor n (CFC.sqrt ρ) * haarMoment n * (leftTensor n (CFC.sqrt ρ))ᴴ at h
213  change leftTensor n (CFC.sqrt ρ) * haarMoment n * leftTensor n (CFC.sqrt ρ) = _
214  rw [hs] at h
215  rw [← h]
216  apply integral_congr_ae
217  apply Filter.Eventually.of_forall
218  intro U
219  simpa only [hs] using (entangledMoment_mul n (CFC.sqrt ρ) (U : Matrix A A ℂ)).symm
