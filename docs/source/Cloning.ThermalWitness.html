20lemma witness_summable {q x : ℝ} (hq0 : 0 < q) (hqx : q < x) (hx1 : x < 1) :
21    Summable (Thermal.witness q x) := by
22  have hx0 : 0 < x := hq0.trans hqx
23  have hr : ‖Real.sqrt (q / x)‖ < 1 := by
24    rw [Real.norm_eq_abs, abs_of_nonneg (Real.sqrt_nonneg _),
25      Real.sqrt_lt' (by norm_num : (0 : ℝ) < 1)]
26    simpa using (div_lt_one hx0).mpr hqx
27  have hs := (summable_geometric_of_norm_lt_one hr).mul_left
28    (Real.sqrt ((1 - q) / (1 - x)))
29  exact hs.congr (fun n => (Thermal.witness_closed hq0.le (hqx.trans hx1) hx0.le hx1 n).symm)

74lemma trace_hasSum_basis {ι : Type*} (b : HilbertBasis ι ℂ H)
75    {T : H →L[ℂ] H} (hT : IsTraceClass T) :
76    HasSum (fun i => ⟪b i, T (b i)⟫_ℂ) (trace T hT) := by
77  let e : ι ≃ Set.range b := Equiv.ofInjective b b.orthonormal.linearIndependent.injective
78  have horth : Orthonormal ℂ (fun i : Set.range b => b (e.symm i)) :=
79    b.orthonormal.comp e.symm e.symm.injective
80  have hrange : Set.range (fun i : Set.range b => b (e.symm i)) = Set.range b := by
81    exact e.symm.surjective.range_comp b
82  let c : HilbertBasis (Set.range b) ℂ H := HilbertBasis.mk horth (by
83    rw [hrange, b.dense_span])
84  have hc : ∀ i, c i = b (e.symm i) := fun i => congrFun (HilbertBasis.coe_mk horth _) i
85  have hs := (summable_trace_diagonal_of_isTraceClass hT c).hasSum

87  have hs' := e.hasSum_iff.mpr hs

155/-- The inverse witness is handled through actual bounded inverses of positive
156regularizations. Their moments are uniformly bounded by the exact thermal affinity. -/
157lemma thermal_regularized_inverse_moment (b : HilbertBasis ℕ ℂ H) {q x ε : ℝ}
158    (hq0 : 0 < q) (hqx : q < x) (hx1 : x < 1) (hε : 0 < ε) :
159    (tracePairing (vectorMixture b (Thermal.geometric q))
160      (CFC.rpow (regularizedWeight (witnessOperator b q x) ε) (-1))).re ≤
161        Thermal.fidelity q x := by
162  rw [diagonalMixture_tracePairing b _ (Thermal.geometric_hasSum hq0.le (hqx.trans hx1)).summable
163    _ _ (regularized_inverse_witness_apply_basis b hq0 hqx hx1 hε)]
164  have hs := Thermal.witness_inverse_moment_hasSum hq0 (hqx.trans hx1) (hq0.trans hqx) hx1
165  have hbound (n : ℕ) : Thermal.geometric q n * (Thermal.witness q x n + ε)⁻¹ ≤
166      Thermal.geometric q n / Thermal.witness q x n := by
167    rw [← div_eq_mul_inv]
168    exact div_le_div_of_nonneg_left (Thermal.geometric_nonneg hq0.le (hqx.trans hx1).le n)
169      (Thermal.witness_pos hq0 (hqx.trans hx1) (hq0.trans hqx) hx1 n)
170      (le_add_of_nonneg_right hε.le)
171  have hn (n : ℕ) : 0 ≤ Thermal.geometric q n * (Thermal.witness q x n + ε)⁻¹ :=
172    mul_nonneg (Thermal.geometric_nonneg hq0.le (hqx.trans hx1).le n)
173      (inv_nonneg.mpr (add_nonneg
174        (Thermal.witness_pos hq0 (hqx.trans hx1) (hq0.trans hqx) hx1 n).le hε.le))
175  have hs' := hs.summable.of_nonneg_of_le hn hbound
176  exact (hs'.tsum_le_tsum hbound hs.summable).trans_eq hs.tsum_eq

196lemma diagonal_regularized_inverse_moment {ι : Type*} (b : HilbertBasis ι ℂ H)
197    (p w : ι → ℝ) (hp : Summable p) (hp0 : ∀ i, 0 ≤ p i) (hw0 : ∀ i, 0 < w i)
198    (W : H →L[ℂ] H) (hW0 : 0 ≤ W) (hW : ∀ i, W (b i) = (w i : ℂ) • b i)
199    {M ε : ℝ} (hs : HasSum (fun i => p i / w i) M) (hε : 0 < ε) :
200    (tracePairing (vectorMixture b p) (CFC.rpow (regularizedWeight W ε) (-1))).re ≤ M := by
201  have hreg (i : ι) : regularizedWeight W ε (b i) = ((w i + ε : ℝ) : ℂ) • b i := by
202    simp only [regularizedWeight, ContinuousLinearMap.add_apply, ContinuousLinearMap.smul_apply,
203      ContinuousLinearMap.one_apply, hW i, Complex.ofReal_add, add_smul, Complex.coe_smul]
204  have hinv (i : ι) : CFC.rpow (regularizedWeight W ε) (-1) (b i) =
205      (((w i + ε)⁻¹ : ℝ) : ℂ) • b i := by
206    simpa only [Complex.ofReal_inv] using inverse_apply_eigenvector
207      (regularizedWeight_strictlyPositive hW0 hε) (add_pos (hw0 i) hε) (hreg i)
208  rw [diagonalMixture_tracePairing b p hp _ _ hinv]
209  have hb (i : ι) : p i * (w i + ε)⁻¹ ≤ p i / w i := by
210    rw [← div_eq_mul_inv]
211    exact div_le_div_of_nonneg_left (hp0 i) (hw0 i) (le_add_of_nonneg_right hε.le)
212  have hn (i : ι) : 0 ≤ p i * (w i + ε)⁻¹ :=
213    mul_nonneg (hp0 i) (inv_nonneg.mpr (add_pos (hw0 i) hε).le)
214  exact ((hs.summable.of_nonneg_of_le hn hb).tsum_le_tsum hb hs.summable).trans_eq hs.tsum_eq

314lemma positive_diagonal_injective {ι : Type*} (b : HilbertBasis ι ℂ H)
315    {W : H →L[ℂ] H} (hW : 0 ≤ W) (w : ι → ℝ) (hw : ∀ i, 0 < w i)
316    (heig : ∀ i, W (b i) = (w i : ℂ) • b i) : Function.Injective W := by
317  have hstar : W.adjoint = W := by
318    simpa only [ContinuousLinearMap.star_eq_adjoint] using (IsSelfAdjoint.of_nonneg hW).star_eq
319  have hker (v : H) (hv : W v = 0) : v = 0 := by
320    apply b.repr.injective
321    ext i
322    rw [map_zero]
323    change b.repr v i = 0
324    rw [b.repr_apply_apply]
325    have h := ContinuousLinearMap.adjoint_inner_left W v (b i)
326    rw [hstar, heig i, inner_smul_left, hv, inner_zero_right] at h
327    simp only [Complex.conj_ofReal] at h
328    exact (mul_eq_zero.mp h).resolve_left (Complex.ofReal_ne_zero.mpr (hw i).ne')
329  intro u v huv
330  apply sub_eq_zero.mp
331  exact hker (u - v) (by rw [map_sub, huv, sub_self])

399lemma single_mixture_witness_moment_le {q x c : ℝ} {a : ℕ → ℝ}
400    (hq0 : 0 < q) (hqx : q < x) (hx1 : x < 1)
401    (ha0 : ∀ l, 0 ≤ a l) (ha : HasSum a c) :
402    (∑' k, (∑' l, a l * BosonicNumberLaw.seededLaw x l k) * Thermal.witness q x k) ≤
403      c * Thermal.fidelity q x := by
404  let e : (Fin 1 → ℕ) ≃ ℕ := Equiv.funUnique (Fin 1) ℕ
405  have he (l : Fin 1 → ℕ) : e l = l 0 := rfl
406  have ha' : HasSum (fun l : Fin 1 → ℕ => a (l 0)) c := by
407    simpa only [Function.comp_apply, he] using e.hasSum_iff.mpr ha
408  have h := BosonicNumberLaw.mixtureLaw_witness_moment_le
409    (q := fun _ : Fin 1 => q) (x := fun _ => x) (a := fun l => a (l 0))
410    (fun _ => hq0) (fun _ => hqx) (fun _ => hx1) (fun l => ha0 (l 0)) ha'
411  simp only [BosonicNumberLaw.mixtureLaw, BosonicNumberLaw.productLaw, Fin.prod_univ_one] at h
412  have hinner (k : ℕ) : (∑' l : Fin 1 → ℕ, a (l 0) * BosonicNumberLaw.seededLaw x (l 0) k) =
413      ∑' l, a l * BosonicNumberLaw.seededLaw x l k :=
414    e.tsum_eq (fun l => a l * BosonicNumberLaw.seededLaw x l k)
415  simp_rw [hinner] at h
416  have houter := e.tsum_eq (fun k =>
417    (∑' l, a l * BosonicNumberLaw.seededLaw x l k) * Thermal.witness q x k)
418  rw [show (∑' k : Fin 1 → ℕ,
419      (∑' l, a l * BosonicNumberLaw.seededLaw x l (k 0)) * Thermal.witness q x (k 0)) =
420      (∑' k, (∑' l, a l * BosonicNumberLaw.seededLaw x l k) * Thermal.witness q x k)
421    from houter] at h
422  exact h
