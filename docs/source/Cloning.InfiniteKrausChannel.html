20/-- A single Kraus summand, bundled as a continuous trace-class map. -/
21def krausTerm (K : H →L[ℂ] H) : TraceClass H →L[ℂ] TraceClass H :=
22  sandwichCLM K (star K)

44/-- Strong normalization of the Kraus family, tested on actual Hilbert vectors. -/
45def KrausComplete (K : ι → H →L[ℂ] H) : Prop :=
46  ∀ x : H, HasSum (fun k => ‖K k x‖ ^ 2) (‖x‖ ^ 2)

48/-- Positive inputs have an absolutely convergent Kraus series, with total
49trace-norm mass exactly their input trace. The proof uses the actual rank-one
50trace-norm decomposition and nonnegative Fubini. -/
51theorem krausTerm_norm_hasSum_of_nonneg (K : ι → H →L[ℂ] H) (hK : KrausComplete K)
52    (A : TraceClass H) (hA : 0 ≤ A.1) :
53    HasSum (fun k => ‖krausTerm (K k) A‖) ‖A‖ := by
54  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
55  let v : w → H := fun i => CFC.sqrt A.1 (b i)
56  have hmass : HasSum (fun i : w => ‖v i‖ ^ 2) ‖A‖ := by
57    simpa only [norm_vectorProjector, TraceClass.norm_eq_trace_re_of_nonneg A hA]
58      using positive_rankOne_mass hA A.2 b
59  have hd : Summable (fun p : w × ι => ‖K p.2 (v p.1)‖ ^ 2) := by
60    apply (summable_prod_of_nonneg (fun p => sq_nonneg ‖K p.2 (v p.1)‖)).mpr
61    refine ⟨fun i => (hK (v i)).summable, ?_⟩
62    simpa only [(hK _).tsum_eq] using hmass.summable
63  have hrow (k : ι) : HasSum (fun i : w => ‖K k (v i)‖ ^ 2)
64      (traceCLM (krausTerm (K k) A)).re := by
65    have h := ((positive_rankOne_series hA A.2 b).mapL (krausTerm (K k))).mapL traceCLM
66    have hr := Complex.hasSum_re h
67    simpa only [krausTerm_vectorProjector, traceCLM_vectorProjector, Complex.ofReal_re] using hr
68  have heq (k : ι) : ‖krausTerm (K k) A‖ = ∑' i : w, ‖K k (v i)‖ ^ 2 := by
69    rw [TraceClass.norm_eq_trace_re_of_nonneg _ (krausTerm_nonneg (K k) A hA)]
70    exact (hrow k).tsum_eq.symm
71  have hs : Summable (fun k => ‖krausTerm (K k) A‖) := by
72    simpa only [heq] using hd.prod_symm.prod
73  have ht : (∑' k, ‖krausTerm (K k) A‖) = ‖A‖ := by
74    simp_rw [heq]
75    rw [hd.tsum_comm]
76    simpa only [(hK _).tsum_eq] using hmass.tsum_eq
77  exact ht ▸ hs.hasSum

79/-- Every trace-class input has a convergent Kraus series. Positivity is
80removed by the actual Jordan and real/imaginary decompositions. -/
81theorem krausTerm_summable (K : ι → H →L[ℂ] H) (hK : KrausComplete K)
82    (A : TraceClass H) : Summable (fun k => krausTerm (K k) A) := by
83  have hpos (B : TraceClass H) (hB : 0 ≤ B.1) :
84      Summable (fun k => krausTerm (K k) B) :=
85    (krausTerm_norm_hasSum_of_nonneg K hK B hB).summable.of_norm
86  have hself (B : TraceClass H) (hB : IsSelfAdjoint B.1) :
87      Summable (fun k => krausTerm (K k) B) := by
88    have hp := hpos (TraceClass.positivePart B hB) (TraceClass.positivePart_nonneg B hB)
89    have hn := hpos (TraceClass.negativePart B hB) (TraceClass.negativePart_nonneg B hB)
90    convert hp.sub hn using 1
91    ext k
92    rw [← map_sub, TraceClass.positivePart_sub_negativePart]
93  have hr := hself (TraceClass.realComponent A) (TraceClass.realComponent_isSelfAdjoint A)
94  have hi := hself (TraceClass.imaginaryComponent A) (TraceClass.imaginaryComponent_isSelfAdjoint A)
95  convert hr.add (hi.const_smul Complex.I) using 1
96  ext k
97  rw [← map_smul, ← map_add, TraceClass.realComponent_add_I_smul_imaginaryComponent]

99/-- The Kraus series is complex linear on the whole trace-class space. -/
100def krausLinearMap (K : ι → H →L[ℂ] H) (hK : KrausComplete K) :
101    TraceClass H →ₗ[ℂ] TraceClass H where
102  toFun A := ∑' k, krausTerm (K k) A
103  map_add' A B := by
104    simp only [map_add]
105    exact (krausTerm_summable K hK A).tsum_add (krausTerm_summable K hK B)
106  map_smul' c A := by
107    simp only [map_smul]
108    exact (krausTerm_summable K hK A).tsum_const_smul c

126lemma krausLinearMap_trace_of_nonneg (K : ι → H →L[ℂ] H) (hK : KrausComplete K)
127    (A : TraceClass H) (hA : 0 ≤ A.1) : traceCLM (krausLinearMap K hK A) = traceCLM A := by
128  have hsource := (krausLinearMap_hasSum K hK A).mapL traceCLM
129  have htarget : HasSum (fun k => traceCLM (krausTerm (K k) A)) (traceCLM A) := by
130    simp only [traceCLM_eq_norm_of_nonneg A hA,
131      traceCLM_eq_norm_of_nonneg _ (krausTerm_nonneg _ A hA)]

135/-- Trace preservation holds for all trace-class operators, including
136non-self-adjoint inputs. -/
137lemma krausLinearMap_trace (K : ι → H →L[ℂ] H) (hK : KrausComplete K)
138    (A : TraceClass H) : traceCLM (krausLinearMap K hK A) = traceCLM A := by
139  let D : TraceClass H →ₗ[ℂ] ℂ :=
140    traceCLM.toLinearMap.comp (krausLinearMap K hK) - traceCLM.toLinearMap
141  have hpos (B : TraceClass H) (hB : 0 ≤ B.1) : D B = 0 :=
142    sub_eq_zero.mpr (krausLinearMap_trace_of_nonneg K hK B hB)
143  have hself (B : TraceClass H) (hB : IsSelfAdjoint B.1) : D B = 0 := by
144    rw [← TraceClass.positivePart_sub_negativePart B hB, map_sub,
145      hpos _ (TraceClass.positivePart_nonneg B hB),
146      hpos _ (TraceClass.negativePart_nonneg B hB), sub_self]
147  have hzero : D A = 0 := by
148    rw [← TraceClass.realComponent_add_I_smul_imaginaryComponent A, map_add, map_smul,
149      hself _ (TraceClass.realComponent_isSelfAdjoint A),
150      hself _ (TraceClass.imaginaryComponent_isSelfAdjoint A)]
151    simp
152  exact sub_eq_zero.mp hzero

154/-- Complete positivity follows from finite Kraus sums and their actual
155trace-norm (hence weak-operator) convergence, at every finite ancilla size. -/
156lemma krausLinearMap_completelyPositive (K : ι → H →L[ℂ] H) (hK : KrausComplete K) :
157    IsCompletelyPositive (krausLinearMap K hK) := by
158  classical
159  let F : Finset ι → TraceClass H →ₗ[ℂ] TraceClass H := fun s =>
160    ∑ k ∈ s, (krausTerm (K k)).toLinearMap
161  have hF (s : Finset ι) : IsCompletelyPositive (F s) := by
162    induction s using Finset.induction_on with
163    | empty =>
164      intro n A hA x
165      simp [F]
166    | @insert k s hk ih =>
167      simpa only [F, Finset.sum_insert hk] using (krausTerm_completelyPositive (K k)).add ih
168  intro n A hA
169  apply BlockPositive.of_weakOperator_tendsto (l := (atTop : Filter (Finset ι)))
170    (fun s i j => F s (A i j)) (fun i j => krausLinearMap K hK (A i j))
171    (fun s => hF s n A hA)
172  intro i j x y
173  have h := (traceClassMatrixCoefficient x y).continuous.tendsto
174    (krausLinearMap K hK (A i j)) |>.comp (krausLinearMap_hasSum K hK (A i j))
175  simpa only [F, LinearMap.sum_apply, traceClassMatrixCoefficient_apply] using h
