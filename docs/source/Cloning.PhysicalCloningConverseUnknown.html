70    (S : Set (SimpleSpectrum (k+1))) (p : SimpleSpectrum (k+1)) (hp : p ∈ interior S)
71    (b : OrthonormalBasis (Fin (k+1)) ℝ (EuclideanSpace ℝ (Fin (k+1))))

75    ∃ δ η : ℕ → ℝ, Tendsto δ atTop (𝓝 0) ∧ Tendsto η atTop (𝓝 0) ∧

77        let u := unknownLocalChart S p (interior_subset hp) b e n ξ
78        (‖(lan.reverse n).map (translatedPositive ξ (reference p e)).1 -
79          (tensorState (orbitState u.1.val u.2) n).1‖ ≤ δ n) ∧
80        (‖(lan.forward (m n)).map (tensorState (orbitState u.1.val u.2) (m n)).1 -
81          (translatedPositive (Real.sqrt g • ξ) (reference p e)).1‖ ≤ η n) := by

83  have hK : IsCompact K := compact_phaseParameters p b e _ (phaseSpaceBox_compact k s _)

85    rintro θ ⟨ξ,hξ,rfl⟩
86    exact phaseParameters_sum_zero p b hb e ξ
87  have hm := output_size_tendsto m g hg hratio
88  obtain ⟨δ,η,hδ,hη,hbound⟩ := scaled_window_approximations p b e lan m hm
89    (fun n => sampleRatio n (m n)) (Real.sqrt g) (sampleRatio_tendsto m g hratio) K hK hz
90  have hKscore : Bornology.IsBounded (Prod.fst '' K) := (hK.image continuous_fst).isBounded

92    rintro h ⟨θ,hθ,rfl⟩
93    exact hz θ hθ

104      (localUnitary p θ n)).matrix = _
105    rw [spectrumInSet, dif_pos hS]
106    exact he θ hθ

123theorem limsup_unknownSpectrumValue_le_of_compactWindowLAN
124    (S : Set (SimpleSpectrum (k+1))) (p : SimpleSpectrum (k+1)) (hp : p ∈ interior S)
125    (b : OrthonormalBasis (Fin (k+1)) ℝ (EuclideanSpace ℝ (Fin (k+1))))
126    (hb : b 0 = sqrtSpectrum p.eigenvalue) (e : Fin s ≃ PairIndex (k+1))
127    (lan : CompactWindowLAN p b e) (m : ℕ → ℕ) (g : ℝ) (hg : 1 < g)
128    (hratio : Tendsto (fun n => (m n : ℝ)/(n : ℝ)) atTop (𝓝 g)) :
129    limsup (fun n => unknownSpectrumValue n (m n) S) atTop ≤ universalValue g p := by
130  letI : Nonempty S := ⟨⟨p,interior_subset hp⟩⟩
131  choose δ η hδ hη hbound using fun L =>
132    unknown_fixedWindow_errors S p hp b hb e lan m g (by linarith) hratio L
133  have hd : ∀ L, ∀ᶠ n in atTop, ∀ᵐ ξ ∂(flatBoxDensity k s).expandingPrior L,
134      ‖(lan.reverse n).map (translatedPositive ξ (reference p e)).1 -
135        (tensorState (orbitState
136          (unknownLocalChart S p (interior_subset hp) b e n ξ).1.val
137          (unknownLocalChart S p (interior_subset hp) b e n ξ).2) n).1‖ ≤ δ L n := by
138    intro L
139    filter_upwards [hbound L] with n hn
140    exact (ae_flatBox_mem L).mono fun ξ hξ => (hn ξ hξ).1
141  have he : ∀ L, ∀ᶠ n in atTop, ∀ᵐ ξ ∂(flatBoxDensity k s).expandingPrior L,
142      ‖(lan.forward (m n)).map (tensorState (orbitState
143          (unknownLocalChart S p (interior_subset hp) b e n ξ).1.val
144          (unknownLocalChart S p (interior_subset hp) b e n ξ).2) (m n)).1 -
145        (translatedPositive (Real.sqrt g • ξ) (reference p e)).1‖ ≤ η L n := by
146    intro L
147    filter_upwards [hbound L] with n hn
148    exact (ae_flatBox_mem L).mono fun ξ hξ => (hn ξ hξ).2
149  have h := limsup_statePayoff_le_modeFactor_of_approximations
150    (fun u : S × unitary (Matrix (Fin (k+1)) (Fin (k+1)) ℂ) => orbitState u.1.val u.2)
151    m (flatBoxDensity k s) (fun _ => 1/2) (fun _ => by norm_num) g hg
152    (fun i => p.ratio (e i)) (fun i => (p.ratio_pos (e i)).le) (fun i => p.ratio_lt_one (e i))
153    lan.reverse (fun n => lan.forward (m n))
154    (unknownLocalChart S p (interior_subset hp) b e) δ η hd he hδ hη
155  have hprod := e.prod_comp (fun ij => Thermal.modeFactor g (p.ratio ij))
156  simpa only [unknownSpectrumValue, spectrumPayoff, universalValue, classicalValue, orbitalValue,
157    Nat.cast_add, Nat.cast_one, add_sub_cancel_right, hprod] using h

161theorem limsup_unknownSpectrumValue_infimum_of_compactWindowLAN
162    (S : Set (SimpleSpectrum (k+1))) (hSne : S.Nonempty)

164    (hLAN : ∀ p ∈ interior S,
165      ∃ b : OrthonormalBasis (Fin (k+1)) ℝ (EuclideanSpace ℝ (Fin (k+1))),
166        b 0 = sqrtSpectrum p.eigenvalue ∧ Nonempty (CompactWindowLAN p b e))
167    (m : ℕ → ℕ) (g : ℝ) (hg : 1 < g)
168    (hratio : Tendsto (fun n => (m n : ℝ)/(n : ℝ)) atTop (𝓝 g)) :
169    limsup (fun n => unknownSpectrumValue n (m n) S) atTop ≤
