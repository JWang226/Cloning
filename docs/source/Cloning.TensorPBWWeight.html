12/-- The Cartan-weight change caused by the lowering matrix unit `E_{ji}`. -/
13def PositiveRoot.weight (r : PositiveRoot d) (a : Fin d) : ℤ :=
14  (if a = r.val.2 then 1 else 0) - (if a = r.val.1 then 1 else 0)

106/-- The exact Cartan/lowering commutator as an identity of actual operators. -/
107theorem cartan_lowering_commutator (a : Fin d) (r : PositiveRoot d) :
108    collectiveGenerator n a a * collectiveGenerator n r.val.2 r.val.1 -
109      collectiveGenerator n r.val.2 r.val.1 * collectiveGenerator n a a =
110        (r.weight a : ℂ) • collectiveGenerator n r.val.2 r.val.1 := by
111  rw [collectiveGenerator_commutator]
112  by_cases haj : a = r.val.2
113  · subst a
114    have hji : r.val.2 ≠ r.val.1 := ne_of_gt r.property
115    simp [PositiveRoot.weight, hji, Ne.symm hji]
116  · by_cases hai : a = r.val.1
117    · subst a
118      simp [PositiveRoot.weight, haj]
119    · simp [PositiveRoot.weight, haj, hai, Ne.symm hai]

121theorem cartan_lowering_apply (a : Fin d) (r : PositiveRoot d)
122    (x : TensorRegister n (Fin d)) :
123    collectiveGenerator n a a (collectiveGenerator n r.val.2 r.val.1 x) =
124      collectiveGenerator n r.val.2 r.val.1 (collectiveGenerator n a a x) +
125        (r.weight a : ℂ) • collectiveGenerator n r.val.2 r.val.1 x := by
126  have he := congrArg
127    (fun T : TensorRegister n (Fin d) →L[ℂ] TensorRegister n (Fin d) => T x)
128    (cartan_lowering_commutator (n := n) a r)
129  simp only [ContinuousLinearMap.sub_apply, ContinuousLinearMap.mul_apply,
130    ContinuousLinearMap.smul_apply] at he
131  exact (sub_eq_iff_eq_add.mp he).trans (add_comm _ _)

133/-- Each actual lowering word is a simultaneous Cartan eigenvector with its
134exact additive weight. No raising, normalization, or nonvanishing premise is needed. -/
135theorem cartan_loweringWord
136    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℂ)
137    (hweight : ∀ a, collectiveGenerator n a a Ω = mu a • Ω)
138    (w : List (PositiveRoot d)) (a : Fin d) :
139    collectiveGenerator n a a (loweringWord Ω w) =
140      (mu a + (loweringWeight w a : ℂ)) • loweringWord Ω w := by
141  induction w with
142  | nil => simpa [loweringWord] using hweight a
143  | cons r w ih =>
144      rw [loweringWord, cartan_lowering_apply, ih, map_smul]
145      rw [← add_smul]
146      congr 1
147      simp only [loweringWeight_cons, Pi.add_apply, Int.cast_add]
148      ring
