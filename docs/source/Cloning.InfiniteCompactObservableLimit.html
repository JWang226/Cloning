3/-!
4# Compact-observable convergence from weak operator convergence
5 
6Uniform trace-norm bounds and convergence of actual matrix coefficients imply
7convergence of traces against every compact bounded observable. Finite-rank
8approximation is proved using the Hilbert-basis projections of the compact
9observable, rather than assumed as a density hypothesis.
10-/

21private theorem tracePairing_opNorm_le (T : TraceClass H) : ‖tracePairing T‖ ≤ ‖T‖ := by
22  have ht : ‖(traceCLM : TraceClass H →L[ℂ] ℂ)‖ ≤ 1 := by
23    apply ContinuousLinearMap.opNorm_le_bound _ zero_le_one
24    intro A
25    simpa using norm_trace_le_traceNorm A.2
26  have hmul : ‖traceClassRightMultiply T‖ ≤ ‖T‖ := by
27    unfold traceClassRightMultiply
28    exact LinearMap.mkContinuous_norm_le _ (norm_nonneg _) _
29  calc
30    ‖tracePairing T‖ ≤ ‖(traceCLM : TraceClass H →L[ℂ] ℂ)‖ * ‖traceClassRightMultiply T‖ :=
31      ContinuousLinearMap.opNorm_comp_le _ _
32    _ ≤ 1 * ‖T‖ := mul_le_mul ht hmul (norm_nonneg _) zero_le_one
33    _ = ‖T‖ := one_mul _

35/-- The closure argument for bounded functionals also works for arbitrary nets. -/
36theorem tendsto_functionals_at_closure_filter {α E : Type*}
37    [NormedAddCommGroup E] [NormedSpace ℂ E] (l : Filter α)
38    (u : α → E →L[ℂ] ℂ) (f : E →L[ℂ] ℂ) (C : ℝ) (hC : 0 ≤ C)
39    (hu : ∀ n, ‖u n‖ ≤ C) (hf : ‖f‖ ≤ C) {S : Set E}
40    (hpoint : ∀ x ∈ S, Tendsto (fun n => u n x) l (𝓝 (f x)))
41    {x : E} (hx : x ∈ closure S) : Tendsto (fun n => u n x) l (𝓝 (f x)) := by
42  rw [Metric.tendsto_nhds]
43  intro ε hε
44  let D := 2 * C + 1
45  have hD : 0 < D := by dsimp [D]; linarith
46  obtain ⟨y, hy, hxy⟩ := Metric.mem_closure_iff.mp hx (ε / (3 * D)) (by positivity)
47  have hevent := (Metric.tendsto_nhds.mp (hpoint y hy)) (ε / 3) (by positivity)
48  filter_upwards [hevent] with n hn
49  have hdiff : ‖u n - f‖ ≤ D := by
50    have := norm_sub_le (u n) f
51    dsimp [D]
52    linarith [hu n]
53  have hbound : ‖(u n - f) (x - y)‖ ≤ D * ‖x - y‖ :=
54    ((u n - f).le_opNorm _).trans (mul_le_mul_of_nonneg_right hdiff (norm_nonneg _))
55  have heq : u n x - f x = (u n - f) (x - y) + (u n y - f y) := by
56    simp only [ContinuousLinearMap.sub_apply, map_sub]
57    ring
58  rw [dist_eq_norm, heq]
59  have htri := norm_add_le ((u n - f) (x - y)) (u n y - f y)
60  rw [dist_eq_norm] at hn hxy
61  have hsmall : D * ‖x - y‖ < ε / 3 := by
62    have h := mul_lt_mul_of_pos_left hxy hD
63    have hd : D * (ε / (3 * D)) = ε / 3 := by field_simp
64    rwa [hd] at h
65  linarith

67/-- Every finite-basis approximation of an observable only tests finitely
68many of the matrix coefficients in the weak operator convergence hypothesis. -/
69theorem tracePairing_basisProjection_mul_tendsto {α ι : Type*}
70    (l : Filter α) (A : α → TraceClass H) (T : TraceClass H)
71    (hlim : ∀ x y : H, Tendsto (fun n => ⟪y, (A n).1 x⟫_ℂ) l (𝓝 ⟪y, T.1 x⟫_ℂ))
72    (b : HilbertBasis ι ℂ H) (s : Finset ι) (K : H →L[ℂ] H) :
73    Tendsto (fun n => tracePairing (A n) (basisProjection b s * K)) l
74      (𝓝 (tracePairing T (basisProjection b s * K))) := by
75  simp only [basisProjection_mul_eq_sum_rankOne, map_sum, tracePairing_rankOne]
76  exact tendsto_finset_sum s (fun i _ => hlim (b i) (K.adjoint (b i)))

78/-- Uniformly trace-norm bounded weak operator convergence implies convergence
79against every compact observable, on an arbitrary complete Hilbert space.
80Neither positivity nor separability is needed for this implication. -/
81theorem tracePairing_compact_tendsto {α : Type*} (l : Filter α)
82    (A : α → TraceClass H) (T : TraceClass H) (C : ℝ)
83    (hbound : ∀ n, ‖A n‖ ≤ C)
84    (hlim : ∀ x y : H, Tendsto (fun n => ⟪y, (A n).1 x⟫_ℂ) l (𝓝 ⟪y, T.1 x⟫_ℂ))
85    (K : H →L[ℂ] H) (hK : IsCompactOperator K) :
86    Tendsto (fun n => tracePairing (A n) K) l (𝓝 (tracePairing T K)) := by
87  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ H
88  let S : Set (H →L[ℂ] H) := Set.range (fun s : Finset w => basisProjection b s * K)
89  have hclosure : K ∈ closure S :=
90    mem_closure_of_tendsto (basisProjection_mul_tendsto b K hK)
91      (Eventually.of_forall (fun s => Set.mem_range_self s))
92  apply tendsto_functionals_at_closure_filter l (fun n => tracePairing (A n))
93    (tracePairing T) (max C ‖T‖) (le_trans (norm_nonneg T) (le_max_right _ _))
94    (fun n => (tracePairing_opNorm_le (A n)).trans ((hbound n).trans (le_max_left _ _)))
95    ((tracePairing_opNorm_le T).trans (le_max_right _ _)) (S := S) _ hclosure
96  rintro _ ⟨s, rfl⟩
97  exact tracePairing_basisProjection_mul_tendsto l A T hlim b s K

99/-- For positive inputs it is enough to assume an upper bound on the actual trace. -/
100theorem tracePairing_compact_tendsto_of_nonneg {α : Type*} (l : Filter α)
101    (A : α → TraceClass H) (T : TraceClass H) (C : ℝ)
102    (hA : ∀ n, 0 ≤ (A n).1)
103    (htrace : ∀ n, (traceCLM (A n)).re ≤ C)
104    (hlim : ∀ x y : H, Tendsto (fun n => ⟪y, (A n).1 x⟫_ℂ) l (𝓝 ⟪y, T.1 x⟫_ℂ))
105    (K : H →L[ℂ] H) (hK : IsCompactOperator K) :
106    Tendsto (fun n => tracePairing (A n) K) l (𝓝 (tracePairing T K)) := by
107  apply tracePairing_compact_tendsto l A T C _ hlim K hK
108  intro n
109  rw [TraceClass.norm_eq_trace_re_of_nonneg _ (hA n)]
110  exact htrace n
