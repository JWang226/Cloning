35/-- Exact product characteristic function, with a full mixed idler and every
36complex trace-class signal input. -/
37theorem preparationChannel_characteristic (σ : DensityState (Fock d))
38    (T : TraceClass (Fock d)) (a b : Fin d → ℂ) :
39    tracePairing ((preparationChannel σ).toLinearMap T) (displacement (Fin.append a b))=
40      tracePairing (idlerTraceClass σ) (displacement b)*tracePairing T (displacement a) := by
41  apply channel_pairing_of_pure
42  intro x
43  have hs := rectangularKraus_pairing_pure_hasSum (preparationKraus σ)
44    (preparationKraus_complete σ) x (displacement (Fin.append a b))
45  have ht := (idlerVector_pairing_hasSum σ (displacement b)).mul_left
46    ⟪x,displacement a x⟫_ℂ
47  have he (i : IdlerIndex d) :
48      ⟪preparationKraus σ i x,displacement (Fin.append a b) (preparationKraus σ i x)⟫_ℂ=
49        ⟪x,displacement a x⟫_ℂ*⟪idlerVector σ i,displacement b (idlerVector σ i)⟫_ℂ := by
50    simp only [preparationKraus,tensorRight_apply,displacement_append_tensor,tensorVector_inner]
51  simp_rw [he] at hs
52  exact hs.unique (by simpa only [mul_comm] using ht)
