3/-!
4# Completely positive repair of trace loss
5 
6A positive scalar functional gives a completely positive replacement map on
7the actual trace-class spaces. Applied to the trace deficit of a completely
8positive, trace-nonincreasing map, this constructs an actual quantum channel.
9No complete-positivity witness for the repair is assumed.
10-/

23/-- Scalar compression shows that every positive linear functional is
24completely positive, at the level of its scalar-valued block matrices. -/
25theorem BlockPositive.functionalMatrix {A : ι → ι → TraceClass H}
26    (hA : BlockPositive A) (f : TraceClass H →ₗ[ℂ] ℂ)
27    (hf : ∀ T, 0 ≤ T.1 → 0 ≤ f T) (c : ι → ℂ) :
28    0 ≤ ∑ i, ∑ j, star (c i) * c j * f (A i j) := by
29  have h := hf (∑ i, ∑ j, (star (c i) * c j) • A i j) (hA.scalarCompression c)
30  simpa only [map_sum, map_smul, smul_eq_mul] using h

32theorem BlockPositive.replacePure_functional {A : ι → ι → TraceClass H}
33    (hA : BlockPositive A) (f : TraceClass H →ₗ[ℂ] ℂ)
34    (hf : ∀ T, 0 ≤ T.1 → 0 ≤ f T) (z : K) :
35    BlockPositive (fun i j => f (A i j) • vectorProjector z) := by
36  intro x
37  convert hA.functionalMatrix f hf (fun i => ⟪z, x i⟫_ℂ) using 1
38  apply Finset.sum_congr rfl
39  intro i hi
40  apply Finset.sum_congr rfl
41  intro j hj

48/-- Infinite-rank replacement is obtained from the convergent rank-one
49expansion of the positive trace-class output operator. -/
50theorem BlockPositive.replace_functional {A : ι → ι → TraceClass H}
51    (hA : BlockPositive A) (f : TraceClass H →ₗ[ℂ] ℂ)
52    (hf : ∀ T, 0 ≤ T.1 → 0 ≤ f T) (σ : TraceClass K) (hσ : 0 ≤ σ.1) :
53    BlockPositive (fun i j => f (A i j) • σ) := by
54  intro x
55  obtain ⟨w, b, _⟩ := exists_hilbertBasis ℂ K
56  let E := replacementExpectation (fun i j => f (A i j)) x
57  have hs := (positive_rankOne_series hσ σ.2 b).mapL E
58  have hnonneg : ∀ k : w, 0 ≤ E (vectorProjector (CFC.sqrt σ.1 (b k))) := by
59    intro k
60    have h := hA.replacePure_functional f hf (CFC.sqrt σ.1 (b k)) x
61    change 0 ≤ replacementExpectation (fun i j => f (A i j)) x _
62    rw [replacementExpectation_apply]
63    convert h using 1
64    apply Finset.sum_congr rfl
65    intro i hi
66    apply Finset.sum_congr rfl
67    intro j hj
68    change _ = ⟪x i, (f (A i j) •
69      (vectorProjector (CFC.sqrt σ.1 (b k))).1) (x j)⟫_ℂ
70    simp only [ContinuousLinearMap.smul_apply, inner_smul_right]
71  have hsum : 0 ≤ E (TraceClass.ofOperator σ.1 σ.2) := by
72    rw [← hs.tsum_eq]
73    exact tsum_nonneg hnonneg
74  change 0 ≤ E σ at hsum
75  rw [replacementExpectation_apply] at hsum
76  convert hsum using 1
77  apply Finset.sum_congr rfl
78  intro i hi
79  apply Finset.sum_congr rfl
80  intro j hj
81  change ⟪x i, (f (A i j) • σ.1) (x j)⟫_ℂ = _
82  simp only [ContinuousLinearMap.smul_apply, inner_smul_right]

90/-- Ordinary positivity follows from complete positivity at ancilla dimension one. -/
91theorem IsCompletelyPositive.map_nonneg {Φ : TraceClass H →ₗ[ℂ] TraceClass K}
92    (hΦ : IsCompletelyPositive Φ) (A : TraceClass H) (hA : 0 ≤ A.1) :
93    0 ≤ (Φ A).1 := by

123theorem traceDefect_nonneg {Φ : TraceClass H →ₗ[ℂ] TraceClass K}

134/-- Repair replaces precisely the missing trace in a fixed density state. -/
135def traceRepairLinear (Φ : TraceClass H →ₗ[ℂ] TraceClass K) (σ : DensityState K) :
136    TraceClass H →ₗ[ℂ] TraceClass K :=
137  Φ + (traceDefect Φ).smulRight (TraceClass.ofOperator σ.op σ.traceClass)

144theorem traceRepair_completelyPositive {Φ : TraceClass H →ₗ[ℂ] TraceClass K}

169/-- Repair increases the image of every positive input in operator order. -/
170theorem le_traceRepair_of_nonneg {Φ : TraceClass H →ₗ[ℂ] TraceClass K}
171    (hΦ : IsCompletelyPositive Φ)
172    (htrace : ∀ A, 0 ≤ A.1 → (traceCLM (Φ A)).re ≤ (traceCLM A).re)
173    (σ : DensityState K) (A : TraceClass H) (hA : 0 ≤ A.1) :
174    (Φ A).1 ≤ (traceRepairLinear Φ σ A).1 := by
175  change (Φ A).1 ≤ (Φ A).1 + traceDefect Φ A • σ.op
176  exact le_add_of_nonneg_right (smul_nonneg (traceDefect_nonneg hΦ htrace A hA) σ.positive)

178/-- The exact trace-norm cost of repair is the lost trace. -/
179theorem norm_traceRepair_sub {Φ : TraceClass H →ₗ[ℂ] TraceClass K}
180    (hΦ : IsCompletelyPositive Φ)
181    (htrace : ∀ A, 0 ≤ A.1 → (traceCLM (Φ A)).re ≤ (traceCLM A).re)
182    (σ : DensityState K) (A : TraceClass H) (hA : 0 ≤ A.1) :
183    ‖traceRepairLinear Φ σ A - Φ A‖ = (traceCLM A).re - (traceCLM (Φ A)).re := by
184  have hσnorm : ‖TraceClass.ofOperator σ.op σ.traceClass‖ = 1 := by
185    rw [TraceClass.norm_eq_trace_re_of_nonneg _ σ.positive]
186    change (trace σ.op σ.traceClass).re = 1
187    rw [σ.trace_one]
188    rfl
189  have hdef : ‖traceDefect Φ A‖ = (traceDefect Φ A).re :=
190    (Complex.re_eq_norm.mpr (traceDefect_nonneg hΦ htrace A hA)).symm
191  change ‖(Φ A + traceDefect Φ A • TraceClass.ofOperator σ.op σ.traceClass) - Φ A‖ = _
192  rw [add_sub_cancel_left, norm_smul, hσnorm, mul_one, hdef, traceDefect_apply,
193    Complex.sub_re]
