31theorem exists_diverging_cutoff_twoWay_gibbs
32    (mu : ℕ → Fin d → ℕ) (hmu : ∀ N, Antitone (mu N))
33    (δ : ℕ → ℝ) (hδ : Tendsto δ atTop atTop)
34    (hgap : ∀ᶠ N in atTop, ∀ a : PositiveRoot d,
35      δ N ≤ (mu N a.val.1 : ℝ) - mu N a.val.2)
36    (pN : ℕ → Fin d → ℝ) (p : Fin d → ℝ) (hp : ∀ a, 0 < p a) (hord : StrictAnti p)
37    (hlim : ∀ a, Tendsto (fun N => pN N a) atTop (𝓝 (p a))) :
38    ∃ R : ℕ → ℕ, Monotone R ∧ Tendsto R atTop atTop ∧
39      Tendsto (fun N => partitionGibbsForwardError (mu N) (hmu N) (pN N) p (R N)) atTop (𝓝 0) ∧
40      Tendsto (fun N => partitionGibbsReverseError (mu N) (hmu N) (pN N) p (R N)) atTop (𝓝 0) := by
41  let bound (r : ℕ) : ℝ := 2 * ‖rootThermalState p - rootThermalCutoff p r‖ + 1 / ((r : ℝ)+1)
42  let P (N r : ℕ) : Prop :=
43    CutoffReady (partitionHighestTensor (mu N) (hmu N)) (mu N) r ∧ (∀ a, 0 < pN N a) ∧
44    gibbsThermalCutoffError (partitionHighestTensor (mu N) (hmu N)) (mu N)
45      (partitionHighestTensor_cartan (mu N) (hmu N))
46      (partitionHighestTensor_raising_zero (mu N) (hmu N)) (pN N) p r ≤ bound r
47  have hpN : ∀ᶠ N in atTop, ∀ a, 0 < pN N a :=
48    Filter.eventually_all.mpr (fun a => (hlim a).eventually (eventually_gt_nhds (hp a)))
49  have hP (r : ℕ) : ∀ᶠ N in atTop, P N r := by

51      (eventually_lt_nhds (show 2 * ‖rootThermalState p - rootThermalCutoff p r‖ < bound r by
52        dsimp [bound]; exact lt_add_of_pos_right _ (by positivity)))
53    filter_upwards [partition_eventually_CutoffReady mu hmu δ hδ hgap r, hpN, hb] with N hr hn hb
54    exact ⟨hr,hn,hb.le⟩
55  obtain ⟨R, hmono, hR, _, he⟩ := exists_common_diverging_cutoff P hP
56  have hb : Tendsto (fun N => bound (R N)) atTop (𝓝 0) := by
