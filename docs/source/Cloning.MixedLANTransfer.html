21/-- The genuine positive trace-class operator underlying an analytic density state. -/
22def statePositive (ρ : DensityState A) : PositiveTraceClass A :=
23  ⟨TraceClass.ofOperator ρ.op ρ.traceClass, ρ.positive⟩

25@[simp] theorem norm_statePositive (ρ : DensityState A) : ‖(statePositive ρ).1‖ = 1 := by
26  rw [TraceClass.norm_eq_trace_re_of_nonneg _ (statePositive ρ).2]
27  change (trace ρ.op ρ.traceClass).re = 1
28  rw [ρ.trace_one]
29  rfl

43/-- The quantum-to-hybrid comparison contracts the input approximation error
44uniformly over the competing quantum channel. -/
45theorem composition_approximation
46    (S : HybridToQuantum G A μ) (M : QuantumChannel A B) (T : QuantumToHybrid B H μ)
47    (ρ : PositiveTraceClass A) (Φ : PositiveL1 G μ) {δ : ℝ}
48    (hδ : ‖S.map Φ.1 - ρ.1‖ ≤ δ) :
49    ‖T.map (M.toLinearMap ρ.1) - (transferred S M T).map Φ.1‖ ≤ δ := by
50  rw [transferred_apply]
51  apply (T.norm_map_sub_le _ _
52    (M.toPositiveTracePreservingMap.map_nonneg ρ.1 ρ.2)
53    (M.toPositiveTracePreservingMap.map_nonneg _ (S.map_nonneg Φ.1 Φ.2))).trans
54  apply (M.toPositiveTracePreservingMap.norm_map_sub_le _ _ ρ.2
55    (S.map_nonneg Φ.1 Φ.2)).trans
56  rwa [norm_sub_rev]

58/-- Sharp mixed-experiment root-fidelity comparison: the error has no dimension,
59competitor, or channel-norm constant. -/
60theorem fidelity_transfer
61    (S : HybridToQuantum G A μ) (M : QuantumChannel A B) (T : QuantumToHybrid B H μ)
62    (ρ : PositiveTraceClass A) (σ : PositiveTraceClass B)
63    (Φ : PositiveL1 G μ) (Ψ : PositiveL1 H μ)
64    (hσ : ‖σ.1‖ = 1) (hΦ : ‖Φ.1‖ = 1) {δ η : ℝ}
65    (hδ : ‖S.map Φ.1 - ρ.1‖ ≤ δ) (hη : ‖T.map σ.1 - Ψ.1‖ ≤ η) :
66    (ρ.map M.toPositiveTracePreservingMap).rootFidelity σ ≤
67      (Φ.map (transferred S M T)).rootFidelity Ψ + Real.sqrt δ + Real.sqrt η := by
68  have hcont := PositiveL1.rootFidelity_continuity
69    (T.mapPositive (ρ.map M.toPositiveTracePreservingMap)) (T.mapPositive σ)
70    (Φ.map (transferred S M T)) Ψ
71  have hBnorm : ‖(T.mapPositive σ).1‖ = 1 :=
72    (T.norm_map_of_nonneg σ.1 σ.2).trans hσ
73  have hCnorm : ‖(Φ.map (transferred S M T)).1‖ = 1 :=
74    (PositiveL1.norm_map _ Φ).trans hΦ
75  rw [hBnorm, hCnorm, Real.sqrt_one, mul_one, mul_one] at hcont
76  have hd := composition_approximation S M T ρ Φ hδ
77  have hcont' :
78      |(T.mapPositive (ρ.map M.toPositiveTracePreservingMap)).rootFidelity (T.mapPositive σ) -
79        (Φ.map (transferred S M T)).rootFidelity Ψ| ≤ Real.sqrt δ + Real.sqrt η :=
80    hcont.trans (add_le_add (Real.sqrt_le_sqrt hd) (Real.sqrt_le_sqrt hη))
81  have hdata := T.fidelity_data_processing (ρ.map M.toPositiveTracePreservingMap) σ
82  have hle := (le_abs_self
83    ((T.mapPositive (ρ.map M.toPositiveTracePreservingMap)).rootFidelity (T.mapPositive σ) -
84      (Φ.map (transferred S M T)).rootFidelity Ψ)).trans hcont'
85  linarith

99/-- Uniform approximation gives the minimax bound against a supremum over all
100actual hybrid channels. The comparison maps are fixed before the competitor. -/
101theorem minimax_le_bayesValue
102    {I P Θ : Type*} [Nonempty I] [MeasurableSpace Θ]
103    (prior : Measure Θ) [IsProbabilityMeasure prior]
104    (S : HybridToQuantum G A μ) (M : I → QuantumChannel A B) (T : QuantumToHybrid B H μ)
105    (ρ : P → DensityState A) (σ : P → DensityState B)
106    (Φ : Θ → PositiveL1 G μ) (Ψ : Θ → PositiveL1 H μ)
107    (hΦ : ∀ θ, ‖(Φ θ).1‖ = 1) (hΨ : ∀ θ, ‖(Ψ θ).1‖ = 1)
108    (localChart : Θ → P) (δ η : ℝ)
109    (hδ : ∀ θ, ‖S.map (Φ θ).1 - (statePositive (ρ (localChart θ))).1‖ ≤ δ)
110    (hη : ∀ θ, ‖T.map (statePositive (σ (localChart θ))).1 - (Ψ θ).1‖ ≤ η)
111    (hintegrable : ∀ N : Hybrid.Channel G H μ,
112      Integrable (fun θ => ((Φ θ).map N).rootFidelity (Ψ θ)) prior) :
113    LAN.minimaxValue (fun i p =>
114      stateFidelity ((M i).toPositiveTracePreservingMap.mapState (ρ p)) (σ p)) ≤
115      LAN.bayesValue prior
116        (fun (N : Hybrid.Channel G H μ) θ => ((Φ θ).map N).rootFidelity (Ψ θ)) +
117          (Real.sqrt δ + Real.sqrt η) := by
118  apply LAN.minimax_le_bayesValue prior _ _ localChart (fun i => transferred S (M i) T) _
119    (fun _ _ => stateFidelity_nonneg _ _)
120  · intro i θ
121    simpa only [add_assoc] using stateFidelity_transfer S (M i) T
122      (ρ (localChart θ)) (σ (localChart θ)) (Φ θ) (Ψ θ) (hΦ θ) (hδ θ) (hη θ)
123  · exact hintegrable
124  · intro N θ
125    simpa only [PositiveL1.norm_map, hΦ θ, hΨ θ, Real.sqrt_one, mul_one] using
126      ((Φ θ).map N).rootFidelity_le_sqrt (Ψ θ)

133/-- Prior-almost-everywhere approximation suffices: a compact-supported prior
134requires estimates only on its compact window, even if the chart is defined on
135the entire phase space. -/
136theorem minimax_le_bayesValue_ae
137    {I P Θ : Type*} [Nonempty I] [MeasurableSpace Θ]
138    (prior : Measure Θ) [IsProbabilityMeasure prior]
139    (S : HybridToQuantum G A μ) (M : I → QuantumChannel A B) (T : QuantumToHybrid B H μ)
140    (ρ : P → DensityState A) (σ : P → DensityState B)
141    (Φ : Θ → PositiveL1 G μ) (Ψ : Θ → PositiveL1 H μ)
142    (hΦ : ∀ θ, ‖(Φ θ).1‖ = 1) (hΨ : ∀ θ, ‖(Ψ θ).1‖ = 1)
143    (localChart : Θ → P) (δ η : ℝ)
144    (hδ : ∀ᵐ θ ∂prior, ‖S.map (Φ θ).1 - (statePositive (ρ (localChart θ))).1‖ ≤ δ)
145    (hη : ∀ᵐ θ ∂prior, ‖T.map (statePositive (σ (localChart θ))).1 - (Ψ θ).1‖ ≤ η)
146    (hintegrable : ∀ N : Hybrid.Channel G H μ,
147      Integrable (fun θ => ((Φ θ).map N).rootFidelity (Ψ θ)) prior) :
148    LAN.minimaxValue (fun i p =>
149      stateFidelity ((M i).toPositiveTracePreservingMap.mapState (ρ p)) (σ p)) ≤
150      LAN.bayesValue prior
151        (fun (N : Hybrid.Channel G H μ) θ => ((Φ θ).map N).rootFidelity (Ψ θ)) +
152          (Real.sqrt δ + Real.sqrt η) := by
153  let f : I → P → ℝ := fun i p =>
154    stateFidelity ((M i).toPositiveTracePreservingMap.mapState (ρ p)) (σ p)
155  let g : Hybrid.Channel G H μ → Θ → ℝ := fun N θ => ((Φ θ).map N).rootFidelity (Ψ θ)
156  have hupper (N : Hybrid.Channel G H μ) (θ : Θ) : g N θ ≤ 1 := by
157    simpa only [g, PositiveL1.norm_map, hΦ θ, hΨ θ, Real.sqrt_one, mul_one] using
158      ((Φ θ).map N).rootFidelity_le_sqrt (Ψ θ)
159  have hbdd : BddAbove (Set.range (fun N => ∫ θ, g N θ ∂prior)) := by
160    refine ⟨1, ?_⟩
161    rintro x ⟨N, rfl⟩
162    have h := integral_mono (hintegrable N) (integrable_const (1 : ℝ)) (hupper N)
163    simpa only [integral_const, probReal_univ, one_smul] using h
164  change (⨆ i, ⨅ p, f i p) ≤ _
165  apply ciSup_le
166  intro i
167  have hbelow : BddBelow (Set.range (f i)) := by
168    refine ⟨0, ?_⟩
169    rintro x ⟨p, rfl⟩
170    exact stateFidelity_nonneg _ _
171  have hpoint : ∀ᵐ θ ∂prior,
172      (⨅ p, f i p) ≤ g (transferred S (M i) T) θ + (Real.sqrt δ + Real.sqrt η) := by
173    filter_upwards [hδ, hη] with θ hd he
174    apply (ciInf_le hbelow (localChart θ)).trans
175    simpa only [f, g, add_assoc] using stateFidelity_transfer S (M i) T
176      (ρ (localChart θ)) (σ (localChart θ)) (Φ θ) (Ψ θ) (hΦ θ) hd he
177  have havg := integral_mono_ae (integrable_const (⨅ p, f i p))
178    ((hintegrable (transferred S (M i) T)).add
179      (integrable_const (Real.sqrt δ + Real.sqrt η))) hpoint
180  simp only [Pi.add_apply] at havg
181  rw [integral_add (hintegrable (transferred S (M i) T)) (integrable_const _)] at havg
182  simp only [integral_const, probReal_univ, one_smul] at havg
183  exact havg.trans (add_le_add (le_ciSup hbdd (transferred S (M i) T)) le_rfl)
