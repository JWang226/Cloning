20theorem highest_tensorLength_eq (Ω : TensorRegister n (Fin d)) (Ψ : TensorRegister m (Fin d))

24    (hΩnorm : ‖Ω‖ = 1) (hΨnorm : ‖Ψ‖ = 1) : n = m := by
