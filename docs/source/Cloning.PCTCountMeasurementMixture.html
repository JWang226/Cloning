19theorem measuredDensity_matrixTensorPower (d N : ℕ) (hN : 0<N)
20    (ρ : Matrix (Fin (d+1)) (Fin (d+1)) ℂ) (p : Fin (d+1) → ℝ)
21    (hp : ∀ i,0≤p i) (hs : ∑ i,p i=1) (hdiag : ∀ i,ρ i i=(p i : ℂ))
22    (x : rootSpace d) : measuredDensity d N (matrixTensorPower ρ N) x=density d N p hp hs x := by
23  rw [measuredDensity_eq_clm d N hN, density_eq_word_sum d N hN]
24  simp only [densityCLM, ContinuousLinearMap.smul_apply, smul_eq_mul, weightCLM_apply_any]
25  congr 1
26  apply Finset.sum_congr rfl
27  intro w _
28  have he (w : Word N (d+1)) :
29      ⟪registerBasis (Word N (d+1)) w,(matrixTensorPower ρ N).1 (registerBasis _ w)⟫_ℂ =
30        (wordWeight p w : ℂ) := by
31    change matrixOf (registerBasis (Word N (d+1)))
32      (ofMatrix (registerBasis (Word N (d+1))) (fun a c => ∏ i,ρ (a i) (c i))) w w = _
33    rw [matrixOf_ofMatrix (registerBasis (Word N (d+1))).orthonormal]
34    simp only [hdiag, ← Complex.ofReal_prod, wordWeight]
35  unfold latticeWordLabel
36  by_cases hw : sampleLabel d N (flatSpectrum (d+1)) x=shapeLattice d N (countShape w)
37  · simp only [hw, if_true, he, Complex.ofReal_re]
38  · simp only [hw, if_false, Ne.symm hw]
