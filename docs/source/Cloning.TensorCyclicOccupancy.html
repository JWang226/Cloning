12theorem occupancy_update (w : Fin n → Fin d) (t : Fin n) (a i : Fin d) :
13    occupancy (Function.update w t a) i + (if w t = i then 1 else 0) =
14      occupancy w i + (if a = i then 1 else 0) := by
15  have he := Finset.sum_congr (s₁ := (Finset.univ : Finset (Fin n))) rfl
16    (fun u _ => show
17      (if Function.update w t a u = i then 1 else 0) +
18        (if u = t then (if w t = i then 1 else 0) else 0) =
19      (if w u = i then 1 else 0) +
20        (if u = t then (if a = i then 1 else 0) else 0) from by
21      by_cases h : u = t
22      · subst u; simp [Nat.add_comm]
23      · simp [h])
24  simpa only [occupancy, Finset.card_filter, Finset.sum_add_distrib,
25    Finset.sum_ite_eq', Finset.mem_univ, if_true] using he

27theorem occupancy_update_dist_le_one (w : Fin n → Fin d) (t : Fin n) (a i : Fin d) :
28    |(occupancy (Function.update w t a) i : ℤ) - occupancy w i| ≤ 1 := by
29  have he := occupancy_update w t a i

33/-- A lowering word can move any individual occupation by at most its total
34positive-root height. This statement is about literal computational coefficients. -/
35theorem loweringWord_occupancy_bound
36    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
37    (hweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
38    (w : List (PositiveRoot d)) (v : Fin n → Fin d)
39    (hv : loweringWord Ω w v ≠ 0) (i : Fin d) :
40    |(occupancy v i : ℤ) - mu i| ≤ loweringHeight w := by
41  induction w generalizing v with
42  | nil =>
43      have he := congrArg (fun x : TensorRegister n (Fin d) => x v) (hweight i)
44      simp only [collectiveGenerator_diagonal, lp.coeFn_smul, Pi.smul_apply, smul_eq_mul] at he
45      have heq : (occupancy v i : ℂ) = (mu i : ℂ) := (mul_right_cancel₀ hv he)
46      have hnat : occupancy v i = mu i := by exact_mod_cast heq
47      simp [hnat, loweringHeight]
48  | cons a w ih =>
49      have hv' : (∑ t : Fin n, if v t = a.val.2 then
50          loweringWord Ω w (Function.update v t a.val.1) else 0) ≠ 0 := by
51        simpa only [loweringWord, collectiveGenerator_apply] using hv
52      obtain ⟨t, ht⟩ : ∃ t : Fin n, (if v t = a.val.2 then
53          loweringWord Ω w (Function.update v t a.val.1) else 0) ≠ 0 := by
54        by_contra! hn
55        exact hv' (Finset.sum_eq_zero (fun t _ => hn t))
56      have ht' : loweringWord Ω w (Function.update v t a.val.1) ≠ 0 := by
57        split_ifs at ht with htval
58        · exact ht
59        · exact False.elim (ht rfl)
60      have hi := ih _ ht'
61      have hd := occupancy_update_dist_le_one v t a.val.1 i
62      have ha := a.height_pos
63      rw [abs_le] at hi hd ⊢
64      simp only [loweringHeight, Nat.cast_add]
65      omega

106/-- The exact Cartan occupation operator differs from its highest weight by
107at most the root-height cutoff. -/
108theorem cartan_defect_norm_le
109    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
110    (hweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
111    (r : ℕ) {x : TensorRegister n (Fin d)} (hx : x ∈ cyclicCutoff Ω r) (i : Fin d) :
112    ‖collectiveGenerator n i i x - (mu i : ℂ) • x‖ ≤ (r : ℝ) * ‖x‖ := by
113  apply register_norm_le_of_pointwise _ x r (Nat.cast_nonneg r)
114  intro v
115  simp only [lp.coeFn_sub, Pi.sub_apply, collectiveGenerator_diagonal,
116    lp.coeFn_smul, Pi.smul_apply, smul_eq_mul, ← sub_mul, norm_mul]
117  by_cases hv : x v = 0
118  · simp [hv]
119  · have hb := cyclicCutoff_occupancy_bound Ω mu hweight r hx v hv i
120    have hb' : |(occupancy v i : ℝ) - mu i| ≤ (r : ℝ) := by exact_mod_cast hb
121    have hc : ‖(occupancy v i : ℂ) - mu i‖ ≤ (r : ℝ) := by
122      simpa only [← Complex.ofReal_natCast, ← Complex.ofReal_sub,
123        Complex.norm_real, Real.norm_eq_abs] using hb'
124    exact mul_le_mul_of_nonneg_right hc (norm_nonneg _)

126theorem cartan_difference_defect_norm_le
127    (Ω : TensorRegister n (Fin d)) (mu : Fin d → ℕ)
128    (hweight : ∀ a, collectiveGenerator n a a Ω = (mu a : ℂ) • Ω)
129    (r : ℕ) {x : TensorRegister n (Fin d)} (hx : x ∈ cyclicCutoff Ω r) (a b : Fin d) :
130    ‖collectiveGenerator n a a x - collectiveGenerator n b b x -
131      ((mu a : ℂ) - mu b) • x‖ ≤ (2 * r : ℝ) * ‖x‖ := by
132  have he : collectiveGenerator n a a x - collectiveGenerator n b b x -
133      ((mu a : ℂ) - mu b) • x =
134      (collectiveGenerator n a a x - (mu a : ℂ) • x) -
135      (collectiveGenerator n b b x - (mu b : ℂ) • x) := by module
136  rw [he]
137  calc
138    _ ≤ ‖collectiveGenerator n a a x - (mu a : ℂ) • x‖ +
139        ‖collectiveGenerator n b b x - (mu b : ℂ) • x‖ := norm_sub_le _ _
140    _ ≤ (r : ℝ) * ‖x‖ + r * ‖x‖ := add_le_add
141      (cartan_defect_norm_le Ω mu hweight r hx a)
142      (cartan_defect_norm_le Ω mu hweight r hx b)
143    _ = _ := by ring
