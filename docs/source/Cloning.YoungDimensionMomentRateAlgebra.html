10/-- A bounded finite product has a quadratic remainder, with a bound that
11allows every finite physical label rather than just a central window. -/
12theorem product_quadratic_remainder_bounded {ι : Type*} (S : Finset ι)
13    (x : ι → ℝ) (C : ℝ) (hC : 1≤C) (hx : ∀i∈S,|1+x i|≤C) :
14    |(∏i∈S,(1+x i))-1-∑i∈S,x i| ≤ C^S.card*(∑i∈S,|x i|)^2 := by
15  induction S using Finset.induction_on with
16  | empty => simp
17  | @insert a S ha ih =>
18    have hi := ih (fun i hi=>hx i (Finset.mem_insert_of_mem hi))
19    have hxa := hx a (Finset.mem_insert_self a S)
20    have hs := Finset.abs_sum_le_sum_abs (s := S) (f := x)
21    have hs0 : 0≤∑i∈S,|x i| := Finset.sum_nonneg (fun _ _=>abs_nonneg _)
22    have hp : 1≤C^S.card := one_le_pow₀ hC
23    have hC0 : 0≤C := zero_le_one.trans hC
24    have hp0 : 0≤C^S.card := zero_le_one.trans hp
25    simp only [Finset.prod_insert ha,Finset.sum_insert ha,Finset.card_insert_of_notMem ha,pow_succ]
26    have he : (1+x a)*(∏i∈S,(1+x i))-1-(x a+∑i∈S,x i)=
27        (1+x a)*((∏i∈S,(1+x i))-1-∑i∈S,x i)+x a*(∑i∈S,x i) := by ring
28    rw [he]
29    calc
30      _ ≤ |1+x a| *|(∏i∈S,(1+x i))-1-∑i∈S,x i|+|x a| *|∑i∈S,x i| := by
31        simpa only [abs_mul] using abs_add_le
32          ((1+x a)*((∏i∈S,(1+x i))-1-∑i∈S,x i)) (x a*(∑i∈S,x i))
33      _ ≤ C*(C^S.card*(∑i∈S,|x i|)^2)+|x a| *(∑i∈S,|x i|) :=
34        add_le_add (mul_le_mul hxa hi (abs_nonneg _) hC0)
35          (mul_le_mul_of_nonneg_left hs (abs_nonneg _))
36      _ ≤ _ := by
37        have hCP : 1≤C^S.card*C := one_le_mul_of_one_le_of_one_le hp hC
38        have hcross := mul_nonneg (show 0≤2*(C^S.card*C)-1 by linarith)
39          (mul_nonneg (abs_nonneg (x a)) hs0)
40        have hsquare := mul_nonneg (mul_nonneg hp0 hC0) (sq_nonneg |x a|)
41        nlinarith

43theorem product_error_le_bounded_square_sum {ι : Type*} (S : Finset ι)
44    (x : ι → ℝ) (C : ℝ) (hC : 1≤C) (hx : ∀i∈S,|1+x i|≤C) :
45    |(∏i∈S,(1+x i))-1| ≤ |∑i∈S,x i|+
46      (C^S.card*S.card)*(∑i∈S,(x i)^2) := by
47  have hr := product_quadratic_remainder_bounded S x C hC hx
48  have hcs := Finset.sum_mul_sq_le_sq_mul_sq S (fun i=>|x i|) (fun _=>(1:ℝ))
49  simp only [mul_one,one_pow,Finset.sum_const,nsmul_eq_mul,mul_one,sq_abs] at hcs
50  have hsplit : (∏i∈S,(1+x i))-1=((∏i∈S,(1+x i))-1-∑i∈S,x i)+(∑i∈S,x i) := by ring
51  rw [hsplit]
52  calc
53    _ ≤ |(∏i∈S,(1+x i))-1-∑i∈S,x i|+|∑i∈S,x i| := abs_add_le _ _
54    _ ≤ C^S.card*(∑i∈S,|x i|)^2+|∑i∈S,x i| := add_le_add hr le_rfl
55    _ ≤ C^S.card*((∑i∈S,(x i)^2)*S.card)+|∑i∈S,x i| :=
56      add_le_add (mul_le_mul_of_nonneg_left hcs (pow_nonneg (zero_le_one.trans hC) _)) le_rfl
57    _ = _ := by ring

59private theorem bounded_second_order_envelope_le (A B C V N : ℝ)
60    (hA : 0≤A) (hB : 0≤B) (hC : 0≤C) (hV : 0≤V) (hN : 1≤N) :
61    A/N+B*V/N^2+C/N^2≤(A+B+C)*(1/N+V/N^2) := by

63  have hNN : N≤N^2 := by nlinarith
64  have hi : 1/N^2≤1/N := one_div_le_one_div_of_le hn hNN
