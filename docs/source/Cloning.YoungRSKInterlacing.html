3/-!
4# Column-strict interlacing under actual RSK insertion
5 
6For count rows, column strictness says that the number of lower-row letters
7at most `k` is no greater than the number of upper-row letters strictly below
8`k`.  The proof below verifies this invariant under a genuine bump propagated
9from an upper row into the next row.
10-/

17def RowInterlaces {d : ℕ} (U V : RowContent d) : Prop :=
18  ∀ k, rowPrefix V k ≤ rowStrictPrefix U k

20theorem rowStrictPrefix_le_prefix {d : ℕ} (A : RowContent d) (k : Fin d) :
21    rowStrictPrefix A k ≤ rowPrefix A k := by
22  apply Finset.sum_le_sum_of_subset
23  intro j hj
24  simp only [Finset.mem_filter, Finset.mem_univ, true_and] at hj ⊢
25  exact hj.le
26 
27theorem RowInterlaces.trans {d : ℕ} {U V W : RowContent d}
28    (hUV : RowInterlaces U V) (hVW : RowInterlaces V W) : RowInterlaces U W := by
29  intro k
30  exact (hVW k).trans ((rowStrictPrefix_le_prefix V k).trans (hUV k))

32theorem rowInserted_strictPrefix_ge {d : ℕ} (A : RowContent d) (x k : Fin d) :
33    rowStrictPrefix A k ≤ rowStrictPrefix (rowInserted A x) k := by
34  cases h : rowBump A x with
35  | none => rw [rowInserted_strictPrefix_of_none A x k h]; omega
36  | some y => rw [rowInserted_strictPrefix_of_bump A x y k h]; omega

38theorem RowInterlaces.insert_upper {d : ℕ} {U V : RowContent d}
39    (hUV : RowInterlaces U V) (x : Fin d) : RowInterlaces (rowInserted U x) V := by
40  intro k
41  exact (hUV k).trans (rowInserted_strictPrefix_ge U x k)

43/-- A bumped letter inserted into the next row preserves strict columns.
44Both bumping and terminal append cases are handled by the actual algorithm. -/
45theorem RowInterlaces.insert_bumped {d : ℕ} {U V : RowContent d}
46    (hUV : RowInterlaces U V) (a b : Fin d) (hb : rowBump U a = some b) :
47    RowInterlaces (rowInserted U a) (rowInserted V b) := by
48  obtain ⟨hab, hbpos, _⟩ := (rowBump_eq_some_iff U a b).mp hb
49  intro k
50  rw [rowInserted_strictPrefix_of_bump U a b k hb]
51  cases hc : rowBump V b with
52  | none =>
53    rw [rowInserted_prefix_of_none V b k hc]
54    by_cases hbk : b ≤ k
55    · rw [if_pos hbk]
56      rcases eq_or_lt_of_le hbk with rfl | hbk
57      · rw [if_pos ⟨hab, le_rfl⟩]
58        exact Nat.add_le_add_right (hUV b) 1
59      · have he := rowPrefix_eq_of_no_entries V b k hbk.le
60          (fun j hbj _ ↦ (rowBump_eq_none_iff V b).mp hc j hbj)
61        rw [he]
62        have hs := rowStrictPrefix_slack U b k hbk hbpos
63        have hu := hUV b
64        omega
65    · rw [if_neg hbk]
66      have hu := hUV k
67      omega
68  | some c =>
69    rw [rowInserted_prefix_of_bump V b c k hc]
70    by_cases hchange : b ≤ k ∧ k < c
71    · rw [if_pos hchange]
72      rcases eq_or_lt_of_le hchange.1 with rfl | hbk
73      · rw [if_pos ⟨hab, le_rfl⟩]
74        exact Nat.add_le_add_right (hUV b) 1
75      · obtain ⟨_, _, hcmin⟩ := (rowBump_eq_some_iff V b c).mp hc
76        have he := rowPrefix_eq_of_no_entries V b k hbk.le (fun j hbj hjk ↦ by
77          by_contra hj
78          have := hcmin j hbj (Nat.pos_of_ne_zero hj)
79          exact (not_le_of_gt hchange.2) (this.trans hjk))
80        rw [he]
81        have hs := rowStrictPrefix_slack U b k hbk hbpos
82        have hu := hUV b
83        omega
84    · rw [if_neg hchange]
85      have hu := hUV k
86      omega

92/-- Terminating below the existing final row is also column-strict. -/
93theorem RowInterlaces.insert_new_row {d : ℕ} (U : RowContent d) (a b : Fin d)
94    (hb : rowBump U a = some b) : RowInterlaces (rowInserted U a) (rowAdd 0 b) := by
95  have h := (rowInterlaces_zero U).insert_bumped a b hb
96  have hz : rowBump (0 : RowContent d) b = none :=
97    (rowBump_eq_none_iff _ _).mpr (fun _ _ ↦ rfl)
98  simpa only [rowInserted, hz] using h
