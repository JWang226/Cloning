15theorem length_le_loweringHeight (w : List (PositiveRoot d)) : w.length≤loweringHeight w := by
16  induction w with

49/-- Uniform exact-weight version of the appendix PBW lemma. The finite
50dimensional Hilbert space, Gram inverse, ONB, completeness, and quantitative
51error all refer to the literal physical sector. -/
52theorem eventually_weightSymmetricBasis_rate (R : ℕ) (c : ℝ) (hc : 0<c) :
53    ∃ C : ℝ,0≤C ∧ ∀ᶠ (N : ℕ) in atTop,
54      ∀ (mu : Fin d→ℕ) (hmu : Antitone mu),
55      (∀ a : PositiveRoot d,c*(N:ℝ)≤rootGap mu a) →
56      ∀ (η : Fin d→ℤ), (∑a,(a.val:ℤ)*η a≤(R:ℤ)) →
57      ∃ b : OrthonormalBasis (WeightOccupation d R η) ℂ
58        ↥(cyclicSector (partitionHighestTensor mu hmu) ⊓ cartanWeightSpace (∑a,mu a) mu η),
59        (∀i,(b i : TensorRegister (∑a,mu a) (Fin d))=weightSymmetricFrame mu hmu R η i) ∧
60        ∀i,‖weightSymmetricFrame mu hmu R η i-weightRawFrame mu hmu R η i‖≤
61          C/Real.sqrt (N:ℝ) := by
62  classical
63  let words : HeightOccupation d R→List (PositiveRoot d) := fun k=>canonicalWord k.val
64  have hwords : ∀i j,(words i).Perm (words j)↔i=j := by
65    intro i j
66    simp only [words,canonicalWord_perm_iff,Subtype.val_inj]
67  have hlen : ∀i,(words i).length≤R := fun i=>
68    (length_le_loweringHeight _).trans (by rw [canonicalWord_height]; exact i.property)
69  let A := wordGramConstant R words*rootErrorRateConstant (R*d)/Real.sqrt c
70  let D : ℝ := Fintype.card (HeightOccupation d R)
71  let C := D^2*A
72  have hA : 0≤A := div_nonneg
73    (mul_nonneg (wordGramConstant_nonneg _ _) (rootErrorRateConstant_nonneg _))
74    (Real.sqrt_nonneg c)
75  have hC : 0≤C := mul_nonneg (sq_nonneg _) hA
76  have hN : Tendsto (fun N : ℕ=>c*(N:ℝ)) atTop atTop :=
77    tendsto_natCast_atTop_atTop.const_mul_atTop hc
78  have hz : Tendsto (fun N : ℕ=>C/Real.sqrt (N:ℝ)) atTop (𝓝 0) :=
79    tendsto_const_nhds.div_atTop (Real.tendsto_sqrt_atTop.comp tendsto_natCast_atTop_atTop)
80  refine ⟨C,hC,?_⟩
81  filter_upwards [hN.eventually (eventually_ge_atTop (max 1 (2*((((R*d:ℕ):ℝ))+1)))),
82    hz.eventually (eventually_lt_nhds zero_lt_one)] with N hlarge hsmall
83  intro mu hmu hgap η hη
84  have hpos : ∀a : PositiveRoot d,0<rootGap mu a := fun a=>
85    (by have := (le_max_left _ _).trans hlarge; linarith : 0<c*(N:ℝ)).trans_le (hgap a)
86  let v := weightRawFrame mu hmu R η
87  have herr : ∀i j,‖⟪v i,v j⟫_ℂ-(if i=j then 1 else 0 : ℂ)‖≤A/Real.sqrt (N:ℝ) := by
88    intro i j
89    have h := physical_words_gram_error_le R words hwords hlen mu hmu
90      ((le_max_right _ _).trans hlarge) hgap i.val j.val
91    have he : (i.val=j.val)↔i=j := Subtype.val_inj
92    rw [he] at h
93    apply h.trans
94    apply (mul_le_mul_of_nonneg_left
95      (rootError_le_div_sqrt (R*d) ((le_max_left _ _).trans hlarge))
96      (wordGramConstant_nonneg _ _)).trans_eq
97    rw [Real.sqrt_mul hc.le]
98    dsimp only [A]
99    ring
100  have hcard : (Fintype.card (WeightOccupation d R η):ℝ)≤D := by
101    dsimp only [D]
102    exact_mod_cast Fintype.card_le_of_injective
103      (fun k : WeightOccupation d R η=>k.val) Subtype.val_injective
104  have hbound : (Fintype.card (WeightOccupation d R η):ℝ)^2*(A/Real.sqrt (N:ℝ))≤
105      C/Real.sqrt (N:ℝ) := by
106    dsimp only [C]
107    rw [mul_div_assoc]
108    apply mul_le_mul_of_nonneg_right _ (div_nonneg hA (Real.sqrt_nonneg _))
109    exact pow_le_pow_left₀ (Nat.cast_nonneg _) hcard 2
110  obtain ⟨hon,hspan,he⟩ := symmetricFrame_properties v _ herr (hbound.trans_lt hsmall)
111  have hs : Submodule.span ℂ (Set.range (symmetricFrame v))=
112      cyclicSector (partitionHighestTensor mu hmu) ⊓ cartanWeightSpace (∑a,mu a) mu η :=

116    hs.le (Submodule.subset_span (Set.mem_range_self i))
