19structure IsNormalizedPositive (K : X → X → ℂ) : Prop where
20  hermitian : ∀ x y, star (K y x) = K x y
21  diagonal : ∀ x, K x x = 1
22  positive : ∀ (s : Finset X) (c : X → ℂ),
23    0 ≤ (∑ x ∈ s, ∑ y ∈ s, star (c x) * c y * K x y).re

66lemma inner_feature (x y : X) : ⟪feature K x, feature K y⟫_ℂ = K x y := by
67  unfold feature
68  rw [UniformSpace.Completion.inner_coe]
69  change ((Finsupp.single x 1 : X →₀ ℂ).sum fun a u =>
70    (Finsupp.single y 1 : X →₀ ℂ).sum fun b v => star u * v * K a b) = _
71  simp

73lemma norm_feature (x : X) : ‖feature K x‖ = 1 := by
74  have h := congrArg Complex.re (inner_feature K x x)
75  rw [hK.out.diagonal, inner_self_eq_norm_sq_to_K] at h
76  norm_cast at h
77  change ‖feature K x‖ ^ 2 = 1 at h
78  nlinarith [norm_nonneg (feature K x)]

80lemma norm_feature_sub (x y : X) :
81    ‖feature K x - feature K y‖ = Real.sqrt (2 - 2 * (K x y).re) := by

115/-- Finite positivity has become positivity and Cauchy--Schwarz for actual
116Bochner integrals. The weight may be signed and the measure need not be finite. -/
117theorem integral_cauchySchwarz
118    (hc : Continuous (fun p : X × X => K p.1 p.2))
119    {w : X → ℝ} (hw : Integrable w μ) (x₀ : X) :
120    ‖∫ x, (w x : ℂ) * K x₀ x ∂μ‖ ^ 2 ≤
121      (∫ x, ∫ y, (w x : ℂ) * (w y : ℂ) * K x y ∂μ ∂μ).re := by
122  let v : X → Space K := fun x => (w x : ℂ) • feature K x
123  have hv : Integrable v μ := integrable_weighted_feature K hc hw
124  have hfirst : (∫ x, (w x : ℂ) * K x₀ x ∂μ) =
125      ⟪feature K x₀, ∫ x, v x ∂μ⟫_ℂ := by
126    convert integral_inner hv (feature K x₀) using 1
127    congr 1
128    funext x
129    simp only [v, inner_smul_right, inner_feature]
130  have hdouble : (∫ x, ∫ y, (w x : ℂ) * (w y : ℂ) * K x y ∂μ ∂μ) =
131      ⟪∫ x, v x ∂μ, ∫ x, v x ∂μ⟫_ℂ := by
132    calc
133      _ = ∫ x, ∫ y, ⟪v x, v y⟫_ℂ ∂μ ∂μ := by
134        congr 1
135        funext x
136        congr 1
137        funext y
138        simp only [v, inner_smul_left, inner_smul_right, inner_feature,
139          Complex.conj_ofReal]
140        ring
141      _ = ∫ x, ⟪v x, ∫ y, v y ∂μ⟫_ℂ ∂μ := by
142        congr 1
143        funext x
144        exact integral_inner hv (v x)
145      _ = star (∫ x, ⟪∫ y, v y ∂μ, v x⟫_ℂ ∂μ) := by
146        simp_rw [← inner_conj_symm (v _) (∫ y, v y ∂μ)]
147        exact integral_conj
148      _ = _ := by rw [integral_inner hv]; exact inner_conj_symm _ _
149  rw [hfirst, hdouble]
150  have h := norm_inner_le_norm (𝕜 := ℂ) (feature K x₀) (∫ x, v x ∂μ)
151  rw [norm_feature, one_mul] at h
152  have hs := pow_le_pow_left₀ (norm_nonneg _) h 2
153  rw [inner_self_eq_norm_sq_to_K]
154  change _ ≤ ((‖∫ x, v x ∂μ‖ : ℂ) ^ 2).re
155  rw [← Complex.ofReal_pow, Complex.ofReal_re]
156  exact hs
